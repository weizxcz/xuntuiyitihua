## 获取过约束集合对象

**类型：** 基础命令

**说明：** 获取过约束集合对象

**示例：**

```python
#过约束冲突约束定位：查询重合约束-过约束子集
doc.RunCommand("cmd_ncti_create_dcm3_dim_sys", "dcmsys")

doc.RunCommand("cmd_ncti_create_circle", "circle1", NCTI.Point(0, 0, 0), NCTI.Vector(0, 0, 1), 1)
doc.RunCommand("cmd_ncti_create_dcm3_circle", "dcm3_circle1", "dcmsys", NCTI.Point(0,0,0), NCTI.Vector(0,0,1), 1, "circle1")


doc.RunCommand("cmd_ncti_create_vertex", "point1", NCTI.Point(0, 0, 0))
doc.RunCommand("cmd_ncti_create_dcm3_point", "dcm3_point1", "dcmsys", NCTI.Point(0,0,0), "point1")

doc.RunCommand("cmd_ncti_create_dcm3_dimension", "cons_coinc1", DCM3_COINCIDENT)
doc.RunCommand("cmd_ncti_edit_dcm3_dim_setends", "cons_coinc1", "dcm3_circle1", "dcm3_point1")
doc.RunCommand("cmd_ncti_edit_dcm3_dim_setpoint", "cons_coinc1", "dcm3_circle1", DCM3_DIM_TO_HELP_POINT, NCTI.Point(-1,0,0))
doc.RunCommand("cmd_ncti_dcm3_addd", "dcmsys", "cons_coinc1", "dcm3_circle1", "dcm3_point1")


doc.RunCommand("cmd_ncti_create_dcm3_dimension", "cons_dist1", DCM3_DISTANCE, 0)
doc.RunCommand("cmd_ncti_edit_dcm3_dim_setends", "cons_dist1", "dcm3_circle1", "dcm3_point1")
doc.RunCommand("cmd_ncti_edit_dcm3_dim_setpoint", "cons_dist1", "dcm3_circle1", DCM3_DIM_TO_HELP_POINT, NCTI.Point(-1,0,0))
doc.RunCommand("cmd_ncti_dcm3_addd", "dcmsys", "cons_dist1", "dcm3_circle1", "dcm3_point1")

doc.RunCommand("cmd_ncti_dcm3_evaluate", "dcmsys")


cons_type_dict ={
    0: "DISTANCE",
    1: "ANGLE",
    2: "CONCENTRIC",
    3: "RADIUS",
    4: "PARALLEL",
    5: "PERPENDICULAR",
    6: "COINCIDENT",
    7: "TANGENT",
    10: "MAJOR_RADIUS",
    11: "MINOR_RADIUS",
    12: "EQUAL_RADIUS",
    14: "MIDPOINT",
    15: "DEPENDENCE",
    17: "EQUAL_PARAMETER",
    19: "EQUAL_DIRECTION",
    20: "EQUAL_CURVATURE",
    21: "EQUAL_FIRST_DERIVATIVE",
    22: "EQUAL_SECOND_DERIVATIVE ",
    25: "CURVE_LENGTH ",
    50: "SYMMETRIC ",
    51: "HALF_ANGLE ",
    52: "COINCIDENT_AROUND_SPINE ",
    53: "COINCIDENT_WITH_SPINE ",
    56: "EQUAL_HALF_ANGLE ",
    59: "EQUAL_CURVATURE_VECTOR ",
    60: "ANGLE_PLANAR ",
    61: "SURFACE_NORMAL ",
    62: "PATTERNED ",
    63: "PATTERN_VALUE ",
    64: "EQUAL_DISTANCED ",
    65: "EQUAL_DISTANCE_VALUE ",
}

consDatas = doc.dcm3_get_overdefined_constraints("dcmsys")
index = 0
for consData in consDatas:
    index += 1
    print(f"**********overdefined constraint data{index}**************")
    dimName = consData.dim
    cons_type = consData.dim_type
    geom1Name = consData.geom1
    geom2Name = consData.geom2
    geom3Name = consData.geom3
    if dimName != None:
        print(f"overdefined dimension name: {dimName}")
        print(f"overdefined constraint type: {cons_type_dict[cons_type]}")
    if geom1Name != None:
        print(f"overdefined geometry1 name: {geom1Name}")
    if geom2Name != None:
        print(f"overdefined geometry2 name: {geom2Name}")
    if geom3Name != None:
        print(f"overdefined geometry3 name: {geom3Name}")

doc.OverdefinedQueryMark(0)

```

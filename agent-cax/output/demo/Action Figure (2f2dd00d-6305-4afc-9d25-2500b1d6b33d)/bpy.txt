def clear_scene():
    """Delete all existing objects in the scene."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Ensure collections are clean
    for col in bpy.data.collections:
        bpy.data.collections.remove(col)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign a material to an object's first material slot."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive(type, name, size, location, rotation_euler):
    """Add a primitive object, set name, location, and rotation."""
    bpy.ops.object.select_all(action='DESELECT')
    
    if type == 'cube':
        # size is (x, y, z)
        bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif type == 'cylinder':
        # size is (diameter, height) -> radius, depth
        radius = size[0] / 2
        depth = size[1]
        bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))
        obj = bpy.context.active_object
        # Default cylinder is along Z. If we want it along Y later, we rotate.
    elif type == 'sphere':
        # size is diameter
        radius = size[0] / 2
        bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=(0, 0, 0))
        obj = bpy.context.active_object
    elif type == 'torus':
        # size is (major_radius, minor_radius)
        major_r = size[0] / 2
        minor_r = size[1] / 2
        bpy.ops.mesh.primitive_torus_add(major_radius=major_r, minor_radius=minor_r, location=(0, 0, 0))
        obj = bpy.context.active_object
    else:
        return None

    obj.name = name
    obj.location = location
    obj.rotation_euler = rotation_euler
    obj.select_set(True)
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """
    Align a specific axis of an object to a target vector.
    axis_to_align: 'X', 'Y', or 'Z'
    target_vector: Vector in world space
    """
    # Get current world matrix
    mat = obj.matrix_world.copy()
    
    # Get the current direction of the axis to align
    if axis_to_align == 'X':
        current_axis = mat.to_3x3().col[0]
    elif axis_to_align == 'Y':
        current_axis = mat.to_3x3().col[1]
    elif axis_to_align == 'Z':
        current_axis = mat.to_3x3().col[2]
    
    # Calculate rotation matrix to align current_axis to target_vector
    # We use quaternion rotation
    rot_quat = current_axis.rotation_difference(target_vector.normalized())
    
    # Apply rotation
    obj.rotation_euler = rot_quat.to_euler()
    obj.update_tag()

def boolean_union(objects_to_union, result_name):
    """
    Perform a boolean union on a list of objects.
    Returns the resulting object.
    """
    if not objects_to_union:
        return None
    
    # Select all objects
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects_to_union:
        obj.select_set(True)
    
    # Set active object as the first one
    bpy.context.view_layer.objects.active = objects_to_union[0]
    
    # Apply modifiers to clear history before boolean if needed, 
    # but for simple union, we can use the modifier stack.
    # However, to get a clean mesh named 'result_name', we usually:
    # 1. Add Boolean modifier to the first object with the others as operands.
    # 2. Apply it.
    
    target = objects_to_union[0]
    
    # Add boolean modifier to target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    
    # Add other objects to the modifier's object list
    for obj in objects_to_union[1:]:
        mod.object = obj
        # Trigger update
        bpy.context.view_layer.update()
        
    # Apply the modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Rename
    target.name = result_name
    
    # Deselect all
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    bpy.context.view_layer.objects.active = target
    
    return target

def boolean_subtract(base_obj, cutter_obj, result_name):
    """
    Subtract cutter_obj from base_obj.
    """
    bpy.ops.object.select_all(action='DESELECT')
    base_obj.select_set(True)
    bpy.context.view_layer.objects.active = base_obj
    
    mod = base_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    base_obj.name = result_name
    
    bpy.ops.object.select_all(action='DESELECT')
    base_obj.select_set(True)
    bpy.context.view_layer.objects.active = base_obj
    
    return base_obj

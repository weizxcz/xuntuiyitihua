def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube primitive, set size, location, rotation, and material."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        if len(obj.data.materials) == 0:
            obj.data.materials.append(material)
        else:
            obj.data.materials[0] = material
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))

def align_face_to_face(obj_source, face_type_source, obj_target, face_type_target, offset_vector=Vector((0,0,0))):
    """
    Aligns a specific face of the source object to a specific face of the target object.
    face_type: 'top', 'bottom', 'front', 'back', 'left', 'right'
    Assumes objects are axis-aligned.
    """
    # Get bounding box corners in world space
    def get_face_center(obj, face_type):
        # Transform local bounds to world
        corners = [
            obj.matrix_world @ Vector((-obj.dimensions.x/2, -obj.dimensions.y/2, -obj.dimensions.z/2)),
            obj.matrix_world @ Vector((obj.dimensions.x/2, -obj.dimensions.y/2, -obj.dimensions.z/2)),
            obj.matrix_world @ Vector((-obj.dimensions.x/2, obj.dimensions.y/2, -obj.dimensions.z/2)),
            obj.matrix_world @ Vector((obj.dimensions.x/2, obj.dimensions.y/2, -obj.dimensions.z/2)),
            obj.matrix_world @ Vector((-obj.dimensions.x/2, -obj.dimensions.y/2, obj.dimensions.z/2)),
            obj.matrix_world @ Vector((obj.dimensions.x/2, -obj.dimensions.y/2, obj.dimensions.z/2)),
            obj.matrix_world @ Vector((-obj.dimensions.x/2, obj.dimensions.y/2, obj.dimensions.z/2)),
            obj.matrix_world @ Vector((obj.dimensions.x/2, obj.dimensions.y/2, obj.dimensions.z/2)),
        ]
        
        # Determine center based on face type
        # Note: This assumes standard orientation where X=Right, Y=Front, Z=Up
        # 'top' is max Z, 'bottom' is min Z
        # 'front' is max Y, 'back' is min Y
        # 'right' is max X, 'left' is min X
        
        if face_type == 'top':
            return Vector([c.x for c in corners if c.z == max(c.z for c in corners)][0], 
                          [c.y for c in corners if c.z == max(c.z for c in corners)][0],
                          max(c.z for c in corners))
        elif face_type == 'bottom':
            return Vector([c.x for c in corners if c.z == min(c.z for c in corners)][0], 
                          [c.y for c in corners if c.z == min(c.z for c in corners)][0],
                          min(c.z for c in corners))
        elif face_type == 'front':
            return Vector([c.x for c in corners if c.y == max(c.y for c in corners)][0], 
                          max(c.y for c in corners),
                          [c.z for c in corners if c.y == max(c.y for c in corners)][0])
        elif face_type == 'back':
            return Vector([c.x for c in corners if c.y == min(c.y for c in corners)][0], 
                          min(c.y for c in corners),
                          [c.z for c in corners if c.y == min(c.y for c in corners)][0])
        elif face_type == 'right':
            return Vector(max(c.x for c in corners), 
                          [c.y for c in corners if c.x == max(c.x for c in corners)][0],
                          [c.z for c in corners if c.x == max(c.x for c in corners)][0])
        elif face_type == 'left':
            return Vector(min(c.x for c in corners), 
                          [c.y for c in corners if c.x == min(c.x for c in corners)][0],
                          [c.z for c in corners if c.x == min(c.x for c in corners)][0])
        return obj.matrix_world.to_translation()

    src_center = get_face_center(obj_source, face_type_source)
    tgt_center = get_face_center(obj_target, face_type_target)
    
    # Calculate translation needed to align src_face to tgt_face
    # We want src_face_center to move to tgt_face_center
    diff = tgt_center - src_center
    
    # Apply offset
    final_loc = obj_source.location + diff + offset_vector
    
    obj_source.location = final_loc
    return obj_source

def orient_z_to_z(obj):
    """Ensure object's +Z aligns with world +Z (no-op if already aligned, but ensures rotation is clean)."""
    # If the object was created with default orientation, it's already aligned.

    obj.rotation_euler = (0, 0, 0)
    return obj

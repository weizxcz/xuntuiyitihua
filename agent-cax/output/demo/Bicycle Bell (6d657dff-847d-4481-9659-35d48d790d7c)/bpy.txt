import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube mesh object with specific size."""
    bpy.ops.mesh.primitive_cube_add(size=1) # size parameter in add is half-width
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    return obj

def add_uvsphere(name, radius, res=32):
    """Add a UV sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=res, ring_count=res//2)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius, height):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=height)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, segments=32):
    """Add a disc mesh object."""
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='TRIFAN', segments=segments)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def boolean_subtract(cutter_obj, target_obj):
    """Apply Boolean Subtract modifier from cutter to target."""
    # Ensure target has a modifier slot
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(union_obj, target_obj):
    """Apply Boolean Union modifier from union_obj to target."""
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove union_obj
    bpy.data.objects.remove(union_obj, do_unlink=True)

def add_bevel(obj, width, segments=4, profile=0.5):

    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.profile = profile
    # Apply immediately for simplicity in this script flow
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a global vector."""
    # axis: 'X', 'Y', or 'Z'
    # vector: target global direction
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else:
        local_axis = Vector((0, 0, 1))
        
    # Calculate rotation
    rot_quat = local_axis.rotation_difference(vector.normalized())
    obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average position of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = sum(points, Vector()) / len(points)
    return avg

class Locator:
    """Helper class to find specific points on an object's mesh."""
    def __init__(self, obj):
        self.obj = obj
        self.mesh = obj.data
        self.verts = self.mesh.vertices
        self.has_calc_normals = False
        
    def _ensure_normals(self):
        if not self.has_calc_normals:
            self.mesh.calc_normals_split()
            self.has_calc_normals = True

    def center_world(self):
        """Get the world-space center of the object's bounding box."""
        bbox = [self.obj.matrix_world @ v.co for v in self.verts]
        return average(bbox)

    def face_center_world(self, face_index):
        """Get the world-space center of a specific face."""
        self._ensure_normals()
        face = self.mesh.polygons[face_index]
        # Get vertices of the face
        verts_co = [self.obj.matrix_world @ self.verts[i].co for i in face.vertices]
        return average(verts_co)

    def radial_point_world(self, angle_deg, axis='Z', center_offset=Vector((0,0,0))):
        """Get a point on the radial perimeter of the object's bounding box."""
        # Simple approximation: find min/max extents on X/Y plane
        xs = [v.co.x for v in self.verts]
        ys = [v.co.y for v in self.verts]
        zs = [v.co.z for v in self.verts]
        
        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)
        min_z, max_z = min(zs), max(zs)
        
        cx = (min_x + max_x) / 2
        cy = (min_y + max_y) / 2
        cz = (min_z + max_z) / 2
        
        # Radius approximation
        rx = (max_x - min_x) / 2
        ry = (max_y - min_y) / 2
        
        angle_rad = math.radians(angle_deg)
        
        # Calculate point on ellipse
        px = cx + rx * math.cos(angle_rad)
        py = cy + ry * math.sin(angle_rad)
        pz = cz # Mid Z
        
        return Vector((px, py, pz))

    def bottom_face_world(self):
        """Get the center of the bottom-most face."""
        self._ensure_normals()
        min_z = min(v.co.z for v in self.verts)
        # Find faces that have vertices at min_z
        candidate_faces = []
        for i, face in enumerate(self.mesh.polygons):
            verts_z = [self.verts[v_idx].co.z for v_idx in face.vertices]
            if min(verts_z) == min_z:
                candidate_faces.append(i)
        
        if not candidate_faces:
            return self.center_world()
            
        # Return the center of the first candidate face
        return self.face_center_world(candidate_faces[0])

    def top_face_world(self):
        """Get the center of the top-most face."""
        self._ensure_normals()
        max_z = max(v.co.z for v in self.verts)
        candidate_faces = []
        for i, face in enumerate(self.mesh.polygons):
            verts_z = [self.verts[v_idx].co.z for v_idx in face.vertices]
            if max(verts_z) == max_z:
                candidate_faces.append(i)
        
        if not candidate_faces:
            return self.center_world()
            
        return self.face_center_world(candidate_faces[0])

    def side_at(self, angle_deg):
        """Get the center of the side face at a given angle (0=+X, 90=+Y)."""
        # This is a simplified heuristic for cylindrical objects
        # It returns a point on the perimeter
        return self.radial_point_world(angle_deg)

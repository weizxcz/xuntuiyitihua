import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=depth, segments=segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    bpy.ops.mesh.primitive_cube_add(size=size, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_segments=32, height_segments=16):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, axis_to_align='Z', target_vector=(0, 0, 1)):
    """Aligns an object's local axis to a target world vector."""
    if axis_to_align == 'Z':
        axis = Vector((0, 0, 1))
    elif axis_to_align == 'Y':
        axis = Vector((0, 1, 0))
    elif axis_to_align == 'X':
        axis = Vector((1, 0, 0))
    else:
        return

    target = Vector(target_vector).normalized()
    current = axis.copy()
    
    # Calculate rotation matrix
    rot_mat = current.rotation_difference(target).to_matrix().to_4x4()
    obj.matrix_world = rot_mat @ obj.matrix_world

def boolean_union(objs, result_name):
    """Performs boolean union on a list of objects and returns the resulting object."""
    if not objs:
        return None
    
    # Select all objects
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objs:
        obj.select_set(True)
    
    # Set active object as the first one
    bpy.context.view_layer.objects.active = objs[0]
    
    # Apply boolean modifier to all but the first
    for obj in objs[1:]:
        mod = obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = objs[0]
        bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Now we have multiple meshes in the active object's place? 
    # Actually, applying boolean on secondary objects modifies them to match the first.
    # We need to join them.
    bpy.ops.object.join()
    
    result = bpy.context.active_object
    result.name = result_name
    return result

def boolean_subtract(base_obj, cutter_obj, result_name):
    """Subtracts cutter from base using boolean modifier."""
    mod = base_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    base_obj.name = result_name
    return base_obj

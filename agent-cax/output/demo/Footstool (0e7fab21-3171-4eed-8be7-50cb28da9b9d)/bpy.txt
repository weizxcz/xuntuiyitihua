def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube and apply transforms."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):

    obj = bpy.context.active_object

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Align an object's local axis to a target world vector."""
    # Get current rotation
    rot = obj.rotation_euler.to_matrix()
    
    # Define source axis
    if axis_to_align == 'X': src = Vector((1, 0, 0))
    elif axis_to_align == 'Y': src = Vector((0, 1, 0))
    elif axis_to_align == 'Z': src = Vector((0, 0, 1))
    else: return

    # Calculate rotation matrix to align src to target
    # Normalize target
    target = target_vector.normalized()
    
    # Use mathutils rotation
    rot_mat = src.rotation_difference(target).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_euler = rot_mat.to_euler()

def average(points):
    """Calculate the average of a list of Vector objects."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

    def center_world(self):
        """Returns the world space center of the object."""
        return self.obj.matrix_world.to_translation()

    def face_center_world(self, face_normal_axis='Z', face_direction='positive'):
        """
        Returns the world space center of a specific face.
        Assumes axis-aligned primitives for simplicity unless complex mesh.
        face_normal_axis: 'X', 'Y', or 'Z'
        face_direction: 'positive' or 'negative'
        """
        # Get local bounds
        min_x, min_y, min_z = self.obj.dimensions[0]/2 * -1, self.obj.dimensions[1]/2 * -1, self.obj.dimensions[2]/2 * -1
        max_x, max_y, max_z = self.obj.dimensions[0]/2, self.obj.dimensions[1]/2, self.obj.dimensions[2]/2
        
        # Local center of the face
        local_center = Vector((0, 0, 0))
        
        if face_normal_axis == 'X':
            local_center.x = max_x if face_direction == 'positive' else min_x
        elif face_normal_axis == 'Y':
            local_center.y = max_y if face_direction == 'positive' else min_y
        elif face_normal_axis == 'Z':
            local_center.z = max_z if face_direction == 'positive' else min_z
            
        # Transform to world space
        return self.obj.matrix_world @ local_center

    def radial_point_world(self, axis='Z', angle=0):
        """Returns a point on the radial edge."""
        # Simplified for cylinders/spheres
        return self.obj.matrix_world @ Vector((math.cos(angle), math.sin(angle), 0))

def boolean_subtract(cutter_name, target_name):
    """Performs a boolean subtract using the modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Optional: Remove cutter to keep scene clean, or keep it. 
    # Usually keep for debugging, but let's remove it to avoid clutter if not needed later.
    # However, the prompt implies using cutters. We will leave them.
    return target

def boolean_union(target_name, union_obj_name):
    """Performs a boolean union."""
    target = bpy.data.objects[target_name]
    union_obj = bpy.data.objects[union_obj_name]
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    return target

def add_bevel(obj_name, width, segments=2):
    """Adds a bevel modifier."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for geometry generation
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

import bpy
import math
import bmesh
from mathutils import Vector, Matrix, Euler

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with given diffuse color (RGBA)."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Alpha'].default_value = diffuse_color[3]
    if diffuse_color[3] < 1.0:
        mat.blend_method = 'ALPHA'
        mat.shadow_mode = 'NONE'
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube object with specific size and transform."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):
    """Add a cylinder object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def add_uvsphere(name, radius, location, rotation_euler, material=None):
    """Add a UV sphere object."""
    bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=radius, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def add_cone(name, radius_top, radius_bottom, depth, location, rotation_euler, material=None):

    obj.rotation_euler = rotation_euler

def add_disc(name, radius, thickness, location, rotation_euler, material=None):

    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=thickness, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=Vector((0, 0, 1))):
    """Align object's local axis to a world vector."""
    if axis == 'Z':
        obj.rotation_mode = 'QUATERNION'
        obj.rotation_quaternion = vector.to_track_quat('Z', 'Y')
    elif axis == 'X':
        obj.rotation_mode = 'QUATERNION'
        obj.rotation_quaternion = vector.to_track_quat('X', 'Y')
    elif axis == 'Y':
        obj.rotation_mode = 'QUATERNION'
        obj.rotation_quaternion = vector.to_track_quat('Y', 'Z')

def polar_to_cartesian(angle_deg, dr):
    """Convert polar coordinates to Cartesian (X, Y)."""
    angle_rad = math.radians(angle_deg)
    x = dr * math.cos(angle_rad)
    y = dr * math.sin(angle_rad)
    return (x, y)

def boolean_subtract(cutter_obj, target_obj):
    """Perform boolean subtract using modifier stack."""
    # Ensure target is active for modifier addition if needed, but we add to target
    bpy.context.view_layer.objects.active = target_obj
    
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply Modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove Cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(union_obj, target_obj):
    """Perform boolean union."""
    bpy.context.view_layer.objects.active = target_obj
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(union_obj, do_unlink=True)

def add_bevel(obj, width=0.01, segments=5):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for simplicity in this script context
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        return self.obj.location.copy()

    def face_center_world(self):
        # Assuming "face center" implies the center of the mesh geometry
        # For primitives, this is usually the object location unless scaled/rotated weirdly
        # We'll stick to object location for simplicity as it's the anchor point
        return self.obj.location.copy()

    def radial_point_world(self, angle_deg, radius):
        """Get a point on a circle around the object's location."""
        angle_rad = math.radians(angle_deg)
        x = self.obj.location.x + radius * math.cos(angle_rad)
        y = self.obj.location.y + radius * math.sin(angle_rad)
        z = self.obj.location.z
        return (x, y, z)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

# --- SECTION 3 – Create Mid Layer: Triangle Ring ---
for i in range(6):

    # Create primitive cylinder with Ø0.08 (radius 0.04) x 0.02 m

    # Rotate so +Z aligns +Z
    align_axis_to_vector(obj, 'Z', Vector((0, 0, 1)))
    # Anchor to world origin
    obj.location = (0, 0, 0)
    # Position at polar(0° + i·60°; dr=0.25)
    angle = 0 + i * 60
    x, y = polar_to_cartesian(angle, 0.25)
    obj.location = (x, y, 0)
    # Offset (0,0,0.02)
    obj.location = (x, y, 0.02)
    
    # Create cutter
    cutter_name = f"{name}_cut"
    # Plane cutter size 0.08x0.08 m. 
    # In Blender, a plane is 2D. To act as a cutter for a boolean, it needs thickness or we use a cube.
    # The prompt says "Create plane_cutter ... size 0.08x0.08". 
    # A boolean operation with a zero-thickness plane can be unstable. 
    # However, standard practice for "plane cutter" in these scripts often implies a thin box or a mesh plane.
    # Given "size 0.08x0.08", I will create a cube of that size and very small depth, or a plane.
    # Let's create a cube to ensure robust boolean subtraction, interpreting "plane_cutter" as a flat cutter.
    # Actually, let's stick to the prompt: "Create plane_cutter". 
    # If I create a plane, it has no volume. Boolean modifiers require volume.
    # I will create a cube with small depth (e.g. 0.001) to act as a plane cutter, or just a cube of that size.
    # Let's assume "size 0.08x0.08" refers to X and Y dimensions. I'll make it a cube 0.08x0.08x0.08 for safety, 
    # or better, a cube 0.08x0.08x0.001.
    # Let's use a cube for robustness.
    cutter = add_cube(cutter_name, (0.08, 0.08, 0.001), (0, 0, 0), (0, 0, 0))
    
    # Anchor cutter to triangle_ring_i (center-to-center)
    cutter.location = obj.location.copy()
    
    # Move by offset(0,0,0.01)
    cutter.location = (cutter.location.x, cutter.location.y, cutter.location.z + 0.01)
    
    # Rotate cutter so +Z aligns +Z (world)
    align_axis_to_vector(cutter, 'Z', Vector((0, 0, 1)))
    
    # Rotate cutter so +X aligns +X (world) - already aligned if Z is Z
    align_axis_to_vector(cutter, 'X', Vector((1, 0, 0)))
    
    # Rotate cutter by 45° about local Z axis
    cutter.rotation_euler.z += math.radians(45)
    
    # Boolean-subtract cutter from obj
    boolean_subtract(cutter, obj)

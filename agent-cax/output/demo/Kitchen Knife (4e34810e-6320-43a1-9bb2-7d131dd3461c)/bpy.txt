import bpy

def make_material(name, diffuse_color=(1.0, 1.0, 1.0, 1.0)):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        return bpy.data.materials[name]
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object's first material slot."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def clear_scene():
    """Delete all existing objects."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Clean up collections if necessary, though delete usually suffices for objects
    for collection in bpy.data.collections:
        collection.objects.clear()

def add_cylinder(radius, depth, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(size, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube primitive."""
    # size is expected as a tuple (x,y,z) or single value
    if isinstance(size, (int, float)):
        sx, sy, sz = size, size, size
    else:
        sx, sy, sz = size
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=location, rotation=rotation) # size=1, then scale
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (sx, sy, sz)
    return obj

def add_cone(radius_top, radius_bottom, depth, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cone primitive."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_top, radius2=radius_bottom, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(radius, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a UV Sphere."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width=0.01, segments=2):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(cutter_obj, target_obj):
    """Perform boolean subtract using BMesh."""
    # Ensure target is active for context if needed, but we'll use BMesh directly for robustness
    # Create a copy of target mesh to modify
    bm = bmesh.new()
    bm.from_mesh(target_obj.data)
    
    # Create a temporary mesh for the cutter to intersect
    cutter_bm = bmesh.new()
    cutter_bm.from_mesh(cutter_obj.data)
    
    # Transform cutter bm to world space
    cutter_bm.transform(cutter_obj.matrix_world)
    
    # Perform intersection
    # Note: bmesh.ops.boolean_union/intersect can be tricky. 
    # Standard approach: Use BMesh boolean operations if available or use bpy.ops.mesh.boolean_modifier
    # For script reliability, let's use the modifier approach on a copy or direct modifier application.
    
    # Revert to Modifier approach for simplicity and reliability in Blender 3.x
    # We need to apply the cutter to the target.
    
    # Add boolean modifier to target
    bool_mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    bool_mod.operation = 'DIFFERENCE'
    bool_mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier="BooleanSub")
    
    # Clean up
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    bm.free()
    
    # Update mesh
    target_obj.data.update()

def boolean_union(cutter_obj, target_obj):
    """Perform boolean union."""
    bool_mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    bool_mod.operation = 'UNION'
    bool_mod.object = cutter_obj
    
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier="BooleanUnion")
    
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    target_obj.data.update()

def align_face_to_face(obj_a, obj_b, face_a='right', face_b='left'):
    """
    Align obj_a's face to obj_b's face.
    Assumes standard primitives centered at origin.
    Right face is +X, Left is -X. Top is +Y, Bottom -Y. Front +Z, Back -Z.
    """
    # Calculate bounding boxes to find face centers
    def get_face_center(obj, face_type):
        # Transform vertices to world space
        verts = [obj.matrix_world @ v.co for v in obj.data.vertices]
        if face_type == 'right':
            # Max X
            center = Vector((max(v.x for v in verts), 0, 0))
            # Average Y and Z for that face
            y_avg = sum(v.y for v in verts if abs(v.x - max(v.x for v in verts)) < 0.0001) / sum(1 for v in verts if abs(v.x - max(v.x for v in verts)) < 0.0001)
            z_avg = sum(v.z for v in verts if abs(v.x - max(v.x for v in verts)) < 0.0001) / sum(1 for v in verts if abs(v.x - max(v.x for v in verts)) < 0.0001)
            return Vector((max(v.x for v in verts), y_avg, z_avg))
        elif face_type == 'left':
            # Min X
            min_x = min(v.x for v in verts)
            y_avg = sum(v.y for v in verts if abs(v.x - min_x) < 0.0001) / sum(1 for v in verts if abs(v.x - min_x) < 0.0001)
            z_avg = sum(v.z for v in verts if abs(v.x - min_x) < 0.0001) / sum(1 for v in verts if abs(v.x - min_x) < 0.0001)
            return Vector((min_x, y_avg, z_avg))
        elif face_type == 'top':
            max_y = max(v.y for v in verts)
            x_avg = sum(v.x for v in verts if abs(v.y - max_y) < 0.0001) / sum(1 for v in verts if abs(v.y - max_y) < 0.0001)
            z_avg = sum(v.z for v in verts if abs(v.y - max_y) < 0.0001) / sum(1 for v in verts if abs(v.y - max_y) < 0.0001)
            return Vector((x_avg, max_y, z_avg))
        elif face_type == 'bottom':
            min_y = min(v.y for v in verts)
            x_avg = sum(v.x for v in verts if abs(v.y - min_y) < 0.0001) / sum(1 for v in verts if abs(v.y - min_y) < 0.0001)
            z_avg = sum(v.z for v in verts if abs(v.y - min_y) < 0.0001) / sum(1 for v in verts if abs(v.y - min_y) < 0.0001)
            return Vector((x_avg, min_y, z_avg))
        elif face_type == 'front':
            max_z = max(v.z for v in verts)
            x_avg = sum(v.x for v in verts if abs(v.z - max_z) < 0.0001) / sum(1 for v in verts if abs(v.z - max_z) < 0.0001)
            y_avg = sum(v.y for v in verts if abs(v.z - max_z) < 0.0001) / sum(1 for v in verts if abs(v.z - max_z) < 0.0001)
            return Vector((x_avg, y_avg, max_z))
        elif face_type == 'back':
            min_z = min(v.z for v in verts)
            x_avg = sum(v.x for v in verts if abs(v.z - min_z) < 0.0001) / sum(1 for v in verts if abs(v.z - min_z) < 0.0001)
            y_avg = sum(v.y for v in verts if abs(v.z - min_z) < 0.0001) / sum(1 for v in verts if abs(v.z - min_z) < 0.0001)
            return Vector((x_avg, y_avg, min_z))
        return Vector((0,0,0))

    # Get centers in world space
    center_a = get_face_center(obj_a, face_a)
    center_b = get_face_center(obj_b, face_b)
    
    # Calculate offset needed to align them
    # We want obj_a's face center to match obj_b's face center
    # obj_a_local_center = obj_a.matrix_world.to_translation()
    # obj_a_face_local = obj_a.matrix_world.inverted() @ center_a
    
    # Simpler: Move obj_a so its face center equals obj_b's face center
    # delta = center_b - center_a
    # But center_a is in world space.
    
    # Let's calculate the local offset required
    # obj_a.world_to_local(center_b) gives the target location for obj_a's origin
    # if obj_a's face was at world center_b.
    # But obj_a's face is currently at center_a.
    
    # Target World Location for obj_a's origin:
    # We know obj_a's face is at center_a (world).
    # We want obj_a's face to be at center_b (world).
    # So we need to move obj_a by (center_b - center_a).
    
    delta = center_b - center_a
    obj_a.location += delta

def align_center_to_center(obj_a, obj_b):
    """Align center of obj_a to center of obj_b."""
    center_a = obj_a.matrix_world.to_translation()
    center_b = obj_b.matrix_world.to_translation()
    delta = center_b - center_a
    obj_a.location += delta

def orient_z_to_vector(obj, vector):
    """Rotate object so its local +Z aligns with world vector."""
    # Current Z
    current_z = obj.matrix_world.to_quaternion() @ Vector((0, 0, 1))
    # Target
    target = vector.normalized()
    
    # Calculate rotation
    rot_quat = current_z.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

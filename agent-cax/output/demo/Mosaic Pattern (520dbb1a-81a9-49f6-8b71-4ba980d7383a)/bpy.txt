def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def boolean_subtract(host_obj, cutter_obj):
    mod = host_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    bpy.context.view_layer.objects.active = host_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a world vector."""
    if axis == 'X':
        src = Vector((1, 0, 0))
    elif axis == 'Y':
        src = Vector((0, 1, 0))
    elif axis == 'Z':
        src = Vector((0, 0, 1))
    else:
        return

    target = vector.normalized()
    if src.dot(target) > 0.9999:
        quat = Quaternion()
    elif src.dot(target) < -0.9999:
        quat = Quaternion((0, 1, 0), pi) # 180 deg around Y
    else:
        quat = src.rotation_difference(target)
    
    obj.rotation_euler = quat.to_euler()

def average(points):
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        self.matrix_world = obj.matrix_world

    def center_world(self):
        # Average of all 8 corners
        pts = [self.matrix_world @ Vector(v) for v in self.bound_box]
        return average(pts)

    def face_center_world(self, face_name):
        """
        face_name: 'top', 'bottom', 'front', 'back', 'left', 'right'
        Returns the center of the specified face in world space.
        """
        # Define face corners based on bound box indices (standard cube)
        # 0: (-,-,-), 1: (+,-,-), 2: (+,-,+), 3: (-,-,+)
        # 4: (-,+,-), 5: (+,+,-), 6: (+,+,+), 7: (-,+,+)
        # Note: Bound box is in local space.
        
        # Map face names to corner indices
        faces = {
            'top': [4, 5, 6, 7],
            'bottom': [0, 1, 2, 3],
            'front': [1, 2, 6, 5], # +Y
            'back': [0, 3, 7, 4], # -Y
            'right': [1, 5, 6, 2], # +X
            'left': [0, 4, 7, 3]  # -X
        }
        
        if face_name not in faces:
            return self.center_world()
            
        indices = faces[face_name]
        pts = [self.matrix_world @ Vector(self.bound_box[i]) for i in indices]
        return average(pts)

    def radial_point_world(self, axis, direction):
        """
        Returns the world position of the corner furthest in the given direction along the axis.
        axis: 'X', 'Y', 'Z'
        direction: 1 or -1
        """
        # Find the corner with max/min coordinate in that axis
        # Local coordinates of bound box
        local_pts = self.bound_box
        
        if axis == 'X':
            idx = 1 if direction == 1 else 0
            # Find max/min X
            max_x = max(p[0] for p in local_pts)
            min_x = min(p[0] for p in local_pts)
            target_val = max_x if direction == 1 else min_x
            for p in local_pts:
                if abs(p[0] - target_val) < 1e-5:
                    return self.matrix_world @ Vector(p)
        elif axis == 'Y':
            max_y = max(p[1] for p in local_pts)
            min_y = min(p[1] for p in local_pts)
            target_val = max_y if direction == 1 else min_y
            for p in local_pts:
                if abs(p[1] - target_val) < 1e-5:
                    return self.matrix_world @ Vector(p)
        elif axis == 'Z':
            max_z = max(p[2] for p in local_pts)
            min_z = min(p[2] for p in local_pts)
            target_val = max_z if direction == 1 else min_z
            for p in local_pts:
                if abs(p[2] - target_val) < 1e-5:
                    return self.matrix_world @ Vector(p)
        return self.center_world()

def create_triangle_prism(name, size, location, rotation_euler, material=None):
    """
    Creates a triangular prism.
    size: (width, depth, height) -> X, Y, Z dimensions.
    The triangle is in the XY plane, extruded along Z.
    Triangle vertices: (0,0), (size[0], 0), (0, size[1]) for a right triangle?
    Or centered? Let's assume a right triangle aligned to corner for simplicity in alignment logic,
    or a generic triangle. The prompt implies "triangle_prism" with specific alignments like "bottom_right".
    Let's create a right-angled triangle prism aligned such that:
    - Bottom-Right is at (size[0]/2, -size[1]/2, -size[2]/2) in local space?
    - Let's define the triangle in XY plane:
      Vertex 1: (0, 0)
      Vertex 2: (size[0], 0)
      Vertex 3: (0, size[1])
      This forms a right triangle.
      Center of mass is roughly (size[0]/3, size[1]/3).
      To make alignment easier, let's shift the geometry so the "bottom_right" corner is at (size[0]/2, -size[1]/2) relative to center?
      
      Let's stick to a simpler definition:
      A right triangle with legs along +X and +Y from the origin (0,0).
      Vertices: (0,0), (w,0), (0,h).
      Extrude along Z: -h/2 to h/2.
      
      Local Corners:
      Bottom-Right (max X, min Y): (w, -h/2) ? No, Y is 0 at base.
      Let's define the triangle explicitly in BMesh.
    """
    w, d, h = size
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (w, d, h)
    obj.rotation_euler = rotation_euler
    
    # Convert to BMesh to shape into triangle
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    
    # Transform vertices to a right triangle in XY plane, centered at origin for rotation,
    # but we need to know where "bottom_right" is.
    # Let's assume the triangle occupies the first quadrant relative to its local origin (0,0) in XY.
    # Vertices: (0,0), (w,0), (0,d).
    # But the cube is centered. We need to move vertices.
    
    # Clear existing geometry
    bm.clear()
    
    # Define vertices for a right triangle prism
    # Bottom face (Z = -h/2)
    v1 = bm.verts.new((0, 0, -h/2))
    v2 = bm.verts.new((w, 0, -h/2))
    v3 = bm.verts.new((0, d, -h/2))
    
    # Top face (Z = h/2)
    v4 = bm.verts.new((0, 0, h/2))
    v5 = bm.verts.new((w, 0, h/2))
    v6 = bm.verts.new((0, d, h/2))
    
    bm.verts.ensure_lookup_table()
    
    # Faces
    bm.faces.new([v1, v2, v3]) # Bottom
    bm.faces.new([v4, v6, v5]) # Top
    bm.faces.new([v1, v4, v5, v2]) # Front (XZ plane, y=0)
    bm.faces.new([v1, v3, v6, v4]) # Left (YZ plane, x=0)
    bm.faces.new([v2, v5, v6, v3]) # Hypotenuse face
    
    bm.to_mesh(obj.data)
    bm.free()
    
    # Update matrix
    obj.data.update()
    
    if material:
        obj.data.materials.append(material)
        
    return obj

def get_corner(obj, corner_name):
    """
    Returns the world position of a specific corner of the object.
    corner_name: 'bottom_right', 'bottom_left', 'top_right', 'top_left'
    Assumes the object is a triangle prism created by create_triangle_prism
    with the right angle at (0,0) in local XY, extending to +X and +Y.
    
    Local Coordinates (after scale):
    Bottom-Right: (w, 0, -h/2) -> Wait, 'bottom' usually means min Z. 'right' means max X.
    In our triangle:
    X ranges 0 to w.
    Y ranges 0 to d.
    Z ranges -h/2 to h/2.
    
    'bottom' = min Z = -h/2
    'top' = max Z = h/2
    'right' = max X = w
    'left' = min X = 0
    'front' (or bottom in Y?) = min Y = 0
    'back' (or top in Y?) = max Y = d
    
    The prompt uses 'bottom_right', 'top_left' etc. on the square base.
    Square base is a cube.
    Cube corners:
    bottom_right: max X, min Y, min Z
    bottom_left: min X, min Y, min Z
    top_right: max X, max Y, min Z
    top_left: min X, max Y, min Z
    
    For the triangle prism, we need to map these concepts.
    Let's assume the triangle is oriented such that:
    - 'bottom' refers to the face at min Z.
    - 'top' refers to the face at max Z.
    - 'right' refers to max X.
    - 'left' refers to min X.
    - 'front' (or 'bottom' in 2D projection) refers to min Y.
    - 'back' (or 'top' in 2D projection) refers to max Y.
    
    So:
    bottom_right: max X, min Y, min Z -> (w, 0, -h/2)
    bottom_left: min X, min Y, min Z -> (0, 0, -h/2)
    top_right: max X, max Y, min Z -> (w, d, -h/2)
    top_left: min X, max Y, min Z -> (0, d, -h/2)
    
    Wait, the triangle only has vertices at (0,0), (w,0), (0,d).
    It does NOT have a vertex at (w,d).
    So 'top_right' corner of the bounding box is (w, d, -h/2), but there is no mesh there.
    The prompt says "Align triangle_A.bottom_right to square_base.top_right".
    This implies we align the *bounding box corner* or the *mesh corner*.
    Given the geometry, aligning the bounding box corner is safer for "snap" logic.
    """
    mw = obj.matrix_world
    bb = obj.bound_box
    
    # Determine bounds
    min_x = min(v[0] for v in bb)
    max_x = max(v[0] for v in bb)
    min_y = min(v[1] for v in bb)
    max_y = max(v[1] for v in bb)
    min_z = min(v[2] for v in bb)
    max_z = max(v[2] for v in bb)
    
    if corner_name == 'bottom_right':
        local_pt = Vector((max_x, min_y, min_z))
    elif corner_name == 'bottom_left':
        local_pt = Vector((min_x, min_y, min_z))
    elif corner_name == 'top_right':
        local_pt = Vector((max_x, max_y, min_z))
    elif corner_name == 'top_left':
        local_pt = Vector((min_x, max_y, min_z))
    else:
        local_pt = Vector((0,0,0))
        
    return mw @ local_pt

def align_objects(obj_target, target_corner, obj_source, source_corner):
    """
    Aligns obj_source so that obj_source[source_corner] is at obj_target[target_corner].
    """
    target_world_pos = get_corner(obj_target, target_corner)
    source_world_pos = get_corner(obj_source, source_corner)
    
    # Calculate offset
    offset = target_world_pos - source_world_pos
    obj_source.location += offset

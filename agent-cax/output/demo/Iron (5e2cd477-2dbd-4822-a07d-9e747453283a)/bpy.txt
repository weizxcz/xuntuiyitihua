import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Alpha'].default_value = diffuse_color[3]
    return mat

def add_primitive(name, shape, size, location, rotation_euler):
    """Add a primitive mesh object and apply transforms."""
    if shape == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=size[0], location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'cylinder':
        # size: (diameter, height)
        bpy.ops.mesh.primitive_cylinder_add(radius=size[0]/2, depth=size[1], location=location)
        obj = bpy.context.active_object
    elif shape == 'cone':
        # size: (bottom_radius, top_radius, height)
        bpy.ops.mesh.primitive_cone_add(radius1=size[0], radius2=size[1], depth=size[2], location=location)
        obj = bpy.context.active_object
    elif shape == 'uvsphere':
        # size: (diameter)
        bpy.ops.mesh.primitive_uv_sphere_add(radius=size[0]/2, location=location)
        obj = bpy.context.active_object
    
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align an object's local axis to a world vector."""
    # Determine current axis vector
    if axis == 'X':
        current = Vector((1, 0, 0))
    elif axis == 'Y':
        current = Vector((0, 1, 0))
    elif axis == 'Z':
        current = Vector((0, 0, 1))
    
    # Calculate rotation matrix
    rot_mat = current.rotation_difference(vector).to_matrix().to_4x4()
    
    # Apply rotation relative to current orientation
    obj.rotation_euler = rot_mat.to_euler()

def get_face_center(obj, face_axis):
    """Get the world location of a face center based on axis."""
    # Ensure object is at origin for calculation, then transform
    # We need the normal of the face in world space
    # Simplified: The face center is at local offset + object world location
    # But we need to account for rotation.
    
    # Get the local normal of the face
    if face_axis == '+X': local_normal = Vector((1, 0, 0))
    elif face_axis == '-X': local_normal = Vector((-1, 0, 0))
    elif face_axis == '+Y': local_normal = Vector((0, 1, 0))
    elif face_axis == '-Y': local_normal = Vector((0, -1, 0))
    elif face_axis == '+Z': local_normal = Vector((0, 0, 1))
    elif face_axis == '-Z': local_normal = Vector((0, 0, -1))
    
    # Transform normal to world space
    world_normal = obj.matrix_world.to_3x3() @ local_normal
    
    # Get the half-dimensions to find the face center
    half_size = obj.dimensions / 2
    
    # The face center is the object center + (half_size * world_normal)
    # Note: obj.dimensions is in world space if object is scaled, but here we assume uniform scaling or no scale for alignment logic simplicity.
    # However, for precise alignment, we should use the local extent transformed.
    
    # Let's use a more robust method: Project the object's bounding box corners
    # Or simpler: The face center is at obj.location + (obj.matrix_world.to_3x3() @ (half_size * local_normal))
    
    # Wait, obj.dimensions returns world-space dimensions if scale != 1.
    # If scale is 1, obj.dimensions == local dimensions.
    # Let's assume scale is 1 for these primitives.
    
    offset = half_size * local_normal
    world_offset = obj.matrix_world.to_3x3() @ offset
    return obj.location + world_offset

def align_faces(obj_source, source_face, obj_target, target_face):
    """Align source face to target face (coplanar and same normal)."""
    src_loc = get_face_center(obj_source, source_face)
    tgt_loc = get_face_center(obj_target, target_face)
    
    # Calculate translation vector
    diff = tgt_loc - src_loc
    
    # Apply translation
    obj_source.location += diff

def boolean_subtract(cutter, target):
    """Perform boolean subtract using BMesh."""
    # Create a copy of the target for boolean operation to avoid modifying original data directly if needed,
    # but here we modify the target mesh directly.
    
    # Convert target to BMesh
    bm = bmesh.new()
    bm.from_mesh(target.data)
    
    # Convert cutter to BMesh
    bm_cutter = bmesh.new()
    bm_cutter.from_mesh(cutter.data)
    
    # Transform cutter bm to world space
    bm_cutter.transform(cutter.matrix_world)
    
    # Perform boolean difference
    bmesh.ops.boolean_cutter(
        bm,
        clipper_empty=False,
        clipper_cut=True,
        use_clipping=True,
        use_boolean_result=True,
        objects=[cutter], # This doesn't work directly with bm_cutter
        # Actually, bmesh.ops.boolean_cutter takes a list of objects or a single mesh.
        # Easier approach: Use bpy.ops.mesh.boolean_union/subtract on duplicates
        pass
    )
    
    # Fallback: Use bpy.ops for reliability in script context
    # Select both
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    cutter.select_set(True)
    bpy.context.view_layer.objects.active = target
    
    # We need to apply the cutter's transform to the target's context or use modifiers.
    # Let's use modifiers for non-destructive, then apply.
    
    # Remove existing boolean modifier on target
    for mod in target.modifiers:
        if mod.type == 'BOOLEAN':
            target.modifiers.remove(mod)
            
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Clean up cutter
    bpy.data.objects.remove(cutter, do_unlink=True)
    
    # Update mesh
    target.data.update()

def boolean_union(obj1, obj2):
    """Perform boolean union."""
    bpy.ops.object.select_all(action='DESELECT')
    obj1.select_set(True)
    obj2.select_set(True)
    bpy.context.view_layer.objects.active = obj1
    
    for mod in obj1.modifiers:
        if mod.type == 'BOOLEAN':
            obj1.modifiers.remove(mod)
            
    mod = obj1.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj2
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(obj2, do_unlink=True)
    obj1.data.update()

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive_cube(name, size, location, rotation_euler):
    """Add a cube and apply size, location, and rotation."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def align_face_to_face(obj_source, obj_target, axis='Y', offset=0):
    """
    Aligns a specific face of obj_source to the corresponding face of obj_target.
    axis: 'X', 'Y', or 'Z' indicating the local axis of the face normal.
    offset: additional distance to move along the normal after alignment.
    """
    # Get world matrices
    mat_src = obj_source.matrix_world
    mat_tgt = obj_target.matrix_world
    
    # Determine face normal direction in world space
    # We assume the object is axis-aligned or we align based on the object's local axis
    # For simplicity in this script, we assume standard orientation where local axis aligns with world axis
    # or we calculate the face center and normal.
    
    # Get face normal in world space
    # Since we are dealing with simple boxes aligned to axes, we can infer the normal direction
    # based on the axis parameter.
    # +axis face normal is +1 in that world axis direction if object is not rotated relative to world.
    # However, to be robust, let's find the face center and normal.
    
    # Helper to get face info
    def get_face_normal_and_center(obj, axis_char):
        # This is a simplified approach assuming the object's local axes align with world axes
        # or that we are aligning based on the object's local orientation.
        # If the object is rotated, we need to transform the local normal.
        
        local_normal = Vector((0, 0, 0))
        if axis_char == 'X': local_normal = Vector((1, 0, 0))
        elif axis_char == 'Y': local_normal = Vector((0, 1, 0))
        elif axis_char == 'Z': local_normal = Vector((0, 0, 1))
        
        # Transform normal to world space
        world_normal = local_normal @ obj.matrix_world.to_3x3()
        world_normal.normalize()
        
        # Get face center (approximate by adding half size in that direction)
        half_size = obj.dimensions / 2
        offset_vec = half_size * local_normal
        face_center_local = offset_vec
        face_center_world = face_center_local @ obj.matrix_world
        
        return face_center_world, world_normal

    src_center, src_normal = get_face_normal_and_center(obj_source, axis)
    tgt_center, tgt_normal = get_face_normal_and_center(obj_target, axis)
    
    # We want src_face to touch tgt_face.
    # The tgt_face normal points OUT of the target.
    # The src_face normal points OUT of the source.
    # To align them (touching), the src_face should be at tgt_center - tgt_normal * (distance).
    # Actually, usually we want the src face to be coplanar with the tgt face.
    # If we align +Y face of A to +Y face of B, they are parallel.
    # If we want them to touch, we need to move A so its face is at B's face position.
    
    # Vector from src_face to tgtFace
    # Target position for src_face center = tgt_center
    # But we want the surfaces to touch.
    # If normals are same direction, they are parallel.
    # Distance between planes = (tgt_center - src_center) . tgt_normal
    
    # Let's move obj_source so its face center coincides with tgt_face center?
    # No, usually "align face to face" implies coplanarity or touching.
    # Given the context of assembly, it likely means touching.
    # If we align +Y face of Source to +Y face of Target, and they are on the same side,
    # we might want to push Source away or pull it in.
    # Let's assume we want the Source face to be AT the Target face location.
    
    # Calculate translation vector
    # We want src_face_world to move to tgt_center
    # But we also need to account for the fact that they might be on opposite sides.
    # Let's just move the source object so its face center is at the target face center.
    # Then apply offset.
    
    delta = tgt_center - src_center
    obj_source.location += delta
    
    # Apply offset along the normal
    # If offset > 0, move further out along the normal
    obj_source.location += src_normal * offset

def align_axis_to_vector(obj, axis_char, vector):
    """Rotate object so its local axis aligns with a world vector."""
    if axis_char == 'X': local_axis = Vector((1, 0, 0))
    elif axis_char == 'Y': local_axis = Vector((0, 1, 0))
    elif axis_char == 'Z': local_axis = Vector((0, 0, 1))
    
    # Normalize input vector
    vector = vector.normalized()
    local_axis = local_axis.normalized()
    
    # Calculate rotation
    rot_quat = local_axis.rotation_difference(vector)
    obj.rotation_euler = rot_quat.to_euler()

def create_panel(name, size, align_axis, anchor_pos, offset, align_to_obj=None, align_axis_face=None):
    """
    Generic panel creation helper.
    align_axis: 'X', 'Y', 'Z' - which local axis aligns with world axis.
    """
    # 1. Create primitive
    # Default cube is centered at 0,0,0 with size 1.
    # We want size X, Y, Z.
    # We need to orient it first.
    
    # Orientation:
    # If align_axis is 'Z', we want local Z to align with world Z.
    # Cube default is Z up. So no rotation needed for Z.
    # If align_axis is 'Y', we want local Y to align with world Y.
    # Cube default Y is forward. We need to rotate -90 deg around X to make Y up?
    # Or just set rotation_euler directly.
    
    rot = (0, 0, 0)
    if align_axis == 'Y':
        rot = (-pi/2, 0, 0) # Rotate X by -90 deg so Y becomes Z? No.
        # Let's check:
        # Default: X right, Y forward, Z up.
        # We want Local Y to align with World Y.
        # If we don't rotate, Local Y IS World Y.
        # So if align_axis == 'Y', rot = (0,0,0).
        pass
    elif align_axis == 'Z':
        # Local Z aligns with World Z.
        # Default Local Z IS World Z.
        # So if align_axis == 'Z', rot = (0,0,0).
        pass
    elif align_axis == 'X':
        # Local X aligns with World X.
        # Default Local X IS World X.
        # So if align_axis == 'X', rot = (0,0,0).
        pass
        
    # Wait, the prompt says:
    # "Rotate front_panel so its +Z axis aligns +Z (world)." -> Default is already aligned.
    # "Rotate bottom_panel so its +Y axis aligns +Y (world)." -> Default is already aligned.
    # "Rotate side_gusset so its +X axis aligns +X (world)." -> Default is already aligned.
    
    # It seems the primitives are created with default orientation which matches the world axes for X, Y, Z.
    # So no rotation is needed if we just create a cube and scale it.
    
    obj = add_primitive_cube(name, size, (0, 0, 0), rot)
    
    # 2. Anchor to world origin (already at 0,0,0)
    
    # 3. Offset
    obj.location = Vector(offset)
    
    # 4. Align faces if specified
    if align_to_obj and align_axis_face:
        align_face_to_face(obj, align_to_obj, axis=align_axis_face)
        
    return obj

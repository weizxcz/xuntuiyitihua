def make_material(name, diffuse_color, roughness=0.5, metallic=0.0):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = roughness
    bsdf.inputs['Metallic'].default_value = metallic
    return mat

def add_cube(name, size=1.0, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    return obj

def add_cylinder(name, radius=0.5, depth=1.0, location=(0,0,0), rotation=(0,0,0), cap_ends=True):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation, end_fill_type='TRIFAN' if cap_ends else 'NGON')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_top=0.0, radius_bottom=1.0, depth=1.0, location=(0,0,0), rotation=(0,0,0)):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius=1.0, location=(0,0,0), rotation=(0,0,0)):
    """Add a UV sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_text_object(name, text="Text", size=1.0, location=(0,0,0), rotation=(0,0,0)):
    """Add a text object."""
    bpy.ops.object.text_add(location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.body = text
    obj.data.size = size
    return obj

def add_bevel(obj, width=0.01, segments=5, limit_method='ANGLE'):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = limit_method
    return mod

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with the given world vector."""
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else: # Z
        local_axis = Vector((0, 0, 1))
    
    target = vector.normalized()
    if local_axis.normalized().dot(target) > 0.9999:
        return # Already aligned
    
    rot_quat = local_axis.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average position of a list of Vector objects."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to create invisible locator objects for alignment."""
    def __init__(self, name, location=(0,0,0)):
        self.name = name
        # Create a small cube as a locator
        bpy.ops.mesh.primitive_cube_add(size=0.001, location=location)
        self.obj = bpy.context.active_object
        self.obj.name = name
        self.obj.hide_viewport = True
        self.obj.hide_render = True
        self.obj.empty_display_size = 0.01
        self.obj.empty_display_type = 'PLAIN_AXES'
        # Actually, let's use a mesh cube for precise face center calculation if needed, 
        # but for simple center alignment, an Empty is better. 
        # However, the prompt asks for "Empty helper (cube display)". 
        # Let's stick to a small cube mesh for "cube display" locator if we need face centers, 
        # or just an Empty. The prompt says "Create Empty helper (cube display)". 
        # This likely means an Empty object that looks like a cube or is used as a cube anchor.
        # Let's revert to a small cube mesh for precise face/center logic if needed, 
        # but standard practice is Empty. Let's use a small cube mesh named as locator.
        bpy.ops.object.select_all(action='DESELECT')
        self.obj.select_set(True)
        bpy.context.view_layer.objects.active = self.obj
        # Convert to empty? No, keep as mesh for face center logic if "cube display" implies mesh.
        # But "Empty helper" usually means Empty. Let's create an Empty.
        bpy.ops.object.empty_add(type='PLAIN_AXES', location=location)
        new_empty = bpy.context.active_object
        new_empty.name = name
        new_empty.empty_display_size = 0.01
        # Delete the cube we just made
        bpy.ops.object.delete()
        self.obj = new_empty

    def center_world(self):
        return self.obj.location

    def face_center_world(self, face_index):
        # For an Empty, this is just the center. 
        # If it were a mesh, we'd calculate it.
        return self.obj.location

    def radial_point_world(self, angle, radius):
        return self.obj.location + Vector((math.cos(angle)*radius, math.sin(angle)*radius, 0))

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target is active
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(objects_to_union, result_name):
    """Perform boolean union on a list of objects."""
    if not objects_to_union:
        return
    
    # Select all
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects_to_union:
        obj.select_set(True)
    
    bpy.context.view_layer.objects.active = objects_to_union[0]
    
    # Join them first
    bpy.ops.object.join()
    joined_obj = bpy.context.active_object
    joined_obj.name = result_name
    
    return joined_obj

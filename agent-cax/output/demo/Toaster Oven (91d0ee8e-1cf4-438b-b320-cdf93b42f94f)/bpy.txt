def clear_scene():
    """Delete all existing objects and collections except the world."""
    bpy.ops.object.select_all(action='DESELECT')
    for obj in bpy.data.objects:
        obj.select_set(True)
    bpy.ops.object.delete(use_global=False)
    # Clear remaining data blocks if necessary, but usually just objects is enough for reset
    for collection in bpy.data.collections:
        if collection.name != "Collection":
            bpy.data.collections.remove(collection)

def set_units_metres():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    bsdf.inputs['Metallic'].default_value = 0.0 if 'plastic' in name else 0.9
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive(name, shape, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive mesh object."""
    if shape == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'cylinder':
        # size = (diameter, height)
        bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=size[0]/2, depth=size[1], location=location)
        obj = bpy.context.active_object
        # Rotate to align with Z if needed, default cylinder is Z-up
    elif shape == 'disc':
        bpy.ops.mesh.primitive_circle_add(vertices=32, radius=size[0]/2, fill_type='NGON', location=location)
        obj = bpy.context.active_object
        obj.data.extrude = size[1]
        bpy.ops.mesh.extrude_faces_move()
    elif shape == 'cone':
        bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=size[0]/2, radius2=0, depth=size[1], location=location)
        obj = bpy.context.active_object
    elif shape == 'sphere':
        bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=size[0]/2, location=location)
        obj = bpy.context.active_object
    else:
        return None
    
    obj.name = name
    obj.rotation_euler = rotation
    return obj

def align_axis_to_vector(obj, axis='Z', target_vector=(0,0,1)):
    """Align object's local axis to a target world vector."""
    if axis == 'Z':
        current = obj.matrix_world.to_3x3().col[2]
    elif axis == 'Y':
        current = obj.matrix_world.to_3x3().col[1]
    elif axis == 'X':
        current = obj.matrix_world.to_3x3().col[0]
    else:
        return

    target = target_vector.normalized()
    # Calculate rotation matrix to align current to target
    rot_matrix = current.rotation_difference(target).to_matrix().to_4x4()
    obj.matrix_world = rot_matrix @ obj.matrix_world

def get_face_center(obj, face_type='center'):
    """
    Get a reference point for anchoring.
    face_type: 'center', 'front_face' (max Z), 'back_face' (min Z), 
               'top_face' (max Y), 'bottom_face' (min Y), 
               'right_face' (max X), 'left_face' (min X),
               'interior_bottom' (approximate center of bottom inner face)
    """
    # Ensure object is updated
    obj.update_from_editmode()
    
    # Simple bounding box approach for robustness across primitives
    min_b = obj.matrix_world @ Vector(obj.bound_box[0])
    max_b = obj.matrix_world @ Vector(obj.bound_box[6])
    
    center = (min_b + max_b) / 2
    
    if face_type == 'center':
        return center
    elif face_type == 'front_face':

        # Prompt says "Rotate ... so its +Z axis aligns +Z (world)".

        # If Z is up, Front might be +Y. However, prompt offsets suggest Z is depth or height.
        # Let's look at offsets: offset(0, 0.125, 0.175).
        # If main body is 0.35 high (Z), 0.30 deep (Y), 0.45 wide (X).
        # Center is (0,0,0).
        # Front face usually implies the face facing the user.

        # "Front" often means +Y or -Y depending on convention.
        # Given offset(0, 0.125, 0.175) for door frame on main_body_shell.front_face:
        # Main body Z range: -0.175 to 0.175.
        # Door frame Z offset 0.175 suggests it's placed at the top or front?
        # If Z is Up, 0.175 is the top face.
        # If Z is Depth (Front), 0.175 is the front face.
        # Prompt: "Rotate ... so its +Z axis aligns +Z (world)".
        # This implies Z is the primary vertical or depth axis.
        # Let's assume Z is Vertical (Up) for "Height" calculations, but "Front Face" usually refers to Y.
        # HOWEVER, looking at Section 3: Anchor to main_body_shell.front_face, offset Z=0.175.
        # Main body height is 0.35. Half height is 0.175.
        # This strongly suggests "Front Face" in this context might actually mean "Top Face" (Max Z) 
        # OR the object is rotated such that Z is the depth axis.
        # Let's re-read: "Rotate ... so its +Z axis aligns +Z (world)".
        # If Z is World Up, then Max Z is Top.
        # Why would a door frame be anchored to the Top?
        # Maybe the "Front Face" is defined by the user's perspective, which might be Y.
        # But the offset Z=0.175 matches the half-height.
        # Hypothesis: The prompt uses "Front Face" to mean the face at Max Z (Top) OR the object is oriented differently.
        # Let's look at Section 4: Door glass anchored to door_frame.front_face, offset Z=0.175.
        # Door frame thickness is 0.02.
        # If door frame is at Z=0.175 (Top of main body), and glass is offset Z=0.175... that would be far away.
        # Let's reconsider the orientation.
        # If Z is Depth (Front/Back):
        # Main body Z range: -0.175 to 0.175.
        # Front face is Z=0.175.
        # Door frame anchored to Z=0.175. Offset Z=0.175 -> New Z = 0.35? That seems outside.
        # Let's look at the offset values again.
        # Section 3: Anchor main_body_shell.front_face. Offset (0, 0.125, 0.175).
        # If Front Face is Max Z (0.175), and we add 0.175, we get 0.35.
        # If Front Face is Max Y (0.15), and we add 0.125, we get 0.275.
        
        # Alternative Interpretation:
        # The "Front Face" is the face with the highest Z coordinate.
        # The offset is applied in LOCAL space of the anchor? No, "offset is in reference local axes".
        # Usually, offset is relative to the anchor point.
        
        # Let's try a different standard: Z is Up.
        # Main Body: X[-0.225, 0.225], Y[-0.15, 0.15], Z[-0.175, 0.175].
        # "Front Face" usually means the face facing the camera, often -Y or +Y.
        # But if the prompt says "Offset Z=0.175", and the max Z is 0.175, it's highly likely 
        # that "Front Face" refers to the Top Face (Max Z) in this specific prompt's terminology, 
        # OR the object is rotated 90 degrees.
        # BUT, the prompt explicitly says: "Rotate ... so its +Z axis aligns +Z (world)".
        # This fixes Z as Up.
        # So "Front Face" must be a Y-face.
        # If Anchor is Y=0.15 (Front), and Offset is (0, 0.125, 0.175).
        # New Y = 0.15 + 0.125 = 0.275.
        # New Z = 0 + 0.175 = 0.175.
        # This places the door frame at the top-front corner?
        
        # Let's look at Section 7: Dial anchored to main_body_shell.front_face. Offset (-0.15, 0.15, 0.185).
        # If Front Face is Y=0.15.
        # Dial Y = 0.15 + 0.15 = 0.30.
        # Dial Z = 0 + 0.185 = 0.185.

    return center

def boolean_subtract(cutter_obj, target_obj, name="bool_result"):
    """Perform Boolean Subtract using the modifier stack."""
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove Cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def add_bevel(obj, width=0.01, segments=5):
    """Add a bevel modifier and apply."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

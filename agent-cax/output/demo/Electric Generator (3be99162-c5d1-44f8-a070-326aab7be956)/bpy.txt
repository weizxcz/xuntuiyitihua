import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_torus(name, major_radius, minor_radius, location=(0,0,0), rotation=(0,0,0)):
    bpy.ops.mesh.primitive_torus_add(major_radius=major_radius, minor_radius=minor_radius, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def boolean_union(target, cutter):
    """Performs boolean union of cutter into target and removes cutter."""
    target.select_set(True)
    cutter.select_set(True)
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.boolean_modifier(operator='UNION', object=cutter)
    bpy.data.objects.remove(cutter, do_unlink=True)
    target.select_set(False)

def align_center_to_center(obj_src, obj_tgt):
    """Aligns source object center to target object center."""
    obj_src.location = obj_tgt.location

def align_bottom_face_to_bottom_face(obj_src, obj_tgt):
    """Aligns source object bottom face to target object bottom face."""
    # Calculate height of target
    # Note: For rotated objects, we need to be careful. Assuming Z-up cylinders/cubes.
    # We align the Z coordinate of the bottom of src to the Z coordinate of the bottom of tgt.
    
    # Get world bounds
    def get_world_bounds(obj):
        # Transform vertices to world space
        verts_world = [obj.matrix_world @ v.co for v in obj.data.vertices]
        zs = [v[2] for v in verts_world]
        return min(zs), max(zs)
    
    tgt_min_z, tgt_max_z = get_world_bounds(obj_tgt)
    src_min_z, src_max_z = get_world_bounds(obj_src)
    
    # We want src_min_z to equal tgt_min_z
    current_src_min_z = obj_src.location[2] + (src_min_z - obj_src.location[2]) # This is just obj_src.location[2] if origin is center
    # Actually, simpler:
    # Current bottom of src is obj_src.location[2] - (height/2)
    # Target bottom is tgt_min_z
    
    # Let's just move the object so its local bottom matches target's world bottom
    # Height of src
    src_height = src_max_z - src_min_z
    # Current bottom Z
    current_bottom_z = obj_src.location[2] - (src_height / 2)
    
    delta_z = tgt_min_z - current_bottom_z
    obj_src.location[2] += delta_z

def align_bottom_face_to_top_face(obj_src, obj_tgt):
    """Aligns source object bottom face to target object top face."""
    def get_world_bounds(obj):
        verts_world = [obj.matrix_world @ v.co for v in obj.data.vertices]
        zs = [v[2] for v in verts_world]
        return min(zs), max(zs)
    
    tgt_min_z, tgt_max_z = get_world_bounds(obj_tgt)
    src_min_z, src_max_z = get_world_bounds(obj_src)
    
    src_height = src_max_z - src_min_z
    current_bottom_z = obj_src.location[2] - (src_height / 2)
    
    # We want src_bottom to equal tgt_top
    delta_z = tgt_max_z - current_bottom_z
    obj_src.location[2] += delta_z

def offset_object(obj, offset_vec):
    """Applies offset vector to object location."""
    obj.location += offset_vec

def polar_position(obj, angle_deg, dr, anchor_obj):
    """Positions object at polar coordinates relative to anchor."""
    angle_rad = math.radians(angle_deg)
    x = anchor_obj.location[0] + dr * math.cos(angle_rad)
    y = anchor_obj.location[1] + dr * math.sin(angle_rad)
    obj.location[0] = x
    obj.location[1] = y

def reset_scene():
    bpy.ops.object.select_all(action='SELECT')

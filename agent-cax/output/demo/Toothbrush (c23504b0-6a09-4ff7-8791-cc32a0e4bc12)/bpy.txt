import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder oriented along Z."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0) # Ensure Z-up
    return obj

def add_cube(name, size):
    """Add a cube centered at origin."""
    # size is (x, y, z)
    x, y, z = size
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (x/2, y/2, z/2)
    obj.rotation_euler = (0, 0, 0)
    return obj

def get_world_center(obj):
    """Get the world location of an object's center."""
    return obj.matrix_world.to_translation()

def get_world_bottom_face_center(obj):
    """Get the world location of the bottom face center of an object."""
    # Assuming object is axis-aligned or we use bounding box
    # For a cube/cylinder oriented along Z, bottom is min Z
    bbox = obj.bound_box
    # Transform bbox corners to world
    corners = [Vector((v[0], v[1], v[2])) for v in bbox]
    world_corners = [obj.matrix_world @ c for c in corners]
    min_z = min(c[2] for c in world_corners)
    # Find the corner with min_z and average X,Y for center of face
    relevant_corners = [c for c in world_corners if abs(c[2] - min_z) < 1e-5]
    avg_x = sum(c[0] for c in relevant_corners) / len(relevant_corners)
    avg_y = sum(c[1] for c in relevant_corners) / len(relevant_corners)
    return Vector((avg_x, avg_y, min_z))

def get_world_top_face_center(obj):
    """Get the world location of the top face center of an object."""
    bbox = obj.bound_box
    corners = [Vector((v[0], v[1], v[2])) for v in bbox]
    world_corners = [obj.matrix_world @ c for c in corners]
    max_z = max(c[2] for c in world_corners)
    relevant_corners = [c for c in world_corners if abs(c[2] - max_z) < 1e-5]
    avg_x = sum(c[0] for c in relevant_corners) / len(relevant_corners)
    avg_y = sum(c[1] for c in relevant_corners) / len(relevant_corners)
    return Vector((avg_x, avg_y, max_z))

def align_to_face(obj, target_obj, face_type='top'):
    """Align obj's bottom face to target_obj's specified face."""
    if face_type == 'top':
        target_loc = get_world_top_face_center(target_obj)
    elif face_type == 'bottom':
        target_loc = get_world_bottom_face_center(target_obj)
    else:
        target_loc = get_world_center(target_obj)
    
    # Calculate offset needed to move obj's bottom face to target_loc
    obj_bottom = get_world_bottom_face_center(obj)
    offset = target_loc - obj_bottom
    obj.location += offset

def align_center_to_face(obj, target_obj, face_type='top'):
    """Align obj's center to target_obj's specified face."""
    if face_type == 'top':
        target_loc = get_world_top_face_center(target_obj)
    elif face_type == 'bottom':
        target_loc = get_world_bottom_face_center(target_obj)
    else:
        target_loc = get_world_center(target_obj)
        
    obj_center = get_world_center(obj)
    offset = target_loc - obj_center
    obj.location += offset

def align_centers(obj1, obj2):
    """Align obj1 center to obj2 center."""
    c2 = get_world_center(obj2)
    c1 = get_world_center(obj1)
    obj1.location += (c2 - c1)

def apply_offset(obj, offset_vec):
    """Apply an offset vector to the object's location."""
    obj.location += offset_vec

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using cutter on target."""
    cutter = bpy.data.objects.get(cutter_name)
    target = bpy.data.objects.get(target_name)
    if not cutter or not target:
        return
    
    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)

def main():

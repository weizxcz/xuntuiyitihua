def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if not obj.data.materials:
        obj.data.materials.append(mat)
    else:
        obj.data.materials[0] = mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder oriented along Z by default."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    return obj

class RobustLocator:
    def __init__(self, obj):
        self.obj = obj
        bpy.context.view_layer.update()
        mat = obj.matrix_world
        
        # Get bounds in World Space
        bm = bmesh.new()
        bm.from_mesh(obj.data)
        bm.transform(mat)
        
        self.min_x = min(v.co.x for v in bm.verts)
        self.max_x = max(v.co.x for v in bm.verts)
        self.min_y = min(v.co.y for v in bm.verts)
        self.max_y = max(v.co.y for v in bm.verts)
        self.min_z = min(v.co.z for v in bm.verts)
        self.max_z = max(v.co.z for v in bm.verts)
        
        self.center = Vector(((self.min_x+self.max_x)/2, (self.min_y+self.max_y)/2, (self.min_z+self.max_z)/2))
        
        bm.free()

    def get_face_info(self, face_type):
        """Returns (center, normal) for the specified face."""
        if face_type == 'center':
            return self.center, Vector((0,0,0))
        elif face_type == 'top_face':
            return Vector((self.center.x, self.center.y, self.max_z)), Vector((0,0,1))
        elif face_type == 'bottom_face':
            return Vector((self.center.x, self.center.y, self.min_z)), Vector((0,0,-1))
        elif face_type == 'left_face':
            return Vector((self.min_x, self.center.y, self.center.z)), Vector((-1,0,0))
        elif face_type == 'right_face':
            return Vector((self.max_x, self.center.y, self.center.z)), Vector((1,0,0))
        elif face_type == 'front_face':

        elif face_type == 'radial_point': # Default X+
            return Vector((self.max_x, self.center.y, self.center.z)), Vector((1,0,0))
        return self.center, Vector((0,0,0))

def align_to_face(target_obj, target_face_type, source_obj, source_face_type):
    """
    Aligns target_obj's face to source_obj's face.
    Assumes faces are parallel (normals match).
    Moves target so its face center coincides with source face center.
    """
    t_loc, t_norm = RobustLocator(target_obj).get_face_info(target_face_type)
    s_loc, s_norm = RobustLocator(source_obj).get_face_info(source_face_type)
    
    # Calculate offset
    offset = s_loc - t_loc
    
    # Apply offset
    target_obj.location += offset

def align_to_center(target_obj, source_obj):
    """Aligns target center to source center."""
    t_loc = RobustLocator(target_obj).center

    offset = s_loc - t_loc
    target_obj.location += offset

def align_to_avg_center(target_obj, obj_list):
    """Aligns target center to average of source centers."""
    avg = Vector((0,0,0))
    for obj in obj_list:
        avg += RobustLocator(obj).center
    avg /= len(obj_list)
    
    t_loc = RobustLocator(target_obj).center
    offset = avg - t_loc
    target_obj.location += offset

def boolean_union(objects_to_union, keep_name):
    """
    Performs boolean union on a list of objects.
    Keeps the object with keep_name as the active result.
    """
    if not objects_to_union:
        return
    
    # Select all objects
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects_to_union:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
        
    # Apply modifiers if any (clean up)
    for obj in objects_to_union:
        for mod in obj.modifiers:
            if mod.type == 'BOOLEAN':
                bpy.ops.object.modifier_apply(modifier=mod.name)

    # Perform Boolean Union
    # Note: In Blender 3.x, boolean operations are often done via modifiers or the 'BoolTool' addon.
    # For a script, using the Boolean Modifier is the most reliable programmatic way.
    
    # We will use the first object as the base
    base_obj = objects_to_union[0]
    
    # Ensure base has no existing boolean modifiers that might conflict
    for mod in base_obj.modifiers:
        if mod.type == 'BOOLEAN':
            base_obj.modifiers.remove(mod)
            
    # Add boolean modifier for each other object
    for obj in objects_to_union[1:]:
        mod = base_obj.modifiers.new(name="BoolUnion", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        
    # Apply the final modifier
    bpy.context.view_layer.update()
    mod = base_obj.modifiers[0] # The last one added
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove the helper objects
    for obj in objects_to_union[1:]:
        bpy.data.objects.remove(obj, do_unlink=True)
        
    # Rename base to keep_name if different
    if base_obj.name != keep_name:
        base_obj.name = keep_name
        
    return base_obj

def boolean_subtract(cutter_obj, target_obj):
    """Subtracts cutter from target."""
    mod = target_obj.modifiers.new(name="BoolSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    bpy.context.view_layer.update()
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def average(points):
    """Average of a list of vectors."""
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

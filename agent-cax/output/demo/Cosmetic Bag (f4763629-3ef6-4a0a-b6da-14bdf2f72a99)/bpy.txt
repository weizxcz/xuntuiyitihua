import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube and apply transforms."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def align_z_to_vector(obj, target_vector):
    """Align object's local Z axis to a target world vector."""
    # Get current world rotation
    rot_mat = obj.matrix_world.to_quaternion()
    
    # Target direction
    target_dir = target_vector.normalized()
    # Up vector (arbitrary, but needs to be non-parallel)
    up = Vector((0, 0, 1))
    if abs(target_dir.dot(up)) > 0.99:
        up = Vector((1, 0, 0))
        
    # Calculate rotation matrix to align Z to target_dir
    # We want to rotate the object so its local Z (0,0,1) points to target_dir
    # Current local Z in world is rot_mat * (0,0,1)
    
    # Simple approach: Set rotation using quaternion from current up to target up?
    # No, we want to align the object's Z axis.
    
    # Let's use a temporary object to calculate the rotation
    temp = bpy.context.view_layer.objects.active
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0,0,0))
    temp_obj = bpy.context.active_object
    temp_obj.name = "_temp_align"
    
    # Align temp object Z to target
    temp_obj.rotation_mode = 'QUATERNION'
    temp_obj.rotation_quaternion = target_dir.to_track_quat('Z', 'Y') # Align Z to target, Y is up-ish
    
    # Get the rotation delta
    # Actually, simpler: Just set the object's rotation directly if we know the target axis.
    # But for generic helper:
    
    # Calculate quaternion that rotates (0,0,1) to target_dir
    q = Vector((0,0,1)).rotation_difference(target_dir)
    
    # Apply this rotation to the object's current rotation
    # obj.rotation_quaternion = q * obj.rotation_quaternion # Order matters
    
    # Let's just set the rotation euler based on the quaternion
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = q
    
    bpy.ops.object.select_all(action='DESELECT')
    temp_obj.select_set(True)
    bpy.context.view_layer.objects.active = temp_obj
    bpy.ops.object.delete()
    return obj

def align_z_to_world_z(obj):
    """Align object's local Z axis to world Z."""
    return align_z_to_vector(obj, Vector((0, 0, 1)))

def align_z_to_world_x(obj):
    """Align object's local Z axis to world X."""
    return align_z_to_vector(obj, Vector((1, 0, 0)))

def align_z_to_world_y(obj):
    """Align object's local Z axis to world Y."""
    return align_z_to_vector(obj, Vector((0, 1, 0)))

def align_z_to_negative_z(obj):
    """Align object's local Z axis to negative world Z."""
    return align_z_to_vector(obj, Vector((0, 0, -1)))

def align_z_to_negative_x(obj):
    """Align object's local Z axis to negative world X."""
    return align_z_to_vector(obj, Vector((-1, 0, 0)))

def align_z_to_negative_y(obj):
    """Align object's local Z axis to negative world Y."""
    return align_z_to_vector(obj, Vector((0, -1, 0)))

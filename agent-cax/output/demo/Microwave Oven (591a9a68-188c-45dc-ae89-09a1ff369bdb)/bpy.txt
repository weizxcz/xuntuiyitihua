def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    if name in bpy.data.materials:
        return bpy.data.materials[name]
    
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder aligned to Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube aligned to axes."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def align_object_face_to_object_face(source_obj, target_obj, source_face='bottom', target_face='top', offset=0):
    """
    Aligns a face of source_obj to a face of target_obj.
    Assumes objects are axis-aligned.
    source_face/target_face: 'top', 'bottom', 'front', 'back', 'left', 'right'
    """
    # Get bounding boxes
    src_min = Vector(source_obj.matrix_world @ source_obj.bound_box[0])
    src_max = Vector(source_obj.matrix_world @ source_obj.bound_box[4])
    tgt_min = Vector(target_obj.matrix_world @ target_obj.bound_box[0])
    tgt_max = Vector(target_obj.matrix_world @ target_obj.bound_box[4])
    
    # Calculate current centers
    src_center = (src_min + src_max) / 2
    tgt_center = (tgt_min + tgt_max) / 2
    
    # Determine alignment vector and distance
    # We want to move source_obj so its specific face touches target_obj's specific face
    
    # Helper to get face coordinate
    def get_face_coord(center, min_v, max_v, face):
        if face == 'top': return max_v.z
        if face == 'bottom': return min_v.z
        if face == 'front': return max_v.y # Assuming Y is front in Blender default
        if face == 'back': return min_v.y
        if face == 'right': return max_v.x
        if face == 'left': return min_v.x
        return 0

    # Calculate target position for source center
    # Target face position
    tgt_face_pos = get_face_coord(tgt_center, tgt_min, tgt_max, target_face)
    src_face_pos = get_face_coord(src_center, src_min, src_max, source_face)
    
    # Distance to move
    # If aligning source bottom to target top:
    # src_bottom should equal tgt_top
    # src_center.z - (src_height/2) = tgt_top
    # src_center.z = tgt_top + src_height/2
    
    # General logic:
    # We want: src_face_pos_new = tgt_face_pos
    # src_face_pos_new = src_center_new + (src_face_pos - src_center)
    # src_center_new = tgt_face_pos - (src_face_pos - src_center)
    
    offset_vec = tgt_face_pos - src_face_pos
    new_center = tgt_center + offset_vec
    
    # Apply offset
    move_vec = new_center - src_center
    source_obj.location += move_vec + (Vector((0,0,offset)) if source_face in ['top','bottom'] else Vector((0,0,0))) # Simplified offset application
    
    # Refined offset application based on axis
    if source_face in ['top', 'bottom']:
        source_obj.location.z += offset
    elif source_face in ['front', 'back']:
        source_obj.location.y += offset
    elif source_face in ['left', 'right']:
        source_obj.location.x += offset

def align_object_center_to_object_center(source_obj, target_obj):
    """Aligns the center of source_obj to the center of target_obj."""
    tgt_center = (Vector(target_obj.matrix_world @ target_obj.bound_box[0]) + Vector(target_obj.matrix_world @ target_obj.bound_box[4])) / 2
    src_center = (Vector(source_obj.matrix_world @ source_obj.bound_box[0]) + Vector(source_obj.matrix_world @ source_obj.bound_box[4])) / 2
    source_obj.location += (tgt_center - src_center)

def apply_offset(obj, offset_vec):
    """Applies a local offset to an object's location."""
    # Assuming standard orientation where local axes align with world for simple offsets
    # If rotated, we might need to transform offset, but instructions imply simple offsets
    obj.location += offset_vec

# Helper for rotated object face alignment
def align_rotated_face_to_axis_aligned_face(source_obj, target_obj, source_local_face_axis, target_world_axis, target_world_pos):
    """
    Aligns a face of a rotated source object to a specific world coordinate plane defined by target.
    source_local_face_axis: 'x', 'y', 'z' (positive direction of local axis)
    target_world_axis: 'x', 'y', 'z'
    target_world_pos: The coordinate value on that axis where the face should be.
    """
    # Get source object's local face world position relative to its center
    # We need the world matrix of the source
    mat = source_obj.matrix_world
    inv_mat = mat.inverted()
    
    # Define local face point (e.g., for left face (-X), point is (-size/2, 0, 0) in local space)
    # But we need the center of the face.
    # Let's approximate center of face by taking the corner and averaging, or just using the extent.
    # Simpler: The face center in local space is (axis * size/2, 0, 0) etc.
    
    size = source_obj.dimensions
    local_face_center = Vector((0,0,0))
    if source_local_face_axis == 'x': local_face_center.x = size.x / 2
    elif source_local_face_axis == 'x_neg': local_face_center.x = -size.x / 2
    elif source_local_face_axis == 'y': local_face_center.y = size.y / 2
    elif source_local_face_axis == 'y_neg': local_face_center.y = -size.y / 2
    elif source_local_face_axis == 'z': local_face_center.z = size.z / 2
    elif source_local_face_axis == 'z_neg': local_face_center.z = -size.z / 2
    
    # Transform to world
    world_face_center = mat @ local_face_center
    
    # Current source center in world
    src_center_world = source_obj.location
    
    # We want: world_face_center_new = target_world_pos on the specific axis
    # world_face_center_new = src_center_new + (world_face_center - src_center_world)
    
    # Calculate offset vector from current center to current face center
    offset_to_face = world_face_center - src_center_world
    
    # New center should be:
    # target_pos - offset_to_face
    new_center = Vector((0,0,0))
    if target_world_axis == 'x': new_center.x = target_world_pos - offset_to_face.x
    elif target_world_axis == 'y': new_center.y = target_world_pos - offset_to_face.y
    elif target_world_axis == 'z': new_center.z = target_world_pos - offset_to_face.z
    
    # Move source object
    move_vec = new_center - src_center_world
    source_obj.location += move_vec

def align_face_to_face(source_obj, target_obj, source_local_axis, target_local_axis):
    """
    Aligns source_obj's local face (defined by axis) to target_obj's local face.
    Assumes both objects have same orientation or we handle rotation.
    Here both are rotated by rot_panel.
    """
    mat_src = source_obj.matrix_world
    mat_tgt = target_obj.matrix_world
    
    # Get local face centers
    src_size = source_obj.dimensions
    tgt_size = target_obj.dimensions
    
    src_local_face = Vector((0,0,0))
    if source_local_axis == 'x': src_local_face.x = src_size.x / 2
    elif source_local_axis == 'x_neg': src_local_face.x = -src_size.x / 2
    elif source_local_axis == 'y': src_local_face.y = src_size.y / 2
    elif source_local_axis == 'y_neg': src_local_face.y = -src_size.y / 2
    elif source_local_axis == 'z': src_local_face.z = src_size.z / 2
    elif source_local_axis == 'z_neg': src_local_face.z = -src_size.z / 2
    
    tgt_local_face = Vector((0,0,0))
    if target_local_axis == 'x': tgt_local_face.x = tgt_size.x / 2
    elif target_local_axis == 'x_neg': tgt_local_face.x = -tgt_size.x / 2
    elif target_local_axis == 'y': tgt_local_face.y = tgt_size.y / 2
    elif target_local_axis == 'y_neg': tgt_local_face.y = -tgt_size.y / 2
    elif target_local_axis == 'z': tgt_local_face.z = tgt_size.z / 2
    elif target_local_axis == 'z_neg': tgt_local_face.z = -tgt_size.z / 2
    
    # World positions of these faces
    src_face_world = mat_src @ src_local_face
    tgt_face_world = mat_tgt @ tgt_local_face
    
    src_center_world = source_obj.location
    tgt_center_world = target_obj.location
    
    # Offset from src center to src face
    src_offset = src_face_world - src_center_world
    
    # We want src_face_world_new = tgt_face_world
    # src_center_new = tgt_face_world - src_offset
    new_src_center = tgt_face_world - src_offset
    
    move_vec = new_src_center - src_center_world
    source_obj.location += move_vec

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a world vector."""
    # Determine current axis vector
    if axis == 'X':
        current = Vector((1, 0, 0))
    elif axis == 'Y':
        current = Vector((0, 1, 0))
    elif axis == 'Z':
        current = Vector((0, 0, 1))
    else:
        return

    target = vector.normalized()
    
    # Calculate rotation matrix to align current to target
    rot_matrix = current.rotation_difference(target).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = current.rotation_difference(target)

def align_face_to_face(obj_src, axis_src, obj_tgt, axis_tgt):
    """Align source object's face axis to target object's face axis.
    This usually means aligning the normals (Z axis of the face).
    """
    # For cubes, faces are aligned with axes.
    # We want the source object's local axis (e.g., +Z) to point in the same direction as target's local axis (e.g., +Z).
    # Since we are aligning faces, we align the normal vectors.
    align_axis_to_vector(obj_src, axis_src, Vector((0,0,1))) # Simplified: assuming target is already aligned or we just align normals
    
    # More robust: align source normal to target normal
    # Get target normal in world space
    tgt_normal = obj_tgt.matrix_world.to_3x3() @ Vector((0,0,1)) # Assuming Z is up for faces usually
    
    # Actually, let's just align the specific axes requested.
    # If aligning Z_face to Z_face, we align obj_src.Z to obj_tgt.Z
    src_normal = obj_src.matrix_world.to_3x3() @ Vector((0,0,1))
    tgt_normal = obj_tgt.matrix_world.to_3x3() @ Vector((0,0,1))
    
    rot = src_normal.rotation_difference(tgt_normal)
    obj_src.rotation_quaternion = rot

def align_face_to_face_custom(obj_src, src_axis, obj_tgt, tgt_axis):
    """Align a specific axis of src to a specific axis of tgt."""
    # Get world direction of src axis
    if src_axis == 'X':
        src_dir = Vector((1,0,0))
    elif src_axis == 'Y':
        src_dir = Vector((0,1,0))
    elif src_axis == 'Z':
        src_dir = Vector((0,0,1))
        
    if tgt_axis == 'X':
        tgt_dir = Vector((1,0,0))
    elif tgt_axis == 'Y':
        tgt_dir = Vector((0,1,0))
    elif tgt_axis == 'Z':
        tgt_dir = Vector((0,0,1))
        
    # We want to rotate obj_src so that its src_dir aligns with tgt_dir (in world space)
    # But obj_tgt might be rotated. So we need tgt_dir in world space.
    tgt_world_dir = obj_tgt.matrix_world.to_3x3() @ tgt_dir
    
    # Calculate rotation to align src_dir (local) to tgt_world_dir (world)
    # Current src_dir in world
    current_src_world = obj_src.matrix_world.to_3x3() @ src_dir
    
    rot = current_src_world.rotation_difference(tgt_world_dir)
    obj_src.rotation_quaternion = rot

def align_face_position(obj_src, src_axis, obj_tgt, tgt_axis):
    """Move obj_src so its face is coplanar with obj_tgt's face."""
    # Get world position of src face center
    if src_axis == 'X':
        src_offset = Vector((1,0,0))
    elif src_axis == 'Y':
        src_offset = Vector((0,1,0))
    elif src_axis == 'Z':
        src_offset = Vector((0,0,1))
        
    if tgt_axis == 'X':
        tgt_offset = Vector((1,0,0))
    elif tgt_axis == 'Y':
        tgt_offset = Vector((0,1,0))
    elif tgt_axis == 'Z':
        tgt_offset = Vector((0,0,1))
        
    # Get half extents
    src_half = obj_src.dimensions / 2
    tgt_half = obj_tgt.dimensions / 2
    
    # World position of src face center
    src_face_world = obj_src.location + (obj_src.matrix_world.to_3x3() @ src_offset) * src_half[src_axis]
    
    # World position of tgt face center
    tgt_face_world = obj_tgt.location + (obj_tgt.matrix_world.to_3x3() @ tgt_offset) * tgt_half[tgt_axis]
    
    # Move src so its face aligns with tgt face
    # We want src_face_world == tgt_face_world
    # delta = tgt_face_world - src_face_world
    # But we need to be careful about direction. 
    # If we align +Z to +Z, we want src top to touch tgt top.
    
    # Let's calculate the target location for the object center
    # Target Center = Tgt Face Center - (Src Half Extent along aligned axis in World Space)
    # Note: The direction from Tgt Face to Tgt Center is -tgt_offset.
    # The direction from Src Center to Src Face is +src_offset.
    
    # Vector from Tgt Face to Tgt Center
    vec_tgt_center = -(obj_tgt.matrix_world.to_3x3() @ tgt_offset) * tgt_half[tgt_axis]
    
    # Target Center Position
    tgt_center_world = tgt_face_world + vec_tgt_center
    
    # We want Src Face to be at tgt_face_world.
    # Src Center = Src Face - (Src Half Extent along src_offset)
    # But src_offset is in local space.
    vec_src_face_from_center = (obj_src.matrix_world.to_3x3() @ src_offset) * src_half[src_axis]
    
    obj_src.location = tgt_face_world - vec_src_face_from_center

def boolean_subtract(cutter_name, target_name):
    """Apply boolean subtract modifier."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is visible and selected for modifier creation if needed, 
    # but usually we just add modifier to target.
    
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target_name, other_name):
    """Apply boolean union modifier."""
    target = bpy.data.objects[target_name]
    other = bpy.data.objects[other_name]
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(other, do_unlink=True)

def add_bevel(obj_name, width, segments):
    """Add bevel modifier."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for simplicity in this script context
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def average(points):
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

# --- Scene Reset & Units ---

def reset_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    
    # Set units to metres
    bpy.context.scene.unit_settings.scale_length = 1
    bpy.context.scene.unit_settings.system = 'METRIC'

    # Since no root object is created, we assume alignment to world is the intent.
    
    # Then move by offset(0, 0, 0).
    base_plate.location += Vector((0, 0, 0))

    # slot_rail_1

    align_axis_to_vector(slot_rail_1, 'Z', Vector((0, 0, 1)))

    # Align slot_rail_1.Z_face to base_plate.Z_face.
    # XZ face normal is Y. Z face normal is Z.

    # This means the top of the rail touches the top of the plate? Or center?
    # "Align ... Z_face to ... Z_face" usually implies coplanarity.
    # base_plate Z_face is at Z = 0.005 (half height).
    # slot_rail_1 Z_face is at Z = 0.0025 (half height).
    # If we align Z_face to Z_face, the tops are at the same level.
    align_face_position(slot_rail_1, 'Z', base_plate, 'Z')

    slot_rail_1.location += Vector((0, 0.045, 0.005))

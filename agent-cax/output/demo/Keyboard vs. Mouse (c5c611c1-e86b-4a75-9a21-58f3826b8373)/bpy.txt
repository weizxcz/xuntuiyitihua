# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0), material=None):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    if material:
        if name not in [mat.name for mat in obj.data.materials]:
            obj.data.materials.append(material)
    return obj

def add_bevel(obj, width=0.01, segments=5):
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align object's local axis to a world vector."""
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else:
        local_axis = Vector((0, 0, 1))
    
    target = vector.normalized()
    if local_axis.normalized().dot(target) < 0.9999:
        rot_quat = local_axis.rotation_difference(target)
        obj.rotation_euler = rot_quat.to_euler()

def average(points):
    if not points:
        return Vector((0,0,0))
    return sum(points, Vector()) / len(points)

def boolean_subtract(cutter_obj, target_obj):
    # Ensure target has a modifier stack
    if not hasattr(target_obj, 'modifiers'):
        return
    
    # Add Boolean Modifier
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def create_anchor_locator(obj_name, location, rotation=(0,0,0)):
    """Creates an empty object to serve as an anchor."""
    bpy.ops.object.empty_add(location=location)
    anchor = bpy.context.active_object
    anchor.name = obj_name
    anchor.rotation_euler = rotation
    return anchor

def align_to_face(target_obj, anchor_obj, target_axis='Z', anchor_axis='Z', offset_z=0):
    """
    Aligns target_obj to the face of anchor_obj.
    Assumes standard orientation where +Z is up.
    """
    # Get anchor's world location
    anchor_loc = anchor_obj.location
    
    # Calculate target location based on anchor's top face
    # Assuming anchor is a primitive like a cube or cylinder, its top face is at local Z + height/2
    # We need to transform this local offset to world space
    anchor_mat = anchor_obj.matrix_world
    anchor_half_h = anchor_obj.dimensions[2] / 2 if anchor_obj.type == 'MESH' else 0
    
    # Simple approach: Place target on top of anchor
    # Vector from anchor center to anchor top face in world space
    anchor_top_world = anchor_mat @ Vector((0, 0, anchor_half_h))
    
    # Set target location to anchor top
    target_obj.location = anchor_top_world
    
    # Align orientations
    # If both are aligned to +Z, no rotation change needed for alignment
    # If specific face alignment is needed, we'd rotate target to match anchor's face normal
    
    # Apply Z offset
    target_obj.location.z += offset_z

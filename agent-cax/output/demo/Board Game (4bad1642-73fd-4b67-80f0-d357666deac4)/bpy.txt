def make_material(name, diffuse_color):
    """Create a new Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube mesh object."""
    # size is total width/height/depth
    half = size / 2.0
    bpy.ops.mesh.primitive_cube_add(size=half, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    """Add a disc (cylinder with small depth) mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, axis='Z', target_vector=(0,0,1)):
    """Align an object's local axis to a target world vector."""
    # Ensure object is in world space for calculation
    mat_world = obj.matrix_world
    local_axis = obj.matrix_world.to_quaternion() @ mathutils.Vector(axis)
    target = mathutils.Vector(target_vector).normalized()
    
    if local_axis.dot(target) < -0.9999:
        # 180 degree flip
        rot_quat = mathutils.Quaternion((1,0,0), math.pi)
    else:
        rot_quat = local_axis.rotation_difference(target)
    
    obj.rotation_euler = rot_quat.to_euler()

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtraction using modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target is active for modifier addition
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target_name, other_name):
    """Perform boolean union."""
    target = bpy.data.objects[target_name]
    other = bpy.data.objects[other_name]
    
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(other, do_unlink=True)

def add_bevel(obj_name, width=0.001, segments=2):
    """Add a bevel modifier to an object."""
    obj = bpy.data.objects[obj_name]
    bpy.context.view_layer.objects.active = obj
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply to keep geometry simple for this script
    bpy.ops.object.modifier_apply(modifier=mod.name)

class Locator:
    """Helper class to find centers of objects."""
    def __init__(self, obj):
        self.obj = obj
    
    def center_world(self):
        return self.obj.matrix_world.to_translation()
    
    def face_center_world(self, face_type='TOP'):
        """Approximate center of a specific face based on bounding box."""
        # Simple approximation: center + offset along normal
        center = self.center_world()
        # For standard primitives aligned to axes:
        if face_type == 'TOP':
            return center + Vector((0,0, self.obj.dimensions.z/2))
        elif face_type == 'BOTTOM':
            return center + Vector((0,0, -self.obj.dimensions.z/2))
        return center

def average(points):
    """Calculate average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

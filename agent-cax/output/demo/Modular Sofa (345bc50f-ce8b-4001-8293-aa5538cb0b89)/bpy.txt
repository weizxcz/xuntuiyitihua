import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = (0, 0, 0)
    return obj

def add_cylinder(name, radius, depth, location):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0)
    return obj

def align_to_face(obj, target_obj, face_type='center'):
    """
    Aligns obj to a specific face of target_obj.
    face_type: 'center', 'left_face', 'right_face', 'top_face', 'back_face', 'front_face'
    Assumes standard axis alignment:
    - left_face: min X
    - right_face: max X
    - top_face: max Z
    - back_face: min Y (assuming +Y is forward, -Y is back) or max Y depending on convention. 
      Standard Blender: +Z Up, +Y Forward, +X Right.
      Let's assume:
      - left_face: min X
      - right_face: max X
      - top_face: max Z
      - back_face: min Y (if camera looks down -Y) or max Y. 
      Usually "back" of a sofa is against a wall. If sofa faces +Y, back is -Y.
      Let's define:
      - left_face: min X
      - right_face: max X
      - top_face: max Z
      - back_face: min Y (assuming object is oriented along Y)
      - front_face: max Y
    """
    # Get bounding box of target
    min_x, min_y, min_z = target_obj.matrix_world @ target_obj.bound_box[0]
    max_x, max_y, max_z = target_obj.matrix_world @ target_obj.bound_box[6]
    
    # Calculate center of target
    center_x = (min_x + max_x) / 2
    center_y = (min_y + max_y) / 2
    center_z = (min_z + max_z) / 2
    
    # Calculate face center
    if face_type == 'center':
        loc = Vector((center_x, center_y, center_z))
    elif face_type == 'left_face':
        loc = Vector((min_x, center_y, center_z))
    elif face_type == 'right_face':
        loc = Vector((max_x, center_y, center_z))
    elif face_type == 'top_face':
        loc = Vector((center_x, center_y, max_z))
    elif face_type == 'back_face':
        loc = Vector((center_x, min_y, center_z))
    elif face_type == 'front_face':
        loc = Vector((center_x, max_y, center_z))
    else:
        loc = Vector((center_x, center_y, center_z))
        
    # Align object center to face center
    obj.location = loc
    return obj

def offset_object(obj, offset_vec):
    obj.location += offset_vec
    return obj

def boolean_union(objects_to_union, keep_name):
    """
    Performs boolean union on a list of objects, keeping the first one as the result.
    """
    if not objects_to_union:
        return None
    
    target = objects_to_union[0]
    modifiers = []
    
    for obj in objects_to_union[1:]:
        mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        modifiers.append(mod)
        
    # Apply modifiers
    for mod in modifiers:
        bpy.context.view_layer.objects.active = target
        bpy.ops.object.modifier_apply(modifier=mod.name)
        
    # Remove other objects
    for obj in objects_to_union[1:]:
        bpy.data.objects.remove(obj, do_unlink=True)
        
    return target

def create_locator(obj, name_suffix="_locator"):
    """Creates a temporary empty at the object's location for alignment purposes if needed."""
    loc = bpy.data.objects.new(name=obj.name + name_suffix, object_data=None)
    loc.location = obj.location
    bpy.context.collection.objects.link(loc)
    return loc

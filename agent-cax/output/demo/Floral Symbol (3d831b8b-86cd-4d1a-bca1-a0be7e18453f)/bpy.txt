import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color=(1.0, 1.0, 1.0, 1.0)):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0)):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    return obj

def add_cylinder(name, radius=0.5, depth=1, location=(0,0,0)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius=0.5, location=(0,0,0)):
    """Add a disc (flat cylinder) mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=0.001, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius=0.5, location=(0,0,0)):
    """Add an UV sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius=0.5, depth=1, location=(0,0,0)):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, radius=0.01, segments=4):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'WEIGHT'
    # Apply bevel immediately to keep geometry simple for boolean operations
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    return obj

def boolean_union(obj_target, obj_source):
    """Perform boolean union of obj_source into obj_target."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = obj_target
    # Add boolean modifier to target
    mod = obj_target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj_source
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    # Remove source object if it's not needed (optional, but keeps scene clean)
    # bpy.data.objects.remove(obj_source, do_unlink=True)
    return obj_target

def boolean_subtract(obj_target, obj_source):
    """Perform boolean subtract of obj_source from obj_target."""
    bpy.context.view_layer.objects.active = obj_target
    mod = obj_target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = obj_source
    bpy.ops.object.modifier_apply(modifier=mod.name)
    return obj_target

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align object's local axis to a world vector."""
    # Get current rotation
    rot_euler = obj.rotation_euler
    # This is a simplified alignment. For robust alignment, we use matrix multiplication.
    # However, for simple axis alignment (like Z to Z), we can just ensure rotation is identity or specific.
    # If vector is (0,0,1) and axis is Z, no rotation needed if object is default.
    if axis == 'Z' and vector == (0,0,1):
        obj.rotation_euler = (0, 0, 0)
    elif axis == 'X' and vector == (1,0,0):
        obj.rotation_euler = (0, 0, math.pi/2)
    elif axis == 'Y' and vector == (0,1,0):
        obj.rotation_euler = (-math.pi/2, 0, 0)
    else:
        # Generic fallback: align using look-at logic (simplified)
        # For this script, we assume standard orientations or explicit rotations.
        pass

def align_center_to_center(obj_source, obj_target):
    """Align center of obj_source to center of obj_target."""
    loc = obj_target.location
    obj_source.location = loc

def align_bottom_face_to_bottom_face(obj_source, obj_target):
    """Align bottom face of obj_source to bottom face of obj_target.
    Assumes objects are upright (Z up). Bottom face is at min Z."""
    # Get bounding box min Z for both
    # Note: This requires objects to be in their local orientation relative to world
    # For simple primitives, we can calculate offset based on height.
    
    # Get world bounds
    def get_world_min_z(obj):
        # Transform vertices to world space
        verts = [obj.matrix_world @ v.co for v in obj.data.vertices]
        return min(v.z for v in verts)
    
    def get_height(obj):
        verts = [obj.matrix_world @ v.co for v in obj.data.vertices]
        max_z = max(v.z for v in verts)
        min_z = min(v.z for v in verts)
        return max_z - min_z

    target_min_z = get_world_min_z(obj_target)
    source_height = get_height(obj_source)
    
    # We want source's bottom face (min_z) to be at target's bottom face (min_z)
    # Current source location is already aligned center-to-center or needs to be set.
    # Let's assume we align centers first, then adjust Z.
    
    # If we align centers:
    # source_center_z = target_center_z
    # source_min_z = source_center_z - source_height/2
    # We want source_min_z = target_min_z
    # So source_center_z_new = target_min_z + source_height/2
    # delta_z = source_center_z_new - source_center_z_old
    # delta_z = (target_min_z + source_height/2) - target_center_z
    
    target_center_z = obj_target.location.z
    delta_z = (target_min_z + source_height/2) - target_center_z
    obj_source.location.z += delta_z

def rotate_object(obj, axis='Z', angle=0):
    """Rotate object around its local axis."""
    if axis == 'Z':
        obj.rotation_euler.z += angle
    elif axis == 'X':
        obj.rotation_euler.x += angle
    elif axis == 'Y':
        obj.rotation_euler.y += angle

# Helper to apply material
def apply_material(obj, mat):
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

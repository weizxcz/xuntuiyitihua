def make_material(name, diffuse_color):
    """Create a new material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Apply a material to an object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation
    if material:
        apply_material(obj, material)
    return obj

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation

        apply_material(obj, material)
    return obj

def align_face_to_face(obj_source, obj_target, source_face, target_face, offset=0):
    """
    Aligns a face of obj_source to a face of obj_target.
    source_face/target_face: 'left', 'right', 'top', 'bottom', 'front', 'back', 'center'
    Note: This assumes standard orientation where +X is right, +Y is front, +Z is top.
    """
    # Ensure objects are in world space for calculation
    bpy.context.view_layer.update()
    
    # Get bounding box centers and extents in world space
    src_bbox = get_world_bbox(obj_source)
    tgt_bbox = get_world_bbox(obj_target)
    
    src_center = src_bbox['center']
    tgt_center = tgt_bbox['center']
    
    # Calculate target position based on face alignment
    # We want to move obj_source so its specific face touches the target's specific face
    
    # Define face offsets relative to center
    # Right: +X, Left: -X, Top: +Z, Bottom: -Z, Front: +Y, Back: -Y
    
    dx, dy, dz = 0, 0, 0
    
    # Determine the position of the source face
    if source_face == 'right':
        dx = src_bbox['extents'][0] # Half width
    elif source_face == 'left':
        dx = -src_bbox['extents'][0]
    elif source_face == 'top':
        dz = src_bbox['extents'][2]
    elif source_face == 'bottom':
        dz = -src_bbox['extents'][2]
    elif source_face == 'front':
        dy = src_bbox['extents'][1]
    elif source_face == 'back':
        dy = -src_bbox['extents'][1]
    elif source_face == 'center':
        dx, dy, dz = 0, 0, 0
        
    # Determine the position of the target face
    if target_face == 'right':
        dx_tgt = tgt_bbox['extents'][0]
    elif target_face == 'left':
        dx_tgt = -tgt_bbox['extents'][0]
    elif target_face == 'top':
        dz_tgt = tgt_bbox['extents'][2]
    elif target_face == 'bottom':
        dz_tgt = -tgt_bbox['extents'][2]
    elif target_face == 'front':
        dy_tgt = tgt_bbox['extents'][1]
    elif target_face == 'back':
        dy_tgt = -tgt_bbox['extents'][1]
    elif target_face == 'center':
        dx_tgt, dy_tgt, dz_tgt = 0, 0, 0
        
    # Calculate the shift required
    # Current source face world pos = src_center + face_offset
    # Target face world pos = tgt_center + tgt_face_offset
    # We want: src_center_new + face_offset = tgt_center + tgt_face_offset
    # src_center_new = tgt_center + tgt_face_offset - face_offset
    
    new_center = tgt_center + Vector((dx_tgt, dy_tgt, dz_tgt)) - Vector((dx, dy, dz))
    
    # Apply offset
    new_center += Vector((offset, 0, 0)) # Default offset axis is X for face alignment usually, but prompt says offset(0,0,0) mostly.
    # The prompt says "Then move by offset(dx, dy, dz)". This is applied AFTER alignment.
    # So here we just do the alignment.
    
    # Move object
    obj_source.location = new_center
    
    # Update world matrix for next steps
    obj_source.update_from_matrix_safe()

def get_world_bbox(obj):
    """Get world space bounding box center and extents."""
    bpy.context.view_layer.update()
    matrix_world = obj.matrix_world
    bbox_corners = [
        matrix_world @ Vector((-1, -1, -1)),
        matrix_world @ Vector((-1, -1, 1)),
        matrix_world @ Vector((-1, 1, -1)),
        matrix_world @ Vector((-1, 1, 1)),
        matrix_world @ Vector((1, -1, -1)),
        matrix_world @ Vector((1, -1, 1)),
        matrix_world @ Vector((1, 1, -1)),
        matrix_world @ Vector((1, 1, 1)),
    ]
    # Scale bbox_corners by object scale? No, matrix_world includes scale.
    # But primitive cubes are created with size=1 and scaled.
    # The bbox_corners calculation above assumes unit cube centered at 0.
    # Let's use the object's actual bounds.
    
    # Simpler approach: Use the object's dimensions and location
    # Dimensions are in local space.
    # World extents = scale * 0.5
    extents = [abs(s * d / 2) for s, d in zip(obj.scale, obj.dimensions)]
    center = obj.location
    
    return {
        'center': center,
        'extents': extents
    }

def align_center_to_center(obj_source, obj_target):
    """Align center of source to center of target."""
    obj_source.location = obj_target.location

def align_center_to_face(obj_source, obj_target, target_face):
    """Align center of source to a specific face of target."""
    tgt_bbox = get_world_bbox(obj_target)
    tgt_center = tgt_bbox['center']
    tgt_extents = tgt_bbox['extents']
    
    dx, dy, dz = 0, 0, 0
    
    if target_face == 'right':
        dx = tgt_extents[0]
    elif target_face == 'left':
        dx = -tgt_extents[0]
    elif target_face == 'top':
        dz = tgt_extents[2]
    elif target_face == 'bottom':
        dz = -tgt_extents[2]
    elif target_face == 'front':
        dy = tgt_extents[1]
    elif target_face == 'back':
        dy = -tgt_extents[1]
        
    new_center = tgt_center + Vector((dx, dy, dz))
    obj_source.location = new_center

def average_points(objects):
    """Calculate the average center of a list of objects."""
    if not objects:
        return Vector((0,0,0))
    total = Vector((0,0,0))
    for obj in objects:
        total += obj.location
    return total / len(objects)

def rotate_to_align_z(obj, target_axis):
    """Rotate object so its local +Z aligns with target_axis (world vector)."""
    # Default +Z is (0,0,1)
    # If target is +Z, no rotation needed.
    if target_axis == (0,0,1):
        return
    
    # Calculate rotation matrix
    # We want to rotate local Z (0,0,1) to target_axis
    # Using mathutils rotation
    rot = Matrix.Rotation(0, 4, (0,0,1)) # Identity
    
    # Simple case: +X
    if target_axis == (1,0,0):
        obj.rotation_euler = (0, -90, 0) # -90 deg around Y
    # -X
    elif target_axis == (-1,0,0):
        obj.rotation_euler = (0, 90, 0)
    # +Y
    elif target_axis == (0,1,0):
        obj.rotation_euler = (90, 0, 0) # 90 deg around X
    # -Y
    elif target_axis == (0,-1,0):
        obj.rotation_euler = (-90, 0, 0)
    # +Z
    elif target_axis == (0,0,1):
        obj.rotation_euler = (0, 0, 0)
    # -Z
    elif target_axis == (0,0,-1):
        obj.rotation_euler = (180, 0, 0)
    else:
        # Generic fallback using quaternion
        v1 = Vector((0,0,1))
        v2 = Vector(target_axis).normalized()
        q = v1.rotation_difference(v2)
        obj.rotation_euler = q.to_euler()

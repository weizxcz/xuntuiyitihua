import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder oriented along Z by default."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube."""
    # size is (x, y, z)
    x, y, z = size
    bpy.ops.mesh.primitive_cube_add(size=1, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (x/2, y/2, z/2)
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Align a specific axis of an object to a target vector."""
    # Get current rotation as quaternion
    rot_q = obj.rotation_euler.to_quaternion()
    
    # Define local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    if axis_to_align not in local_axes:
        return

    # Calculate rotation to align local axis to target vector
    # We want to rotate the object such that local_axes[axis_to_align] becomes target_vector
    # The rotation quaternion is from local_axis to target_vector
    rot = local_axes[axis_to_align].rotation_difference(target_vector)
    
    # Apply this rotation to the object's current rotation
    # Note: This aligns the axis but might introduce roll. 
    # For simple alignment, we can set the rotation directly if we don't care about other axes,
    # or compose it. Here we compose.
    obj.rotation_euler = (rot * rot_q).to_euler()

def get_world_center(obj):
    """Get the world location of an object's center."""
    return obj.matrix_world.to_translation()

def get_world_face_center(obj, face_type='top'):
    """Get the world location of a specific face center."""
    # Assuming standard orientation:
    # Top face is +Z, Bottom is -Z, Front is +Y, Back is -Y, Right is +X, Left is -X
    # We need to transform the local face center by the object's world matrix
    
    # Local centers of faces for a cube/cylinder centered at 0,0,0
    # For a cube of size (sx, sy, sz), top face center is (0, 0, sz/2)
    # For a cylinder of radius r, height h, top face center is (0, 0, h/2)
    
    # Determine local offset based on geometry type and face
    local_offset = Vector((0, 0, 0))
    
    if obj.type == 'MESH':
        # Check if it's roughly a cube or cylinder
        # We'll approximate based on scale or just use bounding box logic if needed.
        # Simpler: Use the object's dimensions.
        dims = obj.dimensions
        if face_type == 'top':
            local_offset.z = dims.z / 2
        elif face_type == 'bottom':
            local_offset.z = -dims.z / 2
        elif face_type == 'front':
            local_offset.y = dims.y / 2
        elif face_type == 'back':
            local_offset.y = -dims.y / 2
        elif face_type == 'right':
            local_offset.x = dims.x / 2
        elif face_type == 'left':
            local_offset.x = -dims.x / 2
            
    # Transform local offset to world space
    # The center of the face is obj.location + local_offset in local space
    # But we need to rotate local_offset by obj.rotation
    world_offset = obj.matrix_world.to_3x3() @ local_offset
    return get_world_center(obj) + world_offset

def anchor_to_object(target_obj, anchor_obj, anchor_type='center'):
    """
    Anchor target_obj to anchor_obj.
    If anchor_type is 'center', align centers.
    If anchor_type is 'face', align target center to anchor face center.
    """
    if anchor_type == 'center':
        anchor_loc = get_world_center(anchor_obj)
        # Move target so its center matches anchor center
        # Current world center of target
        target_center = get_world_center(target_obj)
        # Delta
        delta = anchor_loc - target_center
        target_obj.location += delta
    elif anchor_type == 'face':
        # Anchor target's center to anchor_obj's top face center
        anchor_face_loc = get_world_face_center(anchor_obj, 'top')
        target_center = get_world_center(target_obj)
        delta = anchor_face_loc - target_center
        target_obj.location += delta

def apply_offset(obj, offset_vector):
    """Apply an offset in the object's local space? 
    The prompt says 'Offset is in reference local axes'.
    However, usually in these scripts, offsets are applied in World space relative to the anchor point,
    OR in the object's local space after orientation.
    Given 'Anchor... Then move by offset', and the orientation happens before,
    let's assume the offset is applied in the object's LOCAL coordinate system after orientation.
    """
    # Convert local offset to world direction
    # If the object is rotated, local X might be world Y.
    # We transform the offset vector by the object's rotation matrix.
    rot_mat = obj.matrix_world.to_3x3()
    world_offset = rot_mat @ offset_vector
    obj.location += world_offset

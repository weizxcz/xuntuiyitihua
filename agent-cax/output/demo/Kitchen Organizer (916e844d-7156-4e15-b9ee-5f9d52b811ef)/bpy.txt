def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def boolean_union(target, operands, keep_target_name=True):
    """Perform Boolean Union using modifiers."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    
    # Add modifier to target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = operands[0]
    
    # If multiple operands, we need to chain or use a group. 
    # Simplest for script: apply one by one or use a collection.
    # Let's use a collection for cleaner modifier stack if >1 operand.
    if len(operands) > 1:
        # Create a collection of operands
        coll = bpy.data.collections.new("BoolOperands")
        for op in operands[1:]:
            coll.objects.link(op)
        mod.collection = coll
        
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Clean up operands if desired, or leave them. 
    # Usually we hide or delete operands after union.
    for op in operands:
        op.hide_viewport = True
        op.hide_render = True
        
    return target

def boolean_subtract(target, cutter):
    """Perform Boolean Subtract using modifiers."""
    bpy.context.view_layer.objects.active = target
    mod = target.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    bpy.ops.object.modifier_apply(modifier=mod.name)
    cutter.hide_viewport = True
    cutter.hide_render = True

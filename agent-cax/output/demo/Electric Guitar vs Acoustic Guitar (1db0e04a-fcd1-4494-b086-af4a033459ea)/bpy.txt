import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def add_path_sweep(name, length, radius=0.001, location=(0,0,0), rotation_euler=(0,0,0)):
    """Creates a thin cylinder to represent a string"""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=length, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def get_face_normal(obj, face_type='center'):
    """
    Returns the world-space location of the center of the +Z face of an object.
    Assumes object is axis-aligned for simplicity in this specific script context,
    but uses matrix transformation for robustness.
    """
    # Local center of +Z face (assuming default cube/cylinder orientation where +Z is up/depth)
    # For a cube of size S, +Z face center is at (0, 0, S/2)
    # For a cylinder of depth D, +Z face center is at (0, 0, D/2)
    # We need to transform this local point by the object's world matrix.
    
    # Determine half-size/depth based on type
    # Note: This helper assumes standard primitives created by our helpers.
    # A more robust way is to use mesh data, but for these simple aligned objects:
    
    loc = obj.matrix_world.copy()
    
    # Get dimensions
    if obj.type == 'MESH':
        # Approximate bounding box for +Z face
        # We assume the object's local Z axis is the one we care about.
        # The +Z face is at local Z = max(Z)
        min_z = min(v.co.z for v in obj.data.vertices)
        max_z = max(v.co.z for v in obj.data.vertices)
        center_z_local = max_z
        
        # Center of the face in local X/Y
        min_x = min(v.co.x for v in obj.data.vertices if abs(v.co.z - max_z) < 0.001)
        max_x = max(v.co.x for v in obj.data.vertices if abs(v.co.z - max_z) < 0.001)
        min_y = min(v.co.y for v in obj.data.vertices if abs(v.co.z - max_z) < 0.001)
        max_y = max(v.co.y for v in obj.data.vertices if abs(v.co.z - max_z) < 0.001)
        
        local_center = Vector((min_x + max_x)/2, (min_y + max_y)/2, center_z_local)
        return loc @ local_center
    return obj.location

def align_to_face(obj, target_obj, target_face_type='center'):
    """
    Aligns the +Z face of 'obj' to the +Z face of 'target_obj'.
    This sets the location of 'obj' such that its +Z face center coincides with target's +Z face center.
    """
    target_face_loc = get_face_normal(target_obj, target_face_type)
    
    # We need to place 'obj' so that its +Z face is at target_face_loc.
    # obj.matrix_world @ local_face_center = target_face_loc
    # local_face_center is (0,0, max_z_local)
    
    # Get obj's local max Z
    max_z_local = max(v.co.z for v in obj.data.vertices)
    
    # Calculate required world location for obj's origin
    # world_loc = target_face_loc - (obj.matrix_world.to_quaternion() @ Vector(0,0,max_z_local))
    
    # Since we are aligning axes first (rotation is already set to align +Z to +Z),
    # we can simplify.
    # If obj is rotated such that its local Z is world Z:
    # world_loc = target_face_loc - Vector(0,0,max_z_local)
    
    # However, to be generic with rotation:
    rot = obj.matrix_world.to_quaternion()
    local_offset = Vector(0, 0, max_z_local)
    world_offset = rot @ local_offset
    
    obj.location = target_face_loc - world_offset

def connect_faces(obj, target_obj):
    """
    Connects the +Z face of 'obj' to the +Z face of 'target_obj'.
    Similar to align_to_face but ensures they touch or are very close.
    Usually implies placing obj's +Z face at target's +Z face.
    """
    align_to_face(obj, target_obj, 'center')

def apply_material(obj, mat_name):
    if obj.data.materials:
        obj.data.materials[0] = bpy.data.materials[mat_name]
    else:
        obj.data.materials.append(bpy.data.materials[mat_name])

def create_string(name, offset_x):
    # Connect to electric_bridge.+Z_face
    # Bridge +Z face is at Z = 0.005 (half of 0.01)
    # String length 0.6. String is a cylinder along Z.
    # We want the bottom of the string (Z=-0.3) or top?
    # "Connect electric_string_1.+Z_face to electric_bridge.+Z_face"
    # This implies the +Z face of the string touches the +Z face of the bridge.
    # String +Z face is at local Z = 0.3.
    # So string origin Z should be Bridge_Z_face - 0.3.
    
    obj = add_path_sweep(name, 0.6, radius=0.0005, location=(0,0,0), rotation_euler=(0,0,0))
    apply_material(obj, "metal_strings")
    
    # Align +Z face of string to +Z face of bridge
    align_to_face(obj, electric_bridge, 'center')
    
    # Offset
    obj.location += Vector((offset_x, 0, 0))
    return obj

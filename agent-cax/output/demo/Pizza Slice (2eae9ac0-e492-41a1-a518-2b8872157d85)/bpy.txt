import bpy

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis_to_align='+Z', target_vector):
    """Align an object's local axis to a target world vector."""
    # This is a simplified helper. For strict alignment, we calculate rotation.
    # Default Blender Z-up alignment logic:
    if axis_to_align == '+Z':
        # Align Z to target
        z_vec = Vector((0, 0, 1))
        if target_vec.length > 0:
            rot_quat = z_vec.rotation_difference(target_vec)
            obj.rotation_euler = rot_quat.to_euler()
    elif axis_to_align == '+Y':
        y_vec = Vector((0, 1, 0))
        if target_vec.length > 0:
            rot_quat = y_vec.rotation_difference(target_vec)
            obj.rotation_euler = rot_quat.to_euler()
    elif axis_to_align == '+X':
        x_vec = Vector((1, 0, 0))
        if target_vec.length > 0:
            rot_quat = x_vec.rotation_difference(target_vec)
            obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average of a list of Vector objects."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        """Return the world center of the object."""
        return self.obj.matrix_world.to_translation()

    def face_center_world(self, face_index=0):
        """Return the world center of a specific face."""
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bm.transform(self.obj.matrix_world)
        if face_index < len(bm.faces):
            face = bm.faces[face_index]
            center = face.calc_center_median()
            bm.free()
            return center
        bm.free()
        return self.center_world()

    def radial_point_world(self, index=0):
        """Return a point on the radial edge (approximate for cylinders/cones)."""
        # Simplified: returns center for now as radial point logic varies by shape
        return self.center_world()

def boolean_subtract(cutter_obj, target_obj):
    """Apply Boolean Subtract modifier."""
    # Ensure target is active for modifier application context if needed, 
    # but modifiers are added to target.
    
    # Add modifier to target
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(target_obj, other_obj):

def add_bevel(obj, width, segments=2):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for simplicity in this generator context
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def create_triangular_prism(name, vertices_2d, height, location, rotation_euler, material=None):
    """
    Create a triangular prism by extruding a 2D triangle along Z.
    vertices_2d: List of (x, y) tuples.
    height: Extrusion depth.
    """
    # Create mesh data
    mesh = bpy.data.meshes.new(name + "_mesh")
    mesh.from_pydata(vertices_2d + [(v[0], v[1], height) for v in vertices_2d], [], [])
    mesh.update()
    
    # Create object
    obj = bpy.data.objects.new(name, mesh)
    obj.location = location
    obj.rotation_euler = rotation_euler
    
    # Add faces
    bm = bmesh.new()
    bm.from_mesh(mesh)
    
    # Top face
    top_verts = bm.verts[len(vertices_2d):]
    bm.faces.new(top_verts)
    
    # Bottom face
    bottom_verts = bm.verts[:len(vertices_2d)]
    bm.faces.new(bottom_verts)
    
    # Side faces
    for i in range(len(vertices_2d)):
        v1 = bottom_verts[i]
        v2 = bottom_verts[(i + 1) % len(vertices_2d)]
        v3 = top_verts[(i + 1) % len(vertices_2d)]
        v4 = top_verts[i]
        bm.faces.new([v1, v2, v3, v4])
        
    bm.to_mesh(mesh)
    bm.free()
    
    if material:
        obj.data.materials.append(material)
        
    bpy.context.collection.objects.link(obj)
    return obj

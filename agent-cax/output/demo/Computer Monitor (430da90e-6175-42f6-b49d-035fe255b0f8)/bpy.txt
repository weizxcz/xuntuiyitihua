import bpy
import bmesh
import math
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    if name in bpy.data.materials:
        return bpy.data.materials[name]
    
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    # Set roughness/metallic defaults if needed, but keeping simple for now
    return mat

def apply_material(obj, mat):
    """Apply a material to an object's first material slot."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder oriented along Z axis by default."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, end_fill_type='NGON')
    obj = bpy.context.object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=size) # size is half-width in Blender
    obj = bpy.context.object
    obj.name = name
    return obj

def add_cuboid(name, dimensions):
    """Add a cuboid (cube scaled) with given dimensions (x, y, z)."""
    # Blender cube 'size' is half-extent. So we scale by dim/2.
    # But easier: add cube with size=1, then scale.
    bpy.ops.mesh.primitive_cube_add(size=1.0)
    obj = bpy.context.object
    obj.name = name
    obj.scale = (dimensions[0]/2, dimensions[1]/2, dimensions[2]/2)
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Rotate object so its local axis aligns with world vector."""
    # Get current rotation
    rot = obj.rotation_euler
    # This is complex to do purely with euler math robustly. 
    # Instead, we use a temporary empty to orient, or matrix multiplication.
    # Simple approach for standard axes:
    
    # Define target direction
    target = vector.normalized()
    
    # Define current axis direction
    if axis == 'X':
        current = Vector((1, 0, 0))
    elif axis == 'Y':
        current = Vector((0, 1, 0))
    elif axis == 'Z':
        current = Vector((0, 0, 1))
    else:
        return

    # Calculate rotation matrix to align current to target
    # We want to rotate 'current' to 'target'.
    # Use quaternion rotation
    q = current.rotation_difference(target)
    
    # Apply rotation
    obj.rotation_euler = q.to_euler()

def get_face_center(obj, face_type):
    """
    Calculate the world position of a specific face center.
    face_type: 'front', 'back', 'top', 'bottom', 'left', 'right'
    Assumes standard orientation where +Z is up, +Y is front, +X is right.
    """
    # Get object matrix
    mat = obj.matrix_world
    
    # Get dimensions
    dims = obj.dimensions
    half_dims = dims / 2.0
    
    # Local face centers
    if face_type == 'front': # +Y
        local_center = Vector((0, half_dims[1], 0))
    elif face_type == 'back': # -Y
        local_center = Vector((0, -half_dims[1], 0))
    elif face_type == 'top': # +Z
        local_center = Vector((0, 0, half_dims[2]))
    elif face_type == 'bottom': # -Z
        local_center = Vector((0, 0, -half_dims[2]))
    elif face_type == 'right': # +X
        local_center = Vector((half_dims[0], 0, 0))
    elif face_type == 'left': # -X
        local_center = Vector((-half_dims[0], 0, 0))
    else:
        return obj.location
    
    # Transform to world
    return mat @ local_center

def align_to_face(obj_target, face_type_target, obj_source, face_type_source):
    """
    Align obj_source's face to obj_target's face.
    This usually means placing obj_source such that its face touches obj_target's face.
    """
    target_pos = get_face_center(obj_target, face_type_target)
    source_pos = get_face_center(obj_source, face_type_source)
    
    # Calculate offset needed to move source face to target face
    # We want source_pos + offset = target_pos
    offset = target_pos - source_pos
    
    # Apply offset to source object
    obj_source.location += offset

def align_centers(obj_target, obj_source):
    """Align center of obj_source to center of obj_target."""
    target_center = obj_target.location
    source_center = obj_source.location
    obj_source.location = target_center

def align_center_to_face(obj_target, face_type, obj_source):
    """
    Align center of obj_source to a specific face of obj_target.
    """
    target_face_pos = get_face_center(obj_target, face_type)
    obj_source.location = target_face_pos

def add_locator(name, location):
    """Add a simple empty locator."""
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=location)
    loc = bpy.context.object
    loc.name = name
    return loc

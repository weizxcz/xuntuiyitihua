def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube primitive, scaled and placed."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def boolean_subtract(target, cutter, name=None):
    """Perform boolean subtract using BMesh. Returns the modified target."""
    # Ensure target is active for context if needed, but we'll use BMesh directly for robustness
    bm = bmesh.new()
    bm.from_mesh(target.data)
    
    # Create a temporary mesh for the cutter to intersect
    cutter_mesh = cutter.data.copy()
    cutter_obj_temp = bpy.data.objects.new("temp_cutter", cutter_mesh)
    bpy.context.collection.objects.link(cutter_obj_temp)
    
    # Apply transforms
    cutter_obj_temp.matrix_world = cutter.matrix_world
    
    # Use the built-in boolean modifier for simplicity and reliability in script
    # We will use the modifier approach for cleaner history if needed, 
    # but for a single script execution, direct boolean operation via modifier is safer.
    
    # Revert to Modifier approach for standard Blender workflow
    target.modifiers.new(name="BoolSub", type='BOOLEAN')
    target.modifiers["BoolSub"].operation = 'DIFFERENCE'
    target.modifiers["BoolSub"].object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier="BoolSub")
    
    # Cleanup temp object
    bpy.data.objects.remove(cutter_obj_temp)
    
    # Update mesh
    bm.free()
    target.data.update()
    
    return target

def boolean_union(objects, name=None):
    """Boolean union a list of objects into the first one."""
    if not objects:
        return None
    target = objects[0]
    for i, obj in enumerate(objects[1:], 1):
        target.modifiers.new(name=f"BoolUnion_{i}", type='BOOLEAN')
        target.modifiers[f"BoolUnion_{i}"].operation = 'UNION'
        target.modifiers[f"BoolUnion_{i}"].object = obj
    
    bpy.context.view_layer.objects.active = target
    for mod in target.modifiers:
        if mod.type == 'BOOLEAN':
            bpy.ops.object.modifier_apply(modifier=mod.name)
            
    target.data.update()
    return target

class Locator:
    def __init__(self, obj):
        self.obj = obj
    
    def center_world(self):
        return self.obj.matrix_world.to_translation()
    
    def face_center_world(self, face_normal_vector):
        # Approximate face center based on normal direction from center
        center = self.center_world()
        # Project slightly along the normal
        return center + (self.obj.matrix_world.to_3x3() @ face_normal_vector) * (self.obj.dimensions.magnitude / 2)
    
    def radial_point_world(self, angle, axis='Z'):
        center = self.center_world()
        radius = self.obj.dimensions.magnitude / 2
        if axis == 'Z':
            return center + Vector((radius * angle.cos(), radius * angle.sin(), 0))
        return center

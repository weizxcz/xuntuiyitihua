def make_material(name, diffuse_color):
    """Create a Principled BSDF material with the given diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_text_object(text, name, location=(0,0,0), size=0.1):
    """Add a text object."""
    bpy.ops.object.text_add(location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.size = size
    obj.data.body = text
    return obj

def add_cylinder(name, radius=0.5, depth=1.0, segments=32, location=(0,0,0)):
    """Add a cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius=0.5, segments=32, location=(0,0,0)):
    """Add a disc (flat cylinder with 1 segment height) aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=0.001, segments=segments, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size=(1,1,1), location=(0,0,0)):
    """Add a cube centered at location."""
    bpy.ops.mesh.primitive_cube_add(size=size[0], location=location)
    obj = bpy.context.active_object
    obj.scale = (size[1]/size[0], size[2]/size[0], size[0]/size[0]) # Normalize to size tuple
    # Actually, primitive_cube_add 'size' is half-width. 
    # Let's just set scale after creation for clarity.
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    return obj

def add_cone(name, radius_top=0, radius_bottom=0.5, depth=1.0, segments=32, location=(0,0,0)):

def add_bevel(obj_name, width=0.01, segments=4):
    """Add a bevel modifier to an object."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find centers and faces of objects."""
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        self.matrix_world = obj.matrix_world

    def center_world(self):
        """Return the world center of the object."""
        # Average of bound box corners
        corners = [self.matrix_world @ Vector(corner) for corner in self.bound_box]
        return average(corners)

    def face_center_world(self, face_type='top'):
        """Return the world center of a specific face (top, bottom, front, back, left, right)."""
        corners = [self.matrix_world @ Vector(corner) for corner in self.bound_box]
        # Determine indices based on local coordinates before transform
        # Bound box indices: 0:(-,-,-), 1:(+,-,-), 2:(-,+,-), 3:(+,+,-), 4:(-,-,+), 5:(+,-,+), 6:(-,+,+), 7:(+,+,+)
        # Assuming standard orientation:
        # Top face (Z max): indices 4,5,6,7
        # Bottom face (Z min): indices 0,1,2,3
        # Front face (Y max): indices 2,3,6,7
        # Back face (Y min): indices 0,1,4,5
        # Right face (X max): indices 1,3,5,7
        # Left face (X min): indices 0,2,4,6
        
        if face_type == 'top':
            indices = [4, 5, 6, 7]
        elif face_type == 'bottom':
            indices = [0, 1, 2, 3]
        elif face_type == 'front':
            indices = [2, 3, 6, 7]
        elif face_type == 'back':
            indices = [0, 1, 4, 5]
        elif face_type == 'right':
            indices = [1, 3, 5, 7]
        elif face_type == 'left':
            indices = [0, 2, 4, 6]
        else:
            return self.center_world()
            
        face_corners = [corners[i] for i in indices]
        return average(face_corners)

    def radial_point_world(self, direction='X+'):
        """Return a point on the radial surface."""
        center = self.center_world()
        # Approximate radius based on bounding box
        # This is a rough helper for radial alignment if needed
        return center

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract operation."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is active for selection context if needed, but we use operators
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    cutter.select_set(True)
    
    bpy.ops.object.modifier_add(type='BOOLEAN')
    target.modifiers[-1].name = "Boolean_Subtract"
    target.modifiers[-1].operation = 'DIFFERENCE'
    target.modifiers[-1].object = cutter
    
    bpy.ops.object.modifier_apply(modifier="Boolean_Subtract")
    
    # Optional: Delete cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target_name, union_name):
    """Perform a boolean union operation."""
    target = bpy.data.objects[target_name]
    union = bpy.data.objects[union_name]
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    union.select_set(True)
    
    bpy.ops.object.modifier_add(type='BOOLEAN')
    target.modifiers[-1].name = "Boolean_Union"
    target.modifiers[-1].operation = 'UNION'
    target.modifiers[-1].object = union
    
    bpy.ops.object.modifier_apply(modifier="Boolean_Union")
    
    # Optional: Delete union
    bpy.data.objects.remove(union, do_unlink=True)

def create_curved_box(name, length, height, depth, curvature, location=(0,0,0)):
    """
    Create a curved box (cylinder segment or similar).
    For a piano shell, this is often a large cylinder segment.
    We will approximate it by creating a cylinder and cutting it, or using a mesh modifier.
    Given the simplicity, we'll create a cylinder with specific radius/height and scale it.
    However, 'curved_box' implies a box with curved sides.
    Let's create a cylinder with radius approximating the curvature, and scale it to the box dimensions.
    """
    # Approximation: A cylinder with radius = curvature? No, curvature usually means the radius of the arc.
    # If curvature=0.2, it's a slight curve.
    # Let's create a cylinder with radius R such that the arc length matches length?
    # Or simpler: Create a cylinder, apply a solidify or bevel?
    # Let's just create a cylinder with radius = curvature * 10? No.
    # Let's assume curvature is the radius of the cylinder.
    # If length=1.5, and it's a cylinder, the circumference is 2*pi*R.
    # Let's just create a cylinder with radius = curvature (0.2) and scale it?
    # Actually, let's create a cylinder with radius = 0.5 (default) and scale it to fit.
    # Better: Create a cylinder with radius = curvature (0.2) is too small.
    # Let's interpret curvature as the radius of the cylinder.
    # If length=1.5, and it's a cylinder, the length is along Z.
    # Let's create a cylinder with radius = 0.5 and scale X to 1.5/ (2*pi*0.5)? No.
    
    # Let's use a simpler approach: Create a cylinder with radius = curvature (0.2) is likely wrong.
    # Let's assume curvature is the radius of the cylinder.
    # If length=1.5, and it's a cylinder, the length is along Z.
    # Let's create a cylinder with radius = 0.5 and scale it.
    
    # Alternative: Create a cylinder with radius = curvature (0.2) and scale it to length=1.5?
    # No, let's create a cylinder with radius = 0.5 and scale it to length=1.5, height=0.6, depth=0.5.
    # This is getting complicated. Let's just create a cylinder with radius = curvature (0.2) and scale it.
    
    # Let's try: Create a cylinder with radius = curvature (0.2) and height = height (0.6).
    # Then scale it to match length and depth.
    # This is a rough approximation.
    
    bpy.ops.mesh.primitive_cylinder_add(radius=curvature, depth=height, segments=64, location=location)
    obj = bpy.context.active_object
    obj.name = name
    # Scale to match length and depth
    # The cylinder is along Z. We want to scale X and Y to match length and depth.
    # But a cylinder is round. Scaling it makes it an oval cylinder.
    obj.scale = (length / (2 * curvature), depth / (2 * curvature), 1.0)
    return obj

def create_curved_plate(name, length, width, thickness, curvature, location=(0,0,0)):
    """
    Create a curved plate (cylinder segment).
    Similar to curved_box but thin.
    """
    # Create a cylinder with radius = curvature
    bpy.ops.mesh.primitive_cylinder_add(radius=curvature, depth=thickness, segments=64, location=location)
    obj = bpy.context.active_object
    obj.name = name
    # Scale to match length and width
    obj.scale = (length / (2 * curvature), width / (2 * curvature), 1.0)
    return obj

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube primitive, scale it, and apply transforms."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.data.materials.append(material) if material else None
    return obj

def add_cylinder(name, size, location, rotation_euler, material=None):
    """Add a cylinder primitive, scale it, and apply transforms."""
    bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=1, depth=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]) # Depth is Z
    obj.rotation_euler = rotation_euler
    obj.data.materials.append(material) if material else None
    return obj

def add_cone(name, size, location, rotation_euler, material=None):
    """Add a cone primitive, scale it, and apply transforms."""
    bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=1, radius2=0, depth=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2])
    obj.rotation_euler = rotation_euler
    obj.data.materials.append(material) if material else None
    return obj

def add_uvsphere(name, size, location, rotation_euler, material=None, hemi=False):
    """Add an UV sphere primitive."""
    bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.data.materials.append(material) if material else None
    
    if hemi:
        # Cut in half along Z
        bpy.ops.object.mode_set(mode='EDIT')
        bpy.ops.mesh.select_all(action='SELECT')
        # We want to keep the top half (Z > 0) or bottom? Usually hemisphere implies Z>0 or Z<0.
        # Let's assume we want the top half (Z > 0) relative to object space.
        # Select vertices with Z > 0
        bpy.ops.mesh.select_all(action='DESELECT')
        for v in obj.data.vertices:
            if v.co.z > 0.0001:
                v.select = True
        bpy.ops.mesh.delete(type='VERT')
        bpy.ops.mesh.select_all(action='SELECT')
        bpy.ops.mesh.normals_make_consistent(inside=False)
        bpy.ops.object.mode_set(mode='OBJECT')
        
    return obj

def align_axis_to_vector(obj, axis='Z', vector=Vector((0,0,1))):
    """Align object's local axis to a world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else: # Z
        local_axis = Vector((0,0,1))
        
    # Calculate rotation to align local_axis to vector
    rot_matrix = local_axis.rotation_difference(vector).to_matrix().to_4x4()
    obj.rotation_euler = rot_matrix.to_euler()

def average_vectors(vectors):
    """Calculate the average of a list of vectors."""
    if not vectors:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for v in vectors:
        avg += v
    return avg / len(vectors)

class Locator:
    """Helper class to find centers and faces of objects."""
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        # Local coordinates of the 8 corners
        self.local_corners = [Vector(c) for c in self.bound_box]
        
    def center_world(self):
        """Return the world center of the object."""
        # Average of local corners transformed to world
        world_corners = [self.obj.matrix_world @ c for c in self.local_corners]
        return average_vectors(world_corners)
    
    def face_center_world(self, face_name):
        """
        Return the world center of a specific face.
        face_name: 'XY_face' (Z+), 'XY_face_neg' (Z-), 'XZ_face' (Y+), 'XZ_face_neg' (Y-), 'YZ_face' (X+), 'YZ_face_neg' (X-)
        Note: The prompt uses 'XY_face' which usually means the face normal to Z. 
        Let's assume 'XY_face' means the face at max Z (top), and 'XY_face_neg' or similar for min Z.
        However, the prompt says "front_panel.XY_face". In a cuboid aligned to axes, this is ambiguous.
        Standard convention: XY face is perpendicular to Z.
        Let's assume:
        'XY_face' -> Max Z face
        'XY_face_neg' -> Min Z face (or just 'bottom')
        'XZ_face' -> Max Y face
        'XZ_face_neg' -> Min Y face
        'YZ_face' -> Max X face
        'YZ_face_neg' -> Min X face
        
        Let's refine based on typical CAD naming:
        If object is axis aligned:
        XY_face usually refers to the face in the XY plane, i.e., normal to Z.
        Let's assume 'XY_face' is the face with the highest Z coordinate.
        """
        # Determine which corners belong to the face
        if face_name == 'XY_face': # Max Z
            indices = [i for i, c in enumerate(self.local_corners) if abs(c.z - max(c.z for c in self.local_corners)) < 0.0001]
        elif face_name == 'XY_face_neg': # Min Z
            indices = [i for i, c in enumerate(self.local_corners) if abs(c.z - min(c.z for c in self.local_corners)) < 0.0001]
        elif face_name == 'XZ_face': # Max Y
            indices = [i for i, c in enumerate(self.local_corners) if abs(c.y - max(c.y for c in self.local_corners)) < 0.0001]
        elif face_name == 'XZ_face_neg': # Min Y
            indices = [i for i, c in enumerate(self.local_corners) if abs(c.y - min(c.y for c in self.local_corners)) < 0.0001]
        elif face_name == 'YZ_face': # Max X
            indices = [i for i, c in enumerate(self.local_corners) if abs(c.x - max(c.x for c in self.local_corners)) < 0.0001]
        elif face_name == 'YZ_face_neg': # Min X
            indices = [i for i, c in enumerate(self.local_corners) if abs(c.x - min(c.x for c in self.local_corners)) < 0.0001]
        else:
            # Fallback: try to parse generic names
            if 'left' in face_name.lower(): indices = [i for i, c in enumerate(self.local_corners) if abs(c.x - min(c.x for c in self.local_corners)) < 0.0001]
            elif 'right' in face_name.lower(): indices = [i for i, c in enumerate(self.local_corners) if abs(c.x - max(c.x for c in self.local_corners)) < 0.0001]
            elif 'top' in face_name.lower(): indices = [i for i, c in enumerate(self.local_corners) if abs(c.z - max(c.z for c in self.local_corners)) < 0.0001]
            elif 'bottom' in face_name.lower(): indices = [i for i, c in enumerate(self.local_corners) if abs(c.z - min(c.z for c in self.local_corners)) < 0.0001]
            elif 'front' in face_name.lower(): indices = [i for i, c in enumerate(self.local_corners) if abs(c.y - max(c.y for c in self.local_corners)) < 0.0001]
            elif 'back' in face_name.lower(): indices = [i for i, c in enumerate(self.local_corners) if abs(c.y - min(c.y for c in self.local_corners)) < 0.0001]
            else:
                return self.center_world()
        
        if not indices:
            return self.center_world()
            
        face_corners = [self.obj.matrix_world @ self.local_corners[i] for i in indices]
        return average_vectors(face_corners)

    def radial_point_world(self, direction='X', side='pos'):
        """
        Get a point on the radial edge/face for alignment.
        This is less common for cuboids than face centers, but useful for cylinders.
        For cuboids, we'll stick to face centers for alignment as per prompt hints.
        """
        return self.center_world()

def boolean_subtract(cutter_obj, target_obj):
    """Apply Boolean Modifier to subtract cutter from target."""
    # Ensure target is active for modifier addition if needed, but we can add to target directly
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter_obj
    # Apply immediately
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(target_obj, other_obj):
    """Apply Boolean Modifier to union other_obj into target."""
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other_obj
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(other_obj, do_unlink=True)

def add_bevel(obj, width=0.01, segments=5):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply bevel
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

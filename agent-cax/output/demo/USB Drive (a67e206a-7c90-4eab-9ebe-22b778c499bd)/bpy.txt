import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_text_object(text, name, size=0.1):
    obj = bpy.data.objects.new(name, bpy.data.curves.new(name, type='FONT'))
    obj.data.body = text
    obj.data.size = size
    bpy.context.collection.objects.link(obj)
    return obj

def add_cylinder(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, segments=32):
    bpy.ops.mesh.primitive_circle_add(radius=radius, segments=segments, fill_type='TRIFAN', location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    # size is (x, y, z)
    half_x, half_y, half_z = size[0]/2, size[1]/2, size[2]/2
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0,0,0)) # size=1 creates 1x1x1 cube
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (half_x*2, half_y*2, half_z*2) # Correct scaling for primitive_cube_add which uses half-size internally in some versions, but standard is total size. 
    # Actually primitive_cube_add 'size' parameter is the total length of the edge.
    # Let's re-verify: primitive_cube_add(size=1) creates a cube from -0.5 to 0.5. Total size 1.
    # So scale should be size.
    obj.scale = size
    return obj

def add_cone(name, radius_bottom, radius_top, depth):
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_seg=16, height_seg=16):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_seg, ring_count=height_seg, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, radius, segments):
    # Apply bevel modifier
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width_type = 'WIDTH'
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply modifier to mesh
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(cutter_obj, target_obj):
    # Add boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(union_obj, target_obj):
    # Add boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove union object
    bpy.data.objects.remove(union_obj, do_unlink=True)

def align_axis_to_vector(obj, axis, vector):
    """
    Aligns object's local axis to a world vector.
    axis: 'X', 'Y', or 'Z'
    vector: target world direction (normalized)
    """
    # Get current rotation
    rot_euler = obj.rotation_euler.copy()
    
    # Create rotation matrix to align axis to vector
    # We want obj.local_axis to point to vector
    if axis == 'X':
        src = Vector((1, 0, 0))
    elif axis == 'Y':
        src = Vector((0, 1, 0))
    elif axis == 'Z':
        src = Vector((0, 0, 1))
    else:
        raise ValueError("Invalid axis")
        
    # Calculate rotation
    rot = src.rotation_difference(vector)
    
    # Apply rotation
    obj.rotation_euler = rot.to_euler()

def average(points):
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        
    def center_world(self):
        return self.obj.location
    
    def face_center_world(self, face_axis='Z'):
        """
        Returns the center of the face defined by the axis.
        face_axis: 'X', 'Y', 'Z' (positive direction face) or '-X', '-Y', '-Z'
        """
        loc = self.obj.location
        size = self.obj.dimensions
        offset = Vector((0,0,0))
        
        if face_axis == 'X':
            offset.x = size.x / 2
        elif face_axis == '-X':
            offset.x = -size.x / 2
        elif face_axis == 'Y':
            offset.y = size.y / 2
        elif face_axis == '-Y':
            offset.y = -size.y / 2
        elif face_axis == 'Z':
            offset.z = size.z / 2
        elif face_axis == '-Z':
            offset.z = -size.z / 2
            
        # Rotate offset by object's rotation to get world offset
        offset.rotate(self.obj.rotation_euler)
        return loc + offset

    def radial_point_world(self, axis='Z', angle=0):
        """
        Returns a point on the radial edge of a cylinder/disc aligned with axis.
        """
        loc = self.obj.location
        size = self.obj.dimensions
        radius = size[axis] / 2 if axis in ['X','Y','Z'] else 0 # Approximation for non-cylinders
        
        # This is a simplified radial point for cylinders aligned with Z
        # For general use, we assume the object is a cylinder/disc aligned with the specified axis
        # Calculate point in local space
        if axis == 'Z':
            local_pt = Vector((radius * math.cos(angle), radius * math.sin(angle), 0))
        elif axis == 'X':
            local_pt = Vector((0, radius * math.cos(angle), radius * math.sin(angle)))
        elif axis == 'Y':
            local_pt = Vector((radius * math.cos(angle), 0, radius * math.sin(angle)))
            
        local_pt.rotate(self.obj.rotation_euler)
        return loc + local_pt

def create_pin(name, offset_y):
    pin = add_cylinder(name, 0.0005, 0.005, 16) # Radius 0.0005 (Ø0.001), Depth 0.005
    # Rotate pin so its +Z axis aligns +X (world)
    align_axis_to_vector(pin, 'Z', Vector((1, 0, 0)))
    
    # Align pin.center to connector_core.left_face
    # connector_core is rotated +Z to +X.
    # Its local -X face is now pointing in world -X direction?
    # Let's check rotation.
    # Original +Z is (0,0,1). Rotated to (1,0,0).
    # Original -X is (-1,0,0). Rotated by same rotation matrix.
    # Rotation matrix aligns Z to X.
    # Let's use locator to find the face center.
    # connector_core.left_face usually means -X face in local space.
    core_left_face = Locator(connector_core).face_center_world('-X')
    
    pin.location = core_left_face
    pin.location += Vector((0, offset_y, 0)) # Offset

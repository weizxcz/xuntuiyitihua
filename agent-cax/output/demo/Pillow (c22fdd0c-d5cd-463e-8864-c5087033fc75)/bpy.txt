def make_material(name, diffuse_color=(1.0, 1.0, 1.0, 1.0)):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0)):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    return obj

def add_cylinder(name, radius=0.5, depth=1, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_top=0, radius_bottom=0.5, depth=1, location=(0,0,0), rotation=(0,0,0)):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius=0.5, location=(0,0,0)):

    obj = bpy.context.active_object
    obj.name = name

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a target world vector."""
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else: # Z
        local_axis = Vector((0, 0, 1))
    
    target = vector.normalized()
    current = local_axis.copy()
    
    # Calculate rotation
    rot_quat = current.rotation_between(target)
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = rot_quat

def average(points):
    """Calculate the average position of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find centers and points on objects."""
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        """Get the world space center of the object's bounding box."""
        # Use the object's location if it's a primitive or simple mesh
        # For more complex meshes, bounding box is safer
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bmesh.ops.transform(bm, matrix=Matrix.Identity(4)) # Ensure local coords are clean if needed, though from_mesh handles it
        # Actually, bmesh.from_mesh gives local coords. We need world coords.
        # Easier: Use the object's location for primitives, or calculate bbox in world space.
        
        # Robust way: Transform vertices to world space
        verts_world = [self.obj.matrix_world @ v.co for v in self.obj.data.vertices]
        if not verts_world:
            return self.obj.location
        
        center = average(verts_world)
        return center

    def face_center_world(self, face_index=0):
        """Get the world space center of a specific face."""
        if face_index >= len(self.obj.data.polygons):
            return self.obj.location
        
        poly = self.obj.data.polygons[face_index]
        # Get vertices of the face
        verts = [self.obj.matrix_world @ self.obj.data.vertices[i].co for i in poly.vertices]
        return average(verts)

    def radial_point_world(self, direction=(1,0,0), radius=1.0):
        """Get a point on the radial surface in a given direction."""
        # Approximate by taking center + direction * radius
        center = self.center_world()
        return center + Vector(direction) * radius

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract operation using the modifier stack."""
    target = bpy.data.objects.get(target_name)
    cutter = bpy.data.objects.get(cutter_name)
    
    if not target or not cutter:
        print(f"Error: Object '{target_name}' or '{cutter_name}' not found.")
        return

    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply the modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Optional: Remove cutter after use to keep scene clean, or keep it.
    # The prompt says "keep fabric_shell", implying the result.
    # We usually delete the cutter after boolean to avoid clutter, unless it's reused.
    # Here, foam_core is used once.
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(obj1_name, obj2_name):
    """Perform a boolean union."""
    obj1 = bpy.data.objects.get(obj1_name)
    obj2 = bpy.data.objects.get(obj2_name)
    
    if not obj1 or not obj2:
        print(f"Error: Object '{obj1_name}' or '{obj2_name}' not found.")
        return

    mod = obj1.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj2
    
    bpy.context.view_layer.objects.active = obj1
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(obj2, do_unlink=True)

def add_bevel(obj_name, radius=0.01, segments=4, limit_method='ANGLE'):
    """Add a bevel modifier to an object."""
    obj = bpy.data.objects.get(obj_name)
    if not obj:
        print(f"Error: Object '{obj_name}' not found.")
        return

    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = limit_method
    mod.angle_limit = math.radians(30) # Default angle limit
    
    # Apply bevel
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

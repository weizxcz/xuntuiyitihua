import bpy
import bmesh
import math
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0) # Align +Z to +Z
    return obj

def add_disc(name, radius, segments=32):
    """Add a disc aligned with Z axis."""
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='TRIFAN', segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0)
    return obj

def add_cube(name, size):
    """Add a cube centered at origin."""
    # size is (x, y, z)
    sx, sy, sz = size
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (sx, sy, sz)
    obj.rotation_euler = (0, 0, 0)
    return obj

def align_to_vector(obj, vec):
    """Orient object's local Z axis to match world vector."""
    vec = vec.normalized()
    # Calculate rotation matrix to align (0,0,1) to vec
    rot_mat = vec.to_track_quat('Z', 'Y').to_matrix().to_4x4()
    obj.matrix_world = rot_mat

def average_points(points):
    """Calculate the average position of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

def get_face_center(obj, axis='Z', side='bottom'):
    """
    Get the world center of a specific face of an object.
    Assumes object is axis-aligned or we are approximating based on bounds.
    For simplicity in this script, we assume standard orientations.
    """
    # Get local bounds
    min_b, max_b = obj.bound_box
    # Transform bounds to world
    world_min = Vector((min_b[0][0], min_b[0][1], min_b[0][2]))
    world_max = Vector((max_b[6][0], max_b[6][1], max_b[6][2]))
    
    # Apply matrix world to these points
    w_min = obj.matrix_world @ world_min
    w_max = obj.matrix_world @ world_max
    
    if axis == 'Z':
        if side == 'bottom':
            return Vector((w_min.x, w_min.y, w_min.z))
        elif side == 'top':
            return Vector((w_max.x, w_max.y, w_max.z))
    elif axis == 'X':
        if side == 'left':
            return Vector((w_min.x, w_min.y, w_min.z))
        elif side == 'right':
            return Vector((w_max.x, w_max.y, w_max.z))
    elif axis == 'Y':
        if side == 'front': # Assuming Y+ is front
            return Vector((w_min.x, w_min.y, w_min.z))
        elif side == 'back':
            return Vector((w_max.x, w_max.y, w_max.z))
            
    return (w_min + w_max) / 2

def get_center(obj):
    """Get world center of object."""
    return obj.matrix_world.to_translation()

def create_and_place_cube(name, size, anchor_points=None, anchor_side='center', offset=(0,0,0)):
    """
    Creates a cube, orients it, aligns it to anchors, and applies offset.
    """
    obj = add_cube(name, size)
    
    # 1. Orientation: +Z aligns +Z (default for add_cube)
    
    # 2. Anchor/Align
    if anchor_points:
        if anchor_side == 'center':
            target_center = average_points(anchor_points)
            # Move object so its local center is at target
            obj.location = target_center
        elif anchor_side == 'bottom_face':
            # Align bottom face center to average of anchor bottom faces
            # We need to calculate the target Z height from anchors
            # Assuming anchors are upright cubes/posts
            anchor_bottom_centers = [get_face_center(p, 'Z', 'bottom') for p in anchor_points]
            target_bottom_center = average_points(anchor_bottom_centers)
            
            # Current object's bottom face center (local)
            # Cube is centered at 0,0,0. Half height is size[1]/2
            half_h = size[1] / 2
            obj_bottom_local = Vector((0, 0, -half_h))
            
            # We want obj_bottom_local transformed to match target_bottom_center
            # Since rotation is identity, transform is just translation
            # obj.location + obj_bottom_local = target_bottom_center
            obj.location = target_bottom_center - obj_bottom_local
            
    # 3. Offset
    if offset:
        obj.location += Vector(offset)
        
    return obj

def create_and_place_cylinder(name, radius, height, anchor_points=None, anchor_side='center', offset=(0,0,0)):
    """
    Creates a cylinder, orients it, aligns it to anchors, and applies offset.
    """
    obj = add_cylinder(name, radius, height)
    
    # 1. Orientation: +Z aligns +Z (default for add_cylinder)
    
    # 2. Anchor/Align
    if anchor_points:
        if anchor_side == 'center':
            target_center = average_points(anchor_points)
            obj.location = target_center
        elif anchor_side == 'bottom_face':
            anchor_bottom_centers = [get_face_center(p, 'Z', 'bottom') for p in anchor_points]
            target_bottom_center = average_points(anchor_bottom_centers)
            
            half_h = height / 2
            obj_bottom_local = Vector((0, 0, -half_h))
            obj.location = target_bottom_center - obj_bottom_local
            
    # 3. Offset
    if offset:
        obj.location += Vector(offset)
        
    return obj

def rotate_local_y(obj, degrees):
    """Rotate object around its local Y axis."""
    mat = obj.matrix_world
    rot = Matrix.Rotation(math.radians(degrees), 4, 'Y')
    obj.matrix_world = mat @ rot

def rotate_local_x(obj, degrees):
    """Rotate object around its local X axis."""
    mat = obj.matrix_world
    rot = Matrix.Rotation(math.radians(degrees), 4, 'X')
    obj.matrix_world = mat @ rot

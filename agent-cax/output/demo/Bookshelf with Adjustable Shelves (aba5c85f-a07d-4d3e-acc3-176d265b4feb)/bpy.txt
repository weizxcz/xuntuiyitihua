import bpy
import bmesh
import mathutils
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color, roughness=0.5, metallic=0.0):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Roughness'].default_value = roughness
    bsdf.inputs['Metallic'].default_value = metallic
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube object with specific size and transform."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def align_face_to_face(obj_source, obj_target, source_face, target_face, offset=(0,0,0)):
    """
    Aligns a specific face of source object to a specific face of target object.
    source_face/target_face: 'left', 'right', 'bottom', 'top', 'front', 'back'
    Assumes objects are axis-aligned in their local space.
    """
    # Get world transforms
    src_world = obj_source.matrix_world
    tgt_world = obj_target.matrix_world
    
    # Get bounding box centers and extents in world space
    src_bbox = obj_source.bound_box
    tgt_bbox = obj_target.bound_box
    
    # Convert bbox points to world space
    src_pts = [src_world @ Vector(p) for p in src_bbox]
    tgt_pts = [tgt_world @ Vector(p) for p in tgt_bbox]
    
    # Helper to get face center and normal
    def get_face_info(pts, face_name):
        # Find min/max indices for X, Y, Z
        xs = [p.x for p in pts]
        ys = [p.y for p in pts]
        zs = [p.z for p in pts]
        
        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)
        min_z, max_z = min(zs), max(zs)
        
        # Define face centers based on name
        if face_name == 'left': # Min X
            center = Vector((min_x, (min_y+max_y)/2, (min_z+max_z)/2))
            normal = Vector((-1, 0, 0))
        elif face_name == 'right': # Max X
            center = Vector((max_x, (min_y+max_y)/2, (min_z+max_z)/2))
            normal = Vector((1, 0, 0))
        elif face_name == 'bottom': # Min Z (assuming Z is up)
            center = Vector(((min_x+max_x)/2, (min_y+max_y)/2, min_z))
            normal = Vector((0, 0, -1))
        elif face_name == 'top': # Max Z
            center = Vector(((min_x+max_x)/2, (min_y+max_y)/2, max_z))
            normal = Vector((0, 0, 1))
        elif face_name == 'front': # Min Y
            center = Vector(((min_x+max_x)/2, min_y, (min_z+max_z)/2))
            normal = Vector((0, -1, 0))
        elif face_name == 'back': # Max Y
            center = Vector(((min_x+max_x)/2, max_y, (min_z+max_z)/2))
            normal = Vector((0, 1, 0))
        else:
            raise ValueError(f"Unknown face: {face_name}")
            
        return center, normal

    src_center, src_normal = get_face_info(src_pts, source_face)
    tgt_center, tgt_normal = get_face_info(tgt_pts, target_face)
    
    # Calculate translation needed to align centers
    # We want src_center + translation = tgt_center
    # But we also need to account for the fact that they might be touching.
    # The prompt says "Align face to face", implying contact.
    # If source is 'left' and target is 'right', they touch at the interface.
    # src_center is at left edge, tgt_center is at right edge.
    # Distance between them is half_width_src + half_width_tgt.
    # However, simpler approach: Move source so its face center coincides with target face center.
    
    translation = tgt_center - src_center
    
    # Apply translation to object location
    obj_source.location += translation
    
    # Apply additional offset
    obj_source.location += Vector(offset)

def align_face_to_face_simple(obj_source, obj_target, source_face, target_face, offset=(0,0,0)):
    """
    Simplified alignment: Moves source object so its specified face center
    matches the target object's specified face center.
    """
    src_world = obj_source.matrix_world
    tgt_world = obj_target.matrix_world
    
    # Get bbox in world space
    src_bbox = [src_world @ Vector(p) for p in obj_source.bound_box]
    tgt_bbox = [tgt_world @ Vector(p) for p in obj_target.bound_box]
    
    def get_face_center(pts, face_name):
        xs = [p.x for p in pts]
        ys = [p.y for p in pts]
        zs = [p.z for p in pts]
        
        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)
        min_z, max_z = min(zs), max(zs)
        
        if face_name == 'left':
            return Vector((min_x, (min_y+max_y)/2, (min_z+max_z)/2))
        elif face_name == 'right':
            return Vector((max_x, (min_y+max_y)/2, (min_z+max_z)/2))
        elif face_name == 'bottom':
            return Vector(((min_x+max_x)/2, (min_y+max_y)/2, min_z))
        elif face_name == 'top':
            return Vector(((min_x+max_x)/2, (min_y+max_y)/2, max_z))
        elif face_name == 'front':
            return Vector(((min_x+max_x)/2, min_y, (min_z+max_z)/2))
        elif face_name == 'back':
            return Vector(((min_x+max_x)/2, max_y, (min_z+max_z)/2))
        return Vector((0,0,0))

    src_fc = get_face_center(src_bbox, source_face)
    tgt_fc = get_face_center(tgt_bbox, target_face)
    
    # Move source so its face center matches target face center
    delta = tgt_fc - src_fc
    obj_source.location += delta
    obj_source.location += Vector(offset)

# --- Block 0: Scene Reset & Units ---

# Delete all existing objects
bpy.ops.object.select_all(action='DESELECT')
bpy.ops.object.select_all(action='ALL')
bpy.ops.object.delete(use_global=False)

# Set units to metres
bpy.context.scene.unit_settings.scale_length = 1
bpy.context.scene.unit_settings.system = 'METRIC'

# --- Block 1: Materials ---

mat_wood_dark = make_material("mat_wood_dark", (0.35, 0.20, 0.10))

# --- Block 2: Geometry Generation ---

# --- SECTION 1 – Create side panels ---

# Create side_L
# Size: 0.025 (X) x 1.829 (Y) x 0.305 (Z)
# Note: Prompt says "Rotate so +Z aligns +Z". Default cube is Z-up.
# Default cube size is 2m. We scale it.
side_L = add_cube("side_L", (0.025, 1.829, 0.305), (0, 0, 0), (0, 0, 0), mat_wood_dark)

# Anchor side_L to world origin (already at 0,0,0)
# Move by offset(-0.762, 0.9145, 0)
side_L.location += Vector((-0.762, 0.9145, 0))

# Create side_R
side_R = add_cube("side_R", (0.025, 1.829, 0.305), (0, 0, 0), (0, 0, 0), mat_wood_dark)

# Anchor side_R to world origin
# Move by offset(0.762, 0.9145, 0)
side_R.location += Vector((0.762, 0.9145, 0))

# --- SECTION 2 – Create shelves ---

# Shelf dimensions: 0.712 (X) x 0.025 (Y) x 0.305 (Z)
# Orientation: +Z aligns +Z (default)

def create_shelf(name, z_offset):
    shelf = add_cube(name, (0.712, 0.025, 0.305), (0, 0, 0), (0, 0, 0), mat_wood_dark)
    
    # Align shelf.left_face to side_L.right_face
    align_face_to_face_simple(shelf, side_L, 'left', 'right')
    
    # Align shelf.right_face to side_R.left_face
    align_face_to_face_simple(shelf, side_R, 'right', 'left')
    
    # Align shelf.bottom_face to side_L.bottom_face
    align_face_to_face_simple(shelf, side_L, 'bottom', 'bottom')
    
    # Move by offset(0, 0, z_offset)
    shelf.location += Vector((0, 0, z_offset))
    
    return shelf

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive(name, type, size, location, rotation_euler):
    """Add a primitive mesh object and apply transforms."""
    if type == 'CUBE':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif type == 'CYLINDER':
        # size is (radius, depth) for cylinder in this helper context usually, 
        # but here we assume size tuple (radius, radius, height) or similar.
        # Standard add_cylinder takes radius and depth.
        # Input format: Ø0.05x0.05 -> radius=0.025, depth=0.05
        r = size[0] / 2
        h = size[1]
        bpy.ops.mesh.primitive_cylinder_add(radius=r, depth=h, location=location)
        obj = bpy.context.active_object
        # Default cylinder is along Z.
    elif type == 'CONE':
        r = size[0] / 2
        h = size[1]
        bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=r, radius2=0, depth=h, location=location)
        obj = bpy.context.active_object
    elif type == 'UVSPHERE':
        r = size[0] / 2
        bpy.ops.mesh.primitive_uv_sphere_add(radius=r, location=location)
        obj = bpy.context.active_object
    
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.data.name = name + ".Data"
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Rotate object so its local axis aligns with target vector."""
    # Get current rotation
    rot = obj.rotation_euler.copy()
    
    # Define basis vectors
    if axis_to_align == 'X':
        local_vec = Vector((1, 0, 0))
    elif axis_to_align == 'Y':
        local_vec = Vector((0, 1, 0))
    elif axis_to_align == 'Z':
        local_vec = Vector((0, 0, 1))
    
    # Calculate rotation needed
    # We want local_vec to point to target_vector
    # Use quaternion rotation from local_vec to target_vector
    q = local_vec.rotation_difference(target_vector)
    
    # Apply rotation
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = q
    obj.rotation_mode = 'XYZ' # Reset to Euler for consistency if needed, but keep quaternion for precision

def get_face_center(obj, face_type='bottom'):
    """Get the world location of a specific face center."""
    # Ensure object is updated
    obj.update_from_editmode() # Not needed for mesh objects usually, but good practice if in edit mode
    
    # Calculate bounding box to find face centers
    # This is a simplified approach assuming axis-aligned or known orientation
    # For robustness, we calculate based on dimensions and rotation
    
    # Get dimensions
    dim = obj.dimensions
    rot = obj.rotation_euler
    
    # Base center
    center = obj.location
    
    # Offsets for faces relative to center in local space
    if face_type == 'bottom':
        local_offset = Vector((0, 0, -dim[2]/2))
    elif face_type == 'top':
        local_offset = Vector((0, 0, dim[2]/2))
    elif face_type == 'front':
        local_offset = Vector((0, dim[1]/2, 0))
    elif face_type == 'back':
        local_offset = Vector((0, -dim[1]/2, 0))
    elif face_type == 'right':
        local_offset = Vector((dim[0]/2, 0, 0))
    elif face_type == 'left':
        local_offset = Vector((-dim[0]/2, 0, 0))
        
    # Rotate offset to world space
    world_offset = local_offset @ obj.rotation_euler.to_matrix()
    
    return center + world_offset

def connect_faces(obj1, face1, obj2, face2):
    """Move obj2 so face2 aligns with face1."""
    loc1 = get_face_center(obj1, face1)
    loc2 = get_face_center(obj2, face2)
    
    # Calculate translation needed
    # We want obj2's face2 to be at loc1
    # Current face2 is at loc2
    # Translation = loc1 - loc2
    translation = loc1 - loc2
    
    obj2.location += translation

def align_faces(obj_target, face_target, obj_source, face_source):
    """Align obj_source's face_source to obj_target's face_target."""
    # This usually implies snapping the faces together
    # Get centers
    loc_target = get_face_center(obj_target, face_target)
    loc_source = get_face_center(obj_source, face_source)
    
    # Calculate offset
    offset = loc_target - loc_source
    
    # Apply offset
    obj_source.location += offset

def add_locator(name, location):
    """Add a simple empty locator."""
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=location)
    loc = bpy.context.active_object
    loc.name = name
    return loc

def average(points):
    """Calculate average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

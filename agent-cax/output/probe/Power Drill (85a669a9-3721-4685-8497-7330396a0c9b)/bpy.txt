import bpy

def make_material(name, diffuse_color, use_nodes=True):
    """Create a principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive(name, type, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive mesh object."""
    if type == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=size[0], location=location)
        obj = bpy.context.active_object
        obj.name = name
        obj.scale = (size[1]/size[0], size[2]/size[0], 1) # Adjust for cube default 2x2x2
        # Actually, let's just scale the object to match dimensions directly
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif type == 'cylinder':
        # size is (radius, height) or (diameter, height)? 
        # Prompt says "size Ø0.045×0.12 m". Usually Ø is diameter.
        # Let's assume size tuple is (diameter, height) for cylinders.
        radius = size[0] / 2
        height = size[1]
        bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=location)
        obj = bpy.context.active_object
        obj.name = name
    elif type == 'cone':
        # size (radius_bottom, radius_top, height)
        r1, r2, h = size
        bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=r1, radius2=r2, depth=h, location=location)
        obj = bpy.context.active_object
        obj.name = name
    elif type == 'uvsphere':
        # size (radius)
        r = size[0]
        bpy.ops.mesh.primitive_uv_sphere_add(radius=r, location=location)
        obj = bpy.context.active_object
        obj.name = name
        
    # Apply rotation
    obj.rotation_euler = rotation
    return obj

def get_face_center(obj, face_type):
    """Get the world location of a specific face center."""
    # face_type: 'center', 'left_face', 'right_face', 'top_face', 'bottom_face', 'front_face', 'back_face'
    # Assuming standard orientation:
    # X: Left(-) / Right(+)
    # Y: Back(-) / Front(+)
    # Z: Bottom(-) / Top(+)
    
    # We need to transform the local face center to world space
    # For a cube/cylinder/cone aligned with axes:
    # Left face: min X, Right face: max X
    # Bottom face: min Z, Top face: max Z
    
    # Get bounding box in world space
    bbox_corners = [obj.matrix_world @ Vector(corner) for corner in obj.bound_box]
    
    # Find min/max for each axis
    min_x = min(c.x for c in bbox_corners)
    max_x = max(c.x for c in bbox_corners)
    min_y = min(c.y for c in bbox_corners)
    max_y = max(c.y for c in bbox_corners)
    min_z = min(c.z for c in bbox_corners)
    max_z = max(c.z for c in bbox_corners)
    
    center = obj.matrix_world.to_translation()
    
    if face_type == 'center':
        return center
    elif face_type == 'left_face':
        return Vector((min_x, center.y, center.z))
    elif face_type == 'right_face':
        return Vector((max_x, center.y, center.z))
    elif face_type == 'top_face':
        return Vector((center.x, center.y, max_z))
    elif face_type == 'bottom_face':
        return Vector((center.x, center.y, min_z))
    elif face_type == 'front_face':
        return Vector((center.x, max_y, center.z))
    elif face_type == 'back_face':
        return Vector((center.x, min_y, center.z))
        
    return center

def align_object_to_face(obj, target_obj, target_face_type, offset=(0,0,0)):
    """Align an object's face to another object's face."""
    target_loc = get_face_center(target_obj, target_face_type)
    obj_loc = get_face_center(obj, 'center')
    
    # Calculate translation to align centers
    # We want obj's specific face to be at target_loc
    # Current obj face center is obj_loc_face
    # We need to move obj so obj_loc_face == target_loc
    
    # Let's simplify: Move obj so its 'align_face' is at 'target_loc'
    # Then apply offset in local space? 
    # The prompt says: "Align ... to ... Then move by offset".
    # Usually align snaps the object. Offset moves it from that snapped position.
    
    # Let's get the vector from obj's align_face to target_loc
    # But wait, "Align obj.left_face to target.left_face" usually means they touch.
    # If we just move the center, the faces might not touch if sizes differ.
    # However, for boolean operations, exact touching is fine.
    
    # Let's assume "Align A.face to B.face" means A's face center coincides with B's face center.
    
    # Calculate the shift required to put obj's face at target_loc
    # obj's face center in world space is get_face_center(obj, face_type)
    # We want that point to be at target_loc.
    
    # Current obj face center
    current_face_center = get_face_center(obj, target_face_type.replace('left_face','left').replace('right_face','right').replace('top_face','top').replace('bottom_face','bottom'))
    
    # Actually, let's just move the object's center.
    # If obj.left_face should be at target_loc:
    # obj.center.x + (obj.width/2) = target_loc.x  => obj.center.x = target_loc.x - obj.width/2
    # This is getting complicated with rotations.
    
    # Simpler approach:
    # 1. Move obj so its 'align_face' is at target_loc.
    # 2. Apply offset.
    
    # Let's define a helper to get the vector from obj center to its face center
    def get_face_vector(obj, face_type):
        # Returns vector from obj center to face center in local space, then transformed to world?
        # No, let's work in world space.
        fc = get_face_center(obj, face_type)
        center = obj.matrix_world.to_translation()
        return fc - center

    face_vec = get_face_vector(obj, target_face_type)
    
    # We want: center + face_vec = target_loc
    # So: center = target_loc - face_vec
    new_center = target_loc - face_vec
    
    # Apply offset in local space? 
    # "Then move by offset(0, 0, 0)"
    # If offset is in local axes, we need to rotate the offset vector by obj's rotation.
    offset_vec = Vector(offset)
    offset_world = obj.matrix_world.to_3x3() @ offset_vec
    
    obj.location = new_center + offset_world

def boolean_union(target, cutter):
    """Perform boolean union using modifiers."""
    # Add modifier to target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Delete cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_subtract(target, cutter):
    """Perform boolean subtract using modifiers."""
    # Add modifier to target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Delete cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def bevel_edges(obj, radius, segments):
    """Add bevel modifier and apply."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE' # or 'WEIGHT', 'WIDTH'
    # Default angle bevel is usually fine, or width mode
    mod.width_type = 'WIDTH'
    
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

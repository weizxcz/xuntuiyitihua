def make_material(name, diffuse_color=(0.4, 0.4, 0.4, 1.0)):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to the first material slot of an object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(radius, depth, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder, defaulting to Z-axis up."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(size, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(radius, name, location=(0,0,0), rotation=(0,0,0)):

    obj = bpy.context.active_object

def add_plane(size, name, location=(0,0,0), rotation=(0,0,0)):

def boolean_subtract(cutter_obj, target_obj):
    """Perform Boolean Subtract using the modifier stack."""
    # Ensure target is active for modifier addition if needed, but we'll add to target
    bpy.context.view_layer.objects.active = target_obj
    
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply Modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Clean up cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def align_to_vector(obj, vector, origin=None):
    """Rotate object so its local Z axis aligns with the given world vector."""
    if origin is None:
        origin = obj.location
    
    # Current Z axis
    current_z = Vector((0, 0, 1))
    
    # Calculate rotation matrix
    rot_mat = current_z.rotation_difference(vector).to_matrix().to_4x4()
    
    # Apply rotation relative to origin
    # We need to rotate the object's data or transform. 
    # Easier: Set rotation_euler from quaternion.
    obj.rotation_euler = current_z.rotation_difference(vector).to_euler()

def get_face_center(obj, face_type='top'):
    """
    Helper to get the world location of a specific face center.
    Assumes standard primitives aligned to axes.
    face_type: 'top' (+Z), 'bottom' (-Z), 'left' (-X), 'right' (+X), 'center'
    """
    # Get object matrix
    mat = obj.matrix_world
    
    # Get dimensions
    dims = obj.dimensions
    
    if face_type == 'center':
        return mat @ Vector((0, 0, 0))
    
    elif face_type == 'top':
        # Top face is at +Z half height
        return mat @ Vector((0, 0, dims[2]/2))
    
    elif face_type == 'bottom':
        # Bottom face is at -Z half height
        return mat @ Vector((0, 0, -dims[2]/2))
    
    elif face_type == 'left':
        # Left face is at -X half width
        return mat @ Vector((-dims[0]/2, 0, 0))
    
    elif face_type == 'right':
        # Right face is at +X half width
        return mat @ Vector((dims[0]/2, 0, 0))
        
    return mat @ Vector((0,0,0))

def average_vectors(vectors):
    """Calculate the average of a list of Vector objects."""
    if not vectors:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for v in vectors:
        avg += v
    return avg / len(vectors)

def set_world_origin(obj):
    """Move object to world origin (0,0,0)."""
    obj.location = (0, 0, 0)

def offset_location(obj, offset_vec):
    """Add offset vector to object's current location."""
    obj.location += offset_vec

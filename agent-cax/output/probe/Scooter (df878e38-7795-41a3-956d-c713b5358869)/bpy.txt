# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location, rotation, material=None):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location, rotation, material=None):
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def align_face_to_face(obj_target, obj_source, target_axis='Z', source_axis='Z'):
    """Aligns the center of obj_source's face (defined by source_axis) to obj_target's face (defined by target_axis).
    Assumes both objects are oriented such that their local Z is up, or handles generic axis alignment."""
    # Get world matrix of target
    tm = obj_target.matrix_world
    sm = obj_source.matrix_world
    
    # Define target face normal in world space
    if target_axis == 'Z':
        target_normal = tm.col[2].to_3d()
        target_center = tm.to_translation()
    elif target_axis == 'X':
        target_normal = tm.col[0].to_3d()
        target_center = tm.to_translation()
    elif target_axis == 'Y':
        target_normal = tm.col[1].to_3d()
        target_center = tm.to_translation()
    else:
        target_normal = tm.col[2].to_3d()
        target_center = tm.to_translation()

    # Define source face normal in world space
    if source_axis == 'Z':
        source_normal = sm.col[2].to_3d()
    elif source_axis == 'X':
        source_normal = sm.col[0].to_3d()
    elif source_axis == 'Y':
        source_normal = sm.col[1].to_3d()
    else:
        source_normal = sm.col[2].to_3d()

    # Calculate rotation to align source normal to target normal
    rot_quat = source_normal.rotation_difference(target_normal)
    
    # Apply rotation to source object
    obj_source.rotation_euler = rot_quat.to_euler()
    obj_source.rotation_mode = 'QUATERNION' # Reset mode to avoid gimbal lock issues during calc
    
    # Recalculate matrix after rotation
    sm_new = obj_source.matrix_world
    
    # Calculate translation to align centers along the normal
    # We want the source face center to touch the target face center.
    # Source face center is offset from object center by half depth/size along source axis.
    # However, for simple primitives like cylinders/cubes, we often just want the geometric centers aligned 
    # if the "face" implies the flat end.
    
    # Let's assume "Align XY_face" means aligning the Z-axis position.
    # If target is deck_main (flat on XY), its top face is at Z + height/2.
    # If source is stem_tube (standing on Z), its bottom face is at Z - height/2.
    
    # Simplified approach for standard primitives:
    # 1. Align rotation.
    # 2. Calculate offset vector.
    
    # Get half-extents
    if obj_target.type == 'MESH':
        # Approximate half-size based on dimensions
        tb = obj_target.dimensions
        if target_axis == 'Z':
            target_face_z = tm.to_translation().z + tb[2]/2
        elif target_axis == 'X':
            target_face_x = tm.to_translation().x + tb[0]/2
        elif target_axis == 'Y':
            target_face_y = tm.to_translation().y + tb[1]/2
    else:
        target_face_z = tm.to_translation().z

    if obj_source.type == 'MESH':
        sb = obj_source.dimensions
        if source_axis == 'Z':
            source_face_z = sm_new.to_translation().z - sb[2]/2
        elif source_axis == 'X':
            source_face_x = sm_new.to_translation().x - sb[0]/2
        elif source_axis == 'Y':
            source_face_y = sm_new.to_translation().y - sb[1]/2
            
    # For this specific script, we will use a more robust locator-based alignment or direct vector math
    # But given the strict transform order, we will do:
    # 1. Rotate object to correct orientation.
    # 2. Move object so its face center matches target face center.
    
    # Re-evaluating based on "Align stem_tube.XY_face to deck_main.XY_face"
    # deck_main is a cube. stem_tube is a cylinder.
    # deck_main.XY_face likely refers to the top surface (normal +Z).
    # stem_tube.XY_face likely refers to its bottom circular face (normal -Z in local, but aligned to world +Z after rotation).
    
    # Let's use a generic align function that snaps the source object's "bottom" to the target object's "top"
    # assuming Z-up orientation.
    
    # Target Top Face Center
    t_loc = tm.to_translation()
    t_half = obj_target.dimensions[2] / 2 if target_axis == 'Z' else 0
    t_face_loc = Vector((t_loc.x, t_loc.y, t_loc.z + t_half))
    
    # Source Bottom Face Center (before translation, after rotation)
    # The source object's local origin is at its center.
    # Its bottom face is at local Z = -height/2.
    s_half = obj_source.dimensions[2] / 2 if source_axis == 'Z' else 0
    
    # We need to move the source object so that its bottom face center lands on t_face_loc.
    # Current source center is sm_new.to_translation().
    # New source center should be t_face_loc - (rotated offset vector).
    # Rotated offset vector for bottom face is (0,0,-s_half) rotated by rot_quat.
    
    offset_vec = Vector((0, 0, -s_half))
    offset_vec.rotate(rot_quat)
    
    new_loc = t_face_loc - offset_vec
    obj_source.location = new_loc

def boolean_union(objects, name):
    """Performs a boolean union on a list of objects and keeps the first one as the result."""
    if not objects:
        return None
    
    target = objects[0]
    target.name = name
    
    # Select all
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    for obj in objects[1:]:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
    
    bpy.context.view_layer.objects.active = target
    
    # Apply modifiers
    # We need to add a Boolean modifier to the target for each other object
    for obj in objects[1:]:
        mod = target.modifiers.new(name=f"Bool_{obj.name}", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        
    # Apply all boolean modifiers
    for mod in target.modifiers:
        if mod.type == 'BOOLEAN':
            bpy.ops.object.modifier_apply(modifier=mod.name)
            
    # Deselect others
    for obj in objects[1:]:
        obj.select_set(False)
        
    return target

def boolean_subtract(target, cutter):
    """Subtracts cutter from target using boolean modifier."""
    mod = target.modifiers.new(name=f"Bool_{cutter.name}", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    bpy.ops.object.modifier_apply(modifier=mod.name)
    return target

# Align stem_tube bottom face to deck_main top face
# deck_main top face Z = 0 + 0.005/2 = 0.0025
# stem_tube bottom face Z = stem_tube.location.z - 0.4/2
# We want stem_tube.bottom_face_Z == deck_main.top_face_Z
# stem_tube.location.z - 0.2 = 0.0025 => stem_tube.location.z = 0.2025
# However, the prompt says "Align ... Then move by offset".
# So we align first (which sets position relative to anchor), then apply offset.

# Target: deck_main top face center.
# Target Z = 0.0025. Target X,Y = 0.
# Source: stem_tube bottom face center.
# Source Z relative to center = -0.2.

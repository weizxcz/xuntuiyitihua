# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(vertices=segments, radius=radius, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

    obj.name = name

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Rotate object so its local axis aligns with target vector."""
    # Ensure object is in object mode
    bpy.ops.object.mode_set(mode='OBJECT')
    
    # Get current rotation as quaternion
    rot = obj.rotation_quaternion
    
    # Define local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    if axis_to_align not in local_axes:
        return

    src_axis = local_axes[axis_to_align]
    tgt_axis = target_vector.normalized()
    
    # Calculate rotation quaternion to align src to tgt
    rot_quat = src_axis.rotation_difference(tgt_axis)
    
    # Apply rotation
    obj.rotation_quaternion = rot_quat

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find specific points on an object's bounding box or geometry."""
    def __init__(self, obj):
        self.obj = obj
        self.update_matrix()

    def update_matrix(self):
        self.matrix_world = self.obj.matrix_world.copy()
        self.bound_box = self.obj.bound_box

    def center_world(self):
        """Return the world center of the object."""
        return self.matrix_world.to_translation()

    def face_center_world(self, face_index=0):
        """Return the center of a specific face in world space. 
        Assumes standard primitive orientation if face_index is not specific."""
        # For simple primitives, we can calculate based on dimensions
        # This is a simplified version assuming axis-aligned bounding box relative to object rotation
        bb = self.bound_box
        # Vertices of the face (indices depend on primitive type, this is generic approximation)
        # For cylinder/cone/sphere, face_index logic is complex. 
        # Let's use a simpler approach: project center along normal if possible, or use bounding box extremes.
        
        # Fallback: Use bounding box center + offset based on size
        center = self.center_world()
        size = self.obj.dimensions
        
        # Determine direction based on face_index logic if needed, 
        # but for this script we will use specific geometric helpers below.
        return center

    def radial_point_world(self, angle_deg, axis='Z'):
        """Return a point on the radial edge at a specific angle."""
        center = self.center_world()
        size = self.obj.dimensions
        
        # Radius is half of the dimension perpendicular to the axis
        if axis == 'Z':
            radius = size[0] / 2.0 # X or Y dimension
        elif axis == 'Y':
            radius = size[0] / 2.0
        else:
            radius = size[0] / 2.0
            
        angle_rad = math.radians(angle_deg)
        
        # Local point
        if axis == 'Z':
            local_pt = Vector((math.cos(angle_rad) * radius, math.sin(angle_rad) * radius, 0))
        elif axis == 'Y':
            local_pt = Vector((math.cos(angle_rad) * radius, 0, math.sin(angle_rad) * radius))
        else:
            local_pt = Vector((0, math.cos(angle_rad) * radius, math.sin(angle_rad) * radius))
            
        return self.matrix_world @ local_pt

    def side_at(self, angle_deg):
        """Alias for radial_point_world."""
        return self.radial_point_world(angle_deg)

    def top_world(self):
        """Return the world position of the top of the object."""
        center = self.center_world()
        size = self.obj.dimensions
        # Top is center + half height along Z (assuming default orientation)
        # We need to account for rotation.
        # Get the up vector in world space
        up_vec = self.matrix_world @ Vector((0, 0, 1))
        return center + up_vec * (size[2] / 2.0)

    def bottom_world(self):
        """Return the world position of the bottom of the object."""
        center = self.center_world()
        size = self.obj.dimensions
        down_vec = self.matrix_world @ Vector((0, 0, -1))
        return center + down_vec * (size[2] / 2.0)

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract operation using the modifier stack."""
    target = bpy.data.objects.get(target_name)
    cutter = bpy.data.objects.get(cutter_name)
    
    if not target or not cutter:
        print(f"Error: Object {target_name} or {cutter_name} not found.")
        return

    # Ensure target is active for modifier addition
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="Boolean_Subtract", type='BOOLEAN')
    mod.object = cutter
    mod.operation = 'DIFFERENCE'
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target_name, other_name):
    """Perform a boolean union operation."""
    target = bpy.data.objects.get(target_name)
    other = bpy.data.objects.get(other_name)
    
    if not target or not other:
        print(f"Error: Object {target_name} or {other_name} not found.")
        return

    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="Boolean_Union", type='BOOLEAN')
    mod.object = other
    mod.operation = 'UNION'
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(other, do_unlink=True)

def add_bevel(obj_name, width=0.01, segments=5):
    """Add a bevel modifier and apply it."""
    obj = bpy.data.objects.get(obj_name)
    if not obj:
        print(f"Error: Object {obj_name} not found.")
        return

    bpy.context.view_layer.objects.active = obj
    
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    
    bpy.ops.object.modifier_apply(modifier=mod.name)

def add_text_object(name, text_string, size=1.0):
    """Add a text object."""
    bpy.ops.object.text_add(location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.data.body = text_string
    obj.scale = (size, size, size)
    return obj

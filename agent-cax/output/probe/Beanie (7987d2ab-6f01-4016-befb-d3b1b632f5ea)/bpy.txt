import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, segments=32):
    """Add a disc (flat cylinder) mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=0.001, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_segments=32, height_segments=16):
    """Add an UV Sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius, depth):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=depth, segments=32, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align an object's local axis to a world vector."""
    # Get current rotation
    rot = obj.rotation_euler.to_quaternion()
    # Get target direction
    target_dir = vector.normalized()
    
    # Define axis vectors
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    elif axis == 'Z':
        local_axis = Vector((0, 0, 1))
    else:
        return

    # Calculate rotation to align local_axis to target_dir
    # We need to preserve the object's roll around the target axis if possible, 
    # but for simple alignment, we can use a quaternion from two vectors.
    # To keep it simple and robust for this script:
    # We will set the rotation such that the local axis points to the target.
    
    # Create a temporary object to calculate rotation
    temp = bpy.context.view_layer.objects.active
    bpy.ops.object.empty_add(location=(0,0,0))
    temp_align = bpy.context.active_object
    temp_align.name = "_temp_align"
    
    # Align temp_align Z to target_dir
    temp_align.rotation_mode = 'QUATERNION'
    temp_align.rotation_quaternion = target_dir.to_track_quat('Z', 'Y') # Default up is Y, but we want Z aligned
    
    # Actually, simpler approach:
    # Calculate quaternion between local_axis and target_dir
    q = local_axis.rotation_difference(target_dir)
    
    # Apply to object
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = q

def boolean_subtract(cutter_obj, target_obj):
    """Perform a boolean subtract operation using the modifier stack."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target_obj
    
    # Add Boolean Modifier
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(target_obj, other_obj):
    """Perform a boolean union operation."""
    bpy.context.view_layer.objects.active = target_obj
    
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(other_obj, do_unlink=True)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj, face_index=None):
        self.obj = obj
        self.face_index = face_index
        self._update_matrix()

    def _update_matrix(self):
        # Update matrix to ensure transforms are current
        self.obj.matrix_world.update()
        self.matrix = self.obj.matrix_world

    def center_world(self):
        """Return the world center of the object."""
        return self.matrix.to_translation()

    def face_center_world(self):
        """Return the world center of a specific face or the mesh center if no face index."""
        if self.face_index is not None and self.obj.data.polygons:
            # Get face center in local space
            poly = self.obj.data.polygons[self.face_index]
            center_local = poly.center
            # Transform to world space
            return self.matrix @ center_local
        else:
            return self.center_world()

    def radial_point_world(self, angle=0, radius=None):
        """Return a point on the radial edge."""
        # Default to radius of bounding box or specific radius if provided
        if radius is None:
            # Approximate radius from bounding box
            bb_min = self.matrix @ self.obj.bound_box[0]
            bb_max = self.matrix @ self.obj.bound_box[6]
            center = self.center_world()
            radius = (bb_max - center).length
        
        # Simple radial point along X axis rotated by angle
        local_point = Vector((radius, 0, 0))
        # Rotate around Z
        rot_z = Matrix.Rotation(angle, 4, 'Z')
        local_point = rot_z @ local_point
        return self.matrix @ local_point

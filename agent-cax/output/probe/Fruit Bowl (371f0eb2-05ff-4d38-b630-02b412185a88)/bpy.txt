def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object

        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with the given world vector."""
    # Get current rotation matrix
    rot_mat = obj.matrix_world.to_quaternion()
    
    # Target direction
    target_dir = vector.normalized()
    
    # Current axis direction in world space
    if axis == 'X':
        current_dir = Vector((1, 0, 0))
    elif axis == 'Y':
        current_dir = Vector((0, 1, 0))
    else: # Z
        current_dir = Vector((0, 0, 1))
        
    current_dir = rot_mat @ current_dir
    
    # Calculate rotation quaternion to align current_dir to target_dir
    if current_dir.dot(target_dir) < -0.9999:
        # 180 degree rotation
        rot_quat = Mathutils.Quaternion((0,1,0), math.pi) # Fallback, usually needs specific axis
        # Better approach: use rotation_difference
        rot_quat = current_dir.rotation_difference(target_dir)
    else:
        rot_quat = current_dir.rotation_difference(target_dir)
        
    obj.rotation_euler = rot_quat.to_euler()

def align_face_to_face(obj_target, obj_ref, target_axis='Z', ref_axis='Z'):
    """Align obj_target's face axis to obj_ref's face axis."""
    # This is a simplified alignment assuming standard orientations.
    # For robust face-to-face alignment, we often just need to match the axis direction.
    # If we want to align the *plane* (normal), we align the axis.
    # If we want to align the *edge* or specific face, we might need more complex logic.
    # Here we assume aligning the normal vectors.
    
    # Get world rotation of reference
    ref_rot = obj_ref.matrix_world.to_quaternion()
    
    # Define local axis vectors
    if target_axis == 'Z': t_vec = Vector((0,0,1))
    elif target_axis == 'Y': t_vec = Vector((0,1,0))
    else: t_vec = Vector((1,0,0))
    
    if ref_axis == 'Z': r_vec = Vector((0,0,1))
    elif ref_axis == 'Y': r_vec = Vector((0,1,0))
    else: r_vec = Vector((1,0,0))
    
    # Rotate reference local vector to world space
    r_vec_world = ref_rot @ r_vec
    
    # Calculate rotation to align target local vector to reference world vector
    # We want obj_target's t_vec to point in direction of r_vec_world
    # But usually "align face" implies matching the normal.
    
    # Simple approach: Set rotation to match reference's rotation if axes are same
    # Or calculate difference.
    
    # Let's use a simpler method: Align the object's up vector to the reference's up vector
    # This assumes we are just orienting them similarly.
    
    # More precise: Align the specific axis of target to the specific axis of ref
    # We need to rotate target such that its axis matches ref's axis in world space.
    
    # Get current world axis of target
    t_vec_world = obj_target.matrix_world.to_quaternion() @ t_vec
    
    # Rotation from t_vec_world to r_vec_world
    rot_diff = t_vec_world.rotation_difference(r_vec_world)
    
    # Apply rotation
    # Note: Applying rotation directly to euler can be tricky. 
    # We'll update the matrix world directly or use quaternion.
    obj_target.rotation_euler = rot_diff.to_euler()

def align_face_normal(obj_target, obj_ref, target_axis='Z', ref_axis='Z'):
    """Align the normal of obj_target's face to the normal of obj_ref's face."""
    # This is effectively aligning the axis vectors in world space
    align_axis_to_vector(obj_target, axis=target_axis, vector=(obj_ref.matrix_world.to_quaternion() @ (Vector((0,0,1)) if ref_axis=='Z' else Vector((0,1,0)) if ref_axis=='Y' else Vector((1,0,0)))))

def boolean_subtract(cutter_obj, target_obj):
    """Perform boolean subtract using modifier stack."""
    # Add boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(union_obj, target_obj):
    """Perform boolean union using modifier stack."""
    # Add boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove union_obj
    bpy.data.objects.remove(union_obj, do_unlink=True)

def add_bevel(obj, radius, segments, edge_set=None):
    """Add bevel modifier to object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE' # or 'WEIGHT', 'WIDTH'
    # If edge_set is specified, we might need to select edges and use 'EDGE' limit method
    # For simplicity, we apply global bevel. If specific edges are needed, 
    # we'd need to enter edit mode, select edges, and set limit method to EDGE.
    # Given the prompt "Bevel banana.edge_set", let's try to be more specific if possible.
    # However, without a clear definition of "edge_set" (e.g. top/bottom rings), 
    # a global bevel is the safest interpretation for a cylinder unless specified.
    # Let's assume global bevel for now as "edge_set" is vague.
    
    # To bevel specific edges (like the top rim of a cylinder):
    # We would need to select them.
    if edge_set:
        mod.limit_method = 'EDGE'
        # Note: Implementing specific edge selection in script requires entering edit mode
        # and selecting edges by index or normal. This is complex.
        # We will stick to global bevel for robustness unless indices are provided.
        # If the user meant "bevel the edges of the cylinder", global bevel does that.
        
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def get_world_center(obj):
    """Get the world space center of an object."""
    return obj.matrix_world.to_translation()

def get_world_face_center(obj, axis='Z'):
    """Get the world space center of a specific face (top/bottom)."""
    # For a cylinder/sphere aligned to Z, the "top" face center is at Z + radius/height
    # This is an approximation based on object bounds.
    # A more robust way is to use the object's bounding box.
    bbox = obj.bound_box
    # bbox has 8 points. 
    # We need to identify which points correspond to the face.
    # Assuming standard orientation:
    # Z axis is up.
    # Top face Z is max, Bottom face Z is min.
    
    # Find max and min Z
    z_vals = [p[2] for p in bbox]
    z_max = max(z_vals)
    z_min = min(z_vals)
    
    # Find points at z_max
    top_points = [p for p in bbox if abs(p[2] - z_max) < 1e-5]
    bottom_points = [p for p in bbox if abs(p[2] - z_min) < 1e-5]
    
    # Average X,Y for top and bottom
    if top_points:
        avg_x = sum(p[0] for p in top_points) / len(top_points)
        avg_y = sum(p[1] for p in top_points) / len(top_points)
        top_center_local = Vector((avg_x, avg_y, z_max))
    else:
        top_center_local = Vector((0,0,z_max))
        
    if bottom_points:
        avg_x = sum(p[0] for p in bottom_points) / len(bottom_points)
        avg_y = sum(p[1] for p in bottom_points) / len(bottom_points)
        bottom_center_local = Vector((avg_x, avg_y, z_min))
    else:
        bottom_center_local = Vector((0,0,z_min))
        
    # Transform to world space
    top_center_world = obj.matrix_world @ top_center_local
    bottom_center_world = obj.matrix_world @ bottom_center_local
    
    if axis == 'Z':
        # Return top and bottom
        return top_center_world, bottom_center_world
    elif axis == 'X':
        # Left and Right
        x_vals = [p[0] for p in bbox]
        x_max = max(x_vals)
        x_min = min(x_vals)
        right_points = [p for p in bbox if abs(p[0] - x_max) < 1e-5]
        left_points = [p for p in bbox if abs(p[0] - x_min) < 1e-5]
        if right_points:
            avg_y = sum(p[1] for p in right_points)/len(right_points)
            avg_z = sum(p[2] for p in right_points)/len(right_points)
            right_center_local = Vector((x_max, avg_y, avg_z))
        else:
            right_center_local = Vector((x_max,0,0))
        if left_points:
            avg_y = sum(p[1] for p in left_points)/len(left_points)
            avg_z = sum(p[2] for p in left_points)/len(left_points)
            left_center_local = Vector((x_min, avg_y, avg_z))
        else:
            left_center_local = Vector((x_min,0,0))
        return obj.matrix_world @ right_center_local, obj.matrix_world @ left_center_local
        
    return top_center_world, bottom_center_world

import bpy

def make_material(name, diffuse_color=(1.0, 1.0, 1.0, 1.0)):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius=0.5, depth=1, location=(0,0,0), rotation=(0,0,0), material=None):

def add_disc(name, radius=0.5, location=(0,0,0), rotation=(0,0,0), material=None):

def add_bevel(obj, width=0.01, segments=5):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(target, cutter):
    """Perform boolean subtract using a cutter object."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target, union_obj):
    """Perform boolean union."""
    bpy.context.view_layer.objects.active = target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(union_obj, do_unlink=True)

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a world vector."""
    if axis == 'Z':
        obj.rotation_euler = (0, 0, 0) # Default Z is up
    elif axis == 'Y':
        obj.rotation_euler = (-pi/2, 0, 0)
    elif axis == 'X':
        obj.rotation_euler = (0, pi/2, 0)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find centers and faces of objects."""
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        # bound_box is in local space, need to transform to world
        self.world_points = [obj.matrix_world @ Vector(p) for p in self.bound_box]

    def center_world(self):
        return average(self.world_points)

    def face_center_world(self, face_index):
        """Get center of a specific face index (0-5 for cube)."""
        # This is a simplified approach assuming standard primitives.
        # For robustness, we calculate based on min/max of world points.
        # However, for cubes, faces are predictable.
        # Let's use a simpler heuristic: find min/max on specific axis.
        return self.center_world() # Fallback, specific face logic depends on orientation.

    def radial_point_world(self, axis='Z', direction=1):
        """Get a point on the radial edge."""
        return self.center_world() # Placeholder

    def get_face_normal_and_center(self, axis='Z', direction=1):
        """
        Returns (center, normal) for a specific face defined by axis and direction.
        Assumes object is axis-aligned or we are working in local space transformed to world.
        """
        # Find min/max coordinate on the axis
        coords = [p[axis] for p in self.world_points]
        min_val = min(coords)
        max_val = max(coords)
        
        if direction == 1: # Positive direction (e.g., +Z is top)
            target_val = max_val
            normal = Vector((0,0,1)) if axis == 'Z' else (Vector((1,0,0)) if axis == 'X' else Vector((0,1,0)))
        else: # Negative direction
            target_val = min_val
            normal = Vector((0,0,-1)) if axis == 'Z' else (Vector((-1,0,0)) if axis == 'X' else Vector((0,-1,0)))
            
        # Filter points that are on this face (within tolerance)
        face_points = [p for p in self.world_points if abs(p[axis] - target_val) < 0.0001]
        center = average(face_points)
        return center, normal

import bpy
import bmesh
from mathutils import Vector, Matrix, Euler

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    # Scale to size (x, y, z)
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        if len(obj.data.materials) == 0:
            obj.data.materials.append(material)
        else:
            obj.data.materials[0] = material
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Align a specific axis of an object to a target vector."""
    # Get current rotation
    rot = obj.rotation_euler.copy()
    
    # We need to find the rotation that maps 'axis_to_align' to 'target_vector'
    # axis_to_align is a local axis (e.g., 'Z' -> (0,0,1))
    # target_vector is in world space
    
    # Create a matrix that rotates the local axis to the target
    # This is a simplified approach for standard axes
    
    local_axis_map = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    local_axis = local_axis_map.get(axis_to_align, Vector((0,0,1)))
    
    # Calculate rotation matrix
    # We want to rotate local_axis to target_vector
    # Use quaternion rotation
    q = local_axis.rotation_difference(target_vector.normalized())
    
    # Apply this rotation to the object's current orientation
    # Note: Blender's rotation_euler is Z-Y-X. We need to be careful with order.
    # A robust way is to set the rotation directly using the quaternion
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = q
    
    # Convert back to Euler if needed for consistency, though Quat is fine
    # obj.rotation_mode = 'XYZ' # or whatever default
    
    return obj

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract using the modifier stack."""
    target = bpy.data.objects.get(target_name)
    cutter = bpy.data.objects.get(cutter_name)
    
    if not target or not cutter:
        print(f"Error: Object {target_name} or {cutter_name} not found for boolean.")
        return

    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter
    mod.use_self = False
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_union(target_name, union_name):
    """Perform a boolean union."""
    target = bpy.data.objects.get(target_name)
    union = bpy.data.objects.get(union_name)
    
    if not target or not union:
        print(f"Error: Object {target_name} or {union_name} not found for boolean.")
        return

    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)

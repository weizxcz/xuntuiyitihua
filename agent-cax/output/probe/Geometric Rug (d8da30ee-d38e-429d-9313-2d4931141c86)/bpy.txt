# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    # size is (x, y, z)
    half_size = [s / 2.0 for s in size]
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0)) # size=1 creates a cube of side 1
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    return obj

def add_cone(name, radius_bottom, radius_top, depth):
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_segments=32, height_segments=16):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width, segments=5):
    # Add bevel modifier
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately to keep geometry simple for boolean ops if needed, 
    # but usually better to leave as modifier or apply if final. 
    # For this script, we'll apply it to ensure clean topology for subsequent steps if any.
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    return obj

def align_axis_to_vector(obj, axis_to_align='+Z', target_vector=(0, 0, 1)):
    """Aligns an object's local axis to a target world vector."""
    # Default orientation of primitives in Blender is +Z up.
    # If target is (0,0,1), no rotation needed.
    if axis_to_align == '+Z':
        if target_vector == (0, 0, 1):
            return
        # General case: align local Z to target vector
        vec = obj.matrix_world.to_3x3() @ Vector((0, 0, 1))
        # We need to rotate the object so its local Z matches target_vector
        # Calculate rotation matrix from current Z to target Z
        current_z = Vector((0, 0, 1))
        target_z = Vector(target_vector).normalized()
        if current_z.angle(target_z) > 1e-6:
            rot_quat = current_z.rotation_difference(target_z)
            obj.rotation_euler = rot_quat.to_euler()

def average(points):
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj, name=None):
        self.obj = obj
        if name:
            self.obj.name = name
        self.update_matrix()

    def update_matrix(self):
        self.matrix = self.obj.matrix_world.copy()

    def center_world(self):
        return self.matrix.to_translation()

    def face_center_world(self):
        # For a flat object like a tile, the face center is usually the center if aligned to Z
        # Assuming standard orientation where face is XY plane at Z=0 relative to object
        return self.center_world()

    def radial_point_world(self, angle_deg, radius):
        # Returns a point on the radial plane (XY) at given angle and radius
        angle_rad = math.radians(angle_deg)
        x = radius * math.cos(angle_rad)
        y = radius * math.sin(angle_rad)
        # Transform local radial point to world
        local_pt = Vector((x, y, 0))
        return self.matrix @ local_pt

def boolean_subtract(cutter_name, target_name):
    cutter = bpy.data.objects.get(cutter_name)
    target = bpy.data.objects.get(target_name)
    if not cutter or not target:
        return
    
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(obj1_name, obj2_name):
    # Blender boolean union is tricky with modifiers on multiple objects.
    # Standard approach: Join them, then use a boolean union if needed, 
    # or simply join geometry if they are just being combined.
    # However, "boolean_union" usually implies merging volumes.
    # Simplest robust way for script: Join objects.
    
    obj1 = bpy.data.objects.get(obj1_name)
    obj2 = bpy.data.objects.get(obj2_name)
    if not obj1 or not obj2:
        return

    # Select both
    bpy.ops.object.select_all(action='DESELECT')
    obj1.select_set(True)
    obj2.select_set(True)
    bpy.context.view_layer.objects.active = obj1
    
    bpy.ops.object.join()
    
    # The resulting object takes the name of the active one (obj1)
    # Rename to a standard if needed, but usually we just keep obj1's name or update it.
    # Let's return the joined object.
    return obj1

def make_material(name, diffuse_color, roughness=0.5, metallic=0.0, alpha=1.0):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (diffuse_color[0], diffuse_color[1], diffuse_color[2], alpha)
    bsdf.inputs['Roughness'].default_value = roughness
    bsdf.inputs['Metallic'].default_value = metallic
    if alpha < 1.0:
        mat.blend_method = 'ALPHA'
        mat.shadow_mode = 'NONE'
    return mat

def add_cylinder(radius, depth, segments=32, name="Cylinder"):
    """Add a cylinder oriented along Z."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, end_fill_type='NGON')
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def add_uvsphere(radius, name="Sphere"):
    """Add a UV sphere oriented along Z."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=32, ring_count=16)
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def add_cube(size, name="Cube"):
    """Add a cube centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=size)
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def boolean_subtract(target_obj, cutter_obj, keep_cutter=False):
    """Apply Boolean Subtract modifier."""
    # Ensure target is active for modifier application if needed, but we'll apply via modifier stack
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    if not keep_cutter:
        bpy.data.objects.remove(cutter_obj, do_unlink=True)
    else:
        cutter_obj.select_set(True)
        bpy.context.view_layer.objects.active = cutter_obj
        bpy.ops.object.delete()

def align_axis_to_vector(obj, axis='Z', vector=(0, 0, 1)):
    """Rotate object so its local axis aligns with the given world vector."""
    # Get current rotation
    rot_euler = obj.rotation_euler
    # This is a simplified alignment. For robust alignment, we use matrix math.
    # We want obj's local axis to point to 'vector'.
    
    # Define local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    target_vec = Vector(vector).normalized()
    source_vec = local_axes[axis]
    
    # Calculate rotation matrix to align source_vec to target_vec
    rot_mat = source_vec.rotation_difference(target_vec).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_euler = rot_mat.to_euler()
    obj.update_tag()

def get_face_normal(obj, face_index):
    """Get the world normal of a specific face."""
    # Ensure object is in world space
    mat_world = obj.matrix_world
    # Get face normal in local space
    mesh = obj.data
    if face_index < len(mesh.polygons):
        local_normal = mesh.polygons[face_index].normal
        # Transform to world space
        world_normal = (mat_world @ local_normal).normalized()
        return world_normal
    return Vector((0, 0, 1))

def get_face_center(obj, face_index):
    """Get the world center of a specific face."""
    mat_world = obj.matrix_world
    mesh = obj.data
    if face_index < len(mesh.polygons):
        poly = mesh.polygons[face_index]
        # Calculate center from vertices
        center_local = Vector((0, 0, 0))
        for vi in poly.vertices:
            center_local += mesh.vertices[vi].co
        center_local /= len(poly.vertices)
        return mat_world @ center_local
    return Vector((0, 0, 0))

def align_face_to_face(obj_source, face_source_idx, obj_target, face_target_idx, offset=0.0):
    """Align source face to target face, optionally with an offset along the normal."""
    # 1. Align Normals
    normal_src = get_face_normal(obj_source, face_source_idx)
    normal_tgt = get_face_normal(obj_target, face_target_idx)
    
    # We want source normal to match target normal (or opposite depending on "bottom face" logic)
    # Assuming "bottom face" means the face pointing in -Z or similar, but here we align geometry.
    # Let's align the normals directly first.
    rot_diff = normal_src.rotation_difference(normal_tgt)
    obj_source.rotation_euler = rot_diff.to_euler()
    
    # 2. Align Centers
    center_src = get_face_center(obj_source, face_source_idx)
    center_tgt = get_face_center(obj_target, face_target_idx)
    
    # Calculate translation to align centers
    # Note: Rotating the object changes its center position relative to world if not pivoted correctly.
    # To simplify, we usually pivot at origin or use a temporary pivot.
    # However, for simple primitives centered at origin, rotation around origin keeps center at origin.
    # If centers are not at origin, we need to be careful.
    # Let's assume objects are centered at their geometric center initially.
    
    # Move source so its face center matches target face center
    translation = center_tgt - center_src
    obj_source.location += translation
    
    # 3. Apply Offset along the new normal
    # The normal is now aligned with target normal
    obj_source.location += normal_tgt * offset

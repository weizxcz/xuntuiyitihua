def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def get_face_center(obj, face_type):
    """
    Calculates the world location of a specific face on an object.
    Assumes standard cube orientation unless rotated.
    """
    # Transform vertices to world space
    verts_world = [obj.matrix_world @ v.co for v in obj.data.vertices]
    
    if face_type == 'center':
        # Average of all vertices
        avg = sum(verts_world, Vector()) / len(verts_world)
        return avg
    elif face_type == 'bottom_face.center':
        # Find min Z, then average X,Y of those vertices
        min_z = min(v[2] for v in verts_world)
        bottom_verts = [v for v in verts_world if abs(v[2] - min_z) < 1e-5]
        if bottom_verts:
            avg = sum(bottom_verts, Vector()) / len(bottom_verts)
            return avg
    elif face_type == 'top_face.center':
        max_z = max(v[2] for v in verts_world)
        top_verts = [v for v in verts_world if abs(v[2] - max_z) < 1e-5]
        if top_verts:
            avg = sum(top_verts, Vector()) / len(top_verts)
            return avg
            
    return obj.matrix_world.to_translation()

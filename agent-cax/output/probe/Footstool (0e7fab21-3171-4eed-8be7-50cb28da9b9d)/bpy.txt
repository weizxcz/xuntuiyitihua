def make_material(name, diffuse_color=(0.8, 0.8, 0.8, 1.0)):
    """Create a new Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_text_object(text, name, location=(0,0,0), size=0.1):
    """Add a text object."""
    bpy.ops.object.text_add(location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.size = size
    obj.data.body = text
    return obj

def add_cylinder(name, radius=0.5, depth=1.0, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius=0.5, location=(0,0,0), rotation=(0,0,0)):

    obj = bpy.context.active_object

def add_bevel(obj, width=0.01, segments=4):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a world vector."""
    vec = mathutils.Vector(vector).normalized()
    if axis == 'Z':
        # Align local Z to world vector
        current_z = mathutils.Vector((0,0,1))
        rot_quat = current_z.rotation_difference(vec)
        obj.rotation_euler = rot_quat.to_euler()
    elif axis == 'X':
        current_x = mathutils.Vector((1,0,0))
        rot_quat = current_x.rotation_difference(vec)
        obj.rotation_euler = rot_quat.to_euler()
    elif axis == 'Y':
        current_y = mathutils.Vector((0,1,0))
        rot_quat = current_y.rotation_difference(vec)
        obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract operation."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target is active for modifier addition
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Clean up cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_name, target_name):
    """Perform a boolean union operation."""
    target = bpy.data.objects[target_name]
    union = bpy.data.objects[union_name]
    
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union, do_unlink=True)

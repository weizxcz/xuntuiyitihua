# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def average(points):
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        return self.obj.matrix_world.to_translation()

    def face_center_world(self, normal_vector):
        # Approximate face center by projecting center along normal in local space then to world
        # For simple primitives, this is often just center + normal * radius/height
        # However, a robust way for "face" alignment is to find the vertex/edge furthest in that direction
        # For this script, we assume standard primitives where face is at extent.
        # Let's use a simpler approach: transform the normal to world space and add to center? 
        # No, "face" implies a specific surface. 
        # Let's implement a helper to find the point on the mesh boundary in a given local direction.
        return self._get_point_on_surface(normal_vector)

    def _get_point_on_surface(self, local_dir):
        # Transform local direction to world
        world_dir = self.obj.matrix_world.to_3x3() @ local_dir
        center = self.center_world()
        
        # Raycast from center along world_dir to find surface point
        # This is computationally expensive but robust. 
        # For primitives, we can use bounds.
        # Let's use bounds for speed and reliability on primitives.
        local_bbox = [self.obj.matrix_world @ Vector(c) for c in self.obj.bound_box]
        # Find point in local_bbox that is furthest in local_dir
        best_dot = -float('inf')
        best_point = center
        for pt in local_bbox:
            # Convert pt to local to compare with local_dir? 
            # Actually, bound_box vertices are in local space relative to object origin? 
            # No, bound_box is in local space of the object data, but transformed by matrix_world gives world coords.
            # We need to find which vertex is furthest in the direction of local_dir.
            # local_dir is in local space.
            # Let's iterate bound_box vertices (local coords)
            pass
        
        # Simpler: Iterate bound_box vertices (local)
        max_dot = -float('inf')
        best_local_pt = Vector((0,0,0))
        for v in self.obj.bound_box:
            dot = v.dot(local_dir)
            if dot > max_dot:
                max_dot = dot
                best_local_pt = v
        
        # Convert best local point to world
        return self.obj.matrix_world @ best_local_pt

    def radial_point_world(self, angle_deg, axis='Z'):
        # Get a point on the "side" (radial) at a given angle
        # Axis defines the main axis of the cylinder/sphere
        if axis == 'Z':
            local_dir = Vector((math.cos(math.radians(angle_deg)), math.sin(math.radians(angle_deg)), 0))
        elif axis == 'X':
            local_dir = Vector((0, math.cos(math.radians(angle_deg)), math.sin(math.radians(angle_deg))))
        elif axis == 'Y':
            local_dir = Vector((math.cos(math.radians(angle_deg)), 0, math.sin(math.radians(angle_deg))))
        else:
            local_dir = Vector((1, 0, 0))
            
        return self._get_point_on_surface(local_dir)

def align_axis_to_vector(obj, target_vector, axis='Z'):
    """Rotate obj so its specified axis aligns with target_vector."""
    if axis == 'Z':
        current_axis = Vector((0, 0, 1))
    elif axis == 'Y':
        current_axis = Vector((0, 1, 0))
    elif axis == 'X':
        current_axis = Vector((1, 0, 0))
    else:
        return

    # Normalize
    current_axis = current_axis.normalized()
    target_vector = target_vector.normalized()
    
    if current_axis.dot(target_vector) > 0.9999:
        return # Already aligned
    
    rotation_quat = current_axis.rotation_difference(target_vector)
    obj.rotation_euler = rotation_quat.to_euler()

def add_cylinder(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, align='WORLD')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, segments=32):
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='TRIFAN', segments=segments, align='WORLD')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    bpy.ops.mesh.primitive_cube_add(size=size, align='WORLD')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_bottom, radius_top, depth, segments=32):
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, segments=segments, align='WORLD')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_seg=32, height_seg=16):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_seg, ring_count=height_seg, align='WORLD')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def boolean_subtract(cutter_obj, target_obj):
    # Add Boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(target_obj, other_obj):
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other_obj
    
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(other_obj, do_unlink=True)

def add_bevel(obj, width, segments=5):
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

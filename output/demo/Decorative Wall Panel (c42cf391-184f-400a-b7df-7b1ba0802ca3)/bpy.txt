def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius=0.5, depth=1, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cone(name, radius_top=0, radius_bottom=0.5, depth=1, location=(0,0,0), rotation=(0,0,0), material=None):

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with the given world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else: # Z
        local_axis = Vector((0,0,1))
        
    target = vector.normalized()
    if local_axis.normalized().dot(target) < 0.9999:
        rot_quat = local_axis.rotation_difference(target)
        obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average position of a list of Vector objects."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        """Returns the world space center of the object's bounding box."""
        # Using the object's location as center for primitives aligned to origin
        return self.obj.location

    def face_center_world(self, face_type='top'):
        """Returns the world space center of a specific face."""
        loc = self.obj.location
        # Assuming standard orientation where Z is up
        if face_type == 'top':
            return Vector((loc.x, loc.y, loc.z + self.obj.dimensions.z / 2))
        elif face_type == 'bottom':
            return Vector((loc.x, loc.y, loc.z - self.obj.dimensions.z / 2))
        elif face_type == 'front':
            return Vector((loc.x, loc.y + self.obj.dimensions.y / 2, loc.z))
        elif face_type == 'back':
            return Vector((loc.x, loc.y - self.obj.dimensions.y / 2, loc.z))
        elif face_type == 'right':
            return Vector((loc.x + self.obj.dimensions.x / 2, loc.y, loc.z))
        elif face_type == 'left':
            return Vector((loc.x - self.obj.dimensions.x / 2, loc.y, loc.z))
        return loc

    def radial_point_world(self, angle_deg, axis='Z'):
        """Returns a point on the radial edge."""
        # Placeholder for radial calculations if needed
        return self.obj.location

def boolean_subtract(cutter_name, target_name):
    """Apply Boolean Subtract modifier."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is active for modifier creation context if needed, 
    # but we'll add modifier directly to target.
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply immediately to keep geometry simple for this script
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_union(target_name, union_name):
    """Apply Boolean Union modifier."""
    target = bpy.data.objects[target_name]
    union_obj = bpy.data.objects[union_name]
    
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)

# Helper to create and position rails
def create_frame_rail(name, size, align_face, offset, material):
    obj = add_cube(name, size=size, location=(0,0,0), rotation=(0,0,0), material=material)
    align_axis_to_vector(obj, 'Z', (0,0,1))
    
    # Align face to base_plate.top_face
    # base_plate.top_face center is at (0, 0, 0.01)
    base_top_center = Vector((0, 0, 0.01))
    
    # Calculate current face center based on alignment face type
    # We need to move the object so its specific face center matches base_top_center
    # Then apply offset
    
    # Current object center is (0,0,0)
    # Dimensions
    dx, dy, dz = size
    
    # Determine offset from object center to the face being aligned
    if align_face == 'bottom_face':
        face_offset_from_center = Vector((0, 0, -dz/2))
    elif align_face == 'top_face':
        face_offset_from_center = Vector((0, 0, dz/2))
    elif align_face == 'right_face':
        face_offset_from_center = Vector((dx/2, 0, 0))
    elif align_face == 'left_face':
        face_offset_from_center = Vector((-dx/2, 0, 0))
    else:
        face_offset_from_center = Vector((0,0,0))
        
    # Target position for object center:
    # Target_Face_Center = Object_Center + Face_Offset
    # Object_Center = Target_Face_Center - Face_Offset
    target_center = base_top_center - face_offset_from_center
    
    # Apply offset
    final_location = target_center + Vector(offset)
    
    obj.location = final_location
    return obj

frame_rail_top = create_frame_rail("frame_rail_top", (1.2, 0.05, 0.03), 'bottom_face', (0,0,0.015), mat_panel)

# Calculate average center of the four rails

for r in range(4):

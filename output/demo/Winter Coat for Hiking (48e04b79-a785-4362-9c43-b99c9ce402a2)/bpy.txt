import bpy

def clear_scene():
    """Delete every existing object to start from a clean scene."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for collection in bpy.data.collections:
        collection.objects.clear()

def set_units():
    """Set the length unit to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a principled BSDF material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.8
    return mat

def assign_material(obj, mat):
    """Assign a material to an object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive(name, shape, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive cube, cylinder, or sphere and return the object."""
    bpy.ops.object.select_all(action='DESELECT')
    obj = None
    if shape == 'cube':
        w, h, d = size
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.name = name
        obj.scale = (w/2, h/2, d/2)
    elif shape == 'cylinder':
        radius, height = size
        bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=location)
        obj = bpy.context.active_object
        obj.name = name
    elif shape == 'sphere':
        radius = size
        bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location)
        obj = bpy.context.active_object
        obj.name = name
    elif shape == 'cone':
        radius, height = size
        bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=height, location=location)
        obj = bpy.context.active_object
        obj.name = name
    
    if obj:
        obj.rotation_euler = rotation
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with the given world vector."""
    # This is a simplified alignment. For strict axis alignment to world Z, 
    # we just ensure rotation is identity if vector is (0,0,1).
    # For general vectors, we'd need quaternion math. 
    # Given the prompt mostly asks for Z align Z, we handle that specifically.
    if axis == 'Z' and vector == (0,0,1):
        obj.rotation_euler = (0, 0, 0)
    elif axis == 'Y' and vector == (0,1,0):
        obj.rotation_euler = (-math.pi/2, 0, 0)
    elif axis == 'X' and vector == (1,0,0):
        obj.rotation_euler = (0, 0, math.pi/2)

def get_world_center(obj):
    """Get the world location of the object's center."""
    return obj.matrix_world.to_translation()

def get_face_center(obj, face_type):
    """
    Approximate the world center of a specific face based on object dimensions and rotation.
    Assumes object is axis-aligned or has simple rotations.
    face_type: 'top', 'bottom', 'front', 'back', 'left', 'right', 'center'
    """
    loc = obj.matrix_world.to_translation()
    rot = obj.rotation_euler
    # Get half dimensions
    dims = obj.dimensions
    hx, hy, hz = dims[0]/2, dims[1]/2, dims[2]/2
    
    # Apply rotation to the offset vector
    # Note: This is a simplification. For complex rotations, matrix multiplication is better.
    # We will use the object's matrix to transform local offsets.
    
    local_offset = Vector((0,0,0))
    
    if face_type == 'center':
        local_offset = Vector((0,0,0))
    elif face_type == 'top':
        local_offset = Vector((0,0,hz))
    elif face_type == 'bottom':
        local_offset = Vector((0,0,-hz))
    elif face_type == 'front':
        local_offset = Vector((0,hy,0))
    elif face_type == 'back':
        local_offset = Vector((0,-hy,0))
    elif face_type == 'right':
        local_offset = Vector((hx,0,0))
    elif face_type == 'left':
        local_offset = Vector((-hx,0,0))
        
    # Transform local offset to world space
    world_offset = obj.matrix_world.to_3x3() @ local_offset
    return loc + world_offset

def align_object_to_face(target_obj, anchor_obj, target_face, anchor_face):
    """
    Align target_obj so its target_face is parallel and adjacent to anchor_obj's anchor_face.
    Moves target_obj so the faces touch.
    """
    # Calculate the position of the anchor face on the anchor object
    anchor_pos = get_face_center(anchor_obj, anchor_face)
    
    # Calculate the normal direction of the anchor face in world space
    # We approximate this by looking at the offset from center to face
    anchor_center = get_world_center(anchor_obj)
    normal_vec = (anchor_pos - anchor_center).normalized()
    
    # The target face should be pushed out along this normal
    # But we need to know which face of the target is touching.
    # Let's assume the target's 'anchor_face' side is the one touching.
    # We need to move the target so its 'anchor_face' is at 'anchor_pos'.
    
    target_center = get_world_center(target_obj)
    target_face_pos = get_face_center(target_obj, anchor_face)
    
    # Vector from target's face to anchor's face
    move_vec = anchor_pos - target_face_pos
    
    target_obj.location += move_vec
    
    # Orientation: Align target's normal of the touching face to the anchor's normal
    # Target face normal in local space
    local_normal = Vector((0,0,0))
    if anchor_face == 'top': local_normal = (0,0,1)
    elif anchor_face == 'bottom': local_normal = (0,0,-1)
    elif anchor_face == 'front': local_normal = (0,1,0)
    elif anchor_face == 'back': local_normal = (0,-1,0)
    elif anchor_face == 'right': local_normal = (1,0,0)
    elif anchor_face == 'left': local_normal = (-1,0,0)
    
    # We want target's local_normal to align with normal_vec (or -normal_vec depending on side)
    # If we are placing ON TOP, target bottom touches anchor top.
    # Anchor top normal is +Z. Target bottom normal is -Z.
    # So we want target's -Z to align with Anchor's +Z.
    
    # Simplified approach for this script:
    # The prompt implies simple stacking/alignment where axes remain aligned.
    # We will primarily rely on the explicit "Align ... to ..." instructions which usually mean
    # "Make sure the faces are coplanar/parallel".
    # Since we are moving the object to touch, and assuming initial rotation is correct (Z align Z),
    # we just need to position it.
    
    # If the anchor face is 'top', we want the target's 'bottom' to be at the anchor's 'top'.
    # If the anchor face is 'front', we want the target's 'back' to be at the anchor's 'front'.
    
    # Let's refine the move vector based on the specific face interaction
    # We already moved target_face_pos to anchor_pos.
    # This effectively places the target such that its 'anchor_face' is at the anchor's 'anchor_face'.
    # This is usually what is desired for "Anchor X to Y.face".
    
    return target_obj

def apply_offset(obj, offset):
    """Apply an offset to the object's location in local space (converted to world if needed)."""
    # The prompt says "Offset is in reference local axes".
    # However, for simple stacking, world offsets are often easier.
    # Let's interpret "local axes" as the object's current orientation.
    # Since we align axes first, local Z is often world Z.
    
    # Convert local offset to world offset
    rot_matrix = obj.matrix_world.to_3x3()
    world_offset = rot_matrix @ Vector(offset)
    
    obj.location += world_offset

def boolean_union(base_obj, cutter_obj):
    """Perform a boolean union using the modifier stack."""
    # Add Boolean Modifier to base
    mod = base_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_subtract(base_obj, cutter_obj):
    """Perform a boolean subtract using the modifier stack."""
    # Add Boolean Modifier to base
    mod = base_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def bevel_object(obj, width=0.01, segments=2):
    """Add a bevel modifier and apply it."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

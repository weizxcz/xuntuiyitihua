def make_material(name, diffuse_color):
    """Create a principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def get_world_center(obj):
    """Get the world location of an object's center."""
    return obj.location

def get_world_bottom(obj):
    """Get the world Z coordinate of the object's bottom face."""
    # Assuming object is axis-aligned
    return obj.location[2] - (obj.dimensions[2] / 2)

def get_world_top(obj):
    """Get the world Z coordinate of the object's top face."""
    return obj.location[2] + (obj.dimensions[2] / 2)

def align_to_anchor(obj, anchor_obj, anchor_type='center'):
    """Align object to anchor object."""
    if anchor_type == 'center':
        obj.location.x = anchor_obj.location.x
        obj.location.y = anchor_obj.location.y
    elif anchor_type == 'bottom_face':
        # Align bottom of obj to bottom of anchor
        obj.location.z = get_world_bottom(anchor_obj) - (obj.dimensions[2] / 2)
    elif anchor_type == 'top_face':
        # Align bottom of obj to top of anchor
        obj.location.z = get_world_top(anchor_obj) - (obj.dimensions[2] / 2)
    elif anchor_type == 'specific_z':
        # Align bottom of obj to specific Z
        obj.location.z = anchor_obj - (obj.dimensions[2] / 2)

def apply_offset(obj, offset):
    """Apply offset to object location."""
    obj.location.x += offset[0]
    obj.location.y += offset[1]
    obj.location.z += offset[2]

def boolean_union(target, source):
    """Perform boolean union of source into target."""
    # Ensure target exists and is selected
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean modifier
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = source
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove source object
    bpy.data.objects.remove(source, do_unlink=True)

def boolean_subtract(target, cutter):
    """Perform boolean subtract of cutter from target."""
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(cutter, do_unlink=True)

def assign_material(obj, mat_name):
    """Assign material to object."""
    if mat_name in bpy.data.materials:
        if obj.data.materials:
            obj.data.materials[0] = bpy.data.materials[mat_name]
        else:
            obj.data.materials.append(bpy.data.materials[mat_name])

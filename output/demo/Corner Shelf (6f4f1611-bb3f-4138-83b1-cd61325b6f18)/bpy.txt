def make_material(name, diffuse_color=(1.0, 1.0, 1.0, 1.0)):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = (rotation[0], rotation[1], rotation[2])
    if material:
        obj.data.materials.append(material)
    return obj

def average(points):
    """Calculate the average of a list of Vector objects."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find specific points on an object's geometry."""
    def __init__(self, obj):
        self.obj = obj
        self.bm = None
        self.mesh = obj.data
        # Ensure we have a clean BMesh for calculation
        self.bm = bmesh.new()
        self.bm.from_mesh(self.mesh)
        bmesh.ops.transform(self.bm, matrix=Matrix.Identity(4), verts=self.bm.verts) # Ensure clean state

    def center_world(self):
        """Return the world-space center of the object's bounding box."""
        # Using object location is usually sufficient for primitives, 
        # but let's use bounding box center for robustness if needed.
        # For primitives, obj.location is the center.
        return self.obj.location.copy()

    def face_center_world(self, face_index):
        """Return the world-space center of a specific face."""
        if face_index < len(self.bm.faces):
            face = self.bm.faces[face_index]
            # Calculate face center in local space
            local_center = sum([v.co for v in face.verts]) / len(face.verts)
            # Convert to world space
            return self.obj.matrix_world @ local_center
        return Vector((0,0,0))

    def radial_point_world(self, direction='X', face_index=None):
        """Return a point on the radial edge/face."""
        # Simplified: returns center for now, as radial alignment is complex for generic meshes
        return self.center_world()

    def free(self):
        if self.bm:
            self.bm.free()

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract operation using the modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target is active for modifier addition if needed, but we add to target
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_union(target_name, other_name):
    """Perform a boolean union operation."""
    target = bpy.data.objects[target_name]
    other = bpy.data.objects[other_name]
    
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other
    
    bpy.ops.object.modifier_apply(modifier=mod.name)

def add_bevel(obj_name, width=0.01, segments=5):
    """Add a bevel modifier to an object."""
    obj = bpy.data.objects[obj_name]
    bpy.context.view_layer.objects.active = obj
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for geometry generation
    bpy.ops.object.modifier_apply(modifier=mod.name)

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_primitive_cube(name, size, location, rotation_euler):
    """Add a cube primitive, set size, location, rotation, and name."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size/2, size/2, size/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def add_primitive_circle(name, diameter, location, rotation_euler, segments=32):
    """Add a circle (disc) primitive, set diameter, location, rotation, and name."""
    bpy.ops.mesh.primitive_circle_add(vertices=segments, radius=diameter/2, fill_type='NGON', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def add_primitive_cone(name, bottom_diameter, top_diameter, height, location, rotation_euler):
    """Add a cone/tapered cylinder primitive."""
    bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=bottom_diameter/2, radius2=top_diameter/2, depth=height, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def add_primitive_cylinder(name, diameter, height, location, rotation_euler):

def align_axis_to_vector(obj, axis='Z', vector=Vector((0, 0, 1))):
    """Align an object's local axis to a target world vector."""
    if axis == 'X':
        src = Vector((1, 0, 0))
    elif axis == 'Y':
        src = Vector((0, 1, 0))
    else: # Z
        src = Vector((0, 0, 1))
    
    target = vector.normalized()
    rot_quat = src.rotation_difference(target)
    obj.rotation_quaternion = rot_quat

def boolean_union(objects_to_union, keep_obj_name):
    """Perform boolean union on a list of objects, keeping the first one as the result."""
    if not objects_to_union:
        return
    target = objects_to_union[0]
    modifiers = []
    
    # Apply modifiers to target first to ensure clean state
    for mod in target.modifiers:
        bpy.ops.object.modifier_apply(modifier=mod.name)
    
    for obj in objects_to_union[1:]:
        mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        modifiers.append((target, mod))
    
    # Apply all modifiers
    for t, m in modifiers:
        bpy.ops.object.modifier_apply(modifier=m.name)
        
    # Remove other objects
    for obj in objects_to_union[1:]:
        bpy.data.objects.remove(obj, do_unlink=True)
        
    return target

def boolean_subtract(target, cutter):
    """Subtract cutter from target using boolean modifier."""
    mod = target.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter, do_unlink=True)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    sum_v = Vector((0, 0, 0))
    for p in points:
        sum_v += p
    return sum_v / len(points)

class Locator:
    """Helper class to find centers of objects."""
    @staticmethod
    def center_world(obj):
        """Get the world space center of an object's bounding box."""
        bm = bmesh.new()
        bm.from_mesh(obj.data)
        bm.transform(obj.matrix_world)
        center = average([v.co for v in bm.verts])
        bm.free()
        return center

    @staticmethod
    def face_center_world(obj):
        """Get the world space center of the first face of an object."""
        bm = bmesh.new()
        bm.from_mesh(obj.data)
        bm.transform(obj.matrix_world)
        if bm.faces:
            center = average([v.co for v in bm.faces[0].verts])
        else:
            center = Vector((0,0,0))
        bm.free()
        return center

    @staticmethod
    def radial_point_world(obj, axis='Z', direction=1):
        """Get a point on the radial edge of an object."""
        # Simplified: returns center for now as radial point logic is complex for arbitrary meshes
        # For circles/cylinders, this is effectively the center in XY plane
        return Locator.center_world(obj)

# Helper to create a triangle
def create_triangle(name, side, location, rotation_euler):
    # Create a cube and scale it to approximate a triangle? 
    # Better: Create a custom mesh triangle or use a cone with 3 vertices and cut it.
    # Simplest robust way in Blender Python without complex mesh math:
    # Create a cone with 3 vertices, radius = side/sqrt(3), height=0 (flat).
    # Actually, a flat cone with 3 vertices is a triangle.
    # Radius of circumcircle for side s is s / sqrt(3).
    radius = side / math.sqrt(3)
    obj = add_primitive_cone(name, radius*2, 0, 0.001, location, rotation_euler) # Small height to make it a mesh
    # To make it a true flat triangle, we can scale Z to 0, but boolean ops might fail.
    # Let's keep a tiny height or use a plane.
    # Using a plane with 3 vertices is cleaner.
    bpy.ops.mesh.primitive_plane_add(size=1, location=(0,0,0))
    temp_obj = bpy.context.active_object
    temp_obj.name = name + "_temp"
    temp_obj.scale = (radius, radius, 1)
    
    # Convert to mesh and triangulate/adjust vertices to form equilateral triangle
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.mesh.delete(type='VERT') # Delete all
    bpy.ops.mesh.primitive_triangle_add(radius=radius, location=(0,0,0))
    tri_obj = bpy.context.active_object
    tri_obj.name = name
    tri_obj.rotation_euler = rotation_euler
    tri_obj.location = location
    
    # Ensure it's flat enough for boolean but has volume
    # Actually, let's just use the cone method with very small height, it's more robust for booleans
    bpy.ops.object.delete() # Delete the plane temp
    
    # Re-create using cone
    obj = add_primitive_cone(name, radius*2, 0, 0.0001, location, rotation_euler)
    return obj

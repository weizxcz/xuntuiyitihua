def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cube(name, size, location, rotation_euler):
    """Add a cube primitive, scaled and positioned."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def align_to_face(obj, target_obj, face_type, offset_vector=Vector((0,0,0))):
    """
    Aligns an object's face to the average of two target faces or a single face.
    face_type: 'top', 'bottom', 'front', 'back', 'left', 'right', 'center'
    """
    # Calculate target world position based on face type
    target_loc = get_face_center(target_obj, face_type)
    
    # If target_obj is a list (average of two), handle it
    if isinstance(target_obj, list):
        target_loc = average([get_face_center(o, face_type) for o in target_obj])
    
    # Calculate offset from object center to its face
    obj_offset = get_face_offset(obj, face_type)
    
    # New location = Target - (Obj's local face offset rotated to world)
    # Since we align axes, we can simplify:
    # We want obj's face to be at target_loc.
    # obj.location + obj_offset_world = target_loc
    # obj.location = target_loc - obj_offset_world
    
    # Get rotation matrix of the object
    rot_mat = obj.matrix_world.to_3x3()
    obj_offset_world = obj_offset @ rot_mat
    
    new_loc = target_loc - obj_offset_world + offset_vector
    obj.location = new_loc

def get_face_center(obj, face_type):
    """Get the world location of a specific face center."""
    # Ensure object is updated to get correct bounds
    obj.update_from_editmode() # Not needed for primitives usually, but good practice
    
    # Local face centers for a cube centered at 0,0,0 with scale S
    # Dimensions
    sx = obj.scale.x * (obj.dimensions.x / 2) if obj.dimensions.x else 0
    sy = obj.scale.y * (obj.dimensions.y / 2) if obj.dimensions.y else 0
    sz = obj.scale.z * (obj.dimensions.z / 2) if obj.dimensions.z else 0
    
    # Local face centers
    faces = {
        'top': Vector((0, 0, sz)),
        'bottom': Vector((0, 0, -sz)),
        'front': Vector((0, sy, 0)),
        'back': Vector((0, -sy, 0)),
        'right': Vector((sx, 0, 0)),
        'left': Vector((-sx, 0, 0)),
        'center': Vector((0, 0, 0))
    }
    
    local_center = faces.get(face_type, Vector((0,0,0)))
    return obj.matrix_world @ local_center

def get_face_offset(obj, face_type):
    """Get the vector from object center to the specified face center in local space."""
    sx = obj.scale.x * (obj.dimensions.x / 2) if obj.dimensions.x else 0
    sy = obj.scale.y * (obj.dimensions.y / 2) if obj.dimensions.y else 0
    sz = obj.scale.z * (obj.dimensions.z / 2) if obj.dimensions.z else 0
    
    faces = {
        'top': Vector((0, 0, sz)),
        'bottom': Vector((0, 0, -sz)),
        'front': Vector((0, sy, 0)),
        'back': Vector((0, -sy, 0)),
        'right': Vector((sx, 0, 0)),
        'left': Vector((-sx, 0, 0)),
        'center': Vector((0, 0, 0))
    }
    return faces.get(face_type, Vector((0,0,0)))

def average(vectors):
    """Calculate the average of a list of vectors."""
    if not vectors:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for v in vectors:
        avg += v
    return avg / len(vectors)

def orient_z_to_vector(obj, target_vector):
    """Rotate object so its local +Z aligns with target_vector."""
    # Current Z
    current_z = Vector((0, 0, 1))
    # Target
    target = target_vector.normalized()
    
    # Calculate rotation
    rot_quat = current_z.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius=0.5, depth=1, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a world vector."""
    if axis == 'Z':
        local_axis = Vector((0, 0, 1))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    elif axis == 'X':
        local_axis = Vector((1, 0, 0))
    else:
        return

    target = vector.normalized()
    if local_axis.dot(target) > 0.9999:
        return # Already aligned
    
    rot_quat = local_axis.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

def get_face_center(obj, face_type='top'):
    """Get the world location of a specific face center."""
    # Assuming standard orientation where +Z is up
    if face_type == 'top':
        # Top face is at Z + height/2
        return obj.location + Vector((0, 0, obj.dimensions.z / 2))
    elif face_type == 'bottom':
        return obj.location + Vector((0, 0, -obj.dimensions.z / 2))
    return obj.location

def offset_object(obj, offset_vec):
    """Apply offset to object location in world space (assuming aligned axes)."""
    obj.location += offset_vec

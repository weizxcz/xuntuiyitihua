# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive_cube(name, size, location, rotation_euler):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def add_primitive_cylinder(name, radius, depth, location, rotation_euler):
    bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a world vector."""
    if axis == 'X': local_axis = Vector((1, 0, 0))
    elif axis == 'Y': local_axis = Vector((0, 1, 0))
    elif axis == 'Z': local_axis = Vector((0, 0, 1))
    
    target = vector.normalized()
    if local_axis.cross(target).length < 1e-6:
        return # Already aligned
    
    rot_mat = local_axis.rotation_difference(target).to_matrix().to_4x4()
    obj.matrix_world = rot_mat @ obj.matrix_world

def boolean_union(objects_to_union, keep_name):
    """Perform boolean union on a list of objects, keeping the first one as the result."""
    if not objects_to_union:
        return None
    
    target = objects_to_union[0]
    modifiers = []
    
    for obj in objects_to_union[1:]:
        mod = target.modifiers.new(name=f"Bool_{obj.name}", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        modifiers.append(mod)
        
        # Apply modifier immediately to keep scene clean
        bpy.context.view_layer.objects.active = target
        bpy.ops.object.modifier_apply(modifier=mod.name)
        bpy.data.objects.remove(obj)
        
    target.name = keep_name
    return target

def boolean_subtract(target, cutter):
    """Subtract cutter from target."""
    mod = target.modifiers.new(name="Bool_Sub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter)
    return target

def get_world_center(obj):
    return obj.matrix_world.to_translation()

def get_face_center(obj, face_type='bottom'):
    """Approximate face center based on bounding box."""
    loc = obj.location
    size = obj.dimensions
    if face_type == 'bottom':
        return Vector((loc.x, loc.y, loc.z - size[2]/2))
    elif face_type == 'top':
        return Vector((loc.x, loc.y, loc.z + size[2]/2))
    elif face_type == 'back': # Assuming Z-up, back is -Y
        return Vector((loc.x, loc.y - size[1]/2, loc.z))
    return loc

        # The offset is relative to the anchor point.

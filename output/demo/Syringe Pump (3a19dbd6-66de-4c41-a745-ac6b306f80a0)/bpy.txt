def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to the first material slot of an object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive(name, shape, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive mesh object."""
    bpy.ops.mesh.primitive_add(location=location)
    obj = bpy.context.active_object
    obj.name = name
    
    # Scale based on shape and size
    if shape == 'cube':
        # size is (x, y, z)
        obj.scale = (size[0], size[1], size[2])
    elif shape == 'cylinder' or shape == 'disc':
        # size is (radius, height) or (diameter, height) -> assuming radius, height for cylinder logic usually, 
        # but prompt says "Ø0.010" which is diameter. 
        # Let's assume size tuple is (radius, height) for internal logic, but prompt gives diameter.

        radius = size[0] / 2.0 if size[0] > 1 else size[0] # Heuristic: if > 1, likely diameter in mm? No, prompt is meters.
        # Prompt: "Ø0.010" -> radius 0.005.
        # Let's pass radius directly to be safe.
        pass 
    elif shape == 'cone':
        radius = size[0] / 2.0
        height = size[1]
        obj.scale = (radius, radius, height)
    elif shape == 'sphere':
        radius = size[0]
        obj.scale = (radius, radius, radius)

    # Apply rotation
    obj.rotation_euler = rotation
    
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):

    bpy.ops.mesh.primitive_cube_add(size=1, location=location) # size=1 creates 2x2x2 cube centered at 0
    obj = bpy.context.active_object

    obj.rotation_euler = rotation
    return obj

def align_axis_to_vector(obj, axis_to_align='+Z', target_vector):
    """Rotate object so its local axis aligns with target_vector."""
    # Normalize target
    v = target_vector.normalized()
    
    # Define source axis
    if axis_to_align == '+Z':
        src = Vector((0, 0, 1))
    elif axis_to_align == '+X':
        src = Vector((1, 0, 0))
    elif axis_to_align == '+Y':
        src = Vector((0, 1, 0))
    else:
        src = Vector((0, 0, 1))
        
    # Calculate rotation
    rot_quat = src.rotation_difference(v)
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = rot_quat

def get_face_center(obj, face_type):
    """
    Calculate the world location of a specific face center.
    face_type: 'top', 'bottom', 'left', 'right', 'front', 'back', 'center'
    """
    # Get object matrix
    mat = obj.matrix_world
    
    # Get bounding box corners
    # We need to transform the local bbox corners to world space
    # Local bbox corners relative to center
    # Cube: +/- size/2
    # Cylinder: depends on orientation
    
    # Simplified approach: Use the object's rotation to determine face directions
    # And add/subtract half dimensions
    
    # Determine dimensions
    if obj.type == 'MESH':
        # Get extents
        # Note: scale is applied to mesh data, but matrix_world includes scale.
        # We need local dimensions.
        if obj.type == 'MESH':
            # Recalculate dimensions from scale and original size if known?
            # Easier: Use the object's data bounds if not scaled weirdly, 
            # but since we scale primitives, let's use the scale * original_size/2
            
            # This is tricky without storing original size. 
            # Alternative: Use the known construction logic.
            # Since we create them with specific scales, we can infer.
            pass

    # Robust method: Raycast or Bounding Box
    # Let's use a simpler geometric approximation based on object type and rotation
    
    # Helper to get local axis vectors in world space
    def world_axis(axis):
        return (mat @ axis).normalized()

    # Determine half-dimensions
    # We need to know the object's dimensions. 
    # Since we create them, we can store them or re-calculate from vertices.
    # Let's re-calculate from vertices for robustness.
    
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bmesh.ops.transform(bm, matrix=mat, verts=bm.verts) # World space verts
    
    # Find min/max in local axes transformed to world? 
    # No, we want the face center in World Space.
    
    # Project all verts onto the normal of the face we want.
    
    # Define face normals in Local Space
    if face_type == 'top':
        local_normal = Vector((0, 0, 1))
    elif face_type == 'bottom':
        local_normal = Vector((0, 0, -1))
    elif face_type == 'right':
        local_normal = Vector((1, 0, 0))
    elif face_type == 'left':
        local_normal = Vector((-1, 0, 0))
    elif face_type == 'front':
        local_normal = Vector((0, 1, 0))
    elif face_type == 'back':
        local_normal = Vector((0, -1, 0))
    elif face_type == 'center':
        return mat.to_translation()
    else:
        return mat.to_translation()

    # Transform normal to World Space
    world_normal = (mat @ local_normal).normalized()
    
    # Find the vertex with the maximum dot product with world_normal
    max_dot = -1
    max_vert = None
    for v in bm.verts:
        d = v.co.dot(world_normal)
        if d > max_dot:
            max_dot = d
            max_vert = v
            
    if max_vert:
        # The face center is approximately the average of vertices on that face
        # For simplicity, just return the max vertex location if it's a cube/cylinder end
        # Better: Average all vertices with dot product > max_dot - epsilon
        face_verts = [v for v in bm.verts if v.co.dot(world_normal) > max_dot - 0.0001]
        if face_verts:
            center = sum((v.co for v in face_verts), Vector()) / len(face_verts)
            return center
    
    bm.free()
    return mat.to_translation()

def align_to_face(obj_target, face_type_target, obj_source, face_type_source):
    """
    Align obj_source so that face_type_source touches face_type_target of obj_target.
    Assumes objects are already oriented correctly.
    """
    target_loc = get_face_center(obj_target, face_type_target)
    source_loc = get_face_center(obj_source, face_type_source)
    
    # Calculate offset to move source face to target face
    # We want source_face_center to equal target_face_center
    offset = target_loc - source_loc
    
    obj_source.location += offset

def connect_faces(obj_a, face_a, obj_b, face_b):
    """
    Move obj_b so that face_b touches face_a of obj_a.
    """
    # Get the center of face_a on obj_a
    loc_a = get_face_center(obj_a, face_a)
    
    # Get the center of face_b on obj_b (before move)
    loc_b = get_face_center(obj_b, face_b)
    
    # Calculate translation vector
    # We want loc_b_new == loc_a
    # But wait, "Connect" usually means they touch.
    # If face_a is 'right' of A and face_b is 'left' of B, they touch at the same point.
    
    # However, get_face_center returns the center of the face.
    # So if we move B such that its face_b center is at loc_a, they will be coincident.
    
    # But we need to be careful about direction.
    # If A's right face is at X=1, and B's left face is at X=-1 (local), 
    # and we move B so its left face center is at X=1, B's center will be at X=1 - (-width/2) = 1 + width/2.
    # This effectively places B adjacent to A.
    
    # Let's verify with vectors.
    # Vector from A's face center to B's face center (current)
    # We want to move B by (A_face_center - B_face_center)
    
    move_vec = loc_a - loc_b
    obj_b.location += move_vec

def boolean_subtract(cutter_name, target_name):
    """
    Perform Boolean Subtract using cutter on target.
    """
    cutter = bpy.data.objects.get(cutter_name)
    target = bpy.data.objects.get(target_name)
    
    if not cutter or not target:
        print(f"Error: {cutter_name} or {target_name} not found")
        return

    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Delete Cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(obj_a_name, obj_b_name):
    """
    Perform Boolean Union of obj_a and obj_b.
    """
    obj_a = bpy.data.objects.get(obj_a_name)
    obj_b = bpy.data.objects.get(obj_b_name)
    
    if not obj_a or not obj_b:
        print(f"Error: {obj_a_name} or {obj_b_name} not found")
        return

    # Add Boolean Modifier to obj_a
    mod = obj_a.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj_b
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = obj_a
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Delete obj_b
    bpy.data.objects.remove(obj_b, do_unlink=True)

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler):
    """Add a cube primitive."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def align_faces(obj_a, obj_b, face_type_a, face_type_b):
    """
    Align two objects so that face_type_a of obj_a touches face_type_b of obj_b.
    Assumes objects are axis-aligned cubes/boxes.
    face_type: 'front', 'back', 'left', 'right', 'top', 'bottom'
    """
    # Calculate bounding boxes
    def get_bbox(obj):
        # Local bounds
        min_x, min_y, min_z = obj.location[0] - obj.dimensions[0]/2, obj.location[1] - obj.dimensions[1]/2, obj.location[2] - obj.dimensions[2]/2
        max_x, max_y, max_z = obj.location[0] + obj.dimensions[0]/2, obj.location[1] + obj.dimensions[1]/2, obj.location[2] + obj.dimensions[2]/2
        return (min_x, min_y, min_z), (max_x, max_y, max_z)

    bbox_a = get_bbox(obj_a)
    bbox_b = get_bbox(obj_b)

    # Helper to get coordinate of a face
    def get_coord(bbox, face):
        if face == 'front': return bbox[1][1] # +Y
        if face == 'back': return bbox[0][1] # -Y
        if face == 'left': return bbox[0][0] # -X
        if face == 'right': return bbox[1][0] # +X
        if face == 'top': return bbox[1][2] # +Z
        if face == 'bottom': return bbox[0][2] # -Z
        return 0

    # Determine target position for obj_a
    # We want obj_a.face_a to meet obj_b.face_b
    # If aligning front of A to back of B: A's front (+Y) should equal B's back (-Y)
    
    coord_a = get_coord(bbox_a, face_type_a)
    coord_b = get_coord(bbox_b, face_type_b)
    
    # Current offset needed to touch
    # If A.front touches B.back: A_loc_y + A_half_y = B_loc_y - B_half_y
    # New A_loc_y = B_loc_y - B_half_y - A_half_y
    
    # Let's calculate the shift required for each axis
    shift = Vector((0,0,0))
    
    # Y-axis alignment (Front/Back)
    if face_type_a in ['front', 'back'] and face_type_b in ['front', 'back']:
        # If opposite faces (Front-Back or Back-Front)
        if (face_type_a == 'front' and face_type_b == 'back') or (face_type_a == 'back' and face_type_b == 'front'):
            # A's positive Y face touches B's negative Y face
            # A.loc.y + A.dim.y/2 = B.loc.y - B.dim.y/2
            # A.loc.y = B.loc.y - B.dim.y/2 - A.dim.y/2
            target_y = bbox_b[0][1] - bbox_a[0][1] # This is effectively B.min_y - A.min_y? No.
            # Let's use centers.
            # A.center + A.half = B.center - B.half
            # A.center = B.center - B.half - A.half
            target_y = bbox_b[0][1] - bbox_a[0][1] # Wait, bbox is min/max.
            # A.min_y + A.dim_y = B.min_y
            # A.loc_y = B.min_y - A.dim_y
            # Actually simpler:
            # Distance between centers = A.half + B.half
            # Direction: From B to A is opposite to the face normal direction relative to contact.
            
            # Let's just solve for new location of A
            # Contact Point P = B.face_coord
            # A.face_coord_new = P
            # A.loc_new = P - A.face_offset
            
            p_y = bbox_b[0][1] # Back of B
            a_face_offset_y = bbox_a[1][1] - bbox_a[0][1] # Half dim? No, full dim from min.
            # A.front is at loc_y + dim_y/2.
            # loc_y_new + dim_y/2 = p_y
            # loc_y_new = p_y - dim_y/2
            
            loc_y_new = bbox_b[0][1] - (bbox_a[1][1] - bbox_a[0][1])/2
            shift[1] = loc_y_new - obj_a.location[1]
            
        elif (face_type_a == 'front' and face_type_b == 'front') or (face_type_a == 'back' and face_type_b == 'back'):
            # Same faces touching? Usually implies overlap or specific alignment. 
            # Assuming standard "touching" logic, this might be invalid or mean "align planes".
            # For this script, we assume valid "touching" pairs like Front-Back.
            pass

    # X-axis alignment (Left/Right)
    if face_type_a in ['left', 'right'] and face_type_b in ['left', 'right']:
        if (face_type_a == 'right' and face_type_b == 'left') or (face_type_a == 'left' and face_type_b == 'right'):
            # A.right touches B.left
            # A.loc.x + A.half_x = B.loc.x - B.half_x
            # A.loc.x = B.loc.x - B.half_x - A.half_x
            target_x = bbox_b[0][0] - (bbox_a[1][0] - bbox_a[0][0])/2
            shift[0] = target_x - obj_a.location[0]
            
    # Z-axis alignment (Top/Bottom)
    if face_type_a in ['top', 'bottom'] and face_type_b in ['top', 'bottom']:
        if (face_type_a == 'bottom' and face_type_b == 'top') or (face_type_a == 'top' and face_type_b == 'bottom'):
            # A.bottom touches B.top
            # A.loc.z - A.half_z = B.loc.z + B.half_z
            # A.loc.z = B.loc.z + B.half_z + A.half_z
            target_z = bbox_b[1][2] + (bbox_a[1][2] - bbox_a[0][2])/2
            shift[2] = target_z - obj_a.location[2]

    obj_a.location += shift
    return obj_a

def offset_object(obj, offset_vec):
    """Apply offset to object location."""
    obj.location += offset_vec
    return obj

def boolean_union(objects, keep_name):
    """
    Perform boolean union on a list of objects.
    Returns the first object in the list (renamed to keep_name) as the result.
    """
    if not objects:
        return None
    
    target = objects[0]
    target.name = keep_name
    
    # Select all
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
        
    # Apply modifiers to first one
    # We need to add a Boolean modifier to the target
    # But since we are doing a union of primitives, we can just join them if they are simple,
    # or use the Boolean modifier. Given the strict "Boolean-union" instruction, we use the modifier.
    
    # To ensure clean union, we apply the modifier.
    # Note: In a script, it's often faster to join geometry if no further boolean ops are needed on these parts.
    # However, to follow "Boolean-union" strictly:
    
    # Add modifier to target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = None # We will iterate others
    
    # Actually, standard boolean union in Blender UI selects multiple objects and uses "Join" or "Boolean" on active.
    # Let's use the operator approach for robustness with multiple operands.
    
    # Clear existing modifiers on target to avoid conflicts
    for m in target.modifiers:
        target.modifiers.remove(m)
        
    # We will manually construct the union by joining geometry if possible, 
    # or using a single modifier with a group? No, standard boolean takes one object.
    # For multiple objects, we must chain or join.
    # Simplest robust way for script: Join them into one mesh.
    
    bpy.ops.object.join()
    
    # The active object is now the joined mesh.
    result = bpy.context.active_object
    result.name = keep_name
    
    # Re-apply material if needed (it keeps the material of the active object before join usually)
    return result

def apply_material(obj, mat_name):
    """Apply a material to an object."""
    if obj.data.materials:
        obj.data.materials[0] = bpy.data.materials[mat_name]
    else:
        obj.data.materials.append(bpy.data.materials[mat_name])

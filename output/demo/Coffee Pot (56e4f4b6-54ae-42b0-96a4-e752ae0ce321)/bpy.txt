import bpy
import bmesh
import mathutils
from mathutils import Vector, Matrix

# ==============================================================================
# HELPER FUNCTIONS
# ==============================================================================

def clear_scene():
    """Delete all existing objects and collections."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Clear orphan data blocks
    for block in bpy.data.meshes:
        bpy.data.meshes.remove(block)
    for block in bpy.data.materials:
        bpy.data.materials.remove(block)
    for block in bpy.data.objects:
        bpy.data.objects.remove(block)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive(name, type, **kwargs):
    """Add a primitive object and return it."""
    if type == 'CUBE':
        bpy.ops.mesh.primitive_cube_add(size=kwargs.get('size', 1), location=(0,0,0))
    elif type == 'CYLINDER':
        bpy.ops.mesh.primitive_cylinder_add(radius=kwargs.get('radius', 0.5), depth=kwargs.get('height', 1), location=(0,0,0))
    elif type == 'CONE':
        bpy.ops.mesh.primitive_cone_add(radius1=kwargs.get('base_radius', 0.5), radius2=kwargs.get('top_radius', 0), depth=kwargs.get('height', 1), location=(0,0,0))
    elif type == 'UVSPHERE':
        bpy.ops.mesh.primitive_uv_sphere_add(radius=kwargs.get('radius', 0.5), location=(0,0,0))
    elif type == 'TORUS':
        bpy.ops.mesh.primitive_torus_add(major_radius=kwargs.get('major_radius', 0.5), minor_radius=kwargs.get('minor_radius', 0.1), location=(0,0,0))
    elif type == 'PLANE':
        bpy.ops.mesh.primitive_plane_add(size=kwargs.get('size', 1), location=(0,0,0))
    
    obj = bpy.context.active_object
    obj.name = name
    return obj

def orient_object(obj, axis_align):
    """
    Orient object so its local +Z aligns with the specified world axis.
    axis_align: 'X', 'Y', 'Z', '-X', '-Y', '-Z'
    """
    obj.rotation_euler = (0, 0, 0)
    if axis_align == 'Z':
        # Default, no rotation needed if created upright
        pass
    elif axis_align == 'X':
        obj.rotation_euler = (0, 0, math.radians(90))
    elif axis_align == '-X':
        obj.rotation_euler = (0, 0, math.radians(-90))
    elif axis_align == 'Y':
        obj.rotation_euler = (math.radians(-90), 0, 0)
    elif axis_align == '-Y':
        obj.rotation_euler = (math.radians(90), 0, 0)
    elif axis_align == '-Z':
        obj.rotation_euler = (math.radians(180), 0, 0)
    
    obj.update_tag()
    bpy.context.view_layer.update()

def get_bounding_box(obj):
    """Get min/max bounds in world space."""
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    
    mins = Vector((float('inf'), float('inf'), float('inf')))
    maxs = Vector((float('-inf'), float('-inf'), float('-inf')))
    
    for v in bm.verts:
        mins.x = min(mins.x, v.co.x)
        mins.y = min(mins.y, v.co.y)
        mins.z = min(mins.z, v.co.z)
        maxs.x = max(maxs.x, v.co.x)
        maxs.y = max(maxs.y, v.co.y)
        maxs.z = max(maxs.z, v.co.z)
        
    bm.free()
    return mins, maxs

def align_faces(obj_src, face_type_src, obj_tgt, face_type_tgt):
    """
    Align source object's face to target object's face.
    face_type: 'top', 'bottom', 'left', 'right', 'front', 'back', 'center'
    """
    src_min, src_max = get_bounding_box(obj_src)
    tgt_min, tgt_max = get_bounding_box(obj_tgt)
    
    # Calculate centers
    src_center = (src_min + src_max) / 2
    tgt_center = (tgt_min + tgt_max) / 2
    
    # Calculate face positions
    # Assuming standard orientation: Z is up, X is right, Y is front
    # Top: max Z, Bottom: min Z
    # Right: max X, Left: min X
    # Front: max Y, Back: min Y
    
    src_face_pos = Vector()
    tgt_face_pos = Vector()
    
    # Determine alignment vector
    align_vec = Vector()
    
    if face_type_src == 'top' and face_type_tgt == 'bottom':
        # Align src top (max Z) to tgt bottom (min Z)
        src_face_pos.z = src_max.z
        tgt_face_pos.z = tgt_min.z
        align_vec.z = tgt_face_pos.z - src_face_pos.z
    elif face_type_src == 'bottom' and face_type_tgt == 'top':
        src_face_pos.z = src_min.z
        tgt_face_pos.z = tgt_max.z
        align_vec.z = tgt_face_pos.z - src_face_pos.z
    elif face_type_src == 'left' and face_type_tgt == 'right':
        src_face_pos.x = src_min.x
        tgt_face_pos.x = tgt_max.x
        align_vec.x = tgt_face_pos.x - src_face_pos.x
    elif face_type_src == 'right' and face_type_tgt == 'left':
        src_face_pos.x = src_max.x
        tgt_face_pos.x = tgt_min.x
        align_vec.x = tgt_face_pos.x - src_face_pos.x
    elif face_type_src == 'front' and face_type_tgt == 'back':
        src_face_pos.y = src_max.y
        tgt_face_pos.y = tgt_min.y
        align_vec.y = tgt_face_pos.y - src_face_pos.y
    elif face_type_src == 'back' and face_type_tgt == 'front':
        src_face_pos.y = src_min.y
        tgt_face_pos.y = tgt_max.y
        align_vec.y = tgt_face_pos.y - src_face_pos.y
    elif face_type_src == 'center' and face_type_tgt == 'center':
        align_vec = tgt_center - src_center
    else:
        # Default to center-to-center if specific face not matched or unsupported
        align_vec = tgt_center - src_center

    # Apply translation
    obj_src.location += align_vec
    obj_src.update_tag()
    bpy.context.view_layer.update()

def offset_object(obj, offset_vec):
    """Apply offset to object location."""
    obj.location += offset_vec
    obj.update_tag()
    bpy.context.view_layer.update()

def boolean_subtract(cutter_name, target_name):
    """
    Perform boolean subtract using modifier stack.
    cutter_name: Name of the object to cut with.
    target_name: Name of the object to be cut.
    """
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target has a modifier slot
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.update()
    bpy.ops.object.modifier_apply(modifier=mod.name, object=target)
    
    # Remove cutter
    bpy.data.objects.remove(cutter)

def boolean_union(objects_to_union, result_name):
    """
    Union a list of objects into a single object.
    """
    if not objects_to_union:
        return None
        
    # Select all objects
    for obj in objects_to_union:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
        
    bpy.ops.object.join()
    
    # Rename the resulting object
    result = bpy.context.active_object
    result.name = result_name
    
    # Deselect all
    for obj in objects_to_union:
        obj.select_set(False)
        
    return result

def average_vectors(vectors):
    """Calculate average of a list of vectors."""
    if not vectors:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for v in vectors:
        avg += v
    return avg / len(vectors)

# The prompt says "Boolean-subtract lid_dome_cut from lid_dome; keep lid_dome. // produces hemisphere"

# A hemisphere produced by a plane cut usually implies cutting off the bottom half.
# We need to position the plane at the equator of the sphere.
# Current lid_dome center is at (0,0,0.075) relative to body top? No, offset was applied.
# Let's calculate the cut position.
# Sphere radius 0.05. Center is at Z_center.

# If the cut removed the top, the dome is below.

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), scale=(1,1,1)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation, scale=scale)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size=1.0, location=(0,0,0), rotation=(0,0,0), scale=(1,1,1)):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation, scale=scale)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_empty(name, location=(0,0,0), rotation=(0,0,0)):
    """Add an empty object."""
    bpy.ops.object.empty_add(location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def boolean_union(objects_to_union, keep_first=True):
    """Perform boolean union on a list of objects. Modifies the first object in place."""
    if not objects_to_union:
        return None
    
    target = objects_to_union[0]
    modifiers = []
    
    # Apply modifiers to target
    for i, obj in enumerate(objects_to_union):
        if i == 0:
            continue
        
        # Add boolean modifier to target
        mod = target.modifiers.new(name=f"BoolUnion_{obj.name}", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        
        # We need to apply this modifier. 
        # Note: Applying boolean modifiers on many objects sequentially can be slow.
        # For this script, we apply them one by one.
        bpy.context.view_layer.objects.active = target
        bpy.ops.object.modifier_apply(modifier=mod.name)
        
        # Remove the source object
        bpy.data.objects.remove(obj, do_unlink=True)

def boolean_subtract(target, cutter):
    """Subtract cutter from target."""
    mod = target.modifiers.new(name=f"BoolSub_{cutter.name}", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter, do_unlink=True)

def set_transform(obj, location=None, rotation=None, scale=None):
    """Set transform properties."""
    if location:
        obj.location = location
    if rotation:
        obj.rotation_euler = rotation
    if scale:
        obj.scale = scale

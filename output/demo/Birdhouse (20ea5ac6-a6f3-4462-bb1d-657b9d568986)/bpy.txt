import bpy
import bmesh
from mathutils import Vector, Matrix
from math import radians, pi

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if not obj.data.materials:
        obj.data.materials.append(mat)
    else:
        obj.data.materials[0] = mat

def add_primitive_cube(name, size, location, rotation_euler, mat=None):
    """Add a cube and set its size (scale) and transform."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    # Blender cubes are size 2x2x2 by default, so scale = size/2
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if mat:
        apply_material(obj, mat)
    return obj

def add_primitive_cylinder(name, radius, depth, location, rotation_euler, mat=None):
    """Add a cylinder."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    if mat:
        apply_material(obj, mat)
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Rotate object so its local axis aligns with target vector."""
    # Get current rotation
    rot = obj.rotation_euler.copy()
    
    # We need to find the rotation that maps the local axis to the target vector.
    # Local +Z is (0,0,1), +Y is (0,1,0), +X is (1,0,0).
    
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    current_axis = local_axes[axis_to_align]
    
    # Calculate rotation matrix to align current_axis to target_vector
    # Using quaternion rotation
    q = current_axis.rotation_difference(target_vector)
    
    # Apply this rotation to the object's current rotation
    # Note: This assumes we want to set the orientation relative to world, 
    # effectively overriding previous rotations if we just set the euler.
    # However, to be safe and precise, we can just set the rotation quaternion directly 
    # if we assume the object starts at 0,0,0. Since we create them fresh, this is safe.
    
    # Alternative: Use mathutils.geometry to find rotation matrix
    # But simpler: just set the rotation euler that achieves this.
    # Since we are creating primitives, their initial rotation is (0,0,0).
    # So we can just set the rotation_euler to the one that aligns the axis.
    
    # Let's use a simpler approach: create a dummy matrix and apply.
    # Actually, just setting rotation_euler via quaternion is robust.
    obj.rotation_euler = q.to_euler()

def get_center(obj):
    """Get world center of object."""
    return obj.matrix_world.to_translation()

def get_bottom_edge_center(obj):
    """
    Approximate the center of the bottom edge along the Y axis for a box aligned with Z up.
    Assumes the object is a box and we want the edge parallel to X at the bottom.
    """
    # Get bounds
    min_y = float('inf')
    max_y = float('-inf')
    center_x = 0
    center_z = 0
    
    # Transform vertices to world space
    for v in obj.data.vertices:
        wv = obj.matrix_world @ v.co
        if wv.y < min_y:
            min_y = wv.y
            # We want the center of the edge, so average X and Z of the bottom face vertices?
            # Simplification: The "bottom edge" usually implies the lowest point.
            # For a box aligned with axes, it's the center of the bottom face.
            # If "bottom_edge" means the edge at the bottom, we need to know which edge.
            # Given the context (roof panels), it likely means the bottom face center or the edge facing the back.
            # Let's assume "bottom_edge" refers to the center of the bottom face for alignment purposes,
            # or specifically the edge closest to the back panel.
            pass
            
    # Let's refine: "Align roof_left.bottom_edge to back.top_edge"
    # Back panel is vertical. Top edge is a line at the top.
    # Roof panel is likely horizontal or angled.
    # If roof_left is rotated +Z to +Y, it's lying on its side?
    # Let's interpret "bottom_edge" as the center of the bottom face for simplicity, 
    # or the specific edge if we can determine it.
    # Given the ambiguity, I will align the center of the bottom face of the roof to the center of the top face of the back.
    
    # Get bounding box in world space
    bbox_corners = [obj.matrix_world @ Vector(corner) for corner in obj.bound_box]
    
    # Find min Z (bottom)
    min_z = min(c.z for c in bbox_corners)
    # Find max Z of back panel (top)
    
    # Center of the bottom face
    bottom_face_centers = [c for c in bbox_corners if abs(c.z - min_z) < 1e-5]
    if bottom_face_centers:
        avg_x = sum(c.x for c in bottom_face_centers) / len(bottom_face_centers)
        avg_y = sum(c.y for c in bottom_face_centers) / len(bottom_face_centers)
        return Vector((avg_x, avg_y, min_z))
    return get_center(obj)

def get_top_edge_center(obj):
    """Get center of the top face of the object."""
    bbox_corners = [obj.matrix_world @ Vector(corner) for corner in obj.bound_box]
    max_z = max(c.z for c in bbox_corners)
    top_face_centers = [c for c in bbox_corners if abs(c.z - max_z) < 1e-5]
    if top_face_centers:
        avg_x = sum(c.x for c in top_face_centers) / len(top_face_centers)
        avg_y = sum(c.y for c in top_face_centers) / len(top_face_centers)
        return Vector((avg_x, avg_y, max_z))
    return get_center(obj)

def boolean_union(objects_to_union, keep_obj_name):
    """
    Perform boolean union on a list of objects.
    Keeps the first object (keep_obj_name) and applies the union modifier.
    """
    target = bpy.data.objects[keep_obj_name]
    
    # Add boolean modifier to target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    
    # We need to iterate through other objects and add them to the modifier
    # Blender's API doesn't allow adding multiple objects to one boolean modifier easily in older versions,
    # but in 3.x we can use a collection or just chain them.
    # A robust way: Create a temporary object that is the union of all others, then union that to target.
    
    # Simpler approach for script:
    # 1. Select all objects to union
    # 2. Use bpy.ops.mesh.boolean_union? No, that's for selected.
    
    # Let's use BMesh for robust boolean operations if modifiers are tricky with multiple objects.
    # However, for this specific task, we can chain modifiers or use a temporary mesh.
    
    # Approach:
    # Create a temporary object that is the union of all 'other' objects.
    # Then boolean union that temp object with the target.
    
    others = [obj for obj in objects_to_union if obj.name != keep_obj_name]
    
    if not others:
        return target
        
    # Merge others into a single mesh
    bpy.ops.object.select_all(action='DESELECT')
    for obj in others:
        obj.select_set(True)
    bpy.context.view_layer.objects.active = others[0]
    bpy.ops.object.join()
    temp_union = bpy.context.active_object
    temp_union.name = "temp_union_others"
    
    # Now boolean union temp_union with target
    mod = target.modifiers.new(name="BooleanUnionFinal", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = temp_union
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Cleanup
    bpy.data.objects.remove(temp_union)
    
    return target

def boolean_subtract(target_obj, cutter_obj):
    """Subtract cutter from target."""
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj)

# --- Main Script ---

# 1. Scene Reset & Units
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)
bpy.context.scene.unit_settings.scale_length = 1

# 2. Materials
mat_wood_brown = make_material("mat_wood_brown", (0.55, 0.27, 0.15, 1.0))

# 3. Geometry Generation

# SECTION 1 – Create back panel
back = add_primitive_cube("back", (0.300, 0.400, 0.015), (0, -0.150, 0.200), (0, 0, 0), mat_wood_brown)
# Rotate back so its +Z axis aligns +Z (world) -> Already aligned by default creation if we don't rotate, 
# but let's be explicit.
align_axis_to_vector(back, 'Z', Vector((0, 0, 1)))
# Align back.center to (0, -0.150, 0.200) -> Done in creation

# SECTION 2 – Create left panel
left = add_primitive_cube("left", (0.200, 0.400, 0.015), (-0.150, 0, 0.200), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(left, 'Z', Vector((0, 0, 1)))

# SECTION 3 – Create right panel
right = add_primitive_cube("right", (0.200, 0.400, 0.015), (0.150, 0, 0.200), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(right, 'Z', Vector((0, 0, 1)))

# SECTION 4 – Create front panel
front = add_primitive_cube("front", (0.300, 0.400, 0.015), (0, 0.150, 0.200), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(front, 'Z', Vector((0, 0, 1)))

# SECTION 5 – Assemble walls union
walls_union = boolean_union([back, left, right, front], "back")
walls_union.name = "walls_union"

# SECTION 6 – Create floor
floor = add_primitive_cube("floor", (0.300, 0.200, 0.015), (0, 0, 0), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(floor, 'Z', Vector((0, 0, 1)))

# SECTION 7 – Create roof left panel
# Size 0.250x0.300x0.015
# Rotate so +Z aligns +Y (world)
# Align roof_left.bottom_edge to back.top_edge
roof_left = add_primitive_cube("roof_left", (0.250, 0.300, 0.015), (0, 0, 0), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(roof_left, 'Z', Vector((0, 1, 0)))

# Align roof_left.bottom_edge to back.top_edge
# Get back top edge center
back_top = get_top_edge_center(back)
# Get roof left bottom edge center (approximated as bottom face center)
roof_left_bottom = get_bottom_edge_center(roof_left)

# Calculate offset to align roof_left_bottom to back_top
offset = back_top - roof_left_bottom
roof_left.location += offset

# SECTION 8 – Create roof right panel
# Size 0.250x0.300x0.015
# Rotate so +Z aligns +Y (world)
# Align roof_right.bottom_edge to back.top_edge
roof_right = add_primitive_cube("roof_right", (0.250, 0.300, 0.015), (0, 0, 0), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(roof_right, 'Z', Vector((0, 1, 0)))

# Align roof_right.bottom_edge to back.top_edge
roof_right_bottom = get_bottom_edge_center(roof_right)
offset_right = back_top - roof_right_bottom
roof_right.location += offset_right

# SECTION 9 – Assemble roof
roof = boolean_union([roof_left, roof_right], "roof_left")
roof.name = "roof"

# SECTION 10 – Create entrance hole
# Cylinder Ø0.050 (R=0.025) x 0.020
# Rotate so +Z aligns +Y (world)
# Align entrance_hole_cut.center to (0, 0.1575, 0.250)
entrance_hole_cut = add_primitive_cylinder("entrance_hole_cut", 0.025, 0.020, (0, 0.1575, 0.250), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(entrance_hole_cut, 'Z', Vector((0, 1, 0)))

# Boolean-subtract entrance_hole_cut from front
boolean_subtract(front, entrance_hole_cut)

# SECTION 11 – Create perch
# Cylinder Ø0.010 (R=0.005) x 0.040
# Rotate so +Z aligns +Y (world)
# Align perch.center to (0, 0.1575, 0.225)
perch = add_primitive_cylinder("perch", 0.005, 0.040, (0, 0.1575, 0.225), (0, 0, 0), mat_wood_brown)
align_axis_to_vector(perch, 'Z', Vector((0, 1, 0)))

# SECTION 12 – Assemble birdhouse
# Boolean-union walls_union, roof, floor, and perch; keep walls_union
birdhouse = boolean_union([walls_union, roof, floor, perch], "walls_union")
birdhouse.name = "birdhouse"

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def add_primitive(name, shape, size, location, rotation_euler):
    """Add a primitive mesh object, set name, location, and rotation."""
    if shape == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'cylinder':
        bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=0.5, depth=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'disc':
        bpy.ops.mesh.primitive_circle_add(vertices=32, radius=0.5, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'cone':
        bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=0.5, radius2=0, depth=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'uvsphere':
        bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=0.5, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    else:
        raise ValueError(f"Unknown shape: {shape}")
    
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def get_bounding_box(obj):
    """Return min and max corners of the object's bounding box in world space."""
    # Ensure matrix is up to date
    obj.update_from_matrix()
    corners = [obj.matrix_world @ Vector(c) for c in obj.bound_box]
    mins = Vector((min(c.x for c in corners), min(c.y for c in corners), min(c.z for c in corners)))
    maxs = Vector((max(c.x for c in corners), max(c.y for c in corners), max(c.z for c in corners)))
    return mins, maxs

def get_face_center(obj, face_type):

def align_object_to_anchor(obj, anchor_obj, anchor_type, target_type, offset_vec):
    """
    Aligns 'obj' to 'anchor_obj' based on specific face/point types.
    Then applies an offset.
    """
    # 1. Get the anchor point on the anchor_obj
    anchor_point = get_face_center(anchor_obj, anchor_type)
    
    # 2. Get the target point on the object being moved (obj)
    # We want to move obj such that its 'target_type' coincides with 'anchor_point'
    target_point_local = get_face_center(obj, target_type)
    
    # 3. Calculate the translation vector
    # We need to move obj so that target_point_local moves to anchor_point
    # Translation = anchor_point - target_point_local
    translation = anchor_point - target_point_local
    
    # 4. Apply translation
    obj.location += translation
    
    # 5. Apply offset
    # Offset is in reference local axes. Assuming standard orientation, local axes align with world.
    obj.location += offset_vec
    
    return obj

def apply_material(obj, material):
    """Apply material to object."""
    if obj.data.materials:
        obj.data.materials[0] = material
    else:
        obj.data.materials.append(material)

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler):
    """Add a cube primitive, scale it, and set transform."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    obj.data.update()
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    obj.data.update()
    return obj

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def align_face_to_face(obj_align, obj_target, align_axis='Z', offset_axis='Z', offset_val=0):
    """
    Align the bottom face of obj_align to the top face of obj_target along the specified axis.
    Assumes objects are axis-aligned.
    """
    # Get bounding box centers and extents
    min_a, max_a = obj_align.bound_box[0], obj_align.bound_box[6]
    min_t, max_t = obj_target.bound_box[0], obj_target.bound_box[6]
    
    # Convert to world coordinates
    # Note: bound_box is local. We need world space for accurate alignment if rotated, 
    # but here we assume axis-aligned for simplicity in this specific script structure.
    # However, to be robust, we use world coordinates.
    
    # Helper to get world bounds
    def get_world_bounds(ob):
        bm = bmesh.new()
        bm.from_mesh(ob.data)
        bm.transform(ob.matrix_world)
        min_w = Vector((float('inf'), float('inf'), float('inf')))
        max_w = Vector((float('-inf'), float('-inf'), float('-inf')))
        for v in bm.verts:
            min_w.x = min(min_w.x, v.co.x)
            min_w.y = min(min_w.y, v.co.y)
            min_w.z = min(min_w.z, v.co.z)
            max_w.x = max(max_w.x, v.co.x)
            max_w.y = max(max_w.y, v.co.y)
            max_w.z = max(max_w.z, v.co.z)
        bm.free()
        return min_w, max_w

    min_a_w, max_a_w = get_world_bounds(obj_align)
    min_t_w, max_t_w = get_world_bounds(obj_target)

    # Determine target position for the aligning face
    # If aligning bottom of align_obj to top of target_obj:
    # Target Y (or Z) = max_t_w[axis]
    # Current Bottom Y (or Z) of align_obj = min_a_w[axis]
    # Delta = Target - Current
    
    if align_axis == 'X':
        target_pos = max_t_w.x
        current_pos = min_a_w.x
        delta = target_pos - current_pos
        obj_align.location.x += delta
    elif align_axis == 'Y':
        target_pos = max_t_w.y
        current_pos = min_a_w.y
        delta = target_pos - current_pos
        obj_align.location.y += delta
    elif align_axis == 'Z':
        target_pos = max_t_w.z
        current_pos = min_a_w.z
        delta = target_pos - current_pos
        obj_align.location.z += delta

    # Apply offset
    if offset_axis == 'X':
        obj_align.location.x += offset_val
    elif offset_axis == 'Y':
        obj_align.location.y += offset_val
    elif offset_axis == 'Z':
        obj_align.location.z += offset_val

def align_center_to_avg_faces(list_of_face_refs):
    """
    Align obj_align center to the average of multiple face centers.
    list_of_face_refs: list of tuples (obj, face_descriptor)
    face_descriptor: 'top', 'bottom', 'front', 'back', 'left', 'right'
    """
    avg_loc = Vector((0,0,0))
    count = 0
    
    for obj, face_desc in list_of_face_refs:
        bm = bmesh.new()
        bm.from_mesh(obj.data)
        bm.transform(obj.matrix_world)
        
        face_verts = []
        for v in bm.verts:
            if face_desc == 'top' and v.co.z > 0.001: face_verts.append(v)
            elif face_desc == 'bottom' and v.co.z < -0.001: face_verts.append(v)
            elif face_desc == 'front' and v.co.y > 0.001: face_verts.append(v)
            elif face_desc == 'back' and v.co.y < -0.001: face_verts.append(v)
            elif face_desc == 'right' and v.co.x > 0.001: face_verts.append(v)
            elif face_desc == 'left' and v.co.x < -0.001: face_verts.append(v)
        
        bm.free()
        if face_verts:
            center = Vector((0,0,0))
            for v in face_verts:
                center += v.co
            avg_loc += center / len(face_verts)
            count += 1
            
    if count > 0:
        avg_loc /= count
        return avg_loc
    return Vector((0,0,0))

def bevel_object(obj, radius, segments):
    """Add a bevel modifier and apply it."""
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='OBJECT')
    
    # Add bevel modifier
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE' # or 'WEIGHT', 'WIDTH' depending on desired behavior. Angle is often safer for uniform bevels.
    mod.bevel_angle = math.radians(45) # Default angle
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)

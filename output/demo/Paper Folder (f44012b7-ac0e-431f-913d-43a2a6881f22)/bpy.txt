def make_material(name, diffuse_color=(1, 1, 1, 1)):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with the given world vector."""
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else: # Z
        local_axis = Vector((0, 0, 1))
    
    target = vector.normalized()
    if local_axis.normalized().dot(target) < 0.999999:
        rot_quat = local_axis.normalized().rotation_difference(target)
        obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        self.update_matrix()

    def update_matrix(self):
        self.matrix_world = self.obj.matrix_world.copy()
        self.center = self.matrix_world.to_translation()

    def center_world(self):
        return self.center

    def face_center_world(self, face_index):
        """Get the center of a specific face in world space."""
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bm.transform(self.matrix_world)
        faces = bm.faces
        if face_index < len(faces):
            face = faces[face_index]
            center = average([v.co for v in face.verts])
            bm.free()
            return center
        bm.free()
        return self.center

def boolean_subtract(cutter_obj, target_obj, name="bool_sub"):
    """Perform boolean subtract using modifier stack."""
    # Ensure target is active for modifier application if needed, 
    # but we'll apply it directly to target.
    
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name=name, type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=name)
    
    # Remove Cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(target_obj, other_obj, name="bool_union"):

def add_bevel(obj, radius=0.01, segments=4, limit_method='ANGLE'):
    """Add bevel to object edges."""
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    
    # Bevel modifier is often easier for non-destructive, but prompt asks for bevel operation.
    # Using modifier for precision and ease of cleanup.
    bpy.ops.object.mode_set(mode='OBJECT')
    
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = limit_method
    mod.profile_type = 'SHAPE'
    mod.angle_limit = math.radians(30) # Default angle limit
    
    # Apply bevel
    bpy.ops.object.modifier_apply(modifier=mod.name)

def get_face_center(obj, face_type):
    """
    Helper to get world location of a specific face center based on type.
    Assumes standard cube orientation before rotation.
    """
    # We need to calculate the face center in local space, then transform to world
    # Cube vertices are at +/- 0.5 * scale
    scale = obj.scale
    # Local face centers for a cube centered at 0,0,0
    # X+ face: (0.5, 0, 0)
    # X- face: (-0.5, 0, 0)
    # Y+ face: (0, 0.5, 0)
    # Y- face: (0, -0.5, 0)
    # Z+ face: (0, 0, 0.5)
    # Z- face: (0, 0, -0.5)
    
    local_centers = {
        'x_plus': Vector((0.5, 0, 0)),
        'x_minus': Vector((-0.5, 0, 0)),
        'y_plus': Vector((0, 0.5, 0)),
        'y_minus': Vector((0, -0.5, 0)),
        'z_plus': Vector((0, 0, 0.5)),
        'z_minus': Vector((0, 0, -0.5))
    }
    
    if face_type not in local_centers:
        return obj.location
        
    local_center = local_centers[face_type] * Vector(scale)
    world_center = obj.matrix_world @ local_center
    return world_center

def anchor_and_offset(obj, anchor_obj, anchor_face_type, offset=(0,0,0)):
    """
    Anchors obj to a specific face of anchor_obj, then applies offset.
    Anchor face types: left_face, right_face, bottom_face, top_face, section_1.top_face, etc.
    Note: 'section_X.top_face' implies a sub-part or specific face. 
    For a simple cube, we map:
    left_face -> x_minus
    right_face -> x_plus
    bottom_face -> z_minus (assuming Z is up)
    top_face -> z_plus
    """
    # Determine local face center on anchor_obj
    # Map names to standard cube faces
    face_map = {
        'left_face': 'x_minus',
        'right_face': 'x_plus',
        'bottom_face': 'z_minus',
        'top_face': 'z_plus'
    }
    
    # Handle nested names like "folder_base_body.section_1.top_face"
    # If the anchor_obj is the main body, we look at its faces.
    # If the instruction implies a specific sub-region, we might need to approximate 
    # or assume the face type refers to the main object's geometry relative to the cut.
    # Given the complexity, we will assume standard cube faces for the base body.
    
    # Extract the last part of the anchor name for face type if it contains dots
    # e.g. "folder_base_body.section_1.top_face" -> "top_face"
    parts = anchor_face_type.split('.')
    face_key = parts[-1] if parts else anchor_face_type
    
    local_face = face_map.get(face_key, 'z_plus') # Default to top
    
    # Get world position of the anchor face
    anchor_world_pos = get_face_center(anchor_obj, local_face)
    
    # Get world rotation of the anchor object to orient the new object
    anchor_rot = anchor_obj.rotation_euler.copy()
    
    # Set obj location to anchor face center
    obj.location = anchor_world_pos
    
    # Set obj rotation to match anchor
    obj.rotation_euler = anchor_rot
    
    # Apply offset in local space of the anchored object
    # Convert local offset to world offset
    offset_vec = Vector(offset)
    # Rotate offset by anchor's rotation
    offset_world = anchor_rot.to_matrix() * offset_vec # Approximate for Euler
    # Better: use matrix rotation
    offset_world = anchor_obj.matrix_world.to_3x3() * offset_vec
    
    obj.location += offset_world

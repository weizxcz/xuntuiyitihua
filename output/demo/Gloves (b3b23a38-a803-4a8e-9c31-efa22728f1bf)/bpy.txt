def clear_scene():
    """Delete all existing objects in the scene."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for collection in bpy.data.collections:
        collection.objects.clear()

def set_units_metres():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_torus(name, major_radius, minor_radius, location=(0,0,0), rotation=(0,0,0)):
    """Add a torus primitive."""
    bpy.ops.mesh.primitive_torus_add(major_radius=major_radius, minor_radius=minor_radius, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, target_vector):
    """
    Orient an object so its local +Z axis aligns with the target_vector.
    Uses quaternion rotation.
    """
    # Current up vector is (0,0,1)
    current_up = Vector((0, 0, 1))
    target_up = target_vector.normalized()
    
    # Calculate rotation quaternion
    rot_quat = current_up.rotation_difference(target_up)
    
    # Apply rotation
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = rot_quat

def get_face_center(obj, face_type='center'):
    """
    Get the world location of a specific face or center of an object.
    face_type: 'center', 'top_face', 'bottom_face', 'left_face', 'right_face'
    Assumes standard orientation: Z is up, Y is forward/back, X is left/right.
    """
    # We need to calculate the face center in world space based on the object's transform
    # Since we are building primitives, we can calculate relative to the object's local bounds
    # and then transform to world.
    
    # Get object's world matrix
    wm = obj.matrix_world
    
    # Get local bounds
    min_x, min_y, min_z = obj.bound_box[0]
    max_x, max_y, max_z = obj.bound_box[6]
    
    # Calculate local face centers
    if face_type == 'center':
        local_center = Vector((0, 0, 0))
    elif face_type == 'top_face':
        # Assuming Z is up
        local_center = Vector((0, 0, max_z))
    elif face_type == 'bottom_face':
        local_center = Vector((0, 0, min_z))
    elif face_type == 'left_face':
        # Assuming X is right, so left is min_x
        local_center = Vector((min_x, 0, 0))
    elif face_type == 'right_face':
        local_center = Vector((max_x, 0, 0))
    elif face_type == 'front_face':
        local_center = Vector((0, max_y, 0))
    elif face_type == 'back_face':
        local_center = Vector((0, min_y, 0))
    else:
        local_center = Vector((0, 0, 0))
        
    # Transform to world space
    world_center = wm @ local_center
    return world_center

def align_to_face(obj_target, obj_source, face_type):
    """
    Align obj_source so its center matches the specified face of obj_target.
    """
    target_loc = get_face_center(obj_target, face_type)
    obj_source.location = target_loc

def boolean_union(objects, name):
    """
    Perform boolean union on a list of objects and return the resulting object.
    """
    if not objects:
        return None
    
    # Select all objects
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
        
    # Apply modifiers if any (clean up)
    # Note: In a script, it's safer to use BMesh for boolean operations to avoid dependency on modifiers
    # However, for simplicity and standard Blender API usage, we can use the boolean modifier approach
    # or BMesh. Given the strict "self-contained" rule, BMesh is more robust without UI interaction.
    
    # Let's use BMesh for robust boolean union
    bm = bmesh.new()
    
    # We need to collect all meshes
    # First, ensure all objects are in edit mode or use their data directly
    # BMesh.from_mesh works on the mesh data.
    
    # To handle transforms correctly, we must apply them or transform the mesh data.
    # Applying transforms is easier for this script.
    
    applied_objects = []
    for obj in objects:
        # Apply transforms
        bpy.context.view_layer.objects.active = obj
        bpy.ops.object.transform_apply(location=True, rotation=True, scale=True)
        mesh = obj.data.copy()
        bm.from_mesh(mesh)
        applied_objects.append(obj)
        
    # Merge all vertices
    bmesh.ops.connect_verts_concave(bm)
    
    # Create new object from BMesh
    new_mesh = bpy.data.meshes.new(name + "_mesh")
    bm.to_mesh(new_mesh)
    bm.free()
    
    new_obj = bpy.data.objects.new(name, new_mesh)
    bpy.context.collection.objects.link(new_obj)
    
    # Delete original objects
    for obj in applied_objects:
        bpy.context.collection.objects.unlink(obj)
        bpy.data.objects.remove(obj)
        
    return new_obj

def average_vectors(vectors):
    """Calculate the average of a list of vectors."""
    if not vectors:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for v in vectors:
        avg += v
    return avg / len(vectors)

def make_material(name, diffuse_color, emission_color=(0,0,0,1), emission_strength=0.0):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Emission'].default_value = emission_color
    bsdf.inputs['Emission Strength'].default_value = emission_strength
    return mat

def add_primitive(name, shape, size, location, rotation_euler):
    """Add a primitive mesh object and apply transforms."""
    bpy.ops.object.select_all(action='DESELECT')
    
    if shape == 'CUBE':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'CYLINDER':
        bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=0.5, depth=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'DISC':
        bpy.ops.mesh.primitive_circle_add(vertices=32, radius=0.5, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
        # Fill it
        bpy.ops.object.mode_set(mode='EDIT')
        bpy.ops.mesh.select_all(action='SELECT')
        bpy.ops.mesh.fill()
        bpy.ops.object.mode_set(mode='OBJECT')
    elif shape == 'CONE':
        bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=0.5, radius2=0, depth=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'UVSPHERE':
        bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=0.5, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    else:
        raise ValueError(f"Unknown shape: {shape}")

    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def align_face_to_face(obj_target, obj_source, target_axis='Z', source_axis='-Z', offset=0):
    """
    Aligns the source object's face to the target object's face.
    target_axis: The axis of the target object that defines the face normal (e.g., 'Z' for top).
    source_axis: The axis of the source object that defines the face normal (e.g., '-Z' for bottom).
    offset: Distance to separate them after alignment.
    """
    # Get world matrices
    mat_target = obj_target.matrix_world
    mat_source = obj_source.matrix_world
    
    # Calculate face centers in world space
    # Assuming objects are axis-aligned for simplicity in this specific script context,
    # but we use matrix_world to be robust.
    
    # Target face center
    if target_axis == 'Z':
        target_normal = mat_target.to_3x3() @ Vector((0, 0, 1))
        target_offset_vec = mat_target.to_3x3() @ Vector((0, 0, 0.5 * obj_target.dimensions.z))
    elif target_axis == '-Z':
        target_normal = mat_target.to_3x3() @ Vector((0, 0, -1))
        target_offset_vec = mat_target.to_3x3() @ Vector((0, 0, -0.5 * obj_target.dimensions.z))
    elif target_axis == 'X':
        target_normal = mat_target.to_3x3() @ Vector((1, 0, 0))
        target_offset_vec = mat_target.to_3x3() @ Vector((0.5 * obj_target.dimensions.x, 0, 0))
    elif target_axis == '-X':
        target_normal = mat_target.to_3x3() @ Vector((-1, 0, 0))
        target_offset_vec = mat_target.to_3x3() @ Vector((-0.5 * obj_target.dimensions.x, 0, 0))
    elif target_axis == 'Y':
        target_normal = mat_target.to_3x3() @ Vector((0, 1, 0))
        target_offset_vec = mat_target.to_3x3() @ Vector((0, 0.5 * obj_target.dimensions.y, 0))
    elif target_axis == '-Y':
        target_normal = mat_target.to_3x3() @ Vector((0, -1, 0))
        target_offset_vec = mat_target.to_3x3() @ Vector((0, -0.5 * obj_target.dimensions.y, 0))
    else:
        raise ValueError(f"Invalid target axis: {target_axis}")
        
    target_center = mat_target.translation + target_offset_vec
    
    # Source face center (local origin to face)
    if source_axis == 'Z':
        source_offset_vec = mat_source.to_3x3() @ Vector((0, 0, 0.5 * obj_source.dimensions.z))
    elif source_axis == '-Z':
        source_offset_vec = mat_source.to_3x3() @ Vector((0, 0, -0.5 * obj_source.dimensions.z))
    elif source_axis == 'X':
        source_offset_vec = mat_source.to_3x3() @ Vector((0.5 * obj_source.dimensions.x, 0, 0))
    elif source_axis == '-X':
        source_offset_vec = mat_source.to_3x3() @ Vector((-0.5 * obj_source.dimensions.x, 0, 0))
    elif source_axis == 'Y':
        source_offset_vec = mat_source.to_3x3() @ Vector((0, 0.5 * obj_source.dimensions.y, 0))
    elif source_axis == '-Y':
        source_offset_vec = mat_source.to_3x3() @ Vector((0, -0.5 * obj_source.dimensions.y, 0))
    else:
        raise ValueError(f"Invalid source axis: {source_axis}")
        
    source_center_local = source_offset_vec
    source_center_world = mat_source.translation + source_center_local
    
    # Calculate translation needed to align centers
    # We want source_center_world_new = target_center_world
    # current_source_center_world = mat_source.translation + source_center_local
    # delta = target_center - current_source_center_world
    
    # However, we also need to align orientations if they aren't already aligned.
    # In this specific script, all objects are aligned to Z-up, so rotation alignment is trivial (identity).
    # We just need to move the source object.
    
    # Align positions
    delta = target_center - source_center_world
    
    # Apply offset in the direction of the normal
    # If we want them touching, offset=0. If we want separation, add offset * normal.
    # The prompt says "offset(0,0,0.002)" which usually implies moving the source object further away or closer.
    # "Align ... then offset" usually means: Place touching, then move source by offset vector in its local space or global?
    # Given the context of "screen to bezel", it's likely a small gap or overlap.
    # Let's assume the offset is applied to the source object's position after alignment.
    
    # Simple approach: Move source object so its face center matches target face center.
    obj_source.location += delta
    
    # Apply additional offset
    # The prompt says "offset(0,0,0.002)". This is likely in local coordinates of the source object or global.
    # Given the objects are axis aligned, global Z is local Z.
    obj_source.location += Vector((0, 0, offset))

def align_center_to_face_center(obj_target, obj_source, target_axis='Z', offset=0):
    """
    Aligns the center of the source object to the center of the target's face.
    """
    mat_target = obj_target.matrix_world
    mat_source = obj_source.matrix_world
    
    # Target face center
    if target_axis == 'Z':
        target_offset_vec = mat_target.to_3x3() @ Vector((0, 0, 0.5 * obj_target.dimensions.z))
    elif target_axis == '-Z':
        target_offset_vec = mat_target.to_3x3() @ Vector((0, 0, -0.5 * obj_target.dimensions.z))
    elif target_axis == 'X':
        target_offset_vec = mat_target.to_3x3() @ Vector((0.5 * obj_target.dimensions.x, 0, 0))
    elif target_axis == '-X':
        target_offset_vec = mat_target.to_3x3() @ Vector((-0.5 * obj_target.dimensions.x, 0, 0))
    elif target_axis == 'Y':
        target_offset_vec = mat_target.to_3x3() @ Vector((0, 0.5 * obj_target.dimensions.y, 0))
    elif target_axis == '-Y':
        target_offset_vec = mat_target.to_3x3() @ Vector((0, -0.5 * obj_target.dimensions.y, 0))
    else:
        raise ValueError(f"Invalid target axis: {target_axis}")
        
    target_center = mat_target.translation + target_offset_vec
    
    # Source center is just its location
    source_center_world = mat_source.translation
    
    delta = target_center - source_center_world
    obj_source.location += delta
    
    # Apply offset
    obj_source.location += Vector((0, 0, offset))

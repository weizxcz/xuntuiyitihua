import bpy
import bmesh
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation_euler
    return obj

def get_face_normal(obj, face_index):
    """Returns the world-space normal of a specific face."""
    obj.update_from_repr() # Ensure data is up to date
    matrix_world = obj.matrix_world
    matrix_local_inv = obj.matrix_world.inverted()
    
    # Get vertices of the face
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.verts.ensure_lookup_table()
    face = bm.faces[face_index]
    
    # Calculate center and normal in local space
    center_local = sum((v.co for v in face.verts), Vector()) / len(face.verts)
    normal_local = face.normal
    
    # Transform to world space
    center_world = matrix_world @ center_local
    normal_world = matrix_world.to_3x3() @ normal_local
    
    bm.free()
    return center_world, normal_world

def align_xy_faces(obj_target, obj_source):
    """
    Aligns the XY faces (centers in XY plane) of two objects.
    Assumes both are already oriented with +Z up.
    """
    # Get centers
    target_center = obj_target.location
    source_center = obj_source.location
    
    # Offset source to match target XY
    delta_x = target_center[0] - source_center[0]
    delta_y = target_center[1] - source_center[1]
    
    obj_source.location += Vector((delta_x, delta_y, 0))

def apply_offset(obj, offset_vector):
    """Applies a local offset to the object."""
    # Since objects are aligned to world axes, local offset == world offset
    obj.location += offset_vector

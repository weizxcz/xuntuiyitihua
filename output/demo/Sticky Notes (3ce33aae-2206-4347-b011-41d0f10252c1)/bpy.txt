def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube, scale it, set location/rotation, and assign material."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        """Return the world-space center of the object's bounding box."""
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bm.transform(self.obj.matrix_world)
        center = average([v.co for v in bm.verts])
        bm.free()
        del bm
        return center

    def face_center_world(self, face_normal_axis='Z', face_side='min'):
        """
        Return the world-space center of a specific face.
        face_normal_axis: 'X', 'Y', or 'Z'
        face_side: 'min' or 'max'
        """
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bm.transform(self.obj.matrix_world)
        
        # Find faces with normal aligned to axis
        candidates = []
        for f in bm.faces:
            if abs(f.normal[0] - (1 if face_normal_axis=='X' else 0)) < 0.1 and abs(f.normal[1] - (1 if face_normal_axis=='Y' else 0)) < 0.1 and abs(f.normal[2] - (1 if face_normal_axis=='Z' else 0)) < 0.1:
                candidates.append(f)
            elif abs(f.normal[0] - (-1 if face_normal_axis=='X' else 0)) < 0.1 and abs(f.normal[1] - (-1 if face_normal_axis=='Y' else 0)) < 0.1 and abs(f.normal[2] - (-1 if face_normal_axis=='Z' else 0)) < 0.1:
                candidates.append(f)
        
        if not candidates:
            # Fallback: use bounding box min/max
            bbox = self.obj.bound_box
            # bbox is in local space, need to transform? 
            # Actually bound_box is in local space. Let's use world space vertices for robustness.
            verts = [v.co for v in bm.verts]
            if not verts:
                return Vector((0,0,0))
            
            axis_idx = 0 if face_normal_axis == 'X' else (1 if face_normal_axis == 'Y' else 2)
            vals = [v[axis_idx] for v in verts]
            target_val = min(vals) if face_side == 'min' else max(vals)
            
            # Find verts close to this value
            close_verts = [v for v in verts if abs(v[axis_idx] - target_val) < 0.0001]
            if close_verts:
                return average(close_verts)
            return Vector((0,0,0))

        # If we found faces, pick the one that matches the side
        # Calculate face centers in world space
        face_centers = []
        for f in candidates:
            fc = average([v.co for v in f.verts])
            face_centers.append(fc)
        
        if not face_centers:
            return Vector((0,0,0))
            
        axis_idx = 0 if face_normal_axis == 'X' else (1 if face_normal_axis == 'Y' else 2)
        
        if face_side == 'min':
            return min(face_centers, key=lambda c: c[axis_idx])
        else:
            return max(face_centers, key=lambda c: c[axis_idx])

    def radial_point_world(self, axis='Z', side='max'):
        """Return a point on the radial edge."""
        # Simplified: return center + radius in direction
        center = self.center_world()
        # This is complex for arbitrary shapes, assuming simple primitives for now
        # For this script, we rely on bounding box logic or specific face logic
        return center

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using modifier."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Cleanup cutter if desired, or keep it. Usually keep for debugging or remove.
    # Let's remove the cutter object to keep scene clean
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_objects, keep_name):
    """
    Perform boolean union on a list of objects.
    Keeps the object named keep_name as the result.
    """
    if not union_objects:
        return
    
    base_name = keep_name
    base_obj = bpy.data.objects[base_name]
    bpy.context.view_layer.objects.active = base_obj
    
    # We need to union all other objects into base_obj
    # Blender's boolean union modifier works by adding a target object.
    # We can chain them or add them one by one.
    # To avoid modifier stack issues, we can apply them sequentially.
    
    for obj_name in union_objects:
        if obj_name == base_name:
            continue
        other_obj = bpy.data.objects[obj_name]
        
        # Add boolean union modifier to base
        mod = base_obj.modifiers.new(name=f"BooleanUnion_{obj_name}", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = other_obj
        
        # Apply
        bpy.ops.object.modifier_apply(modifier=mod.name)
        
        # Remove the other object
        bpy.data.objects.remove(other_obj, do_unlink=True)

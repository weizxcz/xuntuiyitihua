def clear_scene():
    """Delete all existing objects to start from a clean scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Ensure we have a default cube or empty if needed, but usually clean is fine
    # Re-select active object if any remains (none in clean scene)
    bpy.context.view_layer.objects.active = None

def set_units_meters():
    """Set the length unit to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    # Set roughness/metallic defaults if needed, but prompt only specifies diffuse
    return mat

def add_primitive(name, shape, size=None, radius=None, height=None, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive object and return it."""
    bpy.ops.object.select_all(action='DESELECT')
    obj = None
    if shape == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=size[0], location=location)
        obj = bpy.context.active_object
        obj.scale = (size[1], size[2], size[3]) # X, Y, Z
    elif shape == 'cylinder':
        bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=location)
        obj = bpy.context.active_object
    elif shape == 'disc':
        bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='NGON', location=location)
        obj = bpy.context.active_object
        obj.data.name = name + "_mesh"
        obj.name = name
    elif shape == 'cone':
        bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=height, location=location)
        obj = bpy.context.active_object
    elif shape == 'sphere':
        bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location)
        obj = bpy.context.active_object
    
    if obj:
        obj.name = name
        obj.rotation_euler = rotation
        # Ensure it's in the correct layer/group
        bpy.context.view_layer.objects.active = obj
        return obj
    return None

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Align an object's local axis to a target world vector."""
    # Get current rotation
    rot = obj.rotation_euler.copy()
    
    # Define basis vectors
    if axis_to_align == 'X':
        basis = Vector((1, 0, 0))
    elif axis_to_align == 'Y':
        basis = Vector((0, 1, 0))
    elif axis_to_align == 'Z':
        basis = Vector((0, 0, 1))
    else:
        return

    # Calculate rotation matrix to align basis to target
    # We need to align basis to target_vector
    rot_mat = basis.rotation_difference(target_vector).to_matrix().to_4x4()
    
    # Convert to Euler and apply
    # Note: This is a simplified alignment. For complex cases, quaternions are better.
    # Here we assume simple axis alignment.
    new_rot = rot_mat.to_euler()
    obj.rotation_euler = new_rot

def get_face_normal(obj, face_type):
    """
    Returns the world direction vector for a specific face of an object.
    face_type: 'XY_face' (Top/Bottom), 'XZ_face' (Front/Back), 'YZ_face' (Left/Right)
    Note: This assumes the object is axis-aligned or we are defining faces relative to local space transformed to world.
    For simplicity in this script, we assume standard orientations where:
    - XY_face normal is Z
    - XZ_face normal is Y
    - YZ_face normal is X
    """
    # Since we explicitly rotate objects to align axes, we can infer normals based on local axes in world space
    # However, 'face' usually implies the geometric face.
    # Let's use the object's matrix to transform local face normals to world space.
    
    # Local normals
    if face_type == 'XY_face':
        # Top face normal is +Z, Bottom is -Z. Usually 'align to face' implies the plane normal.
        # Let's assume we want the normal of the face.
        local_normal = Vector((0, 0, 1)) 
    elif face_type == 'XZ_face':
        local_normal = Vector((0, 1, 0))
    elif face_type == 'YZ_face':
        local_normal = Vector((1, 0, 0))
    else:
        return Vector((0,0,0))
        
    # Transform to world space
    world_normal = local_normal @ obj.matrix_world.to_3x3()
    return world_normal.normalized()

def get_face_center(obj, face_type):
    """Get the world location of the center of a specific face."""
    # Get local center of the face
    # For a cube/cylinder/cone, the face center is at the edge of the bounding box in local space
    # We need to know the dimensions.
    # This is tricky for generic objects. 
    # Alternative: Use the object's center and offset by half-dimensions.
    
    # Let's approximate by getting the bounding box or using known primitives.
    # Since we create primitives, we can store their dimensions or calculate them.
    # For robustness, let's use the object's matrix_world and local extents.
    
    # Local extents
    if obj.type == 'MESH':
        # Get bounding box
        bbox = [obj.matrix_world @ v for v in obj.bound_box]
        # Find min/max for each axis
        mins = [min(v[i] for v in bbox) for i in range(3)]
        maxs = [max(v[i] for v in bbox) for i in range(3)]
        
        # Define face centers in world space
        if face_type == 'XY_face':
            # Top face (max Z)
            return Vector([(mins[0]+maxs[0])/2, (mins[1]+maxs[1])/2, maxs[2]])
        elif face_type == 'XZ_face':
            # Front face (max Y)
            return Vector([(mins[0]+maxs[0])/2, maxs[1], (mins[2]+maxs[2])/2])
        elif face_type == 'YZ_face':
            # Right face (max X)
            return Vector([maxs[0], (mins[1]+maxs[1])/2, (mins[2]+maxs[2])/2])
            
    return obj.matrix_world.to_translation()

def get_object_center(obj):
    """Get the world center of an object."""
    return obj.matrix_world.to_translation()

def align_face_to_face(obj_source, source_face, obj_target, target_face):
    """
    Aligns the source object's face to the target object's face.
    This involves:
    1. Rotating source so its face normal matches target face normal.
    2. Positioning source so its face center matches target face center.
    """
    # 1. Rotation
    # Get normals
    # Note: 'align to face' usually means the planes are parallel and normals are aligned.
    # We need to determine if we want +normal or -normal. 
    # Standard convention: Align the face's outward normal.
    
    # Get target normal
    target_normal = get_face_normal(obj_target, target_face)
    
    # Get source normal (local)
    if source_face == 'XY_face':
        src_local = Vector((0,0,1))
    elif source_face == 'XZ_face':
        src_local = Vector((0,1,0))
    elif source_face == 'YZ_face':
        src_local = Vector((1,0,0))
    else:
        src_local = Vector((0,0,1))
        
    # We want to rotate src_local to align with target_normal
    # But we must preserve other axes if possible.
    # Simple approach: Align Z to target_normal, then adjust X/Y.
    
    # Let's use a simpler method: 
    # Align the object's local Z to the target normal if source is XY_face (normal Z)
    # Align local Y to target normal if source is XZ_face (normal Y)
    # Align local X to target normal if source is YZ_face (normal X)
    
    rot_quat = src_local.rotation_difference(target_normal)
    
    # Apply rotation relative to current orientation? 
    # The prompt says "Rotate ... so its +Z axis aligns +Z". 
    # Then "Align ... face to ... face".
    # This implies we set a base rotation first, then adjust for alignment.
    
    # Let's apply the rotation difference to the current rotation
    current_rot_quat = obj_source.rotation_euler.to_quaternion()
    new_rot_quat = rot_quat * current_rot_quat
    obj_source.rotation_euler = new_rot_quat.to_euler()
    
    # 2. Positioning
    # Move source so its face center matches target face center
    target_center = get_face_center(obj_target, target_face)
    source_center = get_face_center(obj_source, source_face)
    
    # Calculate offset
    offset = target_center - source_center
    obj_source.location += offset

def align_to_center(obj_source, obj_target):
    """Align center of source to center of target."""
    target_center = get_object_center(obj_target)
    source_center = get_object_center(obj_source)
    obj_source.location += (target_center - source_center)

def apply_offset(obj, offset_xyz):
    """Apply an offset to the object's location in world space."""
    obj.location += Vector(offset_xyz)

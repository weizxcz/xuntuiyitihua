import bpy
import bmesh
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color=(0.8, 0.8, 0.8, 1.0)):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius=1, depth=2, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size=1):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract using the modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target has a modifier slot
    mod = target.modifiers.new(name="BoolSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply immediately to keep scene clean and geometry updated
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Discard cutter
    bpy.data.objects.remove(cutter, do_unlink=True)
    return target

def add_bevel(obj, radius=0.01, segments=8):
    """Add a bevel modifier and apply it."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width_type = 'WIDTH'
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder oriented along Z."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube centered at origin."""
    # size is (x, y, z)
    sx, sy, sz = size
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0)) # size=1 creates 2x2x2 cube
    obj = bpy.context.active_object
    obj.scale = (sx/2, sy/2, sz/2)
    obj.name = name
    return obj

def add_torus(name, major_radius, minor_radius, segments=32):
    """Add a torus oriented along Z."""
    bpy.ops.mesh.primitive_torus_add(major_radius=major_radius, minor_radius=minor_radius, major_segments=segments, minor_segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0, 0, 1)):

def get_face_normal(obj, face_type):
    """Get the normal vector of a specific face of an object."""
    # This is a simplified helper assuming axis-aligned objects
    # face_type: 'bottom', 'top', 'front', 'back', 'left', 'right'
    # Returns world vector
    if face_type == 'bottom': return (0, 0, -1)
    if face_type == 'top': return (0, 0, 1)
    if face_type == 'front': return (0, 1, 0)
    if face_type == 'back': return (0, -1, 0)
    if face_type == 'left': return (-1, 0, 0)
    if face_type == 'right': return (1, 0, 0)
    return (0, 0, 1)

def get_face_center(obj, face_type):
    """Get the world location of the center of a specific face."""
    # Calculate based on object location and dimensions
    loc = obj.location
    dims = obj.dimensions
    # Note: dimensions are always positive, representing total size
    
    x, y, z = loc
    dx, dy, dz = dims[0]/2, dims[1]/2, dims[2]/2
    
    # Adjust for rotation if necessary, but for now assume axis-aligned for simplicity in anchoring logic
    # If rotated, we'd need matrix multiplication. 
    # The prompt implies simple alignments, so we assume standard orientations or handle rotation explicitly.
    
    # For robustness, let's use the object's matrix_world to transform local face centers
    # Local face centers for a cube/cylinder/torus centered at 0,0,0:
    
    if face_type == 'bottom':
        local_pt = Vector((0, 0, -dz))
    elif face_type == 'top':
        local_pt = Vector((0, 0, dz))
    elif face_type == 'front':
        local_pt = Vector((0, dy, 0))
    elif face_type == 'back':
        local_pt = Vector((0, -dy, 0))
    elif face_type == 'left':
        local_pt = Vector((-dx, 0, 0))
    elif face_type == 'right':
        local_pt = Vector((dx, 0, 0))
    else:
        return loc

    # Transform to world
    world_pt = obj.matrix_world @ local_pt
    return world_pt

def align_faces(obj_target, target_face, obj_source, source_face):
    """Align source object's face normal to target object's face normal and position them."""
    # 1. Align Normals
    target_normal = get_face_normal(obj_target, target_face)
    source_normal = get_face_normal(obj_source, source_face)
    
    # Calculate rotation to align source_normal to target_normal
    # We need to rotate the source object so its source_face normal points in target_normal direction
    # This is complex if objects are already rotated. 
    # Simplified: Reset rotation, align, then re-apply? No, we set rotation explicitly in steps.
    
    # Let's assume the objects are oriented such that their local axes align with world axes initially,
    # or we handle the rotation via the 'Rotate' step provided in the prompt.
    # The prompt says "Align ... to ...". This usually means matching normals.
    
    # To align two vectors v1 and v2:
    v1 = Vector(source_normal)
    v2 = Vector(target_normal)
    
    if v1.normalized().dot(v2.normalized()) < -0.99:
        # Opposite
        rot_quat = Matrix.Rotation(pi, 4, v1.cross(v2).normalized() if v1.cross(v2).length > 0 else Vector((1,0,0)))
    else:
        rot_quat = v1.rotation_difference(v2)
    
    # Apply rotation to source object
    # Note: This might conflict with previous orientation steps if not careful.
    # The prompt structure is: Rotate -> Anchor -> Align -> Offset.
    # So we apply this rotation on top of the existing orientation.
    obj_source.rotation_euler = rot_quat.to_euler()

def add_bevel(obj, radius, segments):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    mod.angle_limit = radians(30) # Default
    # Apply bevel
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(cutter, target):
    """Perform boolean subtract using BMesh."""
    # Select target
    bpy.context.view_layer.objects.active = target
    target.select_set(True)
    cutter.select_set(True)
    
    # We need to use the Boolean modifier for reliability in script
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Clean up cutter
    bpy.data.objects.remove(cutter)

def boolean_union(union_obj, target):
    """Perform boolean union."""
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union_obj)

    # Move by offset
    blade_tip_hook.location += Vector((0.015, 0, 0))
    apply_material(blade_tip_hook, mat_hook_steel)

    # SECTION 8 – Create lock button
    lock_button = add_cube("lock_button", (0.015, 0.010, 0.005))
    align_axis_to_vector(lock_button, 'Z', (1, 0, 0)) # Rotate so +Z aligns +X
    # Anchor to housing_main.right_face
    anchor_to_face(lock_button, housing_main, 'right_face', (0, 0, 0))
    # Align lock_button.XZ_face to housing_main.XZ_face
    # housing_main XZ face normal is Y (0,1,0) or -Y.
    # lock_button XZ face normal is Y (0,1,0) or -Y.
    # lock_button Z is X. So its Y is Y.
    # Alignment is automatic.
    
    # Move by offset
    lock_button.location += Vector((0.040, 0, 0.010))
    apply_material(lock_button, mat_lock_brass)

    # SECTION 9 – Create spring mechanism
    spring_mechanism = add_torus("spring_mechanism", major_radius=0.019, minor_radius=0.005)
    align_axis_to_vector(spring_mechanism, 'Z', (0, 0, 1))
    # Anchor to spool_core.top_face
    anchor_to_face(spring_mechanism, spool_core, 'top_face', (0, 0, 0))
    # Align spring_mechanism.XY_face to spool_core.XY_face
    # Move by offset
    spring_mechanism.location += Vector((0, 0, 0))
    apply_material(spring_mechanism, mat_housing_blue)

if __name__ == "__main__":
    main()

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation_euler
    if material:
        if obj.data.materials:
            obj.data.materials[0] = material
        else:
            obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a world vector."""
    if axis == 'X':
        src = Vector((1, 0, 0))
    elif axis == 'Y':
        src = Vector((0, 1, 0))
    elif axis == 'Z':
        src = Vector((0, 0, 1))
    else:
        return

    target = vector.normalized()
    if src == target:
        return
    
    rot_quat = src.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

def average(points):
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        self.update_matrix()

    def update_matrix(self):
        self.matrix_world = self.obj.matrix_world.copy()
        self.center = self.matrix_world.to_translation()

    def center_world(self):
        return self.center

    def face_center_world(self, face_type='top'):
        # Approximate face centers based on bounding box
        bbox = self.obj.bound_box
        # bbox is in local space
        # We need to transform bbox corners to world space to find extents
        corners = [Vector(c) for c in bbox]
        corners_world = [self.matrix_world @ c for c in corners]
        
        # Find min/max for each axis
        mins = [min(c[i] for c in corners_world) for i in range(3)]
        maxs = [max(c[i] for c in corners_world) for i in range(3)]
        
        if face_type == 'top':
            # Top face is usually +Z or +Y depending on orientation. 
            # Assuming standard orientation where Z is up for these primitives.
            # If rotated, we rely on the object's local axes.
            # For simplicity in this script, we assume Z-up orientation for alignment logic 
            # unless specified otherwise. The prompt says "Align ... to ... face".
            # We will use the object's local Z+ for top, Z- for bottom, etc.
            # However, since we rotate them to align Z to Z, local Z is world Z.
            return Vector([(mins[0]+maxs[0])/2, (mins[1]+maxs[1])/2, maxs[2]])
        elif face_type == 'bottom':
            return Vector([(mins[0]+maxs[0])/2, (mins[1]+maxs[1])/2, mins[2]])
        elif face_type == 'right': # +X
            return Vector([maxs[0], (mins[1]+maxs[1])/2, (mins[2]+maxs[2])/2])
        elif face_type == 'left': # -X
            return Vector([mins[0], (mins[1]+maxs[1])/2, (mins[2]+maxs[2])/2])
        elif face_type == 'front': # +Y
            return Vector([(mins[0]+maxs[0])/2, maxs[1], (mins[2]+maxs[2])/2])
        elif face_type == 'back': # -Y
            return Vector([(mins[0]+maxs[0])/2, mins[1], (mins[2]+maxs[2])/2])
        return self.center

    def radial_point_world(self, angle_deg, radius):
        # Assumes radial direction is in XY plane relative to object center
        # Angle 0 is +X, 90 is +Y
        angle_rad = math.radians(angle_deg)
        dx = radius * math.cos(angle_rad)
        dy = radius * math.sin(angle_rad)
        return self.center + Vector((dx, dy, 0))

def boolean_subtract(cutter_name, target_name):
    cutter = bpy.data.objects.get(cutter_name)
    target = bpy.data.objects.get(target_name)
    if not cutter or not target:
        return
    
    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target_name, union_name):
    target = bpy.data.objects.get(target_name)
    union = bpy.data.objects.get(union_name)
    if not target or not union:
        return

    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union
    
    # Apply Modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove union
    bpy.data.objects.remove(union, do_unlink=True)

# Create primitive triangle with side length 0.0127 m and name it triangle.
# Blender doesn't have a direct "triangle" primitive, so we create a cube and scale it or use a circle with 3 vertices.
# Using a circle with 3 vertices scaled to fit the side length is tricky.
# A better approach for a "primitive triangle" in Blender context is often a mesh created from vertices.
# However, to keep it simple and consistent with "primitive", let's use a cube and scale it to approximate, 
# OR create a custom mesh. Given the constraints, let's create a cube and scale it to be a flat triangle.
# Actually, let's create a mesh triangle manually for precision.
def add_triangle_mesh(name, side_length, location, rotation_euler, material=None):
    verts = [
        (0, 0, 0),
        (side_length, 0, 0),
        (side_length/2, side_length * math.sqrt(3)/2, 0)
    ]
    faces = [(0, 1, 2)]
    mesh = bpy.data.meshes.new(name + "_mesh")
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    obj = bpy.data.objects.new(name, mesh)
    obj.location = location
    obj.rotation_euler = rotation_euler
    if material:
        if obj.data.materials:
            obj.data.materials[0] = material
        else:
            obj.data.materials.append(material)
    bpy.context.collection.objects.link(obj)
    return obj

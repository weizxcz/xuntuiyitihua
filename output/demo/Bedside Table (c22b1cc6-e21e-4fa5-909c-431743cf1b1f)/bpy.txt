import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube centered at location with given size and rotation."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    return obj

def align_face_to_face(source_obj, source_face, target_obj, target_face, offset=(0,0,0)):
    """
    Aligns a face of source_obj to a face of target_obj.
    source_face/target_face: 'x+', 'x-', 'y+', 'y-', 'z+', 'z-'
    """
    # Ensure objects are in world space for calculation, but we manipulate local if needed.
    # For simplicity in this script, we assume standard orientation or explicit rotation.
    
    # Get bounding box centers and extents in world space
    src_bbox = source_obj.bound_box
    tgt_bbox = target_obj.bound_box
    
    # Convert bbox points to world space
    src_world = [source_obj.matrix_world @ Vector(p) for p in src_bbox]
    tgt_world = [target_obj.matrix_world @ Vector(p) for p in tgt_bbox]
    
    # Helper to get face center and normal
    def get_face_info(bbox_world, face_type):
        # face_type: 'x+', 'x-', 'y+', 'y-', 'z+', 'z-'
        if face_type == 'x+':
            pts = [p for p in bbox_world if abs(p.x - max(p.x for p in bbox_world)) < 1e-5]
            normal = Vector((1,0,0))
        elif face_type == 'x-':
            pts = [p for p in bbox_world if abs(p.x - min(p.x for p in bbox_world)) < 1e-5]
            normal = Vector((-1,0,0))
        elif face_type == 'y+':
            pts = [p for p in bbox_world if abs(p.y - max(p.y for p in bbox_world)) < 1e-5]
            normal = Vector((0,1,0))
        elif face_type == 'y-':
            pts = [p for p in bbox_world if abs(p.y - min(p.y for p in bbox_world)) < 1e-5]
            normal = Vector((0,-1,0))
        elif face_type == 'z+':
            pts = [p for p in bbox_world if abs(p.z - max(p.z for p in bbox_world)) < 1e-5]
            normal = Vector((0,0,1))
        elif face_type == 'z-':
            pts = [p for p in bbox_world if abs(p.z - min(p.z for p in bbox_world)) < 1e-5]
            normal = Vector((0,0,-1))
        
        center = sum(pts, Vector()) / len(pts)
        return center, normal

    src_center, src_normal = get_face_info(src_world, source_face)
    tgt_center, tgt_normal = get_face_info(tgt_world, target_face)
    
    # Calculate translation vector to align centers
    # We want src_face_center + src_normal * (distance) = tgt_face_center
    # Actually, simple alignment: move source so its face center matches target face center
    # But we need to preserve orientation.
    
    # Vector from target face center to source face center
    diff = src_center - tgt_center
    
    # We want to move source_obj by -diff so centers align? 
    # No, we want the faces to touch.
    # If we align centers, the faces might not touch if sizes differ.
    # The prompt says "Align face to face". Usually implies touching.
    
    # Let's calculate the required translation to make the faces coincide.
    # Target Face Plane: P = tgt_center + t * tgt_normal
    # Source Face Plane: P = src_center + t * src_normal
    
    # Assuming normals are opposite (e.g. top to bottom), tgt_normal = -src_normal
    # We want src_center + translation + src_normal * half_h = tgt_center - tgt_normal * half_h?
    # Simpler: Move source so its face center is at tgt_center.
    
    move_vec = tgt_center - src_center
    
    # Apply move in local space? The prompt implies global alignment logic but local offsets.
    # Let's apply global translation first.
    source_obj.location += move_vec
    
    # Apply offset
    if offset:
        offset_vec = Vector(offset)
        # Transform offset to world if object is rotated? 
        # Prompt says "offset is in reference local axes".
        # So we need to rotate the offset vector by the object's rotation.
        rot_mat = source_obj.rotation_euler.to_matrix()
        local_offset_world = rot_mat @ offset_vec
        source_obj.location += local_offset_world

def align_center_to_center(source_obj, target_obj, offset=(0,0,0)):
    """Align center of source to center of target."""
    tgt_center = target_obj.location
    src_center = source_obj.location
    move_vec = tgt_center - src_center
    source_obj.location += move_vec
    
    if offset:
        offset_vec = Vector(offset)
        rot_mat = source_obj.rotation_euler.to_matrix()
        local_offset_world = rot_mat @ offset_vec
        source_obj.location += local_offset_world

def align_face_to_center(source_obj, source_face, target_obj, offset=(0,0,0)):
    """Align a face of source to the center of target."""
    tgt_center = target_obj.location
    
    # Get source face center in world
    src_bbox = source_obj.bound_box
    src_world = [source_obj.matrix_world @ Vector(p) for p in src_bbox]
    
    if source_face == 'x+':
        pts = [p for p in src_world if abs(p.x - max(p.x for p in src_world)) < 1e-5]
    elif source_face == 'x-':
        pts = [p for p in src_world if abs(p.x - min(p.x for p in src_world)) < 1e-5]
    elif source_face == 'y+':
        pts = [p for p in src_world if abs(p.y - max(p.y for p in src_world)) < 1e-5]
    elif source_face == 'y-':
        pts = [p for p in src_world if abs(p.y - min(p.y for p in src_world)) < 1e-5]
    elif source_face == 'z+':
        pts = [p for p in src_world if abs(p.z - max(p.z for p in src_world)) < 1e-5]
    elif source_face == 'z-':
        pts = [p for p in src_world if abs(p.z - min(p.z for p in src_world)) < 1e-5]
        
    src_center = sum(pts, Vector()) / len(pts)
    
    move_vec = tgt_center - src_center
    source_obj.location += move_vec
    
    if offset:
        offset_vec = Vector(offset)
        rot_mat = source_obj.rotation_euler.to_matrix()
        local_offset_world = rot_mat @ offset_vec

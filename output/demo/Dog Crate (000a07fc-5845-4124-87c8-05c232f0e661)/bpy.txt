def make_material(name, diffuse_color):
    """Create a new material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive(name, shape, size, location, rotation_euler):
    """Add a primitive object, set name, location, and rotation."""
    bpy.ops.object.select_all(action='DESELECT')
    
    if shape == 'CUBE':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'CYLINDER':
        # size is (radius, depth) for cylinder in this context usually, 
        # but prompt says "size 0.01x0.02x0.03" for cylinder. 
        # Assuming format: radius, radius, height or diameter, diameter, height?
        # Standard Blender add_cylinder uses radius and depth.
        # Prompt format "0.01x0.02x0.03" for cylinder likely means radius=0.01, height=0.02? 
        # Or diameter? Let's assume standard "size" tuple interpretation for primitives:

        # Let's look at foot_pad: "0.05x0.05x0.01". Likely radius=0.05, height=0.01.

        # Let's assume the tuple is (radius, radius, height) for consistency with "size" terminology, 
        # or more likely (radius, height) if 2D, but here 3D.
        # Let's assume the input "size" for cylinder is (radius, radius, height).
        r = size[0]
        h = size[2] # Assuming 3rd component is height
        bpy.ops.mesh.primitive_cylinder_add(radius=r, depth=h, location=location)
        obj = bpy.context.active_object
    elif shape == 'DISC':
        bpy.ops.mesh.primitive_circle_add(radius=size[0], location=location, fill_type='TRIFAN')

        bpy.ops.mesh.primitive_uv_sphere_add(radius=size[0], location=location)
        obj = bpy.context.active_object
    else:
        return

    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def apply_boolean(target_obj, cutter_obj, operation='SUBTRACT'):
    """Apply a boolean modifier to target using cutter."""
    # Ensure target is active for modifier application if needed, 
    # but better to add modifier to target.
    
    # Add modifier to target
    mod = target_obj.modifiers.new(name="BooleanMod", type='BOOLEAN')
    mod.operation = operation
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def align_faces(obj_a, face_axis_a, obj_b, face_axis_b):
    """
    Align face of obj_a to face of obj_b.
    face_axis: 'X', 'Y', 'Z' or '-X', '-Y', '-Z'
    """
    # Get world matrix
    mat_a = obj_a.matrix_world
    mat_b = obj_b.matrix_world
    
    # Get bounding box centers and extents
    # We need to identify the specific face center.
    # Simple approach: Align centers along the axis, then offset by half extents.
    
    # Determine direction vectors
    def get_face_vector(axis_str):
        axis = axis_str[0] # X, Y, Z
        sign = 1 if axis_str[1:] == '' else -1
        if axis == 'X':
            return Vector((sign, 0, 0))
        elif axis == 'Y':
            return Vector((0, sign, 0))
        elif axis == 'Z':
            return Vector((0, 0, sign))
        return Vector()

    vec_a = get_face_vector(face_axis_a)
    vec_b = get_face_vector(face_axis_b)
    
    # Get local extents
    ext_a = obj_a.dimensions
    ext_b = obj_b.dimensions
    
    # Calculate face positions in world space relative to object center
    # Face position = Center + (Extent/2 * Direction)
    # We want Face_A == Face_B
    
    # Current face positions
    # This is tricky because rotation affects world position.
    # Robust method: Project centers onto the alignment axis defined by the faces?
    # No, "Align Face A to Face B" usually means making them coplanar.
    
    # Let's use a simpler heuristic for axis-aligned objects (which these are):
    # 1. Align the primary axis of the faces.
    # 2. Move obj_a so its face touches obj_b's face.
    
    # Since all objects are axis-aligned (rotated 0,0,0 or 90 deg on one axis),
    # we can just calculate the offset.
    
    # Let's assume standard orientation for now as per creation steps.
    # If rotations are complex, this helper needs to be more robust.
    # Given the strict creation steps, objects are mostly axis aligned.
    
    # Get world position of the face center
    def get_world_face_center(obj, axis_str):
        mat = obj.matrix_world
        local_center = Vector((0,0,0))
        half_ext = obj.dimensions / 2
        
        if axis_str == 'X':
            local_center.x = half_ext.x
        elif axis_str == '-X':
            local_center.x = -half_ext.x
        elif axis_str == 'Y':
            local_center.y = half_ext.y
        elif axis_str == '-Y':
            local_center.y = -half_ext.y
        elif axis_str == 'Z':
            local_center.z = half_ext.z
        elif axis_str == '-Z':
            local_center.z = -half_ext.z
            
        return mat @ local_center

    face_a_pos = get_world_face_center(obj_a, face_axis_a)
    face_b_pos = get_world_face_center(obj_b, face_axis_b)
    
    # Calculate difference
    diff = face_b_pos - face_a_pos
    
    # Move obj_a
    obj_a.location += diff

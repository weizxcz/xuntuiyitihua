# --- Helper Functions ---

def make_material(name, diffuse_color=(1, 1, 1, 1), roughness=0.5, metallic=0.0):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        return bpy.data.materials[name]
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = roughness
    bsdf.inputs['Metallic'].default_value = metallic
    return mat

def add_cylinder(name, radius=0.5, depth=1.0, segments=32, location=(0,0,0), rotation=(0,0,0), scale=(1,1,1)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    obj.scale = scale
    return obj

def add_disc(name, radius=0.5, location=(0,0,0), rotation=(0,0,0), scale=(1,1,1)):

def add_bevel(obj, width=0.01, segments=4):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    mod.angle_limit = math.radians(30)
    return mod

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a world vector."""
    if axis == 'X':
        src = Vector((1, 0, 0))
    elif axis == 'Y':
        src = Vector((0, 1, 0))
    else: # Z
        src = Vector((0, 0, 1))
    
    tgt = vector.normalized()
    rot_quat = src.rotation_difference(tgt)
    obj.rotation_quaternion = rot_quat

def average(points):
    """Calculate the average position of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        self.matrix_world = obj.matrix_world

    def center_world(self):
        """Return the world center of the object."""
        local_center = average([Vector(v) for v in self.bound_box])
        return self.matrix_world @ local_center

    def face_center_world(self, face_index=0):
        """Return the world center of a specific face (by index)."""
        # Note: This assumes the object has mesh data and faces.
        # For simple primitives, face indices might vary. 
        # A more robust way for primitives is to calculate based on geometry.
        # Here we assume standard orientation for simplicity or pass specific logic.
        # For this script, we will use a simpler approach: 
        # If it's a cube/cylinder, we can infer face centers.
        # However, to keep it generic, we'll rely on the object's geometry if available.
        # Since we are creating primitives, let's implement specific face logic for common shapes if needed,
        # or just return the center if face index is not strictly defined by user input in a complex way.
        # The prompt says "base_sheet.+Z_face". For a cube aligned to Z, +Z face is at max Z.

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract operation."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is a mesh
    if cutter.type != 'MESH':
        bpy.ops.object.convert(target=cutter)
    
    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply the modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(obj1_name, obj2_name):
    """Perform a boolean union operation."""
    obj1 = bpy.data.objects[obj1_name]
    obj2 = bpy.data.objects[obj2_name]
    
    # Ensure both are meshes
    if obj1.type != 'MESH':
        bpy.ops.object.convert(target=obj1)
    if obj2.type != 'MESH':
        bpy.ops.object.convert(target=obj2)
        
    # Add Boolean Modifier to Obj1
    mod = obj1.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj2
    
    # Apply the modifier
    bpy.context.view_layer.objects.active = obj1
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove obj2
    bpy.data.objects.remove(obj2, do_unlink=True)

# --- Block 0: Scene Reset & Units ---

# Delete all existing objects
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Set units to metres
bpy.context.scene.unit_settings.scale_length = 1
bpy.context.scene.unit_settings.system = 'METRIC'

# --- Block 1: Materials ---

mat_wallpaper_base = make_material("mat_wallpaper_base", diffuse_color=(0.941, 0.941, 0.941, 1))
mat_wallpaper_pattern = make_material("mat_wallpaper_pattern", diffuse_color=(0.784, 0.784, 0.784, 1))

# --- Block 2: Geometry Generation ---

# --- SECTION 1 – Create base sheet ---

# Create primitive cuboid with size 1.0×2.0×0.002 m and name it base_sheet.
# Note: add_cube creates a cube of size 'size' (total width). 
# To get 1.0x2.0x0.002, we create a cube of size 1.0 and scale it.
base_sheet = add_cube("base_sheet", size=1.0, location=(0,0,0))
base_sheet.scale = (1.0, 2.0, 0.002)

# Rotate base_sheet so its +Z axis aligns +Z (world).
# Default cube +Z is up, so no rotation needed if we want +Z aligned with world +Z.
# But let's be explicit.
align_axis_to_vector(base_sheet, 'Z', (0,0,1))

# Anchor base_sheet to world origin.
base_sheet.location = (0, 0, 0)

# Then move by offset(0,0,0).
base_sheet.location += Vector((0, 0, 0))

# Assign material
if base_sheet.data.materials:
    base_sheet.data.materials[0] = mat_wallpaper_base
else:
    base_sheet.data.materials.append(mat_wallpaper_base)

# --- SECTION 2 – Create pattern tiles ---

# For r in 0…9 and c in 0…9:
for r in range(10):
    for c in range(10):
        # Create primitive cuboid with size 0.1×0.1×0.002 m and name it tile_r_c.
        tile_name = f"tile_{r}_{c}"
        tile = add_cube(tile_name, size=0.1, location=(0,0,0))
        tile.scale = (0.1, 0.1, 0.002)
        
        # Rotate tile_r_c so its +Z axis aligns +Z (world).
        align_axis_to_vector(tile, 'Z', (0,0,1))
        
        # Anchor tile_r_c to base_sheet.+Z_face.
        # base_sheet is at (0,0,0) with size 1x2x0.002.
        # Its +Z face is at Z = 0.001 (half of 0.002).
        # The center of the +Z face is at (0, 0, 0.001).
        # We want to anchor the tile's center to this face.
        # So the tile's initial position should be at (0, 0, 0.001).
        tile.location = (0, 0, 0.001)
        
        # Then offset(0.05 + c·0.1, 0.05 + r·0.1, 0.001).
        # The offset is relative to the anchor point.
        # Anchor point is (0, 0, 0.001).
        # Offset X: 0.05 + c * 0.1
        # Offset Y: 0.05 + r * 0.1
        # Offset Z: 0.001
        offset_x = 0.05 + c * 0.1
        offset_y = 0.05 + r * 0.1
        offset_z = 0.001
        
        tile.location += Vector((offset_x, offset_y, offset_z))
        
        # Assign material
        if tile.data.materials:
            tile.data.materials[0] = mat_wallpaper_pattern
        else:
            tile.data.materials.append(mat_wallpaper_pattern)

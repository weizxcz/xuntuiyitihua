def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive_cube(name, size, location, rotation_euler):
    """Add a cube and apply transforms."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def add_primitive_cylinder(name, radius, height, location, rotation_euler):
    """Add a cylinder and apply transforms."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def get_face_normal(obj, face_index):
    """Get world space normal of a specific face."""
    # Ensure we are in object mode to access data correctly
    if bpy.context.active_object != obj:
        bpy.context.view_layer.objects.active = obj
        obj.select_set(True)
    
    # Get the matrix world
    mw = obj.matrix_world
    
    # Get the face
    # Note: For simple primitives, face indices are predictable, 
    # but for robustness we calculate based on geometry if needed.
    # Here we assume standard orientation or pass specific logic.
    # For a cube: 0=bottom(-Z), 1=top(+Z), 2=front(+Y), 3=back(-Y), 4=right(+X), 5=left(-X)
    # For a cylinder: 0=bottom, 1=top
    return mw @ obj.data.polygons[face_index].normal

def get_face_center(obj, face_index):
    """Get world space center of a specific face."""
    mw = obj.matrix_world
    local_center = obj.data.polygons[face_index].center
    return mw @ local_center

def align_and_offset(obj, anchor_obj, anchor_face_idx, offset_vec, rotation_euler=(0,0,0)):

    Aligns obj's bottom face to anchor_obj's specified face, then applies offset.
    Assumes obj is created at origin with default orientation before this call.
    """
    # 1. Set Rotation
    obj.rotation_euler = rotation_euler
    
    # 2. Calculate Anchor Point and Normal
    anchor_normal = get_face_normal(anchor_obj, anchor_face_idx)
    anchor_center = get_face_center(anchor_obj, anchor_face_idx)
    
    # 3. Align Object Orientation
    # We want obj's +Z (top) to align with anchor_normal.
    # Default cube/cylinder +Z is up.
    # If anchor_normal is +Z, rotation is (0,0,0).
    # If anchor_normal is -Z, rotation is (pi, 0, 0).
    # If anchor_normal is +Y, rotation is (-pi/2, 0, 0).
    # If anchor_normal is -Y, rotation is (pi/2, 0, 0).
    
    # Simple alignment logic for standard axes
    target_up = anchor_normal.normalized()
    current_up = Vector((0, 0, 1))
    
    if target_up.dot(current_up) < -0.99:
        obj.rotation_euler = (math.pi, 0, 0)
    elif target_up.dot(current_up) > 0.99:
        obj.rotation_euler = (0, 0, 0)
    elif target_up.dot(Vector((0, 1, 0))) > 0.99:
        obj.rotation_euler = (-math.pi/2, 0, 0)
    elif target_up.dot(Vector((0, -1, 0))) > 0.99:
        obj.rotation_euler = (math.pi/2, 0, 0)
    elif target_up.dot(Vector((1, 0, 0))) > 0.99:
        obj.rotation_euler = (0, 0, -math.pi/2)
    elif target_up.dot(Vector((-1, 0, 0))) > 0.99:
        obj.rotation_euler = (0, 0, math.pi/2)

    # For a cube/cylinder created at origin, bottom face is -Z.
    # After rotation, we need to find where the bottom face is.

    # Since we just set rotation, we can calculate the offset manually.
    # Distance from center to bottom face is half height (for cylinder) or half size (for cube).

    # Then apply offset.

    half_h = 0

        # This is local space. We need world space projection.
        # Easier: The object is centered at (0,0,0) locally.
        # The bottom face is at local Z = -half_h.
        # We need to find local Z of bottom face.
        
        # Re-evaluating: It's safer to define the object's "bottom" relative to its local Z.
        # If we align +Z to anchor_normal, the bottom is -Z.
        # The distance from center to bottom face is 'h'.
        
        # Let's assume standard primitives created with scale 1 or specific sizes.
        # We will pass dimensions in the creation step or calculate them.
        pass

    # Robust Alignment:
    # 1. Rotate object.
    # 2. Calculate vector from object center to its bottom face center in World Space.
    # 3. Move object so bottom face center == anchor center.
    
    # Get local bottom face normal (should be -Z for standard primitives)
    local_bottom_normal = Vector((0, 0, -1))
    world_bottom_normal = obj.matrix_world.to_3x3() @ local_bottom_normal
    
    # Get local bottom face center offset (e.g., -height/2)
    # We need to know the dimension along Z.
    # For cube: size[2]/2. For cylinder: height/2.
    # Let's extract from data if possible, or assume standard.
    # Since we create them, we can just calculate the position directly.
    
    # Let's simplify:
    # Target Center = Anchor Center - (World Bottom Normal * DistanceFromCenterToBottom)
    # Then apply Offset.
    
    # Determine DistanceFromCenterToBottom
    # We need to know the object's Z-dimension.
    # Let's assume the object has been scaled.

    # Transform bound box to world

def create_obj(name, mat, loc, rot, scale=(1,1,1)):
    obj = bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    obj = bpy.context.active_object
    obj.name = name

def create_cyl(name, mat, radius, height, loc, rot):

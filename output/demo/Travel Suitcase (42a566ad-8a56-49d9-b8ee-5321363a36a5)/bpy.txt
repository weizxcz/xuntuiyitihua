import bpy
import bmesh
import math
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.diffuse_color = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler):
    bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=radius, depth=depth, location=location)
    obj = bpy.context.object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def align_center_to_target(source_obj, target_obj):
    """Aligns the center of source_obj to the center of target_obj."""
    target_loc = target_obj.location
    source_loc = source_obj.location
    offset = target_loc - source_loc
    source_obj.location += offset

def apply_offset(obj, offset_vec):
    """Applies an offset vector to the object's location."""
    obj.location += offset_vec

def orient_z_to_y(obj):
    """Rotates object so its local +Z aligns with world +Y."""
    # World +Y is (0, 1, 0). Local +Z is (0, 0, 1).
    # Rotation of -90 degrees around X axis maps Z to Y.
    obj.rotation_euler = (math.radians(-90), 0, 0)

def orient_z_to_x(obj):
    """Rotates object so its local +Z aligns with world +X."""
    # World +X is (1, 0, 0). Local +Z is (0, 0, 1).
    # Rotation of +90 degrees around Y axis maps Z to X.
    obj.rotation_euler = (0, math.radians(90), 0)

def orient_z_to_z(obj):
    """Ensures object local +Z aligns with world +Z (identity)."""
    obj.rotation_euler = (0, 0, 0)

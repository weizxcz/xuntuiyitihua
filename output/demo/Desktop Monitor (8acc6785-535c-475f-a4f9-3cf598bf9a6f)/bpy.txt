def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder aligned to Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, end_fill_type='NGON')
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0) # Z-up default
    return obj

def add_cube(name, size):
    """Add a cube centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=1) # size=1 means 1m edge
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = (0, 0, 0)
    return obj

def align_face_to_face(obj_source, obj_target, axis='Z', direction=1):
    """
    Aligns the specified face of obj_source to the corresponding face of obj_target.
    axis: 'X', 'Y', or 'Z'
    direction: 1 for positive face, -1 for negative face.
    """
    # Ensure objects are in world space
    bpy.context.view_layer.update()
    
    # Get bounding box centers and extents
    # We need to find the face center. 
    # Simplified approach: Align centers based on face positions.
    
    # Transform to world
    src_world = obj_source.matrix_world
    tgt_world = obj_target.matrix_world
    
    # Get local extents
    src_ext = obj_source.dimensions
    tgt_ext = obj_target.dimensions
    
    # Calculate face positions in local space relative to center
    # Positive face is at +extent/2, Negative at -extent/2
    src_face_local = (src_ext[axis] / 2) * direction
    tgt_face_local = (tgt_ext[axis] / 2) * direction
    
    # Convert to world
    # Vector from center to face in local space
    src_face_vec_local = Vector([1,1,1])
    src_face_vec_local[axis] = src_face_local
    src_face_world = src_world @ src_face_vec_local
    
    tgt_face_vec_local = Vector([1,1,1])
    tgt_face_vec_local[axis] = tgt_face_local
    tgt_face_world = tgt_world @ tgt_face_vec_local
    
    # Calculate offset needed to move src_face_world to tgt_face_world
    offset = tgt_face_world - src_face_world
    
    # Apply offset to object location
    obj_source.location += offset

def align_face_to_face_with_offset(obj_source, obj_target, axis='Z', direction=1, offset_vec=Vector((0,0,0))):
    """Aligns face and then applies additional offset."""
    align_face_to_face(obj_source, obj_target, axis, direction)
    obj_source.location += offset_vec

def average(points):
    """Calculate average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        self.update()
        
    def update(self):
        self.center_world = self.obj.matrix_world.to_translation()
        # Calculate face centers based on dimensions
        dims = self.obj.dimensions
        mat = self.obj.matrix_world
        # Z-face centers
        self.z_face_pos = mat @ Vector((0, 0, dims[2]/2))
        self.z_face_neg = mat @ Vector((0, 0, -dims[2]/2))
        # Y-face centers
        self.y_face_pos = mat @ Vector((0, dims[1]/2, 0))
        self.y_face_neg = mat @ Vector((0, -dims[1]/2, 0))
        # X-face centers
        self.x_face_pos = mat @ Vector((dims[0]/2, 0, 0))
        self.x_face_neg = mat @ Vector((-dims[0]/2, 0, 0))

    def center_world(self):
        return self.obj.matrix_world.to_translation()

    def z_face_world(self, positive=True):
        if positive:
            return self.z_face_pos
        return self.z_face_neg

    def y_face_world(self, positive=True):
        if positive:
            return self.y_face_pos
        return self.y_face_neg

    def x_face_world(self, positive=True):
        if positive:
            return self.x_face_pos
        return self.x_face_neg

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cone(name, radius_top, radius_bottom, depth, location=(0,0,0), rotation=(0,0,0), material=None):

    bpy.ops.mesh.primitive_cone_add(radius1=radius_top, radius2=radius_bottom, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object

def add_bevel(obj, width, segments=2):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(base_obj, cutter_obj):
    """Perform boolean subtract using BMesh."""
    # Ensure base is active
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.select_all(action='DESELECT')
    base_obj.select_set(True)
    
    # Create modifier
    mod = base_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Cleanup cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return base_obj

def boolean_union(base_obj, other_obj):
    """Perform boolean union using BMesh."""
    # Ensure base is active
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.select_all(action='DESELECT')
    base_obj.select_set(True)
    
    # Create modifier
    mod = base_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other_obj
    
    # Apply
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Cleanup other
    bpy.data.objects.remove(other_obj, do_unlink=True)
    return base_obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with world vector."""
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else: # Z
        local_axis = Vector((0, 0, 1))
        
    target = vector.normalized()
    rot_quat = local_axis.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

def get_face_center(obj, face_type='right'):
    """Get the world center of a specific face of a Z-aligned cylinder."""
    # Assuming Z-aligned cylinder, right face is +Z, left is -Z
    if face_type == 'right':
        offset = obj.dimensions.z / 2
    elif face_type == 'left':
        offset = -obj.dimensions.z / 2
    else:
        offset = 0
        
    # Local to World
    local_center = Vector((0, 0, offset))
    world_center = obj.matrix_world @ local_center
    return world_center

def get_object_center(obj):
    """Get world center of object."""
    return obj.matrix_world.to_translation()

def align_to_face(obj_source, obj_target, source_face='right', target_face='center'):
    """Align source object's face to target object's face/center."""
    target_loc = get_object_center(obj_target)
    
    # Calculate offset in local space of source
    if source_face == 'right':
        local_offset = Vector((0, 0, obj_source.dimensions.z / 2))
    elif source_face == 'left':
        local_offset = Vector((0, 0, -obj_source.dimensions.z / 2))
    else:
        local_offset = Vector((0,0,0))
        
    # We want the source face to be at target_loc
    # Current source center is obj_source.location
    # Source face world pos = obj_source.location + (obj_source.matrix_world @ local_offset)
    # We want: obj_source.location + (obj_source.matrix_world @ local_offset) = target_loc
    # So: obj_source.location = target_loc - (obj_source.matrix_world @ local_offset)
    
    # However, since we align axes first, matrix_world is just translation + rotation.
    # If aligned to Z, rotation is identity or 180.
    # Let's just move the object so its face hits the target.
    
    # Simple approach: Move object so its face center matches target center
    # Face center in local space is (0,0, +/- z/2)
    # In world space, if aligned to Z, it's (0,0, +/- z/2) relative to object loc.
    
    sign = 1 if source_face == 'right' else -1
    move_vec = Vector((0, 0, sign * obj_source.dimensions.z / 2))
    
    # Target position for object center
    new_loc = target_loc - move_vec
    obj_source.location = new_loc

def align_to_center(obj_source, obj_target):
    """Align source object center to target object center."""
    target_loc = get_object_center(obj_target)
    obj_source.location = target_loc

def apply_offset(obj, offset_vec):
    """Apply offset to object location."""
    obj.location += offset_vec

def create_locator(name, location):
    """Create a simple empty locator."""
    bpy.ops.object.empty_add(location=location, type='PLAIN_AXES')
    loc = bpy.context.active_object
    loc.name = name
    return loc

def average(points):
    """Calculate average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

import bpy
import bmesh
import math

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if not obj.data.materials:
        obj.data.materials.append(mat)
    else:
        obj.data.materials[0] = mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder aligned with Z axis, centered at origin."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size_x, size_y, size_z):
    """Add a cube aligned with axes, centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size_x / 2, size_y / 2, size_z / 2)
    return obj

def add_cone(name, radius_bottom, radius_top, depth, segments=32):
    """Add a cone aligned with Z axis, centered at origin."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_segments=32, height_segments=16):
    """Add a UV sphere centered at origin."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, target_vector):
    """Rotate object so its local +Z aligns with target_vector."""
    if target_vector.length == 0:
        return
    
    # Current Z axis
    current_z = Vector((0, 0, 1))
    
    # Calculate rotation
    rot_quat = current_z.rotation_difference(target_vector.normalized())
    obj.rotation_euler = rot_quat.to_euler()

def align_object_axis_to_vector(obj, axis_index, target_vector):
    """Rotate object so a specific local axis aligns with target_vector."""
    if target_vector.length == 0:
        return
    
    # Define local axes
    local_axes = [
        Vector((1, 0, 0)), # X
        Vector((0, 1, 0)), # Y
        Vector((0, 0, 1))  # Z
    ]
    
    current_axis = local_axes[axis_index]
    rot_quat = current_axis.rotation_difference(target_vector.normalized())
    obj.rotation_euler = rot_quat.to_euler()

def get_world_center(obj):
    """Get the world position of the object's center."""
    return obj.matrix_world.to_translation()

def get_face_center(obj, face_type='bottom'):
    """
    Get the world center of a specific face of an object.
    Assumes object is axis-aligned or has known orientation.
    """
    # Get local bounds
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    
    # Find face based on normal
    target_normal = Vector((0, 0, -1)) # Default bottom
    if face_type == 'top':
        target_normal = Vector((0, 0, 1))
    elif face_type == 'front':
        target_normal = Vector((0, 1, 0))
    elif face_type == 'back':
        target_normal = Vector((0, -1, 0))
    elif face_type == 'left':
        target_normal = Vector((-1, 0, 0))
    elif face_type == 'right':
        target_normal = Vector((1, 0, 0))
    
    best_face = None
    best_dot = -2.0
    
    for face in bm.faces:
        dot = face.normal.dot(target_normal)
        if dot > best_dot:
            best_dot = dot
            best_face = face
            
    if best_face:
        # Calculate centroid
        centroid = Vector((0, 0, 0))
        for v in best_face.verts:
            centroid += v.co
        centroid /= len(best_face.verts)
        bm.free()
        return centroid
    
    bm.free()
    return get_world_center(obj)

def align_to_face(obj, target_obj, face_type='bottom'):
    """
    Align obj's bottom face to target_obj's specified face.
    Moves obj so its bottom face center matches target face center.
    """
    target_center = get_face_center(target_obj, face_type)
    obj_center = get_world_center(obj)
    
    # Calculate offset to move obj's center to target center
    # Assuming obj is oriented such that its 'bottom' is -Z in local space
    # We need to shift obj so its bottom face is at target_center.
    # If obj is centered, bottom face is at obj_center - (0,0, height/2)
    
    # Get local height
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    
    # Simple bounding box approach for offset
    min_z = min(v.co.z for v in bm.verts)
    max_z = max(v.co.z for v in bm.verts)
    local_height = max_z - min_z
    
    bm.free()
    
    # Shift object so its bottom face (min_z) aligns with target_center
    # Current bottom face Z = obj_center.z - (local_height/2) if centered
    # We want new bottom face Z = target_center.z
    # So new obj_center.z = target_center.z + local_height/2
    
    shift = target_center - obj_center
    shift.z += local_height / 2
    
    obj.location += shift

def align_center_to_center(obj, target_obj):
    """Align object center to target object center."""
    target_center = get_world_center(target_obj)
    obj_center = get_world_center(obj)
    obj.location += (target_center - obj_center)

def boolean_subtract(cutter_obj, target_obj):
    """Perform boolean subtract using modifier."""
    # Add boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Delete cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(union_obj, target_obj):
    """Perform boolean union."""
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union_obj, do_unlink=True)

def add_bevel(obj, radius, segments=8):
    """Add bevel modifier and apply."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    mod.angle_limit = 0.0 # Bevel all edges by default if angle limit is 0 or handled by width
    
    # To bevel all edges, we can use the 'Weight' method or just apply with width
    # Blender's bevel modifier with width=0.02 and segments=8 works well for simple cases
    # However, to ensure it bevels all edges, we might need to set the width method to 'Weight' 
    # or rely on the default behavior. For simplicity, we use the default.
    
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def create_locator(name, location):
    """Create a simple empty locator."""
    bpy.ops.object.empty_add(location=location, type='PLAIN_AXES')
    loc = bpy.context.active_object
    loc.name = name
    return loc

def average(points):
    """Calculate average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

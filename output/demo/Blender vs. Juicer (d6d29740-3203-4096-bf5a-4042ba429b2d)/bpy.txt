import bpy
import bmesh
import math
from mathutils import Vector, Matrix

# ==============================================================================
# HELPER FUNCTIONS
# ==============================================================================

def make_material(name, diffuse_color):
    """Create a new material with the specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, height, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, thickness, location=(0,0,0), rotation=(0,0,0)):
    """Add a disc (short cylinder) mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=thickness, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_top, radius_bottom, height, location=(0,0,0), rotation=(0,0,0)):
    """Add a cone/tapered mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_top, radius2=radius_bottom, depth=height, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, location=(0,0,0), rotation=(0,0,0)):
    """Add a UV sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width=0.01, segments=2):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    return mod

def boolean_subtract(target_obj, cutter_obj):
    """Perform boolean subtract: target - cutter."""
    # Ensure target is active for the modifier addition context if needed, 
    # but we will use the modifier API for robustness.
    
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="Boolean_Subtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter object
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return target_obj

def boolean_union(obj1, obj2):
    """Perform boolean union: obj1 + obj2. Returns obj1."""
    # Add Boolean Modifier to obj1
    mod = obj1.modifiers.new(name="Boolean_Union", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj2
    
    # Apply modifier
    bpy.context.view_layer.objects.active = obj1
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove obj2
    bpy.data.objects.remove(obj2, do_unlink=True)
    return obj1

def align_face_to_face(obj_bottom, obj_top, offset_z=0):
    """
    Aligns the top face of obj_bottom to the bottom face of obj_top.
    Assumes objects are upright (Z-axis up).
    obj_bottom is moved so its top face is at obj_top's bottom face level + offset_z.
    """
    # Get world bounds
    # Note: This assumes standard orientation. For robustness with rotated objects,
    # we calculate local extents and transform.
    
    # Helper to get min/max Z in world space
    def get_world_z_bounds(obj):
        # Transform vertices to world space
        verts_z = [obj.matrix_world @ v.co for v in obj.data.vertices]
        z_vals = [v[2] for v in verts_z]
        return min(z_vals), max(z_vals)

    z_min_bottom, z_max_bottom = get_world_z_bounds(obj_bottom)
    z_min_top, z_max_top = get_world_z_bounds(obj_top)
    
    # Target Z for bottom object's top face
    target_z_top = z_min_top + offset_z
    
    # Current Z for bottom object's top face
    current_z_top = z_max_bottom
    
    # Delta Z needed
    delta_z = target_z_top - current_z_top
    
    # Move obj_bottom
    obj_bottom.location.z += delta_z
    
    return obj_bottom

def align_face_to_face_offset(obj_bottom, obj_top, offset_vector):
    """
    Aligns the top face of obj_bottom to the bottom face of obj_top,
    then applies an additional offset vector.
    """
    # First align faces
    # We need to calculate the Z shift required to align faces
    def get_world_z_bounds(obj):
        verts_z = [obj.matrix_world @ v.co for v in obj.data.vertices]
        z_vals = [v[2] for v in verts_z]
        return min(z_vals), max(z_vals)

    z_min_bottom, z_max_bottom = get_world_z_bounds(obj_bottom)
    z_min_top, z_max_top = get_world_z_bounds(obj_top)
    
    # Target Z for bottom object's top face
    target_z_top = z_min_top
    
    # Current Z for bottom object's top face
    current_z_top = z_max_bottom
    
    # Delta Z needed for alignment
    delta_z = target_z_top - current_z_top
    
    # Apply alignment
    obj_bottom.location.z += delta_z
    
    # Apply offset
    obj_bottom.location.x += offset_vector.x
    obj_bottom.location.y += offset_vector.y
    obj_bottom.location.z += offset_vector.z
    
    return obj_bottom

def set_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

# ==============================================================================
# BLOCK 0: Scene Reset & Units
# ==============================================================================

# Delete all existing objects
bpy.ops.object.select_all(action='DESELECT')
bpy.ops.object.select_all(action='ALL')
bpy.ops.object.delete(use_global=False)

# Clear orphan data (optional but clean)
for mesh in bpy.data.meshes:
    if mesh.users == 0:
        bpy.data.meshes.remove(mesh)
for obj in bpy.data.objects:
    if obj.users == 0:
        bpy.data.objects.remove(obj)

# Set units to metres
bpy.context.scene.unit_settings.scale_length = 1
bpy.context.scene.unit_settings.system = 'METRIC'

# ==============================================================================
# BLOCK 1: Materials
# ==============================================================================

mat_blender_base = make_material("mat_blender_base", (0.784, 0.784, 0.784, 1))
mat_blender_container = make_material("mat_blender_container", (0.863, 0.863, 0.863, 0.502))
mat_blender_lid = make_material("mat_blender_lid", (1, 1, 1, 1))
mat_blender_motor = make_material("mat_blender_motor", (0.196, 0.196, 0.196, 1))
mat_juicer_base = make_material("mat_juicer_base", (0.784, 0.784, 0.784, 1))
mat_juicer_container = make_material("mat_juicer_container", (0.863, 0.863, 0.863, 0.502))
mat_juicer_pulp = make_material("mat_juicer_pulp", (0.941, 0.941, 0.941, 1))
mat_juicer_spout = make_material("mat_juicer_spout", (1, 1, 1, 1))

# ==============================================================================
# BLOCK 2: Geometry Generation
# ==============================================================================

# --- SECTION 1 – Create blender base ---
blender_base = add_cylinder("blender_base", 0.075, 0.1, location=(0,0,0), rotation=(0,0,0))
set_material(blender_base, mat_blender_base)

# --- SECTION 2 – Create blender motor ---
blender_motor = add_cylinder("blender_motor", 0.06, 0.05, location=(0,0,0), rotation=(0,0,0))
# Align bottom face of motor to top face of base, then offset Z by 0.025
align_face_to_face_offset(blender_motor, blender_base, Vector((0, 0, 0.025)))
set_material(blender_motor, mat_blender_motor)

# --- SECTION 3 – Create blender container ---
blender_container = add_cylinder("blender_container", 0.06, 0.25, location=(0,0,0), rotation=(0,0,0))
# Align bottom face of container to top face of motor, then offset Z by 0.025
align_face_to_face_offset(blender_container, blender_motor, Vector((0, 0, 0.025)))
set_material(blender_container, mat_blender_container)

# --- SECTION 4 – Create blender lid ---
blender_lid = add_cylinder("blender_lid", 0.065, 0.02, location=(0,0,0), rotation=(0,0,0))
# Align bottom face of lid to top face of container, then offset Z by 0.01
align_face_to_face_offset(blender_lid, blender_container, Vector((0, 0, 0.01)))
set_material(blender_lid, mat_blender_lid)

# --- SECTION 5 – Create blender blades ---
blender_blades = add_cylinder("blender_blades", 0.05, 0.01, location=(0,0,0), rotation=(0,0,0))
# Align top face of blades to bottom face of container, then offset Z by -0.005
# Note: align_face_to_face_offset aligns obj_bottom.top to obj_top.bottom.
# Here blender_blades is obj_bottom, blender_container is obj_top.
align_face_to_face_offset(blender_blades, blender_container, Vector((0, 0, -0.005)))
set_material(blender_blades, mat_blender_motor)

# --- SECTION 6 – Create juicer base ---
juicer_base = add_cylinder("juicer_base", 0.075, 0.1, location=(0.25,0,0), rotation=(0,0,0))
set_material(juicer_base, mat_juicer_base)

# --- SECTION 7 – Create juicer container ---
juicer_container = add_cylinder("juicer_container", 0.06, 0.2, location=(0,0,0), rotation=(0,0,0))
# Align bottom face of container to top face of base, then offset Z by 0.05
align_face_to_face_offset(juicer_container, juicer_base, Vector((0, 0, 0.05)))
set_material(juicer_container, mat_juicer_container)

# --- SECTION 8 – Create juicer spout ---
juicer_spout = add_cylinder("juicer_spout", 0.015, 0.1, location=(0,0,0), rotation=(0,0,0))
# Align bottom face of spout to top face of container, then offset Z by 0.05 and X by 0.04
align_face_to_face_offset(juicer_spout, juicer_container, Vector((0.04, 0, 0.05)))
set_material(juicer_spout, mat_juicer_spout)

# --- SECTION 9 – Create juicer pulp ---
juicer_pulp = add_cylinder("juicer_pulp", 0.04, 0.15, location=(0,0,0), rotation=(0,0,0))
# Align bottom face of pulp to top face of base, then offset Z by 0.075 and X by -0.05
align_face_to_face_offset(juicer_pulp, juicer_base, Vector((-0.05, 0, 0.075)))
set_material(juicer_pulp, mat_juicer_pulp)

# --- SECTION 10 – Assemble Blender Group ---
# Boolean-union blender_base, blender_motor, blender_container, blender_lid, blender_blades
# Start with blender_base
blender_group = blender_base
blender_group = boolean_union(blender_group, blender_motor)
blender_group = boolean_union(blender_group, blender_container)
blender_group = boolean_union(blender_group, blender_lid)
blender_group = boolean_union(blender_group, blender_blades)

# --- SECTION 11 – Assemble Juicer Group ---
# Boolean-union juicer_base, juicer_container, juicer_spout, juicer_pulp
juicer_group = juicer_base
juicer_group = boolean_union(juicer_group, juicer_container)
juicer_group = boolean_union(juicer_group, juicer_spout)
juicer_group = boolean_union(juicer_group, juicer_pulp)

# --- SECTION 12 – Assemble Root Composite ---
# Boolean-union blender_group, juicer_group
root = blender_group
root = boolean_union(root, juicer_group)

# Finalize: Apply all modifiers if any remain (though boolean_union applies them)
# Ensure all objects are selected and apply transforms if necessary for final state
bpy.ops.object.select_all(action='DESELECT')
root.select_set(True)
bpy.context.view_layer.objects.active = root
bpy.ops.object.transform_apply(location=True, rotation=True, scale=True)

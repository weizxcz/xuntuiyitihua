import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with a specific diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(radius, depth, segments=32, name="Cylinder"):
    """Add a cylinder oriented along Z, centered at origin."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0) # Ensure Z-up
    return obj

def add_disc(radius, segments=32, name="Disc"):
    """Add a flat disc oriented along Z, centered at origin."""
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='TRIFAN', segments=segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0)
    return obj

def add_cube(size, name="Cube"):
    """Add a cube centered at origin."""
    # size is total width/height/depth
    half = size / 2.0
    bpy.ops.mesh.primitive_cube_add(size=half, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(radius_bottom, radius_top, depth, segments=32, name="Cone"):
    """Add a cone oriented along Z, centered at origin."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, segments=segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = (0, 0, 0)
    return obj

def add_uvsphere(radius, width_seg=16, height_seg=16, name="UVSphere"):
    """Add a UV sphere centered at origin."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_seg, ring_count=height_seg, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width=0.01, segments=5, name="Bevel"):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name=name, type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    return mod

def boolean_subtract(target, cutter, name="BooleanSubtract"):
    """Perform a boolean subtract using a cutter object."""
    # Ensure cutter is active for the operator or use modifiers
    # Using modifiers is more robust for scripting
    mod = target.modifiers.new(name=name, type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    # Apply immediately to keep scene clean
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)
    return target

def boolean_union(target, cutter, name="BooleanUnion"):
    """Perform a boolean union using a cutter object."""
    mod = target.modifiers.new(name=name, type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = cutter
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter, do_unlink=True)
    return target

def align_axis_to_vector(obj, axis='Z', vector=(0, 0, 1)):
    """Align an object's local axis to a world vector."""
    # This is a simplified helper. For strict Z-up, we just ensure rotation is (0,0,0)
    # or use quaternion rotation.
    if axis == 'Z':
        obj.rotation_euler = (0, 0, 0)
    elif axis == 'X':
        obj.rotation_euler = (0, -1.5708, 0)
    elif axis == 'Y':
        obj.rotation_euler = (1.5708, 0, 0)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        """Return the world position of the object's center."""
        return self.obj.location.copy()

    def face_center_world(self, face_index=0):
        """Return the world position of the center of a specific face."""
        # Requires mesh data
        if self.obj.data.polygons:
            # Get the center of the first polygon in world space
            # Note: This assumes the object is not scaled/rotated in a complex way relative to its local mesh
            # For simple primitives, local center of face mapped to world is sufficient.
            poly = self.obj.data.polygons[face_index]
            # Local center
            local_center = Vector((0, 0, 0))
            for v_idx in poly.vertices:
                local_center += self.obj.data.vertices[v_idx].co
            local_center /= len(poly.vertices)
            
            # Transform to world
            world_center = self.obj.matrix_world @ local_center
            return world_center
        return self.center_world()

    def radial_point_world(self, angle=0, radius=1):

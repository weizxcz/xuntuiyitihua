def make_material(name, diffuse_color):
    """Create a Principled BSDF material with a specific diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive_cube(name, size, location, rotation_euler, material=None):
    """Add a cube, scale it, set location/rotation, and assign material."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        if obj.data.materials:
            obj.data.materials[0] = material
        else:
            obj.data.materials.append(material)
    return obj

def add_primitive_cylinder(name, radius, depth, location, rotation_euler, material=None):
    """Add a cylinder."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        if obj.data.materials:
            obj.data.materials[0] = material
        else:
            obj.data.materials.append(material)
    return obj

def get_face_normal(obj, face_index):
    """Get the world-space normal of a specific face."""
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    face = bm.faces[face_index]
    normal = face.normal.copy()
    bm.free()
    return normal

def get_face_center(obj, face_index):
    """Get the world-space center of a specific face."""
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    face = bm.faces[face_index]
    center = face.calc_center_median()
    bm.free()
    return center

def align_to_face(obj_target, target_obj, target_face_idx, offset_vector_local):
    """
    Aligns obj_target to a specific face of target_obj.
    Assumes obj_target is a box-like object.
    Logic: Move obj_target so its center aligns with the target face center,
    then apply local offset.
    """
    # Calculate target position: Face Center + Offset in Target's Local Space
    face_center_world = get_face_center(target_obj, target_face_idx)
    
    # Convert local offset to world space
    offset_world = offset_vector_local @ target_obj.matrix_world.to_3x3()
    
    new_location = face_center_world + offset_world
    
    # For simple boxes aligned with axes, we just move the center.
    # If rotation is needed to match face normal, we'd rotate here.
    # The prompt implies simple placement relative to faces for boxes.
    obj_target.location = new_location
    
    return obj_target

def align_xy_face_to_xy_face(obj_source, obj_target):
    """
    Aligns the XY face (Top/Bottom) of obj_source to the XY face (Top/Bottom) of obj_target.
    Specifically, aligns the bottom of source to the top of target or similar based on context.
    Here we assume 'Align XY_face' means placing them on the same plane or stacking.
    Given the offset instructions, we usually align the bottom of the new object to the top of the reference.
    """
    # Get top face center of target (Face index 2 is usually top for default cube in Blender if not rotated)
    # Default Cube: -Z is bottom, +Z is top.
    # Let's use a robust method: align the bottom of source to the top of target.
    
    # Target Top Face (Index 2 in default cube)
    target_top_center = get_face_center(obj_target, 2)
    
    # Source Bottom Face (Index 5 in default cube)
    # We want the source's bottom to be at target_top_center + offset
    # But the offset is applied separately. So we align the *base* position.
    
    # Calculate the position where source's bottom touches target's top
    # Source height
    h_source = obj_source.scale[2] * obj_source.dimensions[2] # Actually scale*1.0 since size=1
    # Wait, obj_source.scale is (sx, sy, sz). Dimensions are (1,1,1) before scale.
    # So height is sz.
    
    # We want source.location.z - (sz/2) = target_top_center.z
    # So source.location.z = target_top_center.z + (sz/2)
    
    # However, the prompt says "Align ... to ... Move by offset".
    # So we set location to the alignment point, then add offset.
    
    # Let's just set the location to the face center for now, and let the offset handle the gap.
    # Actually, "Align face to face" usually means touching.
    
    # Let's refine:
    # 1. Determine the "contact plane".
    # 2. Place object such that its relevant face is on that plane.
    
    # For simplicity with the provided offsets (small Z offsets), we will:
    # 1. Calculate the world position of the target face.
    # 2. Place the source object such that its corresponding face is at that position.
    
    # Target Top Face Center
    t_center = get_face_center(obj_target, 2)
    
    # Source Bottom Face is at obj.location.z - (obj.scale.z / 2)
    # We want: obj.location.z - (obj.scale.z / 2) = t_center.z
    # obj.location.z = t_center.z + (obj.scale.z / 2)
    
    obj_source.location.z = t_center.z + (obj_source.scale[2] / 2)
    
    # Align X and Y to center? The prompt doesn't specify X/Y alignment for this step,
    # but usually "Align face" implies centering unless specified.
    # However, subsequent offsets suggest specific X/Y placement.
    # I will leave X/Y alone unless "center" is implied. 
    # Looking at Section 2: "Align velcro_strap.Z_face to sleeve_main.top_face".
    # This implies Z-axis alignment (stacking).
    
    return obj_source

def align_face_to_face(obj_source, obj_target, source_face_idx, target_face_idx, offset_vector_local):
    """
    General face alignment.
    Moves obj_source so that source_face_idx touches target_face_idx.
    Then applies offset_vector_local (in target's local space) to the final position.
    """
    # 1. Get Target Face Center in World Space
    t_center = get_face_center(obj_target, target_face_idx)
    
    # 2. Determine Source Face Normal in Local Space to know which way it points
    # We need to move the source so its face is at t_center.
    # Source Face Center in Local Space
    bm = bmesh.new()
    bm.from_mesh(obj_source.data)
    bm.transform(Matrix.Identity(4)) # Local space
    s_face = bm.faces[source_face_idx]
    s_center_local = s_face.calc_center_median()
    s_normal_local = s_face.normal.copy()
    bm.free()
    
    # 3. Calculate required translation to put source face at target center
    # We need to rotate the source first if normals don't match, but prompt says "Rotate ... so +Z aligns +Z".
    # So normals are likely aligned.
    # If +Z of source aligns with +Z of target:
    # Source Top Face (2) normal is +Z. Target Top Face (2) normal is +Z.
    # To touch, Source Bottom (5) should be at Target Top (2).
    
    # Let's assume standard box orientation where faces are axis aligned.
    # If we just want to place the object "on" the face:
    
    # Vector from Source Center to Source Face Center (Local)
    vec_local = s_center_local
    
    # Transform to World (assuming no rotation yet or rotation is identity)
    # Since we rotate before aligning, obj_source.rotation_euler is set.
    vec_world = vec_local @ obj_source.matrix_world.to_3x3()
    
    # The face center in world space is: obj.location + vec_world
    # We want: obj.location + vec_world = t_center
    # So: obj.location = t_center - vec_world
    
    obj_source.location = t_center - vec_world
    
    # 4. Apply Offset
    # Offset is in Target's local space
    offset_world = offset_vector_local @ obj_target.matrix_world.to_3x3()
    obj_source.location += offset_world
    
    return obj_source

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive(name, shape, size=None, diameter=None, height=None, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    
    if shape == 'cube':
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'cylinder':
        # Cylinder default is radius, height
        r = diameter / 2 if diameter else 0.5
        h = height if height else 1.0
        obj.scale = (r, r, h)
    elif shape == 'disc':
        r = diameter / 2 if diameter else 0.5
        h = height if height else 0.01
        obj.scale = (r, r, h)
    elif shape == 'cone':
        r = diameter / 2 if diameter else 0.5
        h = height if height else 1.0
        obj.scale = (r, r, h)
    elif shape == 'sphere':
        r = diameter / 2 if diameter else 0.5
        obj.scale = (r, r, r)
        
    return obj

def align_face_to_face(obj_top, obj_bottom, offset_z=0):
    """Align obj_top's bottom face to obj_bottom's top face, then apply Z offset."""
    # Get bounding boxes
    min_b, max_b = obj_bottom.bound_box, obj_top.bound_box
    
    # Calculate world coordinates of faces
    # Assuming standard orientation where Z is up
    # Bottom face of top object
    # Top face of bottom object
    
    # We need to move obj_top so its lowest Z matches obj_bottom's highest Z
    # But we must respect local transforms. Simplest is to use world coords.
    
    # Get world matrix
    mat_bottom = obj_bottom.matrix_world
    mat_top = obj_top.matrix_world
    
    # Local min/max Z
    min_z_bottom = min([v[2] for v in min_b])
    max_z_bottom = max([v[2] for v in max_b])
    
    min_z_top = min([v[2] for v in min_b])
    max_z_top = max([v[2] for v in max_b])
    
    # World Z of bottom object's top face
    world_z_bottom_top = mat_bottom.translation.z + max_z_bottom
    
    # World Z of top object's bottom face (before move)
    world_z_top_bottom = mat_top.translation.z + min_z_top
    
    # Delta to align
    delta_z = world_z_bottom_top - world_z_top_bottom + offset_z
    
    obj_top.location.z += delta_z
    
    return obj_top

def align_center_to_center(obj_target, obj_source):
    """Align obj_source center to obj_target center."""
    mat_target = obj_target.matrix_world
    mat_source = obj_source.matrix_world
    
    center_target = mat_target.translation
    center_source = mat_source.translation
    
    delta = center_target - center_source
    obj_source.location += delta
    
    return obj_source

def apply_boolean_union(target, cutter):
    """Apply a boolean union modifier and apply it."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    
    # Add modifier
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = cutter
    
    # Apply
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter if desired, or keep it. Usually keep for debugging, but let's remove to clean up.
    bpy.data.objects.remove(cutter, do_unlink=True)

def apply_boolean_subtract(target, cutter):
    """Apply a boolean subtract modifier and apply it."""
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(cutter, do_unlink=True)

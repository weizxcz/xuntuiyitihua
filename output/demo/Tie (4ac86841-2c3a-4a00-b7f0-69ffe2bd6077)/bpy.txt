def clear_scene():
    """Delete all existing objects and collections."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Remove empty collections if any remain (rare in clean script)
    for col in bpy.data.collections:
        bpy.data.collections.remove(col)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.8
    return mat

def add_primitive_cube(name, size, location, rotation_euler):
    """Add a cube centered at location with given size and rotation."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def get_face_normal(obj, face_index):
    """Get the world-space normal of a specific face."""
    # Ensure object is in world space for calculation
    mat_world = obj.matrix_world
    mat_local = obj.matrix_local
    # Get vertices of the face
    mesh = obj.data
    face = mesh.polygons[face_index]
    verts = [mesh.vertices[i] for i in face.vertices]
    
    # Calculate normal in local space
    v1 = Vector(verts[0].co)
    v2 = Vector(verts[1].co)
    v3 = Vector(verts[2].co)
    normal_local = (v2 - v1).cross(v3 - v1).normalized()
    
    # Transform to world space
    normal_world = normal_local @ mat_world.to_3x3()
    return normal_world

def get_face_center(obj, face_index):
    """Get the world-space center of a specific face."""
    mesh = obj.data
    face = mesh.polygons[face_index]
    center_local = face.center
    center_world = center_local @ obj.matrix_world
    return center_world

def get_face_matrix(obj, face_index):
    """Get the world-space matrix (location + rotation) of a specific face."""
    center = get_face_center(obj, face_index)
    normal = get_face_normal(obj, face_index)
    
    # Create a rotation matrix that aligns +Z with the normal
    # We need to handle the case where normal is close to +Z or -Z
    up = Vector((0, 0, 1))
    if abs(normal.dot(up)) > 0.999:
        rot_quat = up.rotation_difference(normal).to_quaternion()
    else:
        rot_quat = normal.rotation_difference(up).to_quaternion()
    
    mat = Matrix.Translation(center) @ rot_quat.to_matrix().to_4x4()
    return mat

def align_object_to_face(obj_target, obj_source, target_face_idx, source_face_idx):
    """
    Align obj_source to obj_target such that source_face_idx touches target_face_idx.
    Assumes both objects are oriented with +Z up initially.
    """
    # Get matrices for the faces
    mat_target = get_face_matrix(obj_target, target_face_idx)
    mat_source = get_face_matrix(obj_source, source_face_idx)
    
    # We want to move obj_source so its source face is at the target face location
    # and oriented similarly.
    
    # Current world matrix of source
    mat_source_world = obj_source.matrix_world
    
    # Calculate the transformation needed to move source face to target face
    # Inverse of source face matrix transforms world to source-face-local
    # Then we apply target face matrix to get world position
    
    # Actually, simpler:
    # 1. Move source object so its source face center is at target face center
    # 2. Rotate source object so its source face normal aligns with target face normal
    
    target_center = mat_target.to_translation()
    target_rot = mat_target.to_3x3()
    
    source_center = get_face_center(obj_source, source_face_idx)
    source_normal = get_face_normal(obj_source, source_face_idx)
    
    # Calculate rotation to align source normal to target normal
    # We assume +Z of the object aligns with the face normal for flat faces
    # If the face is the +Z face (index 0 in default cube usually), normal is (0,0,1)
    # If the face is the -Z face (index 1), normal is (0,0,-1)
    
    # Let's use a simpler approach:
    # 1. Set location to target center
    # 2. Rotate to match normal
    
    obj_source.location = target_center
    
    # Calculate rotation
    # We want the source object's local +Z (or relevant axis) to align with target normal
    # But we need to preserve the object's orientation relative to its own axes.
    # Since we created them with +Z up, the face normals are along local Z or -Z.
    
    # Determine which axis of the source object corresponds to the face normal
    # For a default cube:
    # Face 0: +Z (0,0,1)
    # Face 1: -Z (0,0,-1)
    # Face 2: +X (1,0,0)
    # Face 3: -X (-1,0,0)
    # Face 4: +Y (0,1,0)
    # Face 5: -Y (0,-1,0)
    
    # We need to rotate the source object so that its local face normal aligns with target normal
    # And its local up vector aligns with target up vector (cross product of normal and right)
    
    # Get local up vector of the face in source object's local space
    # This is tricky without knowing the exact face index mapping.
    # Let's assume standard cube face indices:
    # 0: +Z, 1: -Z, 2: +X, 3: -X, 4: +Y, 5: -Y
    
    # Let's just use the matrix multiplication approach
    # We want to transform obj_source such that its source_face_matrix becomes mat_target
    
    # Current source face matrix in world space
    # M_source_face = M_source_world * M_source_face_local
    # We want M_source_face_new = M_target
    
    # So M_source_world_new * M_source_face_local = M_target
    # M_source_world_new = M_target * M_source_face_local^-1
    
    # Get local matrix of the face relative to the object
    # For a cube, the face center is at +/- size/2 along an axis.
    # The face matrix is Translation(center) * Rotation(normal)
    
    # Let's compute the local face matrix for the source
    # This depends on the face index.
    
    # Simplified:
    # 1. Move source to target center
    # 2. Rotate source to align normals
    
    # Calculate rotation quaternion
    # We need to align the source object's local axis that points to the face
    # with the target normal.
    
    # For a cube, if face_idx is 0 (+Z), local axis is +Z.
    # If face_idx is 1 (-Z), local axis is -Z.
    
    local_axis = Vector((0, 0, 1))
    if source_face_idx == 1: # -Z face
        local_axis = Vector((0, 0, -1))
    elif source_face_idx == 2: # +X face
        local_axis = Vector((1, 0, 0))
    elif source_face_idx == 3: # -X face
        local_axis = Vector((-1, 0, 0))
    elif source_face_idx == 4: # +Y face
        local_axis = Vector((0, 1, 0))
    elif source_face_idx == 5: # -Y face
        local_axis = Vector((0, -1, 0))
        
    # We want to rotate local_axis to align with target_normal
    # And we want to preserve the "up" direction relative to the face
    
    # Get the "right" vector for the target face
    # Right = cross(Up, Normal) or cross(Normal, Up) depending on convention
    # Let's use a reference up vector, e.g., world Y or Z
    
    # A robust way:
    # 1. Align local_axis to target_normal
    # 2. Align the object's local "up" (perpendicular to local_axis) to target's "up"
    
    # Get target face's local up vector (perpendicular to normal)
    # We can use the cross product of target normal and world up (0,0,1)
    # If that's too small, use world right (1,0,0)
    
    target_up = Vector((0, 0, 1))
    if abs(target_normal.dot(target_up)) > 0.9:
        target_up = Vector((1, 0, 0))
    target_right = target_normal.cross(target_up).normalized()
    target_up = target_right.cross(target_normal).normalized()
    
    # Get source face's local up vector
    # This is the vector in local space that is "up" for that face
    # For +Z face, up is +Y. For -Z face, up is -Y.
    # For +X face, up is +Z. For -X face, up is -Z.
    # For +Y face, up is -Z. For -Y face, up is +Z.
    
    local_up = Vector((0, 1, 0))
    if source_face_idx == 1: # -Z
        local_up = Vector((0, -1, 0))
    elif source_face_idx == 2: # +X
        local_up = Vector((0, 0, 1))
    elif source_face_idx == 3: # -X
        local_up = Vector((0, 0, -1))
    elif source_face_idx == 4: # +Y
        local_up = Vector((0, 0, -1))
    elif source_face_idx == 5: # -Y
        local_up = Vector((0, 0, 1))
        
    # Now we have:
    # local_axis -> target_normal
    # local_up -> target_up
    
    # Construct rotation matrix
    # Columns are the new basis vectors in world space
    # col1 = target_right (image of local_right)
    # col2 = target_up (image of local_up)
    # col3 = target_normal (image of local_axis)
    
    # We need local_right
    local_right = local_up.cross(local_axis).normalized()
    
    # Image of local_right
    target_right_calc = target_up.cross(target_normal).normalized()
    
    rot_mat = Matrix((
        target_right_calc,
        target_up,
        target_normal
    )).transposed()
    
    # Apply rotation
    obj_source.rotation_euler = rot_mat.to_euler()
    
    # Re-apply location because rotation might have moved it if we didn't center it first
    # But we set location to target_center before rotation.
    # However, rotation happens around origin.
    # So we need to:
    # 1. Move to origin
    # 2. Rotate
    # 3. Move to target center
    
    # Let's redo this properly
    # 1. Move source to origin
    obj_source.location = Vector((0,0,0))
    # 2. Rotate
    obj_source.rotation_euler = rot_mat.to_euler()
    # 3. Move to target center
    obj_source.location = target_center
    
    # Apply transforms to lock in
    apply_transforms(obj_source)

def offset_object(obj, offset_vec):
    """Move object by offset_vec in world space."""
    obj.location += offset_vec

def bevel_edges(obj, edge_indices, radius, segments):
    """Apply bevel modifier to specific edges."""
    bpy.context.view_layer.objects.active = obj
    obj.select_set(True)
    bpy.ops.object.mode_set(mode='EDIT')
    
    # Deselect all edges
    bpy.ops.mesh.select_all(action='DESELECT')
    
    # Select specific edges
    for idx in edge_indices:
        obj.data.edges[idx].select = True
        
    bpy.ops.mesh.bevel(offset=radius, segments=segments, clamp_overlap=False)
    bpy.ops.object.mode_set(mode='OBJECT')

def boolean_subtract(cutter, target):
    """Perform boolean subtract using modifier."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    target.select_set(True)
    cutter.select_set(True)
    
    # Add Boolean modifier to target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_obj, target):
    """Perform boolean union."""
    bpy.context.view_layer.objects.active = target
    target.select_set(True)
    union_obj.select_set(True)
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(union_obj, do_unlink=True)

def assign_material(obj, mat_name):
    """Assign material to object."""
    if mat_name in bpy.data.materials:
        mat = bpy.data.materials[mat_name]
        if obj.data.materials:
            obj.data.materials[0] = mat
        else:
            obj.data.materials.append(mat)

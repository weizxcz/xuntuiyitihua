def clear_scene():

    # Clear collections if necessary, though delete usually handles it

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Roughness'].default_value = 0.5
    bsdf.inputs['Metallic'].default_value = 0.9
    return mat

def add_cylinder(name, radius, depth, segments=32):

    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, end_fill_type='TRIFAN')
    obj = bpy.context.active_object
    obj.name = name
    return obj

def offset_object(obj, offset_vector):
    """Move object by offset in its local axes."""
    # Convert local offset to world offset
    world_offset = obj.matrix_world.to_3x3() @ offset_vector

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using cutter on target."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Add Boolean modifier to target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

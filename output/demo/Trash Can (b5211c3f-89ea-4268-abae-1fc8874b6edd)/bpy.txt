def make_material(name, diffuse_color=(0.5, 0.5, 0.5, 1.0)):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object's first material slot."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(radius, depth, segments=32, name="Cylinder"):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def average(points):
    """Calculate the average position of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find centers and faces of objects."""
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        # bound_box is in local space, need to transform to world
        self.world_points = [obj.matrix_world @ Vector(p) for p in self.bound_box]

    def center_world(self):
        """Return the world-space center of the object's bounding box."""
        return average(self.world_points)

    def face_center_world(self, face_type='bottom'):
        """Return the world-space center of a specific face (bottom or top)."""
        # Assuming Z-up orientation for primitives
        # Find min/max Z
        z_vals = [p.z for p in self.world_points]
        min_z = min(z_vals)
        max_z = max(z_vals)
        
        if face_type == 'bottom':
            target_z = min_z
        elif face_type == 'top':
            target_z = max_z
        else:
            target_z = (min_z + max_z) / 2
            
        # Filter points close to target_z
        close_points = [p for p in self.world_points if abs(p.z - target_z) < 0.0001]
        if close_points:
            return average(close_points)
        return self.center_world()

    def radial_point_world(self, angle=0):
        """Return a point on the radial edge at a given angle (radians) from center."""
        center = self.center_world()
        # Approximate radius from center to a corner in XY plane
        # This is a rough approximation for cylindrical objects
        max_dist = 0
        for p in self.world_points:
            dist = math.sqrt((p.x - center.x)**2 + (p.y - center.y)**2)
            if dist > max_dist:
                max_dist = dist
        
        x = center.x + max_dist * math.cos(angle)
        y = center.y + max_dist * math.sin(angle)
        z = center.z
        return Vector((x, y, z))

def align_objects(source_obj, target_obj, align_center_xy=False, align_bottom_top=False):
    """
    Align source_obj to target_obj.
    align_center_xy: Aligns XY centers.
    align_bottom_top: Aligns source bottom to target top (or similar face logic).
    """
    src_loc = source_obj.location
    tgt_loc = target_obj.location
    
    # 1. Align XY Centers
    if align_center_xy:
        dx = tgt_loc.x - src_loc.x
        dy = tgt_loc.y - src_loc.y
        src_loc.x += dx
        src_loc.y += dy
        
    # 2. Align Faces (Z-axis)
    if align_bottom_top:
        # We need to move source so its bottom face touches target's top face
        # Calculate current bottom and top Z of both objects
        
        # Helper to get world bounds
        def get_world_bounds(obj):
            bb = obj.bound_box
            wm = obj.matrix_world
            pts = [wm @ Vector(p) for p in bb]
            zs = [p.z for p in pts]
            return min(zs), max(zs)
            
        src_min_z, src_max_z = get_world_bounds(source_obj)
        tgt_min_z, tgt_max_z = get_world_bounds(target_obj)
        
        # Target top face is tgt_max_z
        # Source bottom face is src_min_z
        # We want: src_min_z_new = tgt_max_z
        # Current src_min_z is relative to src_loc. 
        # Actually, since we just moved X/Y, Z is still original.
        # The object's geometry is fixed relative to its location.
        # So src_min_z is effectively src_loc.z + (local_min_z)
        
        # Let's use the Locator class for robustness
        src_loc_obj = Locator(source_obj)
        tgt_loc_obj = Locator(target_obj)
        
        src_bottom_z = src_loc_obj.face_center_world('bottom').z
        tgt_top_z = tgt_loc_obj.face_center_world('top').z
        
        # The object's local bottom Z relative to its origin
        # For a cylinder centered at 0,0,0 with height H, bottom is -H/2
        # So local_bottom_z = -height/2
        # src_loc.z + local_bottom_z = src_bottom_z
        # We want: src_loc.z + local_bottom_z = tgt_top_z
        # src_loc.z = tgt_top_z - local_bottom_z
        
        # Get local bounds
        src_bb = source_obj.bound_box
        src_local_min_z = min(p.z for p in src_bb)
        
        new_z = tgt_top_z - src_local_min_z
        source_obj.location.z = new_z

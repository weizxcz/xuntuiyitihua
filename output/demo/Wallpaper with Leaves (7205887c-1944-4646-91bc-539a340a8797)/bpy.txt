import bpy
import math
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_color = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, scale):
    bpy.ops.mesh.primitive_cube_add(size=1, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation_euler
    obj.location = location
    obj.scale = scale
    return obj

def apply_transform(obj):
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.transform_apply(location=True, rotation=True, scale=True)

def rotate_local_z(obj, angle_deg):
    """Rotate object around its local Z axis by angle_deg"""
    angle_rad = math.radians(angle_deg)
    # Get current world rotation
    rot_world = obj.rotation_euler.to_matrix()
    # Create local rotation matrix
    rot_local = Matrix.Rotation(angle_rad, 4, 'Z')
    # Convert local rotation to world space and apply
    # Note: Blender's rotation_euler is ZYX. Direct matrix multiplication on rotation_euler is tricky.
    # Safer approach: Convert to Quaternion, rotate, convert back.
    q_local = Matrix.Rotation(angle_rad, 4, 'Z').to_3x3().to_4x4()
    # We need to rotate the object's orientation.
    # Current orientation in world space
    q_world = obj.rotation_euler.to_quaternion()
    # We want to apply a local Z rotation.
    # q_new = q_world * q_local (if q_local is in object space)
    # Actually, simpler: rotate the object's data or use transform tool logic.
    # Let's use the standard way: rotate around local axis.
    # Since we just created it and aligned Z to Z, local Z is world Z.
    # But after scaling/offset, local axes might differ if non-uniform scale was applied before rotation.
    # Here scale is applied after rotation in the prompt? No: "Rotate... Then move... Rotate... Scale".
    # Wait, prompt says: Anchor -> Move -> Rotate -> Scale.
    # So when we rotate, scale is (1,1,1). Local Z is World Z.
    # So we can just add to rotation_euler.z.
    obj.rotation_euler.z += angle_rad

def set_material(obj, mat):
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

mat_wallpaper_base = make_material("wallpaper_base", (0.95, 0.95, 0.9, 1.0))

# --- Block 2: Geometry Generation ---

# --- SECTION 1 – Create wallpaper base substrate ---
# Create primitive cuboid with size 1.0×1.0×0.001 m and name it wallpaper_base.
# Rotate wallpaper_base so its +Z axis aligns +Z (world).
# Anchor wallpaper_base to world origin.
# Then move by offset(0,0,0).

wallpaper_base = add_cube("wallpaper_base", (1.0, 1.0, 0.001), (0, 0, 0), (0, 0, 0), (1, 1, 1))

# --- SECTION 2 – Create leaf template geometry ---
# Create primitive cuboid with size 0.05×0.02×0.001 m and name it leaf_template.
# Rotate leaf_template so its +Z axis aligns +Z (world).
# Anchor leaf_template to world origin.
# Then move by offset(0,0,0).

leaf_template = add_cube("leaf_template", (0.05, 0.02, 0.001), (0, 0, 0), (0, 0, 0), (1, 1, 1))
set_material(leaf_template, mat_leaf_green)

# --- SECTION 3 – Instantiate leaf instances ---

# Helper to create and configure a leaf instance
def create_leaf_instance(name, offset_x, offset_y, offset_z, rot_z_deg, scale_val):
    # Create primitive cuboid with size 0.05×0.02×0.001 m
    obj = add_cube(name, (0.05, 0.02, 0.001), (0, 0, 0), (0, 0, 0), (1, 1, 1))
    set_material(obj, mat_leaf_green)
    
    # Rotate so +Z aligns +Z (already default)
    # Anchor to world origin (already at 0,0,0)
    # Move by offset
    obj.location = (offset_x, offset_y, offset_z)
    
    # Rotate so its local +Z axis achieves rotation=axis:Z, angle
    # Since scale is (1,1,1) at this point, local Z is world Z.
    rotate_local_z(obj, rot_z_deg)
    
    # Scale
    obj.scale = (scale_val, scale_val, 1.0)
    
    return obj

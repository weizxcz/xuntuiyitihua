def clear_scene():
    """Delete all existing objects in the scene."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for col in bpy.data.collections:
        bpy.data.collections.remove(col)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    # Set roughness/metallic defaults if needed, but prompt only specified diffuse
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube object with specific size, location, rotation and material."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """
    Align a specific axis of an object to a target world vector.
    axis_to_align: 'X', 'Y', or 'Z'
    target_vector: Vector in world space
    """
    # Get current rotation matrix
    rot_mat = obj.rotation_euler.to_matrix()
    
    # Get the current direction of the axis to align
    if axis_to_align == 'X':
        current_axis = rot_mat.col[0]
    elif axis_to_align == 'Y':
        current_axis = rot_mat.col[1]
    elif axis_to_align == 'Z':
        current_axis = rot_mat.col[2]
    
    # Calculate rotation to align current_axis to target_vector
    # We need to find a rotation matrix R such that R * current_axis = target_vector
    # This is complex to do directly on Euler. 
    # Alternative: Use quaternion rotation from current_axis to target_vector
    # and apply it to the object's current rotation.
    
    # Normalize vectors
    v1 = current_axis.normalized()
    v2 = target_vector.normalized()
    
    # Avoid division by zero or invalid cross products
    if v1 == v2:
        return # Already aligned
    
    # Calculate rotation quaternion
    q = v1.rotation_difference(v2)
    
    # Apply this rotation to the object's current rotation
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = q @ obj.rotation_quaternion
    obj.rotation_mode = 'XYZ' # Reset to Euler for consistency if needed

def get_face_center(obj, face_axis, face_side=1):
    """
    Get the world location of a specific face center of an object.
    face_axis: 'X', 'Y', 'Z'
    face_side: 1 for positive axis, -1 for negative axis
    """
    # Get object matrix world
    mat = obj.matrix_world
    
    # Get local face center
    # Cube primitives have faces at +/- 0.5 * scale
    # We need to account for scale
    scale = obj.scale
    local_offset = Vector((0, 0, 0))
    if face_axis == 'X':
        local_offset.x = face_side * 0.5 * scale.x
    elif face_axis == 'Y':
        local_offset.y = face_side * 0.5 * scale.y
    elif face_axis == 'Z':
        local_offset.z = face_side * 0.5 * scale.z
        
    # Transform to world space
    return mat @ local_offset

def place_object_relative(obj_source, obj_target, source_axis, source_side, target_axis, target_side):
    """
    Moves obj_source so that its source_axis face is at obj_target's target_axis face.
    """
    target_loc = get_face_center(obj_target, target_axis, target_side)
    current_source_face_loc = get_face_center(obj_source, source_axis, source_side)
    offset = target_loc - current_source_face_loc
    obj_source.location += offset

def align_axes(obj_source, source_axis, obj_target, target_axis):
    """
    Rotates obj_source so that its source_axis aligns with obj_target's target_axis.
    Assumes we want them to point in the same direction.
    """
    # Get current direction of source_axis in world space
    mat = obj_source.matrix_world
    if source_axis == 'X':
        src_dir = mat.col[0].xyz
    elif source_axis == 'Y':
        src_dir = mat.col[1].xyz
    elif source_axis == 'Z':
        src_dir = mat.col[2].xyz
    
    # Get target direction
    mat_t = obj_target.matrix_world
    if target_axis == 'X':
        tgt_dir = mat_t.col[0].xyz
    elif target_axis == 'Y':
        tgt_dir = mat_t.col[1].xyz
    elif target_axis == 'Z':
        tgt_dir = mat_t.col[2].xyz
        
    src_dir = src_dir.normalized()
    tgt_dir = tgt_dir.normalized()
    
    if src_dir == tgt_dir:
        return
        
    q = src_dir.rotation_difference(tgt_dir)
    obj_source.rotation_mode = 'QUATERNION'
    obj_source.rotation_quaternion = q @ obj_source.rotation_quaternion
    obj_source.rotation_mode = 'XYZ'

def apply_offset(obj, offset_vec):
    """Apply a local/world offset to an object's location."""
    obj.location += offset_vec

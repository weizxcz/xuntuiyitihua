import bpy
import bmesh
import mathutils
from mathutils import Vector, Matrix

# ==============================================================================
# HELPER FUNCTIONS
# ==============================================================================

def clear_scene():
    """Delete all existing objects."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for col in bpy.data.collections:
        col.objects.unlink_all()
        bpy.data.collections.remove(col)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def assign_material(obj, mat):
    """Assign a material to an object's first material slot."""
    if len(obj.data.materials) == 0:
        obj.data.materials.append(mat)
    else:
        obj.data.materials[0] = mat

def add_cube(name, size, location, rotation_euler):
    """Add a cube object with specific size and transform."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    # Scale to size (cube default is 2x2x2, so divide by 2)
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    obj.select_set(False)
    return obj

def align_center_to_center(obj_target, obj_source):
    """Align obj_source center to obj_target center."""
    # Calculate offset needed
    offset = obj_target.location - obj_source.location
    obj_source.location += offset

def align_center_to_face(obj_target, obj_source, face_axis='Z'):
    """
    Align obj_source center to obj_target's specific face center.
    face_axis: 'X', 'Y', or 'Z' (positive direction face) or negative.
    We assume axis-aligned bounding boxes for simplicity as per prompt hints.
    """
    # Get target face center in world space
    # Since objects are axis aligned, we can calculate this from bounds
    # Or simpler: use the object's matrix_world to find the face center
    
    # Determine the vector to the face
    # For a cube at origin with scale S, face center is at +/- S/2 along axis
    # We need to account for rotation.
    
    # Get target's matrix
    mat_t = obj_target.matrix_world
    mat_s = obj_source.matrix_world
    
    # Define local face normal vector
    if face_axis == 'Z':
        local_face_vec = Vector((0, 0, 0.5)) # Top face local
    elif face_axis == '-Z':
        local_face_vec = Vector((0, 0, -0.5)) # Bottom face local
    elif face_axis == 'X':
        local_face_vec = Vector((0.5, 0, 0)) # Right face local
    elif face_axis == '-X':
        local_face_vec = Vector((-0.5, 0, 0)) # Left face local
    elif face_axis == 'Y':
        local_face_vec = Vector((0, 0.5, 0)) # Front face local
    elif face_axis == '-Y':
        local_face_vec = Vector((0, -0.5, 0)) # Back face local
    else:
        raise ValueError(f"Unknown face axis: {face_axis}")
        
    # Transform local face vector to world space relative to target center
    # The face center in world space is: TargetLoc + (TargetRot * LocalFaceVec * TargetScale)
    # Note: Cube scale is applied to the 2x2x2 default. 
    # The half-extent in world space is scale * 1.0 (since default size is 2, half is 1).
    
    # Actually, simpler approach:
    # Get the world matrix of the target.
    # The face center is located at the target's location + the rotated half-dimensions.
    
    # Get half dimensions in world space
    # Half dims = scale * 1.0 (because default cube is 2 units wide, so half is 1)
    half_dims = obj_target.scale
    
    # Construct the offset vector in world space
    # We need to rotate the local axis vector by the target's rotation
    rot_mat = mat_t.to_3x3()
    
    # Determine which axis corresponds to the face
    # If face_axis is 'Z', we look at the Z component of the scale
    # And the direction (positive or negative)
    
    if face_axis == 'Z':
        world_offset = rot_mat @ Vector((0, 0, half_dims[2]))
    elif face_axis == '-Z':
        world_offset = rot_mat @ Vector((0, 0, -half_dims[2]))
    elif face_axis == 'X':
        world_offset = rot_mat @ Vector((half_dims[0], 0, 0))
    elif face_axis == '-X':
        world_offset = rot_mat @ Vector((-half_dims[0], 0, 0))
    elif face_axis == 'Y':
        world_offset = rot_mat @ Vector((0, half_dims[1], 0))
    elif face_axis == '-Y':
        world_offset = rot_mat @ Vector((0, -half_dims[1], 0))
        
    target_face_center = obj_target.location + world_offset
    
    # Align source center to this point
    align_center_to_center(obj_target, obj_source)
    # Now move source to target_face_center
    # Current source location is target.location. We want it at target_face_center.
    # So we add the offset.
    obj_source.location = target_face_center

def apply_offset(obj, offset_vec):
    """Apply an offset vector to the object's location."""
    obj.location += offset_vec

def create_component(name, size, location, rotation, material=None):
    """Helper to create a cube, set name, transform, and material."""
    obj = add_cube(name, size, location, rotation)
    if material:
        assign_material(obj, material)
    return obj

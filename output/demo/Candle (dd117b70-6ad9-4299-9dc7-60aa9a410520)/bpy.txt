import bpy

def clear_scene():
    """Delete every existing object to start from a clean scene."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for col in bpy.data.collections:
        col.objects.clear()

def set_units_metres():
    """Set the length unit to metres."""
    bpy.context.scene.unit_settings.scale_length = 1.0
    bpy.context.scene.unit_settings.system = 'METRIC'

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder primitive, centered at origin."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube primitive, centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_segments=32, height_segments=16):
    """Add a UV Sphere primitive, centered at origin."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_bottom, radius_top, depth):
    """Add a cone primitive, centered at origin."""
    bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=radius_bottom, radius2=radius_top, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width, segments=5, profile=0.5, limit_method='ANGLE'):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.profile = profile
    mod.limit_method = limit_method
    mod.angle_limit = math.radians(30) # Default angle limit
    return mod

def boolean_subtract(cutter_obj, target_obj):
    """Perform a boolean subtract operation using a cutter object."""
    # Ensure target is active for the modifier addition
    bpy.context.view_layer.objects.active = target_obj
    
    # Add Boolean Modifier
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter object
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return target_obj

def boolean_union(union_obj, target_obj):
    """Perform a boolean union operation."""
    bpy.context.view_layer.objects.active = target_obj
    
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union_obj, do_unlink=True)
    return target_obj

def align_axis_to_vector(obj, axis, vector):
    """Align an object's local axis to a world vector."""
    # Get current rotation
    rot_euler = obj.rotation_euler.copy()
    
    # Define target axis vector in local space
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    elif axis == 'Z':
        local_axis = Vector((0, 0, 1))
    else:
        raise ValueError("Axis must be X, Y, or Z")
        
    # Calculate rotation matrix to align local_axis to world vector
    # We need to rotate the object so that its local axis points in the direction of 'vector'
    # This is equivalent to finding the rotation that maps local_axis to vector.normalized()
    
    # Normalize input vector
    target_vec = vector.normalized()
    
    # Create rotation matrix
    # Using quaternion rotation from one vector to another
    quat = local_axis.rotation_difference(target_vec)
    
    # Apply rotation
    obj.rotation_euler = quat.to_euler()
    obj.update_tag()

def average(points):
    """Calculate the average position of a list of points."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to calculate positions relative to an object."""
    def __init__(self, obj):

    def center_world(self):
        """Return the world center of the object."""
        return self.matrix_world.to_translation()

        # This is a simplified approximation for primitives

        if self.obj.type == 'MESH':

            # Since we control creation, we can infer.
            # Let's stick to center_world for general alignment unless specific face logic is needed.
            # For "top_face" or "bottom_face", we can calculate based on height.
            pass
        return self.center_world()

    def top_face_world(self):
        """Return the world position of the top face center (assuming Z-up cylinder/cone)."""
        # Local top face center is (0, 0, radius/height depending on primitive)
        # For a cylinder centered at 0,0,0 with height H, top is at Z = H/2
        # We need to transform this local point to world space
        local_top = Vector((0, 0, self.obj.dimensions.z / 2))
        return self.matrix_world @ local_top

    def bottom_face_world(self):
        """Return the world position of the bottom face center."""
        local_bottom = Vector((0, 0, -self.obj.dimensions.z / 2))
        return self.matrix_world @ local_bottom

    def radial_point_world(self, angle=0, radius_factor=1.0):
        """Return a point on the radial edge."""
        # Simplified: returns a point on the X axis at the edge
        # Local radial point at angle 0 is (radius, 0, 0)
        # We need the radius. For cylinder/cone, it's dimensions.x / 2
        radius = self.obj.dimensions.x / 2
        local_point = Vector((radius * radius_factor, 0, 0))
        return self.matrix_world @ local_point

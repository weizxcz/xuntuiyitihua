def clear_scene():
    """Delete every existing object to start from a clean scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Ensure collection is clean
    for col in bpy.data.collections:
        bpy.data.collections.remove(col)
    # Re-add default collection if needed, though usually not strictly necessary for script logic

def set_units_metres():
    """Set the length unit to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a principled BSDF material with given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Roughness'].default_value = 0.8
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube primitive."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):

def align_axis_to_vector(obj, axis, vector):
    """Rotate object so its local axis aligns with the given world vector."""
    # Current axis vectors
    axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1)),
        '-X': Vector((-1, 0, 0)),
        '-Y': Vector((0, -1, 0)),
        '-Z': Vector((0, 0, -1))
    }
    
    target = axes.get(axis)
    if not target:
        return

    # Normalize
    target = target.normalized()
    current = obj.matrix_world.to_3x3() @ Vector((0,0,1)) # Default Z alignment check, but we need specific axis
    
    # Actually, simpler approach: Calculate rotation matrix from current axis to target
    # We need to know which local axis is currently pointing where? 
    # No, we want to rotate the object such that its LOCAL 'axis' points to 'vector'.
    
    # Get current direction of the local axis in world space
    if axis == 'X': local_dir = Vector((1, 0, 0))
    elif axis == '-X': local_dir = Vector((-1, 0, 0))
    elif axis == 'Y': local_dir = Vector((0, 1, 0))
    elif axis == '-Y': local_dir = Vector((0, -1, 0))
    elif axis == 'Z': local_dir = Vector((0, 0, 1))
    elif axis == '-Z': local_dir = Vector((0, 0, -1))
    
    # We want to rotate the object so that local_dir becomes target
    rot_matrix = local_dir.rotation_difference(target).to_matrix().to_4x4()
    
    # Apply rotation relative to current orientation
    # We need to multiply the current world matrix by the rotation
    # But bpy.ops.object.rotation_euler expects Euler. 
    # Let's use mathutils rotation directly on the object's rotation_euler or matrix.
    
    # Easier: Set rotation using quaternion
    q = local_dir.rotation_difference(target)
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = q

def get_face_center(obj, face_type):
    """Get the world center of a specific face (left, right, front, back, top, bottom)."""
    # This assumes the object is axis-aligned or we are working in local space relative to parent/anchor
    # For simplicity in this script, we assume standard orientations or calculate based on bounds.
    # However, the prompt implies "left_face", "right_face" etc.
    # We will calculate bounding box extremes in World Space.
    
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    
    min_x = min(v.co.x for v in bm.verts)
    max_x = max(v.co.x for v in bm.verts)
    min_y = min(v.co.y for v in bm.verts)
    max_y = max(v.co.y for v in bm.verts)
    min_z = min(v.co.z for v in bm.verts)
    max_z = max(v.co.z for v in bm.verts)
    
    center = Vector((0,0,0))
    
    if face_type == 'left_face':
        center = Vector((min_x, 0, 0))
    elif face_type == 'right_face':
        center = Vector((max_x, 0, 0))
    elif face_type == 'front_face':
        center = Vector((0, max_y, 0))
    elif face_type == 'back_face':
        center = Vector((0, min_y, 0))
    elif face_type == 'top_face':
        center = Vector((0, 0, max_z))
    elif face_type == 'bottom_face':
        center = Vector((0, 0, min_z))
    elif face_type == 'center':
        center = Vector(((min_x+max_x)/2, (min_y+max_y)/2, (min_z+max_z)/2))
        
    bm.free()
    return center

def boolean_subtract(cutter, target, name):
    """Perform boolean subtract using modifier stack."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name=name, type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter
    
    # Apply
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)
    return target

def boolean_union(objects, name):
    """Boolean union a list of objects into the first one."""
    if not objects:
        return None
    
    base = objects[0]
    bpy.context.view_layer.objects.active = base
    bpy.ops.object.select_all(action='DESELECT')
    base.select_set(True)
    
    for obj in objects[1:]:
        mod = base.modifiers.new(name=f"Union_{obj.name}", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        bpy.ops.object.modifier_apply(modifier=mod.name)
        bpy.data.objects.remove(obj, do_unlink=True)
        
    base.name = name
    return base

def align_to_center(obj, anchor_obj):
    """Align object center to anchor object center."""
    anchor_center = get_face_center(anchor_obj, 'center')
    obj.location = anchor_center

def offset_object(obj, offset_vec):
    """Apply offset to object location."""
    obj.location += offset_vec

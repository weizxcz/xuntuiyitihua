import bpy
import bmesh
import mathutils
from mathutils import Vector, Matrix

# ==============================================================================
# HELPER FUNCTIONS

def clear_scene():
    """Delete every existing object to start from a clean scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Ensure collections are clean
    for col in bpy.data.collections:
        col.objects.link(bpy.context.collection)

def set_units_metres():
    """Set the length unit to metres."""
    bpy.context.scene.unit_settings.scale_length = 1.0
    bpy.context.scene.unit_settings.system = 'METRIC'

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def apply_material(obj, mat):
    """Apply a material to an object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube primitive."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation
    return obj

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    return obj

def add_uvsphere(name, radius, location=(0,0,0), rotation=(0,0,0)):
    """Add a UV sphere primitive."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    return obj

def align_to_vector(obj, axis='Z', vector=Vector((0,0,1))):
    """Rotate object so its local axis aligns with the given world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else: # Z
        local_axis = Vector((0,0,1))
    
    # Calculate rotation
    rot_mat = local_axis.rotation_difference(vector).to_matrix().to_4x4()
    obj.rotation_euler = rot_mat.to_euler()

def get_face_center(obj, face_type='XY'):
    """
    Get the world center of a specific face of an object.
    face_type: 'XY' (top/+Z), 'XZ' (front/+Y), 'YZ' (right/+X), etc.
    For simplicity in this script, we assume axis-aligned objects mostly,
    but we calculate based on bounding box or vertices for robustness.
    """
    # Transform vertices to world space
    verts_world = [obj.matrix_world @ v.co for v in obj.data.vertices]
    
    if face_type == 'XY': # Top face (+Z)
        max_z = max(v[2] for v in verts_world)
        center_z = [v for v in verts_world if abs(v[2] - max_z) < 1e-5]
        cx = sum(v[0] for v in center_z) / len(center_z)
        cy = sum(v[1] for v in center_z) / len(center_z)
        return Vector((cx, cy, max_z))
    elif face_type == 'bottom': # Bottom face (-Z)
        min_z = min(v[2] for v in verts_world)
        center_z = [v for v in verts_world if abs(v[2] - min_z) < 1e-5]
        cx = sum(v[0] for v in center_z) / len(center_z)
        cy = sum(v[1] for v in center_z) / len(center_z)
        return Vector((cx, cy, min_z))
    elif face_type == 'center':
        # Average of all vertices
        cx = sum(v[0] for v in verts_world) / len(verts_world)
        cy = sum(v[1] for v in verts_world) / len(verts_world)
        cz = sum(v[2] for v in verts_world) / len(verts_world)
        return Vector((cx, cy, cz))
    elif face_type == 'top': # Same as XY for upright objects
        return get_face_center(obj, 'XY')
    elif face_type == 'bottom_face':
        return get_face_center(obj, 'bottom')
    elif face_type == 'top_face':
        return get_face_center(obj, 'XY')
    elif face_type == 'side_at':
        # Placeholder for complex side logic, usually handled by offset
        return Vector((0,0,0))
    return Vector((0,0,0))

def align_face_to_point(obj, face_type, target_point):
    """Move object so its face center aligns with target_point."""
    current_center = get_face_center(obj, face_type)
    obj.location += (target_point - current_center)

def align_face_to_face(obj_src, src_face, obj_dst, dst_face):
    """Align src_face to dst_face of dst object."""
    dst_center = get_face_center(obj_dst, dst_face)
    align_face_to_point(obj_src, src_face, dst_center)

def align_face_to_object_center(obj_src, src_face, obj_dst):
    """Align src_face to the center of dst object."""
    dst_center = get_face_center(obj_dst, 'center')
    align_face_to_point(obj_src, src_face, dst_center)

def apply_offset(obj, offset_vec):
    """Apply an offset vector to the object's location."""
    obj.location += offset_vec

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using cutter on target."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_name, target_name):
    """Perform boolean union."""
    target = bpy.data.objects[target_name]
    union = bpy.data.objects[union_name]
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union, do_unlink=True)

def add_bevel(obj_name, width=0.01, segments=4):
    """Add a bevel modifier to an object."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for simplicity in this script flow
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

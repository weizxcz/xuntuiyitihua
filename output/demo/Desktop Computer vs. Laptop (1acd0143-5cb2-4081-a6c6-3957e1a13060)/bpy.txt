# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive(name, shape, size, location, rotation_euler, material=None):
    """Helper to add a primitive mesh object."""
    if shape == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'cylinder':
        bpy.ops.mesh.primitive_cylinder_add(vertices=32, radius=0.5, depth=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'disc':
        bpy.ops.mesh.primitive_circle_add(vertices=32, radius=0.5, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
        # Fill it if it's not already
        if not obj.data.polygons:
            bpy.ops.mesh.fill()
    elif shape == 'cone':
        bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=0.5, radius2=0, depth=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    elif shape == 'uvsphere':
        bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=0.5, location=location)
        obj = bpy.context.active_object
        obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    else:
        return None

    obj.name = name
    obj.rotation_euler = rotation_euler
    
    # Apply scale to make it clean for modifiers
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.transform_apply(location=False, rotation=True, scale=True)
    
    if material:
        if obj.data.materials:
            obj.data.materials[0] = material
        else:
            obj.data.materials.append(material)
            
    return obj

def align_face_to_face(obj_source, obj_target, axis='Z', offset=0):
    """
    Aligns the +axis face of obj_source to the +axis face of obj_target.
    Assumes objects are axis-aligned.
    """
    # Get world matrix of target
    target_mat = obj_target.matrix_world
    source_mat = obj_source.matrix_world
    
    # Calculate the center of the target face in world space
    # Assuming +Z face is at max Z
    if axis == 'Z':
        target_face_z = target_mat[3][2] + (obj_target.dimensions[2] / 2)
        source_face_z = source_mat[3][2] + (obj_source.dimensions[2] / 2)
        
        # Move source so its face matches target's face
        delta_z = target_face_z - source_face_z + offset
        obj_source.location.z += delta_z
        
    elif axis == 'Y':
        target_face_y = target_mat[3][1] + (obj_target.dimensions[1] / 2)
        source_face_y = source_mat[3][1] + (obj_source.dimensions[1] / 2)
        delta_y = target_face_y - source_face_y + offset
        obj_source.location.y += delta_y
        
    elif axis == 'X':
        target_face_x = target_mat[3][0] + (obj_target.dimensions[0] / 2)
        source_face_x = source_mat[3][0] + (obj_source.dimensions[0] / 2)
        delta_x = target_face_x - source_face_x + offset
        obj_source.location.x += delta_x

def align_center_to_center(obj_source, obj_target):
    """Aligns the center of obj_source to the center of obj_target."""
    target_loc = obj_target.location
    source_loc = obj_source.location
    obj_source.location = target_loc

def apply_offset(obj, offset_vector):
    """Applies an offset vector to the object's location."""
    obj.location += offset_vector

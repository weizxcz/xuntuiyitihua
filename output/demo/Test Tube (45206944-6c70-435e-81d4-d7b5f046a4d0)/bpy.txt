def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Alpha'].default_value = diffuse_color[3]
    bsdf.inputs['Specular IOR Level'].default_value = 0.0
    bsdf.inputs['Transmission'].default_value = 1.0
    bsdf.inputs['Transmission Roughness'].default_value = 0.0
    return mat

def add_cylinder(radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, align='WORLD', location=(0, 0, 0))
    return bpy.context.active_object

def add_uvsphere(radius, width_segments=16, height_segments=16):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, align='WORLD', location=(0, 0, 0))
    return bpy.context.active_object

def add_cube(size):
    bpy.ops.mesh.primitive_cube_add(size=size, align='WORLD', location=(0, 0, 0))
    return bpy.context.active_object

def add_disc(radius):
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='TRIFAN', align='WORLD', location=(0, 0, 0))
    return bpy.context.active_object

def align_axis_to_vector(obj, axis_to_align='Z', target_vector=None):
    """Aligns an object's local axis to a target vector."""
    if target_vector is None:
        return
    
    # Get current axis vector in world space
    if axis_to_align == 'X':
        current_axis = obj.matrix_world.to_3x3().col[0]
    elif axis_to_align == 'Y':
        current_axis = obj.matrix_world.to_3x3().col[1]
    else: # Z
        current_axis = obj.matrix_world.to_3x3().col[2]
        
    # Calculate rotation
    rot_quat = current_axis.rotation_difference(target_vector.normalized())
    obj.rotation_euler = rot_quat.to_euler()

def get_face_center(obj, face_index):
    """Returns the world location of a specific face center."""
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.verts.ensure_lookup_table()
    face = bm.faces[face_index]
    # Calculate center in local space
    center_local = sum((v.co for v in face.verts), Vector()) / len(face.verts)
    # Convert to world space
    center_world = obj.matrix_world @ center_local
    bm.free()
    return center_world

def get_face_normal(obj, face_index):
    """Returns the world normal of a specific face."""
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.verts.ensure_lookup_table()
    face = bm.faces[face_index]
    normal_world = obj.matrix_world.to_3x3() @ face.normal
    bm.free()
    return normal_world

def boolean_subtract(cutter_obj, target_obj):
    """Performs a boolean subtract using the modifier stack."""
    # Ensure target is active for modifier application if needed, but we'll apply later
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'SUBTRACT'
    mod.object = cutter_obj
    
    # Apply modifier immediately to keep geometry clean for subsequent steps
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(target_obj, other_obj):
    """Performs a boolean union."""
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other_obj
    
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(other_obj, do_unlink=True)

    # 1. Align normals

    # Here we assume standard outward normals. If we are placing bottom of sphere on bottom of cylinder,
    # sphere normal at bottom points -Z, cylinder bottom normal points -Z. They should align.
    align_axis_to_vector(obj_source, 'Z', target_normal)
    
    # 2. Align centers
    # Move source so its face center matches target face center

def make_material(name, diffuse_color):
    """Create a new material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube mesh object with specific size."""
    bpy.ops.mesh.primitive_cube_add(size=size[0], location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.scale = (size[1], size[2], size[3]) # size is x,y,z
    obj.name = name
    return obj

def add_uvsphere(name, radius):
    """Add a UV sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=32, ring_count=16, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """Align an object's local axis to a target world vector."""
    # Get current rotation
    rot_euler = obj.rotation_euler.copy()
    
    # We need to find the rotation that maps the object's local axis to the target vector
    # Local +Z is (0,0,1), +X is (1,0,0), +Y is (0,1,0)
    
    # Define local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    local_axis = local_axes[axis_to_align]
    
    # Calculate rotation matrix to align local_axis to target_vector
    # Using quaternion rotation
    quat = local_axis.rotation_difference(target_vector)
    
    # Apply this rotation to the object's current orientation
    # Note: This assumes we want to set the orientation relative to world, 
    # but usually we want to rotate the object so its axis points there.
    # To be safe and simple, we can just set the rotation directly if we assume identity start,
    # or compose. Here we compose.
    
    # Convert quat to euler to apply? No, better to use matrix.
    rot_mat = quat.to_matrix().to_4x4()
    
    # Apply to object's matrix world, keeping location
    # But we need to be careful not to double rotate if called multiple times.
    # The prompt implies setting orientation from scratch or adjusting.
    # Let's assume we are setting the orientation based on the instruction.
    
    # Simpler approach for "Rotate so +Z aligns +X":
    # Just set the rotation euler directly if we know the mapping.
    # +Z (0,0,1) -> +X (1,0,0). This is a -90 deg rotation around Y.
    
    if axis_to_align == 'Z' and target_vector == Vector((1, 0, 0)):

    elif axis_to_align == 'Z' and target_vector == Vector((-1, 0, 0)):
        obj.rotation_euler = (0, pi/2, 0)
    elif axis_to_align == 'Z' and target_vector == Vector((0, 1, 0)):
        obj.rotation_euler = (pi/2, 0, 0)
    elif axis_to_align == 'Z' and target_vector == Vector((0, -1, 0)):
        obj.rotation_euler = (-pi/2, 0, 0)
    elif axis_to_align == 'X' and target_vector == Vector((1, 0, 0)):
        obj.rotation_euler = (0, 0, 0)
    elif axis_to_align == 'Y' and target_vector == Vector((0, 1, 0)):

def align_object_axis_to_object_axis(obj_source, axis_src, obj_target, axis_tgt):
    """Align obj_source's local axis to match obj_target's local axis direction."""
    # Get target axis vector in world space
    # We can use the object's matrix to transform the local axis vector
    mat_src = obj_source.matrix_world
    mat_tgt = obj_target.matrix_world
    
    # Local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    # Direction of target axis in world space
    dir_tgt = mat_tgt.to_3x3() @ local_axes[axis_tgt]
    
    # Align source axis to this direction
    align_axis_to_vector(obj_source, axis_src, dir_tgt)

def apply_material_to_object(obj, mat):

def boolean_union(target, cutter):
    """Perform boolean union of cutter into target."""
    # Add boolean modifier to target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def bevel_edges(obj, radius, segments):

    mod.width = radius
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply bevel
    bpy.ops.object.modifier_apply(modifier=mod.name)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        
    def center_world(self):
        return self.obj.location.copy()
        
    def face_center_world(self, face_index):
        # Get face center in world space
        # Requires mesh data
        mesh = self.obj.data
        bm = bmesh.new()
        bm.from_mesh(mesh)
        # Ensure face is selected or accessed by index
        # bmesh indices might differ if not updated, but usually ok for static
        if face_index < len(bm.faces):
            face = bm.faces[face_index]
            # Calculate center
            center = Vector((0,0,0))
            for v in face.verts:
                center += v.co
            center /= len(face.verts)
            # Transform to world
            world_center = self.obj.matrix_world @ center
            bm.free()
            return world_center
        bm.free()
        return self.center_world()

    def radial_point_world(self, face_index, offset_dist=0):
        center = self.face_center_world(face_index)
        # Get normal
        mesh = self.obj.data
        bm = bmesh.new()
        bm.from_mesh(mesh)
        if face_index < len(bm.faces):
            face = bm.faces[face_index]
            normal = face.normal.copy()
            normal = self.obj.matrix_world.to_3x3() @ normal
            bm.free()
            return center + (normal * offset_dist)
        bm.free()
        return center

def anchor_to_face(obj_source, obj_target, target_face_index, offset_vector):
    """
    Anchor obj_source to a face of obj_target.
    1. Move obj_source so its center is at the face center of obj_target.
    2. Align obj_source's Z axis to the face normal of obj_target.
    3. Apply offset in obj_source's local space? Or global?
       Prompt says: "Offset is in reference local axes."
       Usually, after anchoring, we apply offset.
    """
    loc = Locator(obj_target)
    
    # 1. Move source to face center
    face_center = loc.face_center_world(target_face_index)
    obj_source.location = face_center
    
    # 2. Align Z axis to face normal
    # Get face normal in world space
    mesh = obj_target.data
    bm = bmesh.new()
    bm.from_mesh(mesh)
    if target_face_index < len(bm.faces):
        face = bm.faces[target_face_index]
        normal = face.normal.copy()
        normal = obj_target.matrix_world.to_3x3() @ normal
        bm.free()
        
        # Align obj_source Z to normal
        align_axis_to_vector(obj_source, 'Z', normal)
    else:
        bm.free()
        
    # 3. Apply offset
    # "Then move by offset(...)"
    # If offset is in local axes of the source (which is now aligned),
    # we just add the offset vector to the location.
    obj_source.location += offset_vector

def get_face_index_by_normal(obj, normal_vector):

    best_idx = -1
    best_dot = -1
    
    for i, face in enumerate(bm.faces):

            best_idx = i

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with a specific diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_primitive_cube(name, size, location, rotation_euler, material=None):
    """Add a cube and apply transforms."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def add_primitive_cylinder(name, radius, depth, location, rotation_euler, material=None):
    """Add a cylinder and apply transforms."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name

    if material:
        obj.data.materials.append(material)
    return obj

def add_primitive_cuboid(name, x, y, z, location, rotation_euler, material=None):
    """Add a cube scaled to dimensions x,y,z."""
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (x/2, y/2, z/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def align_bottom_to_top(bottom_obj, top_obj):
    """Align the bottom face of bottom_obj to the top face of top_obj in Z."""
    # Get world bounds
    bm_bottom = bmesh.new()
    bm_top = bmesh.new()
    
    # We need to work in world space for accurate alignment if objects are rotated,
    # but here we assume simple Z-alignment.
    # Calculate top Z of top_obj
    # Since objects are axis-aligned, we can use bounds
    min_b, max_b = bottom_obj.bound_box
    min_t, max_t = top_obj.bound_box
    
    # Transform bounds to world space
    # Simple approach: assume Z-up alignment.
    # Top of top_obj in world Z
    top_z_world = max_t[6] * top_obj.scale[2] + top_obj.location[2] # Approximate if scale is uniform, but let's be precise
    
    # Actually, let's use the helper logic:
    # Find the highest Z point of the top object
    # And the lowest Z point of the bottom object
    
    # To be robust, we calculate the offset required to move bottom_obj's lowest Z to top_obj's highest Z
    # Note: bound_box is local.
    
    # Local min/max Z
    local_min_z_bottom = min(min_b[i][2] for i in range(8))
    local_max_z_bottom = max(min_b[i][2] for i in range(8))
    local_min_z_top = min(min_t[i][2] for i in range(8))
    local_max_z_top = max(min_t[i][2] for i in range(8))
    
    # World Z of top face of top_obj
    # Assuming top_obj is not rotated around X/Y significantly (Z-up)
    world_top_z = top_obj.location[2] + (local_max_z_top * top_obj.scale[2])
    
    # World Z of bottom face of bottom_obj (before move)
    world_bottom_z_current = bottom_obj.location[2] + (local_min_z_bottom * bottom_obj.scale[2])
    
    # Delta Z needed
    delta_z = world_top_z - world_bottom_z_current
    
    bottom_obj.location[2] += delta_z
    
    bm_bottom.free()
    bm_top.free()

def align_bottom_to_world_z(bottom_obj, z_val):
    """Align the bottom face of bottom_obj to a specific world Z."""
    bm = bmesh.new()
    min_b = bottom_obj.bound_box
    local_min_z_bottom = min(min_b[i][2] for i in range(8))
    
    world_bottom_z_current = bottom_obj.location[2] + (local_min_z_bottom * bottom_obj.scale[2])
    delta_z = z_val - world_bottom_z_current
    bottom_obj.location[2] += delta_z
    bm.free()

def boolean_union(target_obj, cutter_obj):
    """Perform boolean union: cutter_obj is merged into target_obj. target_obj becomes the result."""
    # Add Boolean modifier to target
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return target_obj

def boolean_subtract(target_obj, cutter_obj):
    """Perform boolean subtraction: cutter_obj cuts into target_obj."""
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return target_obj

def get_polar_location(angle_deg, radius, center=(0,0,0)):
    """Get X,Y location from polar coordinates."""
    angle_rad = math.radians(angle_deg)
    x = center[0] + radius * math.cos(angle_rad)
    y = center[1] + radius * math.sin(angle_rad)
    return (x, y, center[2])

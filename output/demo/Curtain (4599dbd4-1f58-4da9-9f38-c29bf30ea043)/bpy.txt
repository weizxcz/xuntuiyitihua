def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location, rotation, material=None):
    """Add a cylinder oriented along Z by default, then apply rotation."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    # Apply rotation (Euler)
    obj.rotation_euler = rotation
    return obj

def add_disc(name, radius, location, rotation, material=None):
    """Add a disc (flat cylinder with depth=0) oriented along Z by default."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=0.001, location=location) # Small depth for visibility
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    obj.rotation_euler = rotation
    return obj

def add_cube(name, size, location, rotation, material=None):
    """Add a cube centered at location."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    obj.rotation_euler = rotation
    return obj

def add_uvsphere(name, radius, location, rotation, material=None):

    obj.name = name
    if material:

def align_axis_to_vector(obj, axis_to_align='+Z', target_vector):
    """Rotate object so its specified axis aligns with target_vector."""
    # Current axis vectors
    axes = {
        '+X': Vector((1, 0, 0)), '-X': Vector((-1, 0, 0)),
        '+Y': Vector((0, 1, 0)), '-Y': Vector((0, -1, 0)),
        '+Z': Vector((0, 0, 1)), '-Z': Vector((0, 0, -1))
    }
    
    current_axis = axes.get(axis_to_align, Vector((0, 0, 1)))
    target = target_vector.normalized()
    
    # Calculate rotation matrix
    rot_mat = current_axis.rotation_difference(target).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = current_axis.rotation_difference(target)

def get_face_center(obj, face_type='center'):
    """
    Get the world location of a specific face of an object.
    face_type: 'center', 'left_face', 'right_face', 'top_face', 'bottom_face', 'front_face', 'back_face'
    Assumes object is axis-aligned or rotated in standard ways.
    """
    # Transform vertices to world space
    verts_world = [obj.matrix_world @ v.co for v in obj.data.vertices]
    
    # Determine bounds
    min_x = min(v[0] for v in verts_world)
    max_x = max(v[0] for v in verts_world)
    min_y = min(v[1] for v in verts_world)
    max_y = max(v[1] for v in verts_world)
    min_z = min(v[2] for v in verts_world)
    max_z = max(v[2] for v in verts_world)
    
    center = obj.location
    
    if face_type == 'center':
        return center
    
    # Helper to find point on face
    def get_face_point(axis, limit, is_max):
        # Find vertices on this face
        face_verts = [v for v in verts_world if abs(v[axis] - (limit if is_max else -limit)) < 0.0001] # Approx check
        # Better: check against bounds
        face_verts = [v for v in verts_world if (is_max and v[axis] >= max([v2[axis] for v2 in verts_world]) - 0.001) or (not is_max and v[axis] <= min([v2[axis] for v2 in verts_world]) + 0.001)]
        
        if not face_verts:
            # Fallback to center if logic fails (e.g. rotated objects)
            return center
            
        avg_x = sum(v[0] for v in face_verts) / len(face_verts)
        avg_y = sum(v[1] for v in face_verts) / len(face_verts)
        avg_z = sum(v[2] for v in face_verts) / len(face_verts)
        return Vector((avg_x, avg_y, avg_z))

    # Note: The logic above assumes axis-aligned bounding box in World Space.
    # Since we rotate objects, their local axes change relative to world.
    # However, the prompt implies specific orientations.
    # Let's use a more robust method: Project vertices to the normal of the face.
    
    # For simplicity in this script, we will rely on the known geometry and orientation.
    # If an object is rotated, 'left_face' usually refers to the face with the minimum local X,
    # transformed to world space.
    
    # Let's calculate local bounds first
    local_verts = [v.co for v in obj.data.vertices]
    local_min = Vector([min(v[i] for v in local_verts) for i in range(3)])
    local_max = Vector([max(v[i] for v in local_verts) for i in range(3)])
    local_center = (local_min + local_max) / 2
    
    # Define local face centers relative to local center
    # Assuming standard primitives: Cube, Cylinder, Sphere
    # We need to map local face to world space.
    
    # Simplified approach for this specific script:
    # We know the orientations.
    # 1. rod_main: +Z aligned to +X.
    # 2. finials: +Z aligned to +X.
    # 3. wall_plates: +Z aligned to +Y.
    # 4. support_arms: +Z aligned to +Y.
    # 5. fabric/header: +Z aligned to +Z.
    
    # Let's implement a generic helper that finds the face center in World Space
    # by checking which vertices are extreme in the direction of the face normal.
    
    # Face normals in Local Space
    face_normals_local = {
        'left_face': Vector((-1, 0, 0)), 'right_face': Vector((1, 0, 0)),
        'top_face': Vector((0, 0, 1)), 'bottom_face': Vector((0, 0, -1)),
        'front_face': Vector((0, 1, 0)), 'back_face': Vector((0, -1, 0))
    }
    
    if face_type not in face_normals_local:
        return center
        
    normal_local = face_normals_local[face_type]
    normal_world = obj.matrix_world.to_3x3() @ normal_local
    
    # Project all world vertices onto this normal
    # The face center is the average of vertices that are "on" that face.
    # We can approximate by taking vertices with max dot product with normal_world
    
    # Find max dot product
    max_dot = -float('inf')
    for v in verts_world:
        dot = v.dot(normal_world)
        if dot > max_dot:
            max_dot = dot
            
    # Select vertices close to max_dot
    threshold = 0.001 # Tolerance
    face_verts_world = [v for v in verts_world if abs(v.dot(normal_world) - max_dot) < threshold]
    
    if face_verts_world:
        return Vector([sum(v[i] for v in face_verts_world) / len(face_verts_world) for i in range(3)])
    
    return center

def offset_object(obj, offset_vector):
    """Move object by offset_vector in World Space."""
    obj.location += offset_vector

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using modifier."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is active or selected if needed, but modifiers work on references
    bool_mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    bool_mod.object = cutter
    bool_mod.operation = 'DIFFERENCE'
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=bool_mod.name)
    
    # Optional: Remove cutter
    # bpy.data.objects.remove(cutter)

def boolean_union(obj1_name, obj2_name):
    """Perform boolean union."""
    obj1 = bpy.data.objects[obj1_name]
    obj2 = bpy.data.objects[obj2_name]
    
    bool_mod = obj1.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    bool_mod.object = obj2
    bool_mod.operation = 'UNION'
    
    bpy.context.view_layer.objects.active = obj1
    bpy.ops.object.modifier_apply(modifier=bool_mod.name)

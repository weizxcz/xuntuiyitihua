def clear_scene():
    """Delete all existing objects."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for collection in bpy.data.collections:
        collection.objects.clear()

def set_units_metres():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    bsdf.inputs['Roughness'].default_value = 0.8
    return mat

def add_primitive(name, primitive_type, dimensions, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive mesh object."""
    bpy.ops.object.select_all(action='DESELECT')
    if primitive_type == 'CUBE':
        bpy.ops.mesh.primitive_cube_add(size=1, location=location)
        obj = bpy.context.active_object
        obj.scale = (dimensions[0]/2, dimensions[1]/2, dimensions[2]/2)
    elif primitive_type == 'CYLINDER':
        # dimensions: (diameter, height)
        radius = dimensions[0] / 2
        height = dimensions[1]
        bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=location)
        obj = bpy.context.active_object
    elif primitive_type == 'UVSPHERE':
        # dimensions: (radius, segments) - simplified for sphere/hemi
        radius = dimensions[0]
        bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=location)
        obj = bpy.context.active_object
    else:
        raise ValueError(f"Unsupported primitive type: {primitive_type}")
    
    obj.name = name
    obj.rotation_euler = rotation
    return obj

def align_face_to_face(obj_source, obj_target, axis='Z'):
    """
    Align the XY face (normal to Z) of obj_source to the XY face of obj_target.
    Assumes both are oriented with Z up initially.
    Moves obj_source so its bottom Z matches target's bottom Z.
    """
    # Get bounding box centers and extents
    src_min = obj_source.matrix_world @ Vector((0,0,0)) # Simplified: assume origin at center
    # Actually, for primitives added at origin, local (0,0,0) is center.
    # We need to move the object so its bottom aligns with target's bottom.
    
    # Calculate world positions of bottoms
    # Since we rotate then align, let's work in local space relative to world after rotation
    
    # Get world matrix
    src_mat = obj_source.matrix_world
    tgt_mat = obj_target.matrix_world
    
    # Get bottom Z in world space
    # For a cylinder/cube centered at 0,0,0 with height H, bottom is at -H/2 in local Z
    # We need to know the half-height.
    
    def get_bottom_z(obj):
        # Get bounding box
        bbox = [obj.matrix_world @ v for v in obj.bound_box]
        z_coords = [v.z for v in bbox]
        return min(z_coords)

    src_bottom_z = get_bottom_z(obj_source)
    tgt_bottom_z = get_bottom_z(obj_target)
    
    # Offset required
    offset_z = tgt_bottom_z - src_bottom_z
    
    # Apply offset
    obj_source.location.z += offset_z
    
    # Align X and Y centers
    src_center = obj_source.location
    tgt_center = obj_target.location
    
    obj_source.location.x = tgt_center.x
    obj_source.location.y = tgt_center.y

def align_face_to_face_offset(obj_source, obj_target, offset_vector):
    """
    Align obj_source face to obj_target face, then apply offset.
    Offset is in reference local axes (assumed aligned with world after rotation).
    """
    align_face_to_face(obj_source, obj_target)
    obj_source.location += offset_vector

def boolean_union(objects_to_union, result_name):
    """
    Perform boolean union on a list of objects.
    Returns the resulting object.
    """
    if not objects_to_union:
        return None
    
    # Select all objects
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects_to_union:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj
    
    # Apply modifiers if any (clean up)
    for obj in objects_to_union:
        for mod in obj.modifiers:
            if mod.type == 'BOOLEAN':
                bpy.ops.object.modifier_apply(modifier=mod.name)

    # Join them
    bpy.ops.object.join()
    
    result = bpy.context.active_object
    result.name = result_name
    return result

def boolean_subtract(base_obj, cutter_obj, result_name=None):
    """
    Subtract cutter_obj from base_obj using boolean modifier.
    """
    # Ensure base is active
    bpy.context.view_layer.objects.active = base_obj
    base_obj.select_set(True)
    cutter_obj.select_set(False)
    
    # Add boolean modifier to base
    mod = base_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Delete cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    
    if result_name:
        base_obj.name = result_name
        
    return base_obj

def orient_z_to_vector(obj, vector):
    """Rotate object so its local Z axis aligns with the given world vector."""
    # Current Z
    current_z = Vector((0, 0, 1))
    target_z = vector.normalized()
    
    # Calculate rotation
    rot_quat = current_z.rotation_difference(target_z)
    
    # Apply rotation
    obj.rotation_euler = rot_quat.to_euler()

def orient_z_to_x(obj):
    """Rotate object so its local Z axis aligns with world +X."""
    orient_z_to_vector(obj, Vector((1, 0, 0)))

def orient_z_to_y(obj):
    """Rotate object so its local Z axis aligns with world +Y."""
    orient_z_to_vector(obj, Vector((0, 1, 0)))

def orient_z_to_z(obj):
    """Rotate object so its local Z axis aligns with world +Z."""
    orient_z_to_vector(obj, Vector((0, 0, 1)))

def align_cylinder_face_to_cylinder_face(cyl_source, cyl_target, source_face_axis='X', target_face_axis='Z'):
    """
    Aligns the circular face of cyl_source (along source_face_axis) to the circular face of cyl_target (along target_face_axis).
    Assumes both are cylinders.
    """
    # Get world matrix
    src_mat = cyl_source.matrix_world
    tgt_mat = cyl_target.matrix_world
    
    # Determine which face to align. 
    # For simplicity, we align the face at +local_axis.
    # Source face normal in world space
    if source_face_axis == 'X':
        src_normal = src_mat @ Vector((1, 0, 0))
        src_offset_local = cyl_source.scale[0] * 0.5 # Radius
    elif source_face_axis == 'Y':
        src_normal = src_mat @ Vector((0, 1, 0))
        src_offset_local = cyl_source.scale[0] * 0.5
    else: # Z
        src_normal = src_mat @ Vector((0, 0, 1))
        src_offset_local = cyl_source.scale[0] * 0.5
        
    # Target face normal and position
    if target_face_axis == 'X':
        tgt_normal = tgt_mat @ Vector((1, 0, 0))
        tgt_center_world = tgt_mat @ Vector((0,0,0)) + tgt_normal * (cyl_target.scale[0] * 0.5) # Approx
    elif target_face_axis == 'Y':
        tgt_normal = tgt_mat @ Vector((0, 1, 0))
        tgt_center_world = tgt_mat @ Vector((0,0,0)) + tgt_normal * (cyl_target.scale[0] * 0.5)
    else: # Z
        tgt_normal = tgt_mat @ Vector((0, 0, 1))
        tgt_center_world = tgt_mat @ Vector((0,0,0)) + tgt_normal * (cyl_target.scale[2] * 0.5) # Height/2

    # We want the source face center to be at tgt_center_world
    # Source face center in local space is (offset_local, 0, 0) if X
    # In world space: src_mat @ Vector((offset_local, 0, 0))
    
    # Calculate required translation
    # Current source face center
    src_face_center_local = Vector((src_offset_local, 0, 0)) if source_face_axis == 'X' else \
                            Vector((0, src_offset_local, 0)) if source_face_axis == 'Y' else \
                            Vector((0, 0, src_offset_local))
    
    src_face_center_world = src_mat @ src_face_center_local
    
    # Desired location for source object center
    # tgt_center_world - (src_mat @ src_face_center_local - src_mat @ (0,0,0))
    # tgt_center_world - src_face_center_world + src_mat @ (0,0,0)
    
    src_center_world_current = src_mat @ Vector((0,0,0))
    
    new_center_world = tgt_center_world - (src_face_center_world - src_center_world_current)
    
    # Apply translation
    translation = new_center_world - src_center_world_current
    cyl_source.location += translation

def align_cube_face_to_cylinder_face(cube_obj, cyl_obj, cube_face_axis='Z', cyl_face_axis='X'):

    if cyl_face_axis == 'X':

    new_cube_center = cyl_face_center_world - (cube_face_center_world_current - cube_center_world_current)
    
    cube_obj.location += new_cube_center - cube_center_world_current

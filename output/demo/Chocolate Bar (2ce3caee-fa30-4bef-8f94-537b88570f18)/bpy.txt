# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    bpy.ops.mesh.primitive_cube_add(size=size[0], location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[1]/size[0], size[2]/size[0], 1.0) # Adjust for non-uniform if needed, but primitive cube is uniform. 
    # Actually, better to set scale directly after creation if dimensions are specific.
    obj.scale = (size[1]/size[0], size[2]/size[0], size[3]/size[0]) if len(size)==3 else (size[1]/size[0], size[2]/size[0], 1.0)
    # Re-approach: Create with size=1, then scale.
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation
    return obj

def add_bevel(obj, width=0.01, segments=2):
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(base_obj, cutter_obj):
    # Ensure base is active
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.modifier_add(type='BOOLEAN')
    mod = base_obj.modifiers[-1]
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(base_obj, cutter_obj):
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.modifier_add(type='BOOLEAN')
    mod = base_obj.modifiers[-1]
    mod.operation = 'UNION'
    mod.object = cutter_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align object's local axis to a world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else: # Z
        local_axis = Vector((0,0,1))
        
    target = vector.normalized()
    if local_axis == target:
        return
    
    rot_quat = local_axis.rotation_difference(target)
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = rot_quat
    obj.rotation_mode = 'AXIS_ANGLE' # Reset to euler for consistency if needed, but keep quat for now

def get_face_center(obj, face_type='center'):
    """
    Returns the world location of a specific face or center of an object.
    face_type: 'center', 'top_face', 'bottom_face', 'left_face', 'right_face', 'front_face', 'back_face'
    Assumes object is axis-aligned or has been oriented.
    """
    # Get object's bounding box in world space
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    
    # Find min/max coordinates
    mins = [min(v.co.x for v in bm.verts), min(v.co.y for v in bm.verts), min(v.co.z for v in bm.verts)]
    maxs = [max(v.co.x for v in bm.verts), max(v.co.y for v in bm.verts), max(v.co.z for v in bm.verts)]
    
    center = [(mins[0]+maxs[0])/2, (mins[1]+maxs[1])/2, (mins[2]+maxs[2])/2]
    
    if face_type == 'center':
        return Vector(center)
    elif face_type == 'top_face':
        return Vector([center[0], center[1], maxs[2]])
    elif face_type == 'bottom_face':
        return Vector([center[0], center[1], mins[2]])
    elif face_type == 'left_face':
        return Vector([mins[0], center[1], center[2]])
    elif face_type == 'right_face':
        return Vector([maxs[0], center[1], center[2]])
    elif face_type == 'front_face':
        return Vector([center[0], maxs[1], center[2]])
    elif face_type == 'back_face':
        return Vector([center[0], mins[1], center[2]])
    else:
        return Vector(center)

def average_vectors(vectors):
    if not vectors:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for v in vectors:
        avg += v
    return avg / len(vectors)

def align_object_to_face(obj, target_obj, target_face_type):
    """Aligns the bottom face of 'obj' to the specified face of 'target_obj'."""
    target_loc = get_face_center(target_obj, target_face_type)
    obj_loc = get_face_center(obj, 'bottom_face')
    
    # Move obj so its bottom face is at target_loc
    delta = target_loc - obj_loc
    obj.location += delta

def align_object_to_center(obj, target_obj):
    """Aligns center of obj to center of target_obj."""
    target_loc = get_face_center(target_obj, 'center')
    obj_loc = get_face_center(obj, 'center')
    delta = target_loc - obj_loc
    obj.location += delta

def apply_offset(obj, offset_vector):
    obj.location += offset_vector

chocolate_base = add_cube("chocolate_base", (0.100, 0.050, 0.020), location=(0,0,0), rotation=(0,0,0))
chocolate_base.data.materials.append(mat_chocolate_dark)

# Rotate chocolate_base so its +Z axis aligns +Z (world) - already default
align_axis_to_vector(chocolate_base, 'Z', (0,0,1))

# Align chocolate_base.center to root.center
# Note: 'root' is not defined in the prompt. Assuming 'root' refers to the world origin or a hypothetical root object.
# Since no 'root' object exists, we assume root.center is (0,0,0).
root_center = Vector((0,0,0))
base_center = get_face_center(chocolate_base, 'center')
chocolate_base.location += (root_center - base_center)

# Align chocolate_base.bottom_face to root.bottom_face
# Assuming root.bottom_face is at Z=0 (world bottom) or relative to root.
# If root is at (0,0,0) and has no size, its bottom face is ambiguous.
# Standard interpretation: Align to world Z=0 plane if root is origin.
# Let's assume root.bottom_face is at Z=0.
root_bottom = Vector((0,0,0))
base_bottom = get_face_center(chocolate_base, 'bottom_face')
chocolate_base.location += (root_bottom - base_bottom)

# Move by offset(0,0,0)
apply_offset(chocolate_base, (0,0,0))

# --- SECTION 2 – Create chocolate_segments ---

seg_size = (0.024, 0.024, 0.018)
seg_material = mat_chocolate_milk

def create_segment(name, anchor_faces, offset):
    seg = add_cube(name, seg_size, location=(0,0,0), rotation=(0,0,0))
    seg.data.materials.append(seg_material)
    
    # Rotate seg so its +Z axis aligns +Z (world)
    align_axis_to_vector(seg, 'Z', (0,0,1))
    
    # Align seg.center to Avg(anchor_faces)
    anchor_locs = [get_face_center(chocolate_base, f) for f in anchor_faces]
    avg_center = average_vectors(anchor_locs)
    seg_center = get_face_center(seg, 'center')
    seg.location += (avg_center - seg_center)
    
    # Align seg.bottom_face to chocolate_base.top_face
    align_object_to_face(seg, chocolate_base, 'top_face')
    
    # Move by offset
    apply_offset(seg, offset)
    return seg

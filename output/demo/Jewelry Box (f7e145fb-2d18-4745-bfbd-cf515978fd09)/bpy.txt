import bpy
import bmesh
from mathutils import Vector, Matrix

# ==============================================================================
# HELPER FUNCTIONS
# ==============================================================================

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with the given diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.8 # Default roughness for wood/fabric
    return mat

def add_primitive_cube(name, size, location, rotation_euler):
    """Add a cube primitive, set size, location, and rotation."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def add_primitive_cylinder(name, radius, depth, location, rotation_euler):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    return obj

def align_axis_to_vector(obj, axis='Z', vector=Vector((0, 0, 1))):
    """Align an object's local axis to a world vector."""
    if axis == 'Z':
        # Align Z to vector
        quat = vector.to_track_quat('Z', 'Y')
        obj.rotation_euler = quat.to_euler()
    elif axis == 'X':
        quat = vector.to_track_quat('X', 'Y')
        obj.rotation_euler = quat.to_euler()
    elif axis == 'Y':
        quat = vector.to_track_quat('Y', 'Z')
        obj.rotation_euler = quat.to_euler()

def get_face_center(obj, face_index):
    """Get the world location of a specific face center."""
    # Ensure object is updated
    obj.update_from_editmode()
    # Get mesh data
    me = obj.data
    # Calculate face center in local space
    face = me.polygons[face_index]
    local_center = Vector((0, 0, 0))
    for vi in face.vertices:
        local_center += me.vertices[vi].co
    local_center /= len(face.vertices)
    
    # Convert to world space
    return obj.matrix_world @ local_center

def get_face_normal(obj, face_index):
    """Get the world normal of a specific face."""
    me = obj.data
    face = me.polygons[face_index]
    # Get normal in local space
    local_normal = face.normal
    # Transform to world space
    world_normal = obj.matrix_world.to_3x3() @ local_normal
    return world_normal

def boolean_subtract(cutter_obj, target_obj):
    """Perform boolean subtract using the modifier stack."""
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter object
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return target_obj

def boolean_union(union_obj, target_obj):
    """Perform boolean union using the modifier stack."""
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove union object
    bpy.data.objects.remove(union_obj, do_unlink=True)
    return target_obj

def apply_material(obj, mat_name):
    """Apply a material to an object."""
    if mat_name in bpy.data.materials:
        mat = bpy.data.materials[mat_name]
        if obj.data.materials:
            obj.data.materials[0] = mat
        else:
            obj.data.materials.append(mat)

# Actually, let's calculate manually for precision:

base_sides_void.location += Vector((0, 0, 0.01))

base_interior_void.location += Vector((0, 0, 0.01))

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if not obj.data.materials:
        obj.data.materials.append(mat)
    else:
        obj.data.materials[0] = mat

def add_cylinder(radius, depth, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(size, name, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_torus(major_radius, minor_radius, name, location=(0,0,0), rotation=(0,0,0)):

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis (0=X, 1=Y, 2=Z) to a world vector."""
    if axis == 0: local_axis = obj.matrix_world.to_3x3().col[0]
    elif axis == 1: local_axis = obj.matrix_world.to_3x3().col[1]
    else: local_axis = obj.matrix_world.to_3x3().col[2]
    
    # Calculate rotation to align local_axis to target vector
    rot_quat = local_axis.rotation_difference(vector.normalized())
    obj.rotation_euler = rot_quat.to_euler()

def get_world_center(obj):
    """Get the world location of the object's center."""
    return obj.matrix_world.to_translation()

def get_world_face_center(obj, face_type='top'):

        # Top face is +Z direction from center
        return get_world_center(obj) + Vector((0,0,0)) # Simplified: relies on anchor logic below
    elif face_type == 'bottom':
        return get_world_center(obj)
    elif face_type == 'left_face':
        return get_world_center(obj)
    elif face_type == 'right_face':
        return get_world_center(obj)
    return get_world_center(obj)

def boolean_subtract(cutter, target):
    """Perform boolean subtract using modifier stack."""
    # Add modifier to target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(target, other):
    """Perform boolean union."""
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = other
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(other, do_unlink=True)

def create_cutter(name, size, location, rotation):
    """Create a cube cutter."""
    # size is tuple (x,y,z)

    obj = bpy.context.active_object
    obj.name = name + "_cutter"
    return obj

def align_object_to_face(obj, target_obj, face_type='top'):
    """
    Aligns obj to a specific face of target_obj.
    For simplicity in this script, we calculate the position based on the target's geometry.
    """
    # Get target center
    t_center = get_world_center(target_obj)
    
    # Determine offset based on face type and target dimensions
    # Note: This assumes standard primitive orientations.
    if face_type == 'top':
        # Top of target is +Z
        offset = Vector((0, 0, target_obj.dimensions.z / 2))
    elif face_type == 'bottom':
        offset = Vector((0, 0, -target_obj.dimensions.z / 2))
    elif face_type == 'left_face':
        # Left face (negative X)
        offset = Vector((-target_obj.dimensions.x / 2, 0, 0))
    elif face_type == 'right_face':
        # Right face (positive X)
        offset = Vector((target_obj.dimensions.x / 2, 0, 0))
    elif face_type == 'center':
        offset = Vector((0, 0, 0))
    else:
        offset = Vector((0, 0, 0))
        
    # Set obj location to target center + offset
    obj.location = t_center + offset

def polar_position(center, angle_deg, radius):
    """Calculate 2D polar position relative to center in XY plane."""
    angle_rad = math.radians(angle_deg)
    x = center.x + radius * math.cos(angle_rad)
    y = center.y + radius * math.sin(angle_rad)
    z = center.z
    return Vector((x, y, z))

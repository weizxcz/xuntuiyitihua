import bpy

def clear_scene():
    """Delete every existing object to start from a clean scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    # Remove orphan data blocks
    for block in bpy.data.meshes:
        bpy.data.meshes.remove(block)
    for block in bpy.data.materials:
        bpy.data.materials.remove(block)

def set_units():
    """Set the length unit to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube primitive."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def get_face_center(obj, face_type):
    """
    Returns the world location of a specific face center of an object.
    Assumes standard orientation or applies rotation.
    face_type: 'bottom', 'top', 'left', 'right', 'front', 'back'
    """
    # Get object's world matrix
    wm = obj.matrix_world
    
    # Local face centers for a cube/cylinder aligned with axes
    # Note: For cylinders, top/bottom are Z, sides are X/Y depending on rotation.
    # The prompt implies standard axis alignment after rotation.
    
    # We need to account for the object's rotation to find the correct local face
    # However, since we rotate objects to align axes, we can often just use local coordinates
    # transformed by world matrix.
    
    # Let's define local extents
    if obj.type == 'MESH':
        # Calculate bounding box to be safe, or use known geometry
        # For cubes:
        if 'cube' in obj.data.name or obj.type == 'MESH' and len(obj.data.polygons) <= 6: # Simple heuristic
            # Cube local centers
            # Assuming scale is applied to mesh or object? We scale object.
            # Local extents are +/- 1 in scaled space? No, scale is on object.
            # Local geometry is size 1.
            # Let's use the object's dimensions.
            dim = obj.dimensions
            # Local centers relative to origin
            # Top: (0,0, dim[2]/2)
            # Bottom: (0,0, -dim[2]/2)
            # Right: (dim[0]/2, 0, 0)
            # Left: (-dim[0]/2, 0, 0)
            # Front: (0, dim[1]/2, 0)
            # Back: (0, -dim[1]/2, 0)
            
            # But wait, rotation_euler rotates the object.
            # We need to rotate these local vectors by the object's rotation.
            
            rot_mat = obj.rotation_euler.to_matrix()
            
            if face_type == 'top':
                local_pt = Vector((0, 0, dim[2]/2))
            elif face_type == 'bottom':
                local_pt = Vector((0, 0, -dim[2]/2))
            elif face_type == 'right':
                local_pt = Vector((dim[0]/2, 0, 0))
            elif face_type == 'left':
                local_pt = Vector((-dim[0]/2, 0, 0))
            elif face_type == 'front':
                local_pt = Vector((0, dim[1]/2, 0))
            elif face_type == 'back':
                local_pt = Vector((0, -dim[1]/2, 0))
            else:
                return obj.location # Fallback
            
            # Transform to world
            world_pt = wm @ local_pt
            return world_pt
            
        elif 'cylinder' in obj.data.name or obj.type == 'MESH':
            # Cylinder local centers
            # Top/Bottom are along Z
            # Sides are radial.
            # 'right' usually means +X side, 'left' -X, 'front' +Y, 'back' -Y
            dim = obj.dimensions
            rot_mat = obj.rotation_euler.to_matrix()
            
            if face_type == 'top':
                local_pt = Vector((0, 0, dim[2]/2))
            elif face_type == 'bottom':
                local_pt = Vector((0, 0, -dim[2]/2))
            elif face_type == 'right':
                local_pt = Vector((dim[0]/2, 0, 0))
            elif face_type == 'left':
                local_pt = Vector((-dim[0]/2, 0, 0))
            elif face_type == 'front':
                local_pt = Vector((0, dim[0]/2, 0)) # Radius is dim[0]/2
            elif face_type == 'back':
                local_pt = Vector((0, -dim[0]/2, 0))
            else:
                return obj.location
                
            world_pt = wm @ local_pt
            return world_pt
            
    return obj.location

def align_and_offset(obj, anchor_obj, anchor_face, align_obj, align_face, offset):
    """
    1. Anchor: Move obj so anchor_face center matches anchor_obj's anchor_face center.
    2. Align: Rotate obj so align_face normal matches align_obj's align_face normal.
       (Note: The prompt says "Align ... to ...", usually implying normal alignment or face coplanarity.
        Given the simple structure, we assume normal alignment for orientation and position for anchor).
        However, the prompt separates "Anchor" and "Align".
        "Anchor frame_base_left.bottom_face to world.bottom_face" -> Position.
        "Align frame_base_left.left_face to world.left_face" -> Orientation?
        
        Let's interpret "Align face A to face B" as making the normals parallel and pointing in same direction.
        And "Anchor face A to face B" as moving the object so the centers coincide.
        
        Strict order: Orientation -> Create -> Anchor/Align -> Offset.
        The prompt says: "Rotate ... so +Z aligns +Z". This is explicit orientation.
        Then "Anchor ... to ...". This is position.
        Then "Align ... to ...". This might be redundant if orientation is fixed, or it might refine rotation.
        Given "Rotate ... so +Z aligns +Z", the rotation is fixed.
        "Align left_face to world.left_face" might just mean ensuring the face is on the correct side, 
        or it's a constraint that is already satisfied by the rotation.
        
        Let's implement:
        1. Set Rotation (Explicitly stated).
        2. Anchor: Translate object so anchor_face_center == target_face_center.
        3. Align: If "Align" implies rotation, we adjust rotation. If it implies position (sliding along face), we adjust position.
           Given the context of "Align ... to ... face", and the previous rotation step, 
           I will treat "Align" as a secondary position adjustment if normals don't match, 
           OR simply ensuring the face is coplanar/aligned.
           
           Actually, looking at Stage 2:
           "Anchor frame_base_top.top_face to frame_base_left.top_face"
           "Align frame_base_top.front_face to frame_base_left.front_face"
           
           If I anchor top to top, the Z position is set.
           If I align front to front, I might need to rotate around Z to match Y alignment?
           Or does it mean "Move so front faces are aligned"?
           
           Let's assume "Align face A to face B" means:
           - If normals are different, rotate object so normal A == normal B.
           - Then translate so centers coincide? No, anchor did that.
           
           Let's stick to the most robust interpretation for these simple blocks:
           1. Rotation is explicitly set to align +Z to +Z.
           2. Anchor sets the position of the specific face center.
           3. "Align" in this context likely ensures the object is not rotated incorrectly around the Z axis (if it wasn't fixed) or ensures coplanarity.
           Since +Z is fixed, rotation around Z is the only free variable.
           "Align front_face to front_face" likely means ensuring the Y-axis (front) is aligned.
           
           Implementation:
           - Calculate target rotation to align align_face normal to target align_face normal.
           - Apply this rotation.
           - Then Anchor (position).
           - Then Offset.
           
           Wait, the prompt says:
           "Rotate ... so +Z aligns +Z"
           "Anchor ..."
           "Align ..."
           "Then move by offset"
           
           I will perform:
           1. Set Rotation to (0,0,0) or as specified.
           2. Anchor: Move object so anchor_face_center == target_face_center.
           3. Align: Check if align_face normal matches target. If not, rotate. 
              (In this specific scene, all rotations are +Z to +Z, so normals are already aligned. 
               I will add a check/rotation step just in case).
           4. Apply Offset.
    """
    
    # 1. Orientation (Explicit Rotation)
    # The prompt explicitly states rotation. We set it.
    # Note: We assume the object was just created with default orientation or previous rotation.
    # We overwrite it.
    
    # 2. Anchor
    anchor_target_loc = get_face_center(anchor_obj, anchor_face)
    current_anchor_loc = get_face_center(obj, anchor_face)
    
    # Vector to move anchor face to target
    move_vec = anchor_target_loc - current_anchor_loc
    obj.location += move_vec
    
    # 3. Align
    # Get normals
    # Normal of a face in world space
    def get_face_normal(obj, face_type):
        wm = obj.matrix_world
        rot = obj.rotation_euler.to_matrix()
        
        # Local normals
        if face_type == 'top':
            local_n = Vector((0, 0, 1))
        elif face_type == 'bottom':
            local_n = Vector((0, 0, -1))
        elif face_type == 'right':
            local_n = Vector((1, 0, 0))
        elif face_type == 'left':
            local_n = Vector((-1, 0, 0))
        elif face_type == 'front':
            local_n = Vector((0, 1, 0))
        elif face_type == 'back':
            local_n = Vector((0, -1, 0))
        else:
            return Vector((0,0,1))
            
        # Rotate to world
        return rot @ local_n

    obj_align_n = get_face_normal(obj, align_face)
    target_align_n = get_face_normal(align_obj, align_face)
    
    # If normals are not aligned, we need to rotate the object around Z (since Z is fixed)
    # Or generally. But since Z is fixed, we only rotate around Z.
    if abs(obj_align_n.dot(target_align_n)) < 0.99:
        # Calculate angle between projected vectors on XY plane?
        # Or just use quaternion rotation.
        # Since Z is fixed, we want to align the "Front" vector.
        # Front vector is Y axis.
        
        # Let's just rotate the object so its align_face normal matches the target.
        # We must preserve the Z alignment.
        
        # Simple approach: Rotate around Z axis.
        # Current align face normal (projected to XY if it's a side face)
        # Target align face normal (projected to XY)
        
        # For 'front'/'back'/'left'/'right', the normal is in XY.
        # For 'top'/'bottom', the normal is Z.
        
        if align_face in ['front', 'back', 'left', 'right']:
            # Project normals to XY
            curr_xy = obj_align_n.copy()
            curr_xy.z = 0
            if curr_xy.length < 0.01: curr_xy = Vector((1,0,0)) # Fallback
            
            tgt_xy = target_align_n.copy()
            tgt_xy.z = 0
            if tgt_xy.length < 0.01: tgt_xy = Vector((1,0,0))
            
            curr_xy.normalize()
            tgt_xy.normalize()
            
            angle = curr_xy.angle(tgt_xy)
            # Determine direction
            cross = curr_xy.cross(tgt_xy)
            if cross.z < 0:
                angle = -angle
                
            # Rotate object around Z
            rot_z = Matrix.Rotation(angle, 4, 'Z')
            # Apply to rotation matrix
            # This is tricky with Euler. Let's use Quaternion.
            q_curr = obj.rotation_euler.to_quaternion()
            q_rot = Matrix.Rotation(angle, 4, 'Z').to_quaternion()
            q_new = q_rot * q_curr
            obj.rotation_euler = q_new.to_euler()
            
            # Re-anchor because rotation moved the face center!
            current_anchor_loc = get_face_center(obj, anchor_face)
            move_vec = anchor_target_loc - current_anchor_loc
            obj.location += move_vec

    # 4. Offset
    obj.location += offset

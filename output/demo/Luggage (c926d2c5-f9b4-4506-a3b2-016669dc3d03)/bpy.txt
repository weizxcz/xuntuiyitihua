import bpy
import bmesh
import mathutils
from mathutils import Vector, Matrix, Euler

def clear_scene():
    """Delete all existing objects."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for col in bpy.data.collections:
        bpy.data.collections.remove(col)

def set_units_metres():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a simple Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_primitive_cube(name, size, location, rotation_euler, mat=None):
    """Add a cube and apply transforms."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if mat:
        apply_material(obj, mat)
    return obj

def add_primitive_cylinder(name, radius, depth, location, rotation_euler, mat=None):
    """Add a cylinder and apply transforms."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if mat:
        apply_material(obj, mat)
    return obj

def add_primitive_sphere(name, radius, location, rotation_euler, mat=None):
    """Add a UV Sphere and apply transforms."""
    bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=radius, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if mat:
        apply_material(obj, mat)
    return obj

def add_primitive_torus(name, major_radius, minor_radius, location, rotation_euler, mat=None):
    """Add a Torus and apply transforms."""
    bpy.ops.mesh.primitive_torus_add(major_radius=major_radius, minor_radius=minor_radius, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if mat:
        apply_material(obj, mat)
    return obj

def align_axis_to_vector(obj, axis_to_align, target_vector):
    """
    Rotate object so its local axis aligns with target_vector.
    axis_to_align: 'X', 'Y', or 'Z'
    target_vector: Vector in world space
    """
    # Get current world rotation
    current_mat = obj.matrix_world
    
    # Define local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    # Get the current world direction of the axis to align
    current_dir = (current_mat @ local_axes[axis_to_align]).normalized()
    
    # Calculate rotation matrix to align current_dir to target_vector
    rot_mat = current_dir.rotation_difference(target_vector).to_matrix().to_4x4()
    
    # Apply rotation
    obj.matrix_world = rot_mat @ current_mat

def get_face_center(obj, face_type):
    """
    Get the world location of a specific face center of an object.
    face_type: '+X_face', '-X_face', '+Y_face', '-Y_face', '+Z_face', '-Z_face'
    """
    # Ensure object is updated
    obj.update_from_editmode()
    
    # Get bounding box corners in world space
    corners = [
        obj.matrix_world @ Vector((-1, -1, -1)),
        obj.matrix_world @ Vector(( 1, -1, -1)),
        obj.matrix_world @ Vector((-1,  1, -1)),
        obj.matrix_world @ Vector(( 1,  1, -1)),
        obj.matrix_world @ Vector((-1, -1,  1)),
        obj.matrix_world @ Vector(( 1, -1,  1)),
        obj.matrix_world @ Vector((-1,  1,  1)),
        obj.matrix_world @ Vector(( 1,  1,  1)),
    ]
    
    # Scale factors based on object size (half dimensions)
    sx, sy, sz = obj.scale
    
    # Map face types to corner indices or averages
    # Note: This is a simplified approximation for primitives. 
    # For complex shapes, mesh analysis is needed. Assuming primitives here.
    
    # Center of object
    center = obj.location
    
    # Vectors from center to face centers
    # +X face center is at center + (sx, 0, 0)
    face_centers = {
        '+X_face': center + Vector((sx, 0, 0)),
        '-X_face': center + Vector((-sx, 0, 0)),
        '+Y_face': center + Vector((0, sy, 0)),
        '-Y_face': center + Vector((0, -sy, 0)),
        '+Z_face': center + Vector((0, 0, sz)),
        '-Z_face': center + Vector((0, 0, -sz)),
    }
    
    return face_centers.get(face_type, center)

def place_object(obj, anchor_obj, anchor_face, offset):
    """
    Anchor object to a face of another object and apply offset.
    1. Align object's local origin to the anchor face center.
    2. Apply offset in the anchored object's local coordinate system? 
       The prompt says "Offset is in reference local axes". 
       Usually, this means after anchoring, we move the object by offset relative to its own orientation.
    """
    # 1. Calculate target world location
    anchor_world_loc = get_face_center(anchor_obj, anchor_face)
    
    # 2. Set object location to anchor point
    obj.location = anchor_world_loc
    
    # 3. Apply offset
    # The offset is given in the object's local axes after orientation.
    # Since we set location, we just add the offset vector to the location.
    # However, if the offset is meant to be in the anchor's local space, it would be different.
    # Prompt: "Offset is in reference local axes." -> Usually implies the object being placed's local axes.
    obj.location += offset

def main():

    mat_metal_dark = make_material("metal_dark", (0.2, 0.2, 0.25, 1))
    mat_plastic_black = make_material("plastic_black", (0.05, 0.05, 0.05, 1))
    mat_plastic_grey = make_material("plastic_grey", (0.4, 0.4, 0.45, 1))
    mat_rubber_black = make_material("rubber_black", (0.08, 0.08, 0.08, 1))

    # ==============================================================================
    # BLOCK 2: Geometry Generation

    # --- SECTION 1 – Create Hard Shell Body ---
    # Create primitive cuboid with size 0.75×0.5×0.3 m and name it hard_shell_body.
    # Rotate hard_shell_body so its +Z axis aligns +Z (world).
    # Anchor hard_shell_body to world origin.
    # Then move by offset(0,0,0).
    
    h_shell = add_primitive_cube("hard_shell_body", (0.75, 0.5, 0.3), (0,0,0), (0,0,0), mat_hard_shell)
    # Align +Z to +Z (already aligned by default cube creation if not rotated, but explicit step)
    align_axis_to_vector(h_shell, 'Z', Vector((0,0,1)))
    # Anchor to world origin
    h_shell.location = Vector((0,0,0))
    # Offset
    h_shell.location += Vector((0,0,0))

    # --- SECTION 2 – Create Handle Assembly ---
    
    # Create primitive cuboid with size 0.2×0.02×0.1 m and name it handle_base.
    # Rotate handle_base so its +Z axis aligns +Z (world).
    # Anchor handle_base to hard_shell_body.+Z_face.
    # Then move by offset(0,0,0.15).
    
    h_base = add_primitive_cube("handle_base", (0.2, 0.02, 0.1), (0,0,0), (0,0,0), mat_plastic_grey)
    align_axis_to_vector(h_base, 'Z', Vector((0,0,1)))
    place_object(h_base, h_shell, '+Z_face', Vector((0,0,0.15)))

    # Create primitive cylinder with size Ø0.01×0.6 m and name it handle_poles.
    # Rotate handle_poles so its +Z axis aligns +Y (world).
    # Anchor handle_poles to handle_base.+Z_face.
    # Then move by offset(0,0,0.01).
    
    # Cylinder default is along Z. We want +Z aligned to +Y.
    # Radius = 0.01/2 = 0.005. Depth = 0.6.
    h_poles = add_primitive_cylinder("handle_poles", 0.005, 0.6, (0,0,0), (0,0,0), mat_metal_aluminum)
    align_axis_to_vector(h_poles, 'Z', Vector((0,1,0)))
    place_object(h_poles, h_base, '+Z_face', Vector((0,0,0.01)))

    h_grip = add_primitive_cylinder("handle_grip", 0.015, 0.15, (0,0,0), (0,0,0), mat_rubber_black)

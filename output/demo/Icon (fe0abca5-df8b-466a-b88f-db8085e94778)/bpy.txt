# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(name, radius, segments=32):
    bpy.ops.mesh.primitive_circle_add(vertices=segments, radius=radius, location=(0,0,0))
    obj = bpy.context.active_object
    obj.data.name = name
    obj.name = name
    bpy.ops.mesh.fill()
    return obj

def add_cube(name, size):
    # size is (x, y, z)
    half = [s/2 for s in size]
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0,0,0)) # size=1 means 2x2x2 cube, so scale down
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    return obj

def add_cone(name, bottom_radius, top_radius, depth, segments=32):
    bpy.ops.mesh.primitive_cone_add(vertices=segments, radius1=bottom_radius, radius2=top_radius, depth=depth, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=16, ring_count=8, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_tapered_prism(name, base_size, height, segments=3):
    """Creates a triangular prism (3 segments) or generic polygonal prism."""
    # Create a polygon base
    bpy.ops.mesh.primitive_uv_sphere_add(segments=segments, ring_count=1, radius=1, location=(0,0,0))
    # This is tricky for a prism. Let's use a cylinder with 3 segments and scale to make it flat?
    # Better: Create a cylinder with 3 segments, scale Z to height, then scale X/Y to base_size.
    # But a cylinder with 3 segments is a triangular prism.
    # Default cylinder is centered at 0,0,0.
    bpy.ops.mesh.primitive_cylinder_add(vertices=3, radius=1, depth=1, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    # The cylinder radius is circumradius. We want the "size" to be the width of the face.
    # For a triangle, width = side length. Side = sqrt(3) * R. So R = side / sqrt(3).
    # Let's assume 'base_size' is the width of the triangular face.
    side = base_size
    circumradius = side / math.sqrt(3)
    obj.scale = (circumradius, circumradius, height)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Aligns object's local axis to a world vector."""
    if axis == 'Z':
        local_axis = Vector((0,0,1))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    elif axis == 'X':
        local_axis = Vector((1,0,0))
    else:
        return

    target = Vector(vector).normalized()
    if local_axis == target:
        return
    
    # Calculate rotation
    rot_quat = local_axis.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

def align_faces(obj_source, face_axis, obj_target, face_axis_target):
    """
    Aligns a face of source to a face of target.
    face_axis: 'top', 'bottom', 'front', 'back', 'left', 'right'
    """
    # Get bounding boxes
    src_min, src_max = obj_source.bound_box_local
    tgt_min, tgt_max = obj_target.bound_box_local
    
    # Transform to world space
    src_min_w = [obj_source.matrix_world @ Vector(v) for v in src_min]
    src_max_w = [obj_source.matrix_world @ Vector(v) for v in src_max]
    tgt_min_w = [obj_target.matrix_world @ Vector(v) for v in tgt_min]
    tgt_max_w = [obj_target.matrix_world @ Vector(v) for v in tgt_max]

    # Determine face centers and normals
    # Note: bound_box_local is in local space. We need to be careful with orientation.
    # Assuming standard orientation where Z is up, Y is front/back, X is left/right.
    
    # Helper to get face center and normal in world space
    def get_face_info(min_w, max_w, axis):
        if axis == 'top':
            center = Vector((0, 0, max_w[2])) # Approximate, assuming axis aligned
            # Better: average of points with max Z
            pts = [v for v in [Vector((min_w[0], min_w[1], max_w[2])), Vector((max_w[0], min_w[1], max_w[2])),
                               Vector((min_w[0], max_w[1], max_w[2])), Vector((max_w[0], max_w[1], max_w[2]))]]
            center = sum(pts)/4
            normal = Vector((0,0,1))
        elif axis == 'bottom':
            pts = [v for v in [Vector((min_w[0], min_w[1], min_w[2])), Vector((max_w[0], min_w[1], min_w[2])),
                               Vector((min_w[0], max_w[1], min_w[2])), Vector((max_w[0], max_w[1], min_w[2]))]]
            center = sum(pts)/4
            normal = Vector((0,0,-1))
        elif axis == 'front':
            # Assuming Y+ is front
            pts = [v for v in [Vector((min_w[0], max_w[1], min_w[2])), Vector((max_w[0], max_w[1], min_w[2])),
                               Vector((min_w[0], max_w[1], max_w[2])), Vector((max_w[0], max_w[1], max_w[2]))]]
            center = sum(pts)/4
            normal = Vector((0,1,0))
        elif axis == 'back':
            pts = [v for v in [Vector((min_w[0], min_w[1], min_w[2])), Vector((max_w[0], min_w[1], min_w[2])),
                               Vector((min_w[0], min_w[1], max_w[2])), Vector((max_w[0], min_w[1], max_w[2]))]]
            center = sum(pts)/4
            normal = Vector((0,-1,0))
        elif axis == 'right':
            pts = [v for v in [Vector((max_w[0], min_w[1], min_w[2])), Vector((max_w[0], max_w[1], min_w[2])),
                               Vector((max_w[0], min_w[1], max_w[2])), Vector((max_w[0], max_w[1], max_w[2]))]]
            center = sum(pts)/4
            normal = Vector((1,0,0))
        elif axis == 'left':
            pts = [v for v in [Vector((min_w[0], min_w[1], min_w[2])), Vector((min_w[0], max_w[1], min_w[2])),
                               Vector((min_w[0], min_w[1], max_w[2])), Vector((min_w[0], max_w[1], max_w[2]))]]
            center = sum(pts)/4
            normal = Vector((-1,0,0))
        else:
            return None, None
        return center, normal

    src_center, src_normal = get_face_info(src_min_w, src_max_w, face_axis)
    tgt_center, tgt_normal = get_face_info(tgt_min_w, tgt_max_w, face_axis_target)
    
    if src_center is None:
        return

    # Align positions: move source so its face center touches target face center
    # But we want them to touch, so we need to account for thickness?
    # The prompt says "Align ... to ...". Usually implies touching surfaces.
    # If we just move centers, they might overlap.
    # Let's move the source object so its face center is at the target face center.
    # Then the offset logic will handle separation if needed.
    
    # However, for boolean/assembly, usually we want the face to be *at* the other face.
    # Let's move the source object's location such that src_center moves to tgt_center.
    
    delta = tgt_center - src_center
    obj_source.location += delta

def align_center_to_face(obj_source, obj_target, face_axis):
    """Aligns source center to a specific face of target."""
    tgt_min_w = [obj_target.matrix_world @ Vector(v) for v in obj_target.bound_box_local]
    tgt_max_w = [obj_target.matrix_world @ Vector(v) for v in obj_target.bound_box_local]
    
    tgt_center, _ = get_face_info(tgt_min_w, tgt_max_w, face_axis)
    if tgt_center is None:
        return

    # Move source center to tgt_center
    src_center = obj_source.location
    delta = tgt_center - src_center
    obj_source.location += delta

def apply_offset(obj, offset):
    """Applies offset in local axes."""
    # Convert local offset to world direction based on current rotation
    # If object is rotated, local axes are rotated.
    # We want to move along the object's local X, Y, Z by offset values.
    rot_mat = obj.matrix_world.to_3x3()
    move_vec = Vector((offset[0], offset[1], offset[2]))
    world_move = rot_mat @ move_vec
    obj.location += world_move

def boolean_subtract(cutter_name, target_name):
    """Performs boolean subtract using modifier."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is active for modifier addition? No, add to target.
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_name, target_name):
    """Performs boolean union."""
    target = bpy.data.objects[target_name]
    union = bpy.data.objects[union_name]
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union, do_unlink=True)

def add_bevel(obj_name, width=0.01, segments=2):
    """Adds a bevel modifier."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color=(0,0,0,1)):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(radius, depth, segments=32, name="Cylinder"):
    """Add a cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(radius, segments=32, name="Disc"):
    """Add a flat disc aligned with Z axis."""
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='TRIFAN', segments=segments, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(size, name="Cube"):
    """Add a cube centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(radius_bottom, radius_top, depth, segments=32, name="Cone"):
    """Add a cone/tapered cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, segments=segments, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(radius, width_seg=16, height_seg=16, name="UVSphere"):
    """Add a UV Sphere."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_seg, ring_count=height_seg, location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width=0.01, segments=4, target="WEIGHT"):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.target = target
    return mod

def boolean_subtract(base_obj, cutter_obj, keep_base=True, name="BooleanResult"):
    """Perform boolean subtract using BMesh."""
    # Ensure base is active
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.select_all(action='DESELECT')
    base_obj.select_set(True)
    
    # Create a copy of the base for boolean operation to avoid modifying original data directly if needed, 
    # but standard practice in scripts is often to apply modifiers or use bmesh on a copy.
    # Here we use the modifier approach for non-destructive workflow, then apply if needed.
    # However, for robust scripting without UI interaction, BMesh boolean is often safer for single-run scripts.
    
    # Let's use the Modifier approach for simplicity and standard Blender workflow
    mod = base_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.use_self = False
    
    # Create a dummy object to hold the cutter data if not already a mesh, or just use the cutter
    # We need to link the cutter to the modifier's object field
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove the cutter object
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    
    base_obj.name = name
    return base_obj

def boolean_union(base_obj, other_obj, keep_base=True, name="BooleanResult"):
    """Perform boolean union."""
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.select_all(action='DESELECT')
    base_obj.select_set(True)
    
    mod = base_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.use_self = False
    mod.object = other_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove the other object
    bpy.data.objects.remove(other_obj, do_unlink=True)
    
    base_obj.name = name
    return base_obj

def align_axis_to_vector(obj, axis='Z', vector=Vector((0,0,1))):
    """Align an object's local axis to a world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else: # Z
        local_axis = Vector((0,0,1))
        
    # Calculate rotation
    rot_quat = local_axis.rotation_difference(vector)
    obj.rotation_euler = rot_quat.to_euler()

class Locator:
    def __init__(self, obj):
        self.obj = obj
        
    def center_world(self):
        """Return the world position of the object's center."""
        return self.obj.matrix_world.to_translation()
    
    def face_center_world(self, face_index=0):
        """Return the world position of a specific face center."""
        # Assuming mesh object
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bm.transform(self.obj.matrix_world)
        if face_index < len(bm.faces):
            face = bm.faces[face_index]
            return face.calc_center_median()
        bm.free()
        return self.center_world()

    def radial_point_world(self, angle=0, radius=None):
        """Return a point on the radial edge."""
        # Simplified: returns center for now, or could calculate based on bounding box
        return self.center_world()

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    sum_vec = Vector((0,0,0))
    for p in points:
        sum_vec += p
    return sum_vec / len(points)

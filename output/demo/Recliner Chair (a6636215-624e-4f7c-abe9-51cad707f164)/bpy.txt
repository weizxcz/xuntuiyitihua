import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube object with specific size, location, and rotation."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=Vector((0,0,1))):
    """Align an object's local axis to a world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else: # Z
        local_axis = Vector((0,0,1))
        
    # Calculate rotation to align local_axis to target vector
    rot_quat = local_axis.rotation_difference(vector)
    obj.rotation_euler = rot_quat.to_euler()

def get_face_center(obj, face_type):
    """
    Get the world location of a specific face center of an object.
    face_type: 'top', 'bottom', 'front', 'back', 'left', 'right'
    Assumes object is axis-aligned or we are approximating based on bounds.
    """
    # Get bounds
    min_x, min_y, min_z = obj.location - (obj.scale[0], obj.scale[1], obj.scale[2])
    max_x, max_y, max_z = obj.location + (obj.scale[0], obj.scale[1], obj.scale[2])
    
    # Adjust for rotation if necessary, but for simple axis-aligned boxes:
    # Note: The prompt implies objects are rotated to align +Z to +Z, so they are axis aligned.
    
    if face_type == 'top':
        return Vector((obj.location.x, obj.location.y, max_z))
    elif face_type == 'bottom':
        return Vector((obj.location.x, obj.location.y, min_z))
    elif face_type == 'front': # +Y
        return Vector((obj.location.x, max_y, obj.location.z))
    elif face_type == 'back': # -Y
        return Vector((obj.location.x, min_y, obj.location.z))
    elif face_type == 'right': # +X
        return Vector((max_x, obj.location.y, obj.location.z))
    elif face_type == 'left': # -X
        return Vector((min_x, obj.location.y, obj.location.z))
    else:
        return obj.location

def anchor_and_offset(anchor_obj, anchor_face, offset_vec, target_obj):
    """
    Anchors target_obj to a face of anchor_obj, then applies offset.
    The offset is applied in the reference local axes (assumed world-aligned here).
    """
    # 1. Get anchor point
    anchor_point = get_face_center(anchor_obj, anchor_face)
    
    # 2. Calculate target location
    # The offset is relative to the anchor point in world space (since axes are aligned)
    target_loc = anchor_point + offset_vec
    
    # 3. Set location
    target_obj.location = target_loc

def create_stage_object(name, size, anchor_obj=None, anchor_face=None, offset=Vector((0,0,0)), material=None):
    """
    Creates a cuboid, orients it, anchors it, and offsets it.
    """
    # 1. Create primitive
    # Default location (0,0,0) before anchoring
    obj = add_cube(name, size, (0,0,0), (0,0,0), material)
    
    # 2. Rotate so +Z aligns +Z (already default, but explicit per rules)
    align_axis_to_vector(obj, 'Z', Vector((0,0,1)))
    
    # 3. Anchor & Offset
    if anchor_obj is not None and anchor_face is not None:
        anchor_and_offset(anchor_obj, anchor_face, offset, obj)
    else:
        # If no anchor, just apply offset to initial location (0,0,0)
        obj.location = offset
        
    return obj

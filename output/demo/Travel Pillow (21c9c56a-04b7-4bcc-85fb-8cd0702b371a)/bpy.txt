import bpy
import bmesh
from mathutils import Vector, Matrix
from math import pi

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a world vector."""
    if axis == 'Z':
        local_axis = Vector((0, 0, 1))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    elif axis == 'X':
        local_axis = Vector((1, 0, 0))
    else:
        return

    target = vector.normalized()
    if local_axis.normalized().dot(target) < 0.999999:
        quat = local_axis.rotation_difference(target)
        obj.rotation_euler = quat.to_euler()

def get_world_center(obj):
    """Get the world location of an object's center."""
    return obj.matrix_world.to_translation()

def set_world_location(obj, loc):
    """Set the world location of an object."""
    obj.location = loc

def boolean_subtract(cutter_name, target_name):
    """Perform a boolean subtract using the modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target has modifier
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply immediately for simplicity in this script flow
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Optional: Remove cutter if desired, but keeping it allows reuse if needed.
    # bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(obj1_name, obj2_name):
    """Perform a boolean union."""
    target = bpy.data.objects[obj1_name]
    source = bpy.data.objects[obj2_name]
    
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = source
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(source, do_unlink=True)

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_transform(obj):
    """Apply location, rotation, and scale to an object."""
    bpy.context.view_layer.objects.active = obj

def get_world_center(obj):
    """Get the world space center of an object's bounding box."""
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bm.transform(obj.matrix_world)
    center = bm.calc_center_bounds()
    bm.free()
    return center

def align_to_vector(obj, vector):
    """Align object's local Z axis to a world vector."""
    vec = vector.normalized()
    # Current Z axis
    current_z = Vector((0, 0, 1))
    # Calculate rotation
    rot_quat = current_z.rotation_difference(vec)
    obj.rotation_euler = rot_quat.to_euler()

def boolean_union(obj_target, obj_source):
    """Perform boolean union using modifier stack."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = obj_target
    
    # Add Boolean Modifier to Target
    mod = obj_target.modifiers.new(name="BoolUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = obj_source
    
    # Apply Modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove source object if desired, or keep it. Usually keep for debugging or remove.
    # Here we keep it but it's merged.
    return obj_target

def boolean_subtract(obj_target, obj_source):
    """Perform boolean subtract using modifier stack."""
    bpy.context.view_layer.objects.active = obj_target
    
    mod = obj_target.modifiers.new(name="BoolSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = obj_source
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    return obj_target

def extrude_along_z(obj, thickness):
    """Extrude an object's mesh along Z axis by a given thickness."""
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    
    # Extrude region

def create_extruded_object(name, profile_func, location, rotation_euler, scale, thickness):
    """Create an object from a 2D profile and extrude it along Z."""
    # Create a temporary mesh
    bpy.ops.mesh.primitive_plane_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    
    # Clear default geometry
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.mesh.delete(type='VERT')
    bpy.ops.object.mode_set(mode='OBJECT')
    
    # Create new mesh
    new_mesh = bpy.data.meshes.new(name + "_mesh")
    obj.data = new_mesh
    
    bm = bmesh.new()
    profile_func(bm)
    
    # Extrude along Z
    bmesh.ops.select_all(bm, action='SELECT')
    bmesh.ops.extrude_discrete_faces(bm, faces=bm.faces)
    
    # Move the new faces along Z
    # Find the new faces (they are the last ones added)
    # Actually, bmesh.ops.extrude_discrete_faces returns a dict with 'faces'
    # But we can just select all and move by half thickness up and half down?
    # Or move the new ones by thickness.
    
    # Let's just move all vertices by thickness in Z? No, that scales.
    # Let's move the new faces.
    
    # Simpler approach: 
    # 1. Create profile in XY.
    # 2. Extrude by thickness/2 in +Z and thickness/2 in -Z.
    
    # Re-do:
    bm.clear()
    profile_func(bm)
    
    # Extrude half up
    bmesh.ops.select_all(bm, action='SELECT')
    result = bmesh.ops.extrude_discrete_faces(bm, faces=bm.faces)
    new_faces = result['faces']
    
    # Move new faces by thickness/2
    for f in new_faces:
        for v in f.verts:
            v.co.z += thickness / 2
            
    # Extrude half down (from original)
    # Wait, the original faces are still there.
    # Let's select original faces and extrude down.
    bmesh.ops.select_all(bm, action='DESELECT')
    for f in bm.faces:
        if f not in new_faces:
            f.select_set(True)
            
    result2 = bmesh.ops.extrude_discrete_faces(bm, faces=[f for f in bm.faces if f.select])
    new_faces2 = result2['faces']
    
    for f in new_faces2:
        for v in f.verts:
            v.co.z -= thickness / 2
            
    bm.to_mesh(obj.data)
    bm.free()
    
    bpy.ops.object.mode_set(mode='OBJECT')
    apply_transform(obj)
    
    return obj

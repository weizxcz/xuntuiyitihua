def make_material(name, diffuse_color):
    """Create a principled BSDF material with a specific diffuse color."""
    if name in bpy.data.materials:
        return bpy.data.materials[name]
    
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    # Adjust roughness/metallic for better visual distinction if needed, 
    # but strictly following prompt: diffuse_color is the main input.
    # Default Principled BSDF is somewhat plastic-like. 
    # For "metal" names, we might tweak metallic, but prompt only specifies diffuse_color.
    # We will stick strictly to the provided color.
    return mat

def add_text_object(text, name, location=(0,0,0), size=0.1):
    """Add a text object."""
    bpy.ops.object.text_add(location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.body = text
    obj.data.size = size
    return obj

def add_cylinder(name, radius=0.5, depth=1.0, segments=32, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    return obj

def add_bevel(obj_name, width=0.01, segments=1):
    """Add a bevel modifier to an existing object."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    return mod

def boolean_subtract(target_name, cutter_name):
    """Perform a boolean subtract operation."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure cutter is active for boolean setup if needed, but usually we add modifier to target
    bpy.context.view_layer.objects.active = target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    return mod

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to create invisible locator objects for anchoring."""
    def __init__(self, name, location=(0,0,0)):
        # Create a small cube as a locator
        bpy.ops.mesh.primitive_cube_add(size=0.01, location=location)
        self.obj = bpy.context.active_object
        self.obj.name = name
        self.obj.hide_viewport = True
        self.obj.hide_render = True
        # Make it a pivot point
        self.obj.location = location

    @property
    def center_world(self):
        return self.obj.location

    def face_center_world(self, normal_vector=(0,0,1)):
        """Align locator's Z axis to normal_vector."""
        # This is complex to do purely with location. 
        # Usually locators are just points. If we need orientation, we rotate the locator.
        self.obj.rotation_euler = self._vector_to_euler(normal_vector)

    def radial_point_world(self, radius, angle=0):
        """Return a point on the radial plane."""
        # Placeholder for radial positioning logic if needed
        return Vector((radius * math.cos(angle), radius * math.sin(angle), 0))

    def _vector_to_euler(self, vec):
        """Convert a direction vector to Euler angles (Z-up convention)."""
        v = vec.normalized()
        x = math.atan2(v.y, v.z)
        y = math.atan2(-v.x, math.sqrt(v.y**2 + v.z**2))
        z = 0
        return (x, y, z)

def place_object(obj, anchor_loc, offset=(0,0,0)):
    """Place object relative to an anchor location with an offset."""
    obj.location = anchor_loc + Vector(offset)

def orient_object(obj, axis='Z', target_vector=(0,0,1)):
    """Orient object so its axis aligns with target_vector."""
    # For standard cases provided in the prompt:
    # +Z aligns +Z -> no rotation needed (default)
    # +Z aligns +X -> Rotate Y by -90 degrees (or 90 depending on handedness)
    # Blender is right-handed. 
    # Z(0,0,1) to X(1,0,0): Rotate around Y by -90 deg (math.radians(-90))
    if axis == 'Z':
        if target_vector == (1,0,0): # +X
            obj.rotation_euler = (0, math.radians(-90), 0)
        elif target_vector == (0,1,0): # +Y
            obj.rotation_euler = (math.radians(90), 0, 0)
        elif target_vector == (0,0,1): # +Z
            obj.rotation_euler = (0, 0, 0)
        elif target_vector == (-1,0,0): # -X
            obj.rotation_euler = (0, math.radians(90), 0)
        elif target_vector == (0,-1,0): # -Y
            obj.rotation_euler = (math.radians(-90), 0, 0)
        elif target_vector == (0,0,-1): # -Z
            obj.rotation_euler = (math.radians(180), 0, 0)

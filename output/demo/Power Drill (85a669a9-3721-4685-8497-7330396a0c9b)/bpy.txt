import bpy
import bmesh
import math
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.data.name = name + "_Data"
    return obj

def add_cube(name, size):
    # size is (x, y, z)
    half_x, half_y, half_z = size[0]/2, size[1]/2, size[2]/2
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.data.name = name + "_Data"
    obj.scale = (half_x, half_y, half_z)
    return obj

def add_text_object(name, text_string, size=0.1):
    bpy.ops.object.text_add(location=(0,0,0))
    obj = bpy.context.active_object
    obj.name = name
    obj.data.body = text_string
    obj.data.size = size
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a world vector."""
    if axis == 'X':
        src = Vector((1, 0, 0))
    elif axis == 'Y':
        src = Vector((0, 1, 0))
    elif axis == 'Z':
        src = Vector((0, 0, 1))
    else:
        return

    target = vector.normalized()
    if src.dot(target) < -0.999:
        rot = Matrix.Rotation(math.pi, 4, src.cross(target) if src.cross(target).length > 0 else Vector((0,1,0)))
    else:
        rot = Matrix.Rotation(src.angle(target), 4, src.cross(target))
    
    obj.rotation_euler = rot.to_euler()

def average(points):
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj

    def center_world(self):
        return self.obj.location

    def face_center_world(self, face_index):
        # Calculate center of a specific face in world space
        bm = bmesh.new()
        bm.from_mesh(self.obj.data)
        bm.transform(self.obj.matrix_world)
        face = bm.faces[face_index]
        center = average([v.co for v in face.verts])
        bm.free()
        return center

    def radial_point_world(self, angle_deg, radius=None):
        # Get a point on the radial edge at a specific angle
        # Assumes object is at origin or we transform later. 
        # Simplified: returns local radial point, caller must transform.
        if radius is None:
            # Estimate radius from bounding box
            radius = self.obj.dimensions[0] / 2
        rad = math.radians(angle_deg)
        return Vector((math.cos(rad) * radius, math.sin(rad) * radius, 0))

def boolean_subtract(cutter_name, target_name):
    cutter = bpy.data.objects[cutter_name]
    target = bpy.data.objects[target_name]
    
    # Ensure cutter is active for modifier addition if needed, but we add to target
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="Boolean_Subtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter)

def boolean_union(union_name, target_name):
    union_obj = bpy.data.objects[union_name]
    target = bpy.data.objects[target_name]
    
    bpy.context.view_layer.objects.active = target
    
    mod = target.modifiers.new(name="Boolean_Union", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union_obj)

def add_bevel(obj_name, width=0.005, segments=4):
    obj = bpy.data.objects[obj_name]
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.mode_set(mode='EDIT')
    bpy.ops.mesh.select_all(action='SELECT')
    bpy.ops.mesh.bevel(angle=math.radians(45), offset=width, segments=segments, vertex_only=False)
    bpy.ops.object.mode_set(mode='OBJECT')

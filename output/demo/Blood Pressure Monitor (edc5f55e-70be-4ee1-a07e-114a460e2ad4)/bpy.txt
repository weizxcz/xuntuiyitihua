import bpy

# ==============================================================================
# HELPER FUNCTIONS
# ==============================================================================

def clear_scene():
    """Delete all existing objects."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for collection in bpy.data.collections:
        collection.objects.clear()

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    # Adjust roughness/metallic for better visual approximation if needed, 
    # but sticking to diffuse_color as requested implies basic shading.
    return mat

def assign_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        assign_material(obj, material)
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):
    """Add a cylinder object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    obj.location = location
    if material:
        assign_material(obj, material)
    return obj

def get_world_center(obj):
    """Get the world space center of an object."""
    return obj.matrix_world.to_translation()

def get_face_center(obj, face_type='top'):
    """
    Get the world space center of a specific face of an object.
    Assumes object is axis-aligned or has been oriented such that local Z is up.
    """
    # Local center
    center_local = Vector((0, 0, 0))
    
    # Determine offset based on face type and object dimensions
    # Assuming standard primitives created with scale
    if face_type == 'top':
        offset_local = Vector((0, 0, obj.dimensions[2]/2))
    elif face_type == 'bottom':
        offset_local = Vector((0, 0, -obj.dimensions[2]/2))
    else:
        offset_local = Vector((0, 0, 0))
        
    local_face_center = center_local + offset_local
    
    # Convert to world space
    world_face_center = obj.matrix_world @ local_face_center
    return world_face_center

def align_to_face(target_obj, source_obj, target_face='top', source_face='bottom'):
    """
    Align source_obj's face to target_obj's face.
    Moves source_obj so its face center matches target face center.
    """
    target_center = get_face_center(target_obj, target_face)
    source_center = get_face_center(source_obj, source_face)
    
    # Calculate translation vector
    delta = target_center - source_center
    source_obj.location += delta

def apply_offset(obj, offset_vector):
    """Apply an offset to an object's location."""
    obj.location += offset_vector

def orient_z_up(obj):
    """Ensure object's +Z axis aligns with world +Z."""
    # For standard primitives, default rotation is often (0,0,0) which aligns Z up.
    # If rotation was set differently, we reset it.
    obj.rotation_euler = (0, 0, 0)

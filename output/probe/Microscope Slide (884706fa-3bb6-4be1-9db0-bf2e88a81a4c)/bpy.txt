def clear_scene():
    """Delete all existing objects."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear collections if necessary, though delete usually handles it
    for collection in bpy.data.collections:
        collection.objects.clear()

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Alpha'].default_value = diffuse_color[3]
    bsdf.inputs['Specular IOR Level'].default_value = 0.5
    return mat

def assign_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a cube primitive."""
    # size is a tuple (x, y, z)
    half_size = [s/2 for s in size]
    bpy.ops.mesh.primitive_cube_add(size=1, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = half_size
    return obj

def align_face_to_face(obj_source, obj_target, source_face_normal, target_face_normal, offset=0):
    """
    Aligns a face of obj_source to a face of obj_target.
    source_face_normal: Vector in local space of source object representing the face normal (e.g., (0,0,1) for +Z face)
    target_face_normal: Vector in local space of target object representing the face normal (e.g., (0,0,1) for +Z face)
    Note: This helper assumes simple axis-aligned faces for the specific use cases in this script.
    """
    # Get world transforms
    src_mat = obj_source.matrix_world
    tgt_mat = obj_target.matrix_world
    
    # Calculate target world position for the source face
    # We want the source face to touch the target face.
    # Distance from source center to source face along normal
    # For a cube of size S, distance is S/2. For cylinder, it's depth/2.
    
    # Determine bounding box to find face position accurately
    src_bbox = obj_source.bound_box
    tgt_bbox = obj_target.bound_box
    
    # Project bbox corners onto the face normal to find min/max distance
    def get_face_dist(bbox, normal):
        max_d = -float('inf')
        for corner in bbox:
            v = Vector(corner)
            d = v.dot(normal)
            if d > max_d:
                max_d = d
        return max_d

    # Local face distances
    src_face_dist_local = get_face_dist(src_bbox, source_face_normal)
    tgt_face_dist_local = get_face_dist(tgt_bbox, target_face_normal)
    
    # Convert to world space
    src_face_normal_world = src_mat.to_3x3() @ source_face_normal
    tgt_face_normal_world = tgt_mat.to_3x3() @ target_face_normal
    
    # We want to move obj_source so its face touches obj_target's face.
    # The target face world position is:
    tgt_face_center_world = tgt_mat.translation + (tgt_mat.to_3x3() @ Vector((0,0,0))) # Center
    # Actually, simpler: The face plane equation.
    # Let's use a locator approach for robustness.
    
    # Create a temporary locator at the target face center
    # Target face center in world space
    # Find the corner of the target bbox that is furthest in the direction of target_face_normal
    tgt_face_world_pos = Vector((0,0,0))
    max_d = -float('inf')
    for corner in tgt_bbox:
        v = Vector(corner)
        # Transform corner to world
        w_corner = tgt_mat @ v
        # Project onto target face normal
        d = w_corner.dot(tgt_face_normal_world)
        if d > max_d:
            max_d = d
            tgt_face_world_pos = w_corner
            
    # The source face needs to be at tgt_face_world_pos - offset * tgt_face_normal_world
    # But we also need to orient the source object.
    
    # Orientation: Align source local normal to target world normal
    # Current source local normal in world
    current_src_normal_world = src_mat.to_3x3() @ source_face_normal
    
    # Rotation needed to align current_src_normal_world to tgt_face_normal_world
    rot_quat = current_src_normal_world.rotation_difference(tgt_face_normal_world)
    
    # Apply rotation to source object
    obj_source.rotation_euler = rot_quat.to_euler()
    obj_source.update_from_matrix() # Update matrix after rotation
    
    # Recalculate source face position after rotation
    # We need to place the source object such that its face is at tgt_face_world_pos
    # Source center to face vector in local space
    src_center_to_face_local = source_face_normal * src_face_dist_local
    src_center_to_face_world = obj_source.matrix_world.to_3x3() @ src_center_to_face_local
    
    # Target position for source center
    target_src_center_world = tgt_face_world_pos - src_center_to_face_world - (offset * tgt_face_normal_world)
    
    obj_source.location = target_src_center_world

def align_face_to_face_simple(obj_source, obj_target, source_axis, target_axis, offset=0):
    """
    Simplified alignment for axis-aligned primitives.
    source_axis: 'X', 'Y', or 'Z' (positive direction)
    target_axis: 'X', 'Y', or 'Z' (positive direction)
    """
    # Determine normals
    normals = {
        'X': Vector((1,0,0)),
        'Y': Vector((0,1,0)),
        'Z': Vector((0,0,1)),
        '-X': Vector((-1,0,0)),
        '-Y': Vector((0,-1,0)),
        '-Z': Vector((0,0,-1))
    }
    
    src_norm = normals[source_axis]
    tgt_norm = normals[target_axis]
    
    # Align orientation
    # We want src_norm (local) to align with tgt_norm (world)
    # Get current src_norm in world
    src_mat = obj_source.matrix_world
    current_src_norm_world = src_mat.to_3x3() @ src_norm
    
    # Calculate rotation
    rot_quat = current_src_norm_world.rotation_difference(tgt_norm)
    obj_source.rotation_euler = rot_quat.to_euler()
    obj_source.update_from_matrix()
    
    # Calculate positions
    # Get bounding box extents
    src_bbox = obj_source.bound_box
    tgt_bbox = obj_target.bound_box
    
    # Find max extent in the normal direction for both
    def get_max_extent(bbox, normal):
        max_d = -float('inf')
        for corner in bbox:
            v = Vector(corner)
            d = v.dot(normal)
            if d > max_d:
                max_d = d
        return max_d

    src_ext_local = get_max_extent(src_bbox, src_norm)
    tgt_ext_local = get_max_extent(tgt_bbox, tgt_norm)
    
    # Target face world position
    tgt_mat = obj_target.matrix_world
    tgt_face_world_pos = Vector((0,0,0))
    max_d = -float('inf')
    for corner in tgt_bbox:
        w_corner = tgt_mat @ Vector(corner)
        d = w_corner.dot(tgt_norm)
        if d > max_d:
            max_d = d
            tgt_face_world_pos = w_corner
            
    # Source face world position should be tgt_face_world_pos - offset * tgt_norm
    # Source center to face vector in world (after rotation)
    src_center_to_face_world = obj_source.matrix_world.to_3x3() @ (src_norm * src_ext_local)
    
    target_src_center_world = tgt_face_world_pos - src_center_to_face_world - (offset * tgt_norm)
    obj_source.location = target_src_center_world

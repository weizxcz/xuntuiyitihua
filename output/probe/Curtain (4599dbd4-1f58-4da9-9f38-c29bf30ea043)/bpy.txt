import bpy
import bmesh
import math
from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(radius, depth, name, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder oriented along Z by default."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(size, name, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube centered at location."""
    # size is total width/height/depth
    half = size / 2.0
    bpy.ops.mesh.primitive_cube_add(size=half, location=location) # size arg in 3.2+ is half-width
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def add_plane(size_x, size_y, name, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a plane."""
    bpy.ops.mesh.primitive_plane_add(size=size_x, location=location)
    obj = bpy.context.active_object
    obj.name = name
    # Scale Y to match size_y (size arg is half-width)
    obj.scale = (1.0, size_y / size_x, 1.0)
    obj.rotation_euler = rotation
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis_to_align='Z', target_vector=(0,0,1)):
    """Rotate object so its local axis aligns with target vector."""
    # Normalize target
    v = Vector(target_vector).normalized()
    
    # Define source axis
    if axis_to_align == 'X':
        src = Vector((1, 0, 0))
    elif axis_to_align == 'Y':
        src = Vector((0, 1, 0))
    else: # Z
        src = Vector((0, 0, 1))
        
    # Calculate rotation
    rot_quat = src.rotation_difference(v)
    obj.rotation_euler = rot_quat.to_euler()

def set_material(obj, mat):
    if mat:
        obj.data.materials.append(mat)

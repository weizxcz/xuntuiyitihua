def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    return mat

def add_cylinder(radius, depth, name, fill_type='NGON'):
    """Add a cylinder aligned with Z axis."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, fill_type=fill_type, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_disc(radius, name):
    """Add a disc aligned with Z axis."""
    bpy.ops.mesh.primitive_circle_add(radius=radius, fill_type='NGON', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(size, name):
    """Add a cube centered at origin."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(radius_bottom, radius_top, depth, name):
    """Add a cone aligned with Z axis."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(radius, name):
    """Add a UV sphere centered at origin."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_torus(major_radius, minor_radius, angle_deg, name):
    """Add a torus segment aligned with Z axis (default 360)."""
    angle_rad = math.radians(angle_deg)
    bpy.ops.mesh.primitive_torus_add(major_radius=major_radius, minor_radius=minor_radius, major_segments=64, minor_segments=32, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    # Torus default is 360 degrees. We need to cut it.
    # Simplest way in script is to add full torus and use a boolean cutter or mesh cut.
    # However, for simplicity and robustness in this generator, we will add a full torus
    # and if angle < 360, we will use a boolean cutter to slice it.
    if angle_deg < 360:
        cutter_name = f"{name}_torus_cutter"
        # Create a plane cutter to slice the torus
        bpy.ops.mesh.primitive_plane_add(size=1.0, location=(0, 0, 0))
        cutter = bpy.context.active_object
        cutter.name = cutter_name
        cutter.scale = (1.0, 1.0, 1.0)
        
        # Orient cutter to slice along Y axis (keeping XZ plane)
        cutter.rotation_euler = (0, 0, 0) 
        
        # We need to position the cutter to keep the desired angle.
        # Default torus starts at X axis (0 rad) and goes counter-clockwise.
        # To keep 0 to angle_deg, we cut at angle_deg.
        # The cut plane should be at angle angle_deg/2 + 90? 
        # Actually, let's just use a boolean modifier for simplicity if needed, 
        # but for a single script without complex boolean trees, 
        # it's often easier to just add the full torus and note it, 
        # OR use a mesh cut operator.
        # Let's use a boolean modifier approach for robustness.
        
        # Add boolean modifier to Torus
        mod = obj.modifiers.new(name="Cut", type='BOOLEAN')
        mod.object = cutter
        mod.operation = 'DIFFERENCE'
        
        # Position cutter to slice off the unwanted part.
        # We want to keep 0 to angle_deg.
        # Cut plane normal should be at angle_deg/2 + 90 degrees?
        # Let's place the cutter plane at the bisector of the unwanted region.
        # Unwanted region is angle_deg to 360. Bisector is angle_deg + (360-angle_deg)/2.
        cut_angle = math.radians(angle_deg + (360 - angle_deg) / 2)
        dist = major_radius + minor_radius + 0.1
        cutter.location = (math.cos(cut_angle) * dist, math.sin(cut_angle) * dist, 0)
        # Rotate cutter to face center
        cutter.rotation_euler = (0, 0, cut_angle - math.pi/2)
        
        # Apply modifier
        bpy.context.view_layer.objects.active = obj
        bpy.ops.object.modifier_apply(modifier=mod.name)
        
        # Remove cutter
        bpy.data.objects.remove(cutter)
        
    return obj

def align_axis_to_vector(obj, axis, vector):
    """Align object's local axis to a world vector."""
    # Current axis vectors
    if axis == 'X':
        current = Vector((1, 0, 0))
    elif axis == 'Y':
        current = Vector((0, 1, 0))
    elif axis == 'Z':
        current = Vector((0, 0, 1))
    else:
        return

    target = vector.normalized()
    
    # Calculate rotation
    rot_quat = current.rotation_difference(target)
    obj.rotation_euler = rot_quat.to_euler()

def get_face_center(obj, face_type='center'):
    """Get the world location of a specific face or center."""
    obj.update_from_editmode() # Ensure mesh is updated
    matrix_world = obj.matrix_world
    
    if face_type == 'center':
        return matrix_world.translation
    
    # We need to find the specific face. 
    # Since we just created primitives, we can assume standard orientations.
    # However, after rotation, faces move.
    # Robust way: iterate faces, find normal closest to world axis, then project.
    
    # Helper to get face center in world space
    def get_face_world_center(face):
        bm = bmesh.new()
        bm.from_mesh(obj.data)
        bmesh.ops.transform(bm, matrix=matrix_world, verts=bm.verts)
        # Calculate center
        co = Vector((0,0,0))
        for v in face.verts:
            co += v.co
        co /= len(face.verts)
        bm.free()
        return co

    # Determine target normal direction based on face_type
    # 'top_face' usually means +Z local face
    # 'bottom_face' usually means -Z local face
    # 'right_face' usually means +X local face
    # 'left_face' usually means -X local face
    
    # We need to find the face whose normal is closest to the rotated axis.
    # But obj.rotation_euler is applied.
    # Let's transform the target axis to local space or vice versa.
    
    # Easier: Transform world axes to local space using inverse matrix
    inv_mat = matrix_world.inverted()
    
    if face_type == 'top_face':
        target_local = inv_mat @ Vector((0, 0, 1))
    elif face_type == 'bottom_face':
        target_local = inv_mat @ Vector((0, 0, -1))
    elif face_type == 'right_face':
        target_local = inv_mat @ Vector((1, 0, 0))
    elif face_type == 'left_face':
        target_local = inv_mat @ Vector((-1, 0, 0))
    else:
        return matrix_world.translation

    # Find face with normal closest to target_local
    best_face = None
    best_dot = -1
    
    # We need a bmesh from the mesh
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    bmesh.ops.transform(bm, matrix=matrix_world, verts=bm.verts) # Not needed for normal check if we use local normals
    
    # Check local normals
    for face in bm.faces:
        # Local normal
        local_normal = face.normal
        dot = local_normal.dot(target_local)
        if dot > best_dot:
            best_dot = dot
            best_face = face
            
    bm.free()
    
    if best_face:
        # Calculate world center of this face
        co = Vector((0,0,0))
        for v in best_face.verts:
            co += v.co
        co /= len(best_face.verts)
        return matrix_world @ co # Wait, v.co is already in world space if we transformed? 
        # No, bmesh.from_mesh gives local coords. 
        # Let's recalculate properly.
        
        # Recalculate world center
        world_co = Vector((0,0,0))
        for v in best_face.verts:
            world_co += matrix_world @ v.co
        world_co /= len(best_face.verts)
        return world_co
        
    return matrix_world.translation

def boolean_subtract(cutter, target):
    """Perform boolean subtract using modifier."""
    # Ensure target is active
    bpy.context.view_layer.objects.active = target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.object = cutter
    mod.operation = 'DIFFERENCE'
    bpy.ops.object.modifier_apply(modifier=mod.name)
    # Remove cutter
    if cutter.name in bpy.data.objects:
        bpy.data.objects.remove(cutter)

def boolean_union(union_obj, target):
    """Perform boolean union."""
    bpy.context.view_layer.objects.active = target
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.object = union_obj
    mod.operation = 'UNION'
    bpy.ops.object.modifier_apply(modifier=mod.name)
    if union_obj.name in bpy.data.objects:
        bpy.data.objects.remove(union_obj)

def align_to_face(obj, target_obj, face_type):
    """Align obj's center to target_obj's face center."""
    target_center = get_face_center(target_obj, face_type)
    obj_center = obj.matrix_world.translation
    
    # Move obj so its center matches target face center
    delta = target_center - obj_center
    obj.location += delta

def offset_object(obj, offset_vec):
    """Apply offset to object location."""
    obj.location += offset_vec

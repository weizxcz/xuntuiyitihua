def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    return obj

def boolean_union(objects_to_union, result_name):
    """
    Performs a boolean union on a list of objects.
    Returns the resulting object.
    """
    if not objects_to_union:
        return None
    
    # Select all objects
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects_to_union:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj

    # Apply modifiers to first object
    target = objects_to_union[0]
    
    # We need to apply the boolean modifier to the target for each other object
    # However, standard boolean modifier chain is tricky in script without clearing history.
    # Better approach: Use BMesh or simple mesh join if no complex topology changes expected.
    # But requirement says "Boolean-union". Let's use the modifier approach carefully.
    
    # To ensure clean union, we'll apply the union of all into the first object
    # and delete the rest.
    
    # Note: Applying boolean modifiers sequentially can be slow and error-prone.
    # For static geometry, mesh joining is often preferred, but we follow instructions.
    
    # Let's use a simpler approach for robustness in script:
    # Join all meshes, then use a single boolean modifier if needed? 
    # No, let's just join them. Since they are simple boxes, joining is topologically equivalent to union 
    # if they don't intersect in complex ways. 
    # BUT, to strictly follow "Boolean-union", we will use the modifier.
    
    # Actually, for a script generator, applying modifiers to multiple objects in a loop is complex.
    # A more reliable way for "discard intermediate parts" is to join them into one mesh.
    # If the prompt implies visual union, joining is fine. If it implies boolean logic for further cuts, we need modifiers.
    # Given the complexity of chaining boolean modifiers in a single script run without user intervention,
    # and the fact that these are simple boxes, I will use Mesh Joining for the "Union" steps 
    # as it is the standard procedural way to combine primitives in Blender scripts unless specific boolean topology is required.
    # However, the prompt explicitly says "Boolean-union". 
    # Let's try to implement a clean boolean union using modifiers.
    
    # Strategy:
    # 1. Make a copy of the first object to act as the base.
    # 2. Add boolean modifier to base for each other object.
    # 3. Apply all modifiers.
    # 4. Delete originals.
    
    base = target.copy()
    base.data = target.data.copy()
    base.name = result_name
    base.location = target.location
    base.rotation_euler = target.rotation_euler
    base.scale = target.scale
    
    # Link to scene
    bpy.context.collection.objects.link(base)
    
    # Add boolean modifiers
    for other in objects_to_union[1:]:
        mod = base.modifiers.new(name="BoolUnion", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = other
    
    # Apply modifiers
    bpy.context.view_layer.objects.active = base
    bpy.ops.object.modifier_apply(modifier=base.modifiers[0].name) # Apply first
    # Re-select base to apply remaining
    bpy.context.view_layer.objects.active = base
    for mod in base.modifiers:
        bpy.ops.object.modifier_apply(modifier=mod.name)
        
    # Delete original objects
    for obj in objects_to_union:
        bpy.data.objects.remove(obj, do_unlink=True)
        
    return base

def boolean_union_simple_join(objects_to_union, result_name):
    """
    Alternative: Join meshes. This is faster and more stable for simple primitives.
    """
    if not objects_to_union:
        return None
        
    # Select all
    bpy.ops.object.select_all(action='DESELECT')
    for obj in objects_to_union:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = objects_to_union[0]
        
    bpy.ops.object.join()
    result = bpy.context.active_object
    result.name = result_name
    return result

# Using the simple join for stability in this specific context of box primitives, 
# as chaining boolean modifiers in a script can be fragile without clearing history properly.
# If strict boolean topology is needed, the modifier approach above is better, but join is safer for "discard intermediate".
# I will use the modifier approach for correctness to the prompt "Boolean-union".

def boolean_union_strict(objects_to_union, result_name):
    if not objects_to_union:
        return None
    
    # Clone the first object to be the result
    src = objects_to_union[0]
    res = src.copy()
    res.data = src.data.copy()
    res.name = result_name
    bpy.context.collection.objects.link(res)
    
    # Add boolean modifiers for all other objects
    for other in objects_to_union[1:]:
        mod = res.modifiers.new(name="Bool", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = other
        
    # Apply all modifiers
    bpy.context.view_layer.objects.active = res
    for mod in res.modifiers:
        bpy.ops.object.modifier_apply(modifier=mod.name)
        
    # Remove originals
    for obj in objects_to_union:
        bpy.data.objects.remove(obj, do_unlink=True)
        
    return res

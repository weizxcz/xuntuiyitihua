# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.data.update()
    return obj

def add_cylinder(name, radius, depth, location):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def apply_material(obj, mat):
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def get_face_normal(obj, face_index):
    """Returns the world-space normal of a specific face."""
    obj.update_from_editmode() # Ensure data is current
    mat_world = obj.matrix_world
    # Get the face in local space
    # Note: For primitives, face indices are predictable but let's use a robust method
    # For a cylinder, top face is usually the last one or specific index.
    # A safer generic approach for "top face" of a Z-aligned primitive:
    # We will assume standard orientation helpers handle the "top face" logic via locator.
    return None

class Locator:
    def __init__(self, obj, face_type='center'):
        self.obj = obj
        self.face_type = face_type
        self.update()

    def update(self):
        self.obj.update_from_editmode()
        self.center = self.obj.matrix_world.to_translation()
        
        # Determine face center/normal based on type
        # For simplicity with primitives aligned to Z:
        # Top face center is center + (0,0, height/2)
        # Bottom face center is center - (0,0, height/2)
        
        # Get dimensions
        if self.obj.type == 'MESH':
            # Approximate dimensions from bounds
            bbox = self.obj.bound_box
            # Calculate local extents
            min_x = min(v[0] for v in bbox)
            max_x = max(v[0] for v in bbox)
            min_y = min(v[1] for v in bbox)
            max_y = max(v[1] for v in bbox)
            min_z = min(v[2] for v in bbox)
            max_z = max(v[2] for v in bbox)
            
            local_height = max_z - min_z
            local_center = Vector([(min_x+max_x)/2, (min_y+max_y)/2, (min_z+max_z)/2])
            
            # Transform to world
            self.face_center_world = self.obj.matrix_world @ (local_center + Vector((0, 0, local_height/2)))
            self.face_normal_world = self.obj.matrix_world @ Vector((0, 0, 1)) # Assuming Z-up primitive
        else:
            self.face_center_world = self.center
            self.face_normal_world = Vector((0,0,1))

    def get_offset_vector(self, target_locator):
        """Calculates offset to snap this locator's face to target_locator's face."""
        # Vector from this face center to target face center
        return target_locator.face_center_world - self.face_center_world

def place_object(obj, anchor_obj, anchor_face_type, align_face_type, offset_xyz):
    """
    1. Anchor obj to anchor_obj's face.
    2. Align obj's face to anchor_obj's face (orientation).
    3. Apply offset.
    """
    # Create locator for the anchor object's face
    anchor_loc = Locator(anchor_obj, face_type=anchor_face_type)
    
    # Create locator for the new object (initially at origin or created location)
    # We need to place the object such that its face is at the anchor's face.
    
    # Step 1: Position object so its face is at the anchor's face center
    # We need to know the object's local face offset.
    # For a Z-aligned cylinder/cube, the "top" face is at +Z half-height.
    
    # Get object dimensions
    bbox = obj.bound_box
    min_z = min(v[2] for v in bbox)
    max_z = max(v[2] for v in bbox)
    local_face_offset_z = max_z # Distance from center to top face
    
    # Target position for object center
    target_center = anchor_loc.face_center_world - (obj.matrix_world @ Vector((0,0,local_face_offset_z)))
    
    # Apply position
    obj.location = target_center
    
    # Step 2: Align orientation (already Z-aligned by creation, but ensure)
    # Align obj's +Z to anchor's +Z (which is world Z)
    # No rotation needed if both are Z-up.
    
    # Step 3: Apply offset
    offset_vec = Vector(offset_xyz)
    obj.location += offset_vec
    
    return obj

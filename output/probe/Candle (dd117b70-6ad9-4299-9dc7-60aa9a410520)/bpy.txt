def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size=1.0):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_bottom, radius_top, depth):

    obj = bpy.context.active_object
    obj.name = name
    return obj

def align_axis_to_vector(obj, axis='Z', vector=Vector((0, 0, 1))):
    """Align an object's local axis to a world vector."""
    axis_map = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    local_axis = axis_map.get(axis, Vector((0, 0, 1)))
    
    # Calculate rotation to align local_axis to target vector
    rot_quat = local_axis.rotation_difference(vector)
    obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find specific points on an object's geometry."""
    def __init__(self, obj):
        self.obj = obj
        self.mesh = obj.data
        self.verts = self.mesh.vertices
        self.has_calc = False
        self._calc_bounds()

    def _calc_bounds(self):
        """Calculate bounding box and centers."""
        if not self.verts:
            self.min = Vector((0,0,0))
            self.max = Vector((0,0,0))
            self.center = Vector((0,0,0))
            return

        # Transform vertices to world space
        world_verts = [self.obj.matrix_world @ v.co for v in self.verts]
        
        self.min = Vector((min(v.x for v in world_verts), min(v.y for v in world_verts), min(v.z for v in world_verts)))
        self.max = Vector((max(v.x for v in world_verts), max(v.y for v in world_verts), max(v.z for v in world_verts)))
        self.center = (self.min + self.max) / 2.0

    def center_world(self):
        """Return the world-space center of the object's bounding box."""
        return self.center

    def face_center_world(self, face_type='TOP'):
        """Return the world-space center of a specific face (TOP, BOTTOM, FRONT, BACK, LEFT, RIGHT)."""
        if face_type == 'TOP':
            return Vector((self.center.x, self.center.y, self.max.z))
        elif face_type == 'BOTTOM':
            return Vector((self.center.x, self.center.y, self.min.z))
        elif face_type == 'FRONT':
            return Vector((self.center.x, self.max.y, self.center.z))
        elif face_type == 'BACK':
            return Vector((self.center.x, self.min.y, self.center.z))
        elif face_type == 'RIGHT':
            return Vector((self.max.x, self.center.y, self.center.z))
        elif face_type == 'LEFT':
            return Vector((self.min.x, self.center.y, self.center.z))
        return self.center

    def radial_point_world(self, direction='X', height_ratio=0.5):
        """Return a point on the radial surface."""
        if direction == 'X':
            return Vector((self.max.x, self.center.y, self.center.z))
        elif direction == '-X':
            return Vector((self.min.x, self.center.y, self.center.z))
        elif direction == 'Y':
            return Vector((self.center.x, self.max.y, self.center.z))
        elif direction == '-Y':
            return Vector((self.center.x, self.min.y, self.center.z))
        return self.center

def boolean_subtract(cutter_obj, target_obj):
    """Perform a boolean subtract operation using the cutter on the target."""
    # Ensure target is active for modifier application if needed,

    # Create a new object for the boolean result to keep history clean or just apply
    # Here we apply the modifier directly to the target for simplicity in script generation
    
    # Check if modifier already exists
    existing_mod = None
    for mod in target_obj.modifiers:
        if mod.type == 'BOOLEAN' and mod.object == cutter_obj:
            existing_mod = mod
            break
            
    if not existing_mod:
        mod = target_obj.modifiers.new(name="BooleanSubtract", type='BOOLEAN')
        mod.operation = 'SUBTRACT'
        mod.object = cutter_obj
    else:
        mod = existing_mod

    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Optional: Delete cutter after use to clean up

    return target_obj

def align_face_to_face(obj, anchor_obj, obj_face='BOTTOM', anchor_face='TOP'):
    """Align a face of obj to a face of anchor_obj."""
    loc_obj = Locator(obj)
    loc_anchor = Locator(anchor_obj)
    
    # Get world positions of the faces
    obj_face_pos = loc_obj.face_center_world(obj_face)
    anchor_face_pos = loc_anchor.face_center_world(anchor_face)
    
    # Calculate offset required to move obj_face_pos to anchor_face_pos
    # We want obj_face_pos + offset = anchor_face_pos
    offset = anchor_face_pos - obj_face_pos
    
    obj.location += offset

def align_center_to_center(obj, anchor_obj):
    """Align center of obj to center of anchor_obj."""
    loc_obj = Locator(obj)
    loc_anchor = Locator(anchor_obj)
    
    obj_center = loc_obj.center_world()
    anchor_center = loc_anchor.center_world()
    
    offset = anchor_center - obj_center
    obj.location += offset

def apply_offset(obj, offset_vector):
    """Apply an offset vector to the object's location."""
    obj.location += offset_vector

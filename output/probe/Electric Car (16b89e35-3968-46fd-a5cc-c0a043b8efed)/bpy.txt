def make_material(name, diffuse_color):
    """Create a Principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if not obj.data.materials:
        obj.data.materials.append(mat)
    else:
        obj.data.materials[0] = mat

def add_primitive(name, shape, size, location=(0,0,0), rotation=(0,0,0)):
    """Add a primitive mesh object."""
    bpy.ops.object.select_all(action='DESELECT')
    if shape == 'cube':
        bpy.ops.mesh.primitive_cube_add(size=size[0], location=location)
        obj = bpy.context.active_object
        obj.name = name
        obj.scale = (size[1]/size[0], size[2]/size[0], 1) # Adjust for cube default size 2
        # Actually, let's just set dimensions directly after creation for precision
        obj.dimensions = size
    elif shape == 'cylinder':
        # Cylinder default: radius=size[0]/2, depth=size[2]
        # We want diameter=size[0], length=size[2]
        bpy.ops.mesh.primitive_cylinder_add(radius=size[0]/2, depth=size[2], location=location)
        obj = bpy.context.active_object
        obj.name = name
    elif shape == 'disc':
        bpy.ops.mesh.primitive_circle_add(radius=size[0]/2, fill_type='NGON', location=location)
        obj = bpy.context.active_object
        obj.name = name
        obj.data.extrude = size[2]
        bpy.ops.mesh.extrude_faces_move()
    elif shape == 'cone':
        bpy.ops.mesh.primitive_cone_add(radius1=size[0]/2, radius2=0, depth=size[2], location=location)
        obj = bpy.context.active_object
        obj.name = name
    elif shape == 'sphere':
        bpy.ops.mesh.primitive_uv_sphere_add(radius=size[0]/2, location=location)
        obj = bpy.context.active_object
        obj.name = name
    
    # Ensure rotation is applied if needed, though we usually rotate after
    obj.rotation_euler = rotation
    return obj

def align_face_to_face(obj, target, axis='Z', face_type='positive'):
    """
    Aligns a face of 'obj' to the corresponding face of 'target'.
    axis: 'X', 'Y', or 'Z'
    face_type: 'positive' (max) or 'negative' (min)
    """
    # Get world transforms
    obj_mat = obj.matrix_world
    tgt_mat = target.matrix_world
    
    # Get bounding box centers and extents in world space
    # We need to align the specific face.
    # Simple approach: Move object so its face center matches target face center along the normal.
    
    # Local face centers
    # Positive Z face center in local: (0, 0, 0.5 * height)
    # Negative Z face center in local: (0, 0, -0.5 * height)
    
    # Determine local offset for the face
    local_offset = Vector((0,0,0))
    if axis == 'X':
        local_offset.x = 0.5 if face_type == 'positive' else -0.5
    elif axis == 'Y':
        local_offset.y = 0.5 if face_type == 'positive' else -0.5
    elif axis == 'Z':
        local_offset.z = 0.5 if face_type == 'positive' else -0.5
        
    # Scale by dimensions
    local_offset *= obj.dimensions
    
    # World position of obj's face center
    obj_face_center = obj_mat @ local_offset
    
    # Target face center
    tgt_local_offset = Vector((0,0,0))
    if axis == 'X':
        tgt_local_offset.x = 0.5 if face_type == 'positive' else -0.5
    elif axis == 'Y':
        tgt_local_offset.y = 0.5 if face_type == 'positive' else -0.5
    elif axis == 'Z':
        tgt_local_offset.z = 0.5 if face_type == 'positive' else -0.5
        
    tgt_local_offset *= target.dimensions
    tgt_face_center = tgt_mat @ tgt_local_offset
    
    # Calculate translation vector
    translation = tgt_face_center - obj_face_center
    
    # Apply translation
    obj.location += translation

def align_axis_to_vector(obj, axis='Z', vector=Vector((0,0,1))):
    """Rotate object so its local axis aligns with the given world vector."""
    if axis == 'X':
        local_axis = Vector((1,0,0))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    else:
        local_axis = Vector((0,0,1))
        
    # Calculate rotation matrix
    rot_mat = local_axis.rotation_difference(vector).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_euler = rot_mat.to_euler()

def offset_object(obj, offset_vec):
    """Apply offset in object's local space."""
    # Convert local offset to world space based on current rotation
    world_offset = obj.matrix_world.to_3x3() @ offset_vec
    obj.location += world_offset

def boolean_subtract(base_obj, cutter_obj):
    """Perform boolean subtract using modifier."""
    # Add Boolean Modifier
    mod = base_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    # Apply modifier
    bpy.context.view_layer.objects.active = base_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

def boolean_union(base_obj, cutter_obj):
    """Perform boolean union using modifier."""

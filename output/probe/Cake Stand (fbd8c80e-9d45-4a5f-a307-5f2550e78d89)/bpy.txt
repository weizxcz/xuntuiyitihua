import bpy
import bmesh
import mathutils
from mathutils import Vector, Matrix

def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_color = diffuse_color
    return mat

def add_cylinder(radius, depth, segments=32, name="Cylinder"):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, end_fill_type='NGON')
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def add_cube(size, name="Cube"):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size)
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def add_cone(radius_top, radius_bottom, depth, segments=32, name="Cone"):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_top, radius2=radius_bottom, depth=depth, segments=segments)
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def add_uvsphere(radius, width_segments=16, height_segments=16, name="UVSphere"):
    """Add a UV sphere mesh object."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments)
    obj = bpy.context.active_object
    obj.name = name
    obj.location = (0, 0, 0)
    return obj

def add_bevel(obj, width=0.01, segments=5, profile=0.5, name="Bevel"):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name=name, type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.profile = profile
    mod.limit_method = 'ANGLE'
    return mod

def boolean_subtract(cutter_obj, target_obj, name="BooleanSubtract"):
    """Perform a boolean subtract operation using a cutter object."""
    # Ensure target is active for the modifier addition if needed, 
    # but usually we add modifier to target and set cutter as object.
    mod = target_obj.modifiers.new(name=name, type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    # Apply immediately to keep scene clean and transform-independent
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    # Remove cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)
    return target_obj

def boolean_union(union_obj, target_obj, name="BooleanUnion"):
    """Perform a boolean union operation."""
    mod = target_obj.modifiers.new(name=name, type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(union_obj, do_unlink=True)
    return target_obj

def align_axis_to_vector(obj, axis_to_align='Z', vector_to_align=(0, 0, 1)):
    """Rotate object so its local axis aligns with the given world vector."""
    # Get current rotation
    rot_euler = obj.rotation_euler
    # We need to calculate the rotation that aligns the object's local axis to the target vector.
    # For simplicity in standard primitives, we often just set rotation.
    # However, to be robust:
    
    # Define local axis vectors
    local_axes = {
        'X': Vector((1, 0, 0)),
        'Y': Vector((0, 1, 0)),
        'Z': Vector((0, 0, 1))
    }
    
    target_vec = Vector(vector_to_align).normalized()
    local_vec = local_axes[axis_to_align]
    
    # Calculate rotation matrix
    rot_mat = local_vec.rotation_difference(target_vec).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = local_vec.rotation_difference(target_vec)

def average(points):
    """Calculate the average point of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to create invisible locator objects for alignment."""
    def __init__(self, name="Locator"):
        self.name = name
        # Create an empty object
        bpy.ops.object.empty_add(type='PLAIN_AXES', align='WORLD', location=(0, 0, 0))
        self.obj = bpy.context.active_object
        self.obj.name = name
        self.obj.empty_display_size = 0.1

    def center_world(self, points):
        """Position locator at the average of the given points."""
        loc = average(points)
        self.obj.location = loc
        return self.obj

    def face_center_world(self, obj):
        """Position locator at the center of an object's face (assuming Z-up cylinder/cube)."""
        # For a cylinder or cube, the "top face" center is at local Z = height/2
        # We need world coordinates.
        # Assuming standard orientation where +Z is up.
        if obj.type == 'MESH':
            # Get bounding box to find top face center
            # Transform vertices to world space
            verts = [obj.matrix_world @ v.co for v in obj.data.vertices]
            # Find max Z
            max_z = max(v.z for v in verts)
            # Find center of vertices at max_z
            top_verts = [v for v in verts if abs(v.z - max_z) < 1e-5]
            center = average(top_verts)
            self.obj.location = center
        return self.obj

    def radial_point_world(self, obj, angle=0):
        """Position locator at a radial point on the object's circumference."""
        if obj.type == 'MESH':
            # Assume cylinder-like
            verts = [obj.matrix_world @ v.co for v in obj.data.vertices]
            # Find bottom center and radius
            min_z = min(v.z for v in verts)
            bottom_verts = [v for v in verts if abs(v.z - min_z) < 1e-5]
            if bottom_verts:
                center_bottom = average(bottom_verts)
                # Calculate radius
                if len(bottom_verts) > 1:
                    # Average distance from center
                    dists = [((v - center_bottom).xy).length for v in bottom_verts]
                    radius = sum(dists) / len(dists)
                else:
                    radius = 0
                # Calculate radial point
                x = radius * math.cos(angle)
                y = radius * math.sin(angle)
                loc = Vector((center_bottom.x + x, center_bottom.y + y, center_bottom.z))
                self.obj.location = loc
        return self.obj

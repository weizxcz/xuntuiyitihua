import bpy

from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.rotation_euler = rotation_euler
    return obj

def apply_transform(obj):
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.transform_apply(location=True, rotation=True, scale=True)

def get_world_center(obj):
    # Transform the local origin (0,0,0) by the object's world matrix
    return obj.matrix_world.to_translation()

def boolean_subtract(cutter_obj, target_obj):
    # Ensure target is active for modifier application if needed, 
    # but standard approach: add modifier to target, apply.
    bpy.context.view_layer.objects.active = target_obj
    
    # Add Boolean Modifier
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply Modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Cleanup cutter
    bpy.data.objects.remove(cutter_obj, do_unlink=True)

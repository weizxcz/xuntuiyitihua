def clear_scene():
    """Delete all existing objects in the scene."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Ensure collections are clean
    for col in bpy.data.collections:
        col.objects.link(bpy.context.collection)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size, location, rotation_euler, material=None):
    """Add a cube object with specific size and transform."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0]/2, size[1]/2, size[2]/2)
    obj.rotation_euler = rotation_euler
    if material:
        obj.data.materials.append(material)
    return obj

def add_cylinder(name, radius, depth, location, rotation_euler, material=None):

def get_world_center(obj):
    """Get the world location of an object's center."""
    return obj.matrix_world.to_translation()

def get_face_center(obj, face_type='top'):
    """
    Get the world location of a specific face center based on object orientation.
    Assumes standard axis alignment for simplicity unless complex logic is needed.
    """
    loc = get_world_center(obj)
    # Dimensions
    sx, sy, sz = obj.dimensions
    
    # Offsets based on face type relative to object center
    # Note: This assumes the object's local axes align with world axes or are handled by rotation.
    # For robustness, we calculate offset in local space then transform.
    
    offset_local = Vector((0, 0, 0))
    
    if face_type == 'top_face':
        offset_local.z = sz / 2
    elif face_type == 'bottom_face':
        offset_local.z = -sz / 2
    elif face_type == 'front_face':
        # Assuming +Y is front for standard Blender cube if rotated 0,0,0
        # However, standard convention often varies. Let's assume +Y is front.
        offset_local.y = sy / 2
    elif face_type == 'back_face':
        offset_local.y = -sy / 2
    elif face_type == 'right_face':
        offset_local.x = sx / 2
    elif face_type == 'left_face':
        offset_local.x = -sx / 2
    elif face_type == 'center':
        offset_local = Vector((0, 0, 0))
        
    # Transform local offset to world space
    offset_world = obj.matrix_world * offset_local - loc
    return loc + offset_world

def align_object_to_vector(obj, vector):
    """Rotate object so its local +Z aligns with the given world vector."""
    if vector.length == 0:
        return
    
    # Current up vector
    current_up = Vector((0, 0, 1))
    
    # Calculate rotation
    rot_quat = current_up.rotation_difference(vector)
    
    # Apply rotation
    obj.rotation_euler = rot_quat.to_euler()

def anchor_and_offset(obj_target, obj_source, anchor_type, offset_vec):
    """
    Anchor obj_source to obj_target based on anchor_type, then apply offset.
    anchor_type: 'center', 'top_face', 'bottom_face', 'front_face', etc.
    offset_vec: Vector to add after anchoring.
    """
    target_loc = get_world_center(obj_target)
    
    if anchor_type == 'center':
        anchor_loc = target_loc
    else:
        anchor_loc = get_face_center(obj_target, anchor_type)
        
    # Move source to anchor
    obj_source.location = anchor_loc
    
    # Apply offset
    obj_source.location += offset_vec

def boolean_subtract(cutter_obj, target_obj):
    """Perform boolean subtract using modifier stack."""
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_union(union_obj, target_obj):
    """Perform boolean union using modifier stack."""
    # Add Boolean Modifier to Target
    mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

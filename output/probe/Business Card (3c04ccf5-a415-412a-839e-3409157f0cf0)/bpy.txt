def make_material(name, diffuse_color=(1.0, 1.0, 1.0, 1.0)):
    """Create a principled BSDF material."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cube(name, size=(1,1,1), location=(0,0,0), rotation=(0,0,0)):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (size[0], size[1], size[2])
    obj.rotation_euler = rotation
    return obj

def add_cylinder(name, radius=0.5, depth=1, location=(0,0,0), rotation=(0,0,0)):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_top=0, radius_bottom=0.5, depth=1, location=(0,0,0), rotation=(0,0,0)):

    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object

    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align an object's local axis to a world vector."""
    if axis == 'Z':
        # Align Z to vector
        z_vec = obj.matrix_world.to_3x3().col[2]
        if z_vec.dot(vector) < 0:
            vector = -vector
        # Simple rotation to align Z with vector (assuming vector is normalized)
        rot_mat = z_vec.rotation_difference(vector).to_matrix().to_4x4()
        obj.matrix_world = rot_mat @ obj.matrix_world
    elif axis == 'X':
        x_vec = obj.matrix_world.to_3x3().col[0]
        if x_vec.dot(vector) < 0:
            vector = -vector
        rot_mat = x_vec.rotation_difference(vector).to_matrix().to_4x4()
        obj.matrix_world = rot_mat @ obj.matrix_world
    elif axis == 'Y':
        y_vec = obj.matrix_world.to_3x3().col[1]
        if y_vec.dot(vector) < 0:
            vector = -vector
        rot_mat = y_vec.rotation_difference(vector).to_matrix().to_4x4()
        obj.matrix_world = rot_mat @ obj.matrix_world

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using modifier stack."""
    target = bpy.data.objects[target_name]
    cutter = bpy.data.objects[cutter_name]
    
    # Ensure target has no existing boolean modifier named 'Boolean'
    for mod in target.modifiers:
        if mod.name == 'Boolean':
            target.modifiers.remove(mod)
            
    bool_mod = target.modifiers.new(name='Boolean', type='BOOLEAN')
    bool_mod.object = cutter
    bool_mod.operation = 'DIFFERENCE'
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=bool_mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_name, target_name):
    """Perform boolean union using modifier stack."""
    target = bpy.data.objects[target_name]
    union_obj = bpy.data.objects[union_name]
    
    for mod in target.modifiers:
        if mod.name == 'Boolean':
            target.modifiers.remove(mod)
            
    bool_mod = target.modifiers.new(name='Boolean', type='BOOLEAN')
    bool_mod.object = union_obj
    bool_mod.operation = 'UNION'
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=bool_mod.name)
    
    bpy.data.objects.remove(union_obj, do_unlink=True)

def add_bevel(obj_name, width=0.01, segments=5):
    """Add a bevel modifier to an object."""
    obj = bpy.data.objects[obj_name]
    mod = obj.modifiers.new(name='Bevel', type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    mod.angle_limit = 0.0 # Default, can be adjusted
    # Apply immediately for simplicity in this script context
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

        self.obj = obj
        
    def center_world(self):
        return self.obj.location

        # Approximate face center by averaging vertices of the face
        # Note: This requires mesh data to be evaluated
        mesh = self.obj.data
        bm = bmesh.new()
        bm.from_mesh(mesh)
        bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
        
        if face_index < len(bm.faces):
            face = bm.faces[face_index]

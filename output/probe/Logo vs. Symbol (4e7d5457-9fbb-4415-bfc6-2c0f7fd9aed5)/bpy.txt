import bpy
import bmesh

def clear_scene():
    """Delete all existing objects and collections."""
    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.object.select_all(action='ALL')
    bpy.ops.object.delete(use_global=False)
    # Clear remaining collections if any (mostly for safety)
    for col in bpy.data.collections:
        col.objects.unlink_all()
        bpy.data.collections.remove(col)

def set_units():
    """Set scene units to metres."""
    bpy.context.scene.unit_settings.system = 'METRIC'
    bpy.context.scene.unit_settings.scale_length = 1.0

def make_material(name, diffuse_color):
    """Create a simple Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    bsdf.inputs['Roughness'].default_value = 0.5
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder primitive."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube primitive."""
    # size is (x,y,z)
    x, y, z = size
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.scale = (x, y, z)
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_text_object(name, text, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a text object."""
    bpy.ops.object.text_add(location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    obj.data.body = text
    # Scale text to fit approximate size (text size is in points, roughly)
    # This is a rough approximation; for precise control, font scaling is needed.
    # Assuming size is in meters, and default text is ~0.1m high at scale 1.
    # We'll just set the name and material for now, as text geometry is complex.
    if material:
        obj.data.materials.append(material)
    return obj

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Rotate object so its local axis aligns with the given world vector."""
    # For standard primitives aligned to world, this is often identity if vector is world axis.
    # If we need to align a specific local axis to a world vector:
    if axis == 'Z':
        target = Vector(vector)
        current = Vector((0, 0, 1))
        rot_quat = current.rotation_difference(target)
        obj.rotation_euler = rot_quat.to_euler()
    elif axis == 'X':
        target = Vector(vector)
        current = Vector((1, 0, 0))
        rot_quat = current.rotation_difference(target)
        obj.rotation_euler = rot_quat.to_euler()
    elif axis == 'Y':
        target = Vector(vector)
        current = Vector((0, 1, 0))
        rot_quat = current.rotation_difference(target)
        obj.rotation_euler = rot_quat.to_euler()

def boolean_union(objs_to_union, keep_obj_name):
    """
    Perform boolean union on a list of objects.
    Keeps the object named keep_obj_name.
    """
    if not objs_to_union:
        return
    
    # Select all objects to union
    for obj in objs_to_union:
        obj.select_set(True)
        bpy.context.view_layer.objects.active = obj

    # Apply boolean modifier to the first object (or the one we want to keep)
    # We need to ensure the 'keep' object is the target of the boolean.
    # Strategy: Add boolean modifier to keep_obj_name using others as modifiers.
    
    keep_obj = None
    for obj in objs_to_union:
        if obj.name == keep_obj_name:
            keep_obj = obj
            break
    
    if not keep_obj:
        keep_obj = objs_to_union[0]
        keep_obj_name = keep_obj.name

    # Ensure keep_obj is active
    bpy.context.view_layer.objects.active = keep_obj
    keep_obj.select_set(True)
    
    # Add boolean modifier to keep_obj for each other object
    for obj in objs_to_union:
        if obj == keep_obj:
            continue
        
        # Add boolean modifier
        mod = keep_obj.modifiers.new(name=f"Bool_{obj.name}", type='BOOLEAN')
        mod.operation = 'UNION'
        mod.object = obj
        
    # Apply all modifiers
    for mod in keep_obj.modifiers:
        if mod.type == 'BOOLEAN':
            bpy.ops.object.modifier_apply(modifier=mod.name)
            
    # Deselect and remove the helper objects
    for obj in objs_to_union:
        if obj != keep_obj:
            obj.select_set(True)
    keep_obj.select_set(True)
    bpy.ops.object.delete()
    
    # Return the kept object
    return keep_obj

def boolean_subtract(cutter_obj, target_obj):
    """Subtract cutter_obj from target_obj using boolean."""
    mod = target_obj.modifiers.new(name="Bool_Subtract", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter_obj
    bpy.ops.object.modifier_apply(modifier=mod.name)
    bpy.data.objects.remove(cutter_obj)

def align_to_xy_face(obj, target_xy=(0,0)):
    """
    Align the center of the object's XY face (top face for Z-up cylinder/cube) 
    to the target XY coordinates.
    Assumes object is Z-up.
    """
    # Get current location
    x, y, z = obj.location
    # Set new X, Y
    obj.location.x = target_xy[0]
    obj.location.y = target_xy[1]
    # Z remains unchanged unless specified otherwise. 
    # The prompt says "Align graphic_element.XY_face to (-0.10, 0.02)".
    # This usually means the center of the face in XY plane.

def offset_object(obj, offset=(0,0,0)):
    """Apply offset to object location."""
    obj.location.x += offset[0]
    obj.location.y += offset[1]
    obj.location.z += offset[2]

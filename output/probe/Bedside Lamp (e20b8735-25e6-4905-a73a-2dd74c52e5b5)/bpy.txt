# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius, depth, segments=32):
    """Add a cone mesh object."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius, radius2=0, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def get_face_center(obj, face_type='center'):
    """
    Get the world location of a specific face or center of an object.
    face_type: 'center', 'bottom_face', 'top_face'
    Assumes standard orientation: Z is up.
    """
    # Ensure we are working with the object's data
    me = obj.data
    # Find the face with the normal closest to +Z (top) or -Z (bottom)
    # Note: This is a simplified approach assuming simple primitives.
    # For robustness in generated scripts, we often rely on known geometry properties.
    
    # Calculate bounds to determine top/bottom faces reliably
    # Transform vertices to world space
    verts_world = [obj.matrix_world @ v.co for v in me.vertices]
    
    min_z = min(v[2] for v in verts_world)
    max_z = max(v[2] for v in verts_world)
    
    if face_type == 'center':
        return obj.matrix_world @ Vector((0, 0, 0))
    elif face_type == 'bottom_face':
        # Find a vertex at min_z, project to XY plane for center
        # Assuming symmetry, the center X/Y is the object's X/Y
        return Vector((obj.location.x, obj.location.y, min_z))
    elif face_type == 'top_face':
        return Vector((obj.location.x, obj.location.y, max_z))
        
    return obj.matrix_world @ Vector((0, 0, 0))

def align_and_offset(obj, anchor_obj, anchor_type, offset=(0,0,0)):
    """
    Aligns obj to anchor_obj based on anchor_type, then applies offset.
    anchor_type examples: 'center', 'bottom_face', 'top_face'
    """
    # 1. Orientation: Ensure +Z aligns +Z (already done by creation usually, but explicit here)
    # Reset rotation to identity for clean alignment calculation
    obj.rotation_euler = (0, 0, 0)
    
    # 2. Anchor/Align
    if anchor_type == 'center':
        target_loc = anchor_obj.matrix_world @ Vector((0,0,0))
    elif anchor_type == 'bottom_face':
        target_loc = get_face_center(anchor_obj, 'bottom_face')
    elif anchor_type == 'top_face':
        target_loc = get_face_center(anchor_obj, 'top_face')
    else:
        target_loc = anchor_obj.matrix_world @ Vector((0,0,0))
        
    # Move object so its reference point matches the target
    # obj.location is currently (0,0,0) relative to its origin.
    # We want obj.location such that its reference point is at target_loc.
    # Reference point in local space:
    if anchor_type == 'center':
        ref_local = Vector((0,0,0))
    elif anchor_type == 'bottom_face':
        # For a cylinder/cone created at origin, bottom is at -depth/2
        ref_local = Vector((0, 0, -obj.dimensions.z / 2))
    elif anchor_type == 'top_face':
        ref_local = Vector((0, 0, obj.dimensions.z / 2))
    else:
        ref_local = Vector((0,0,0))
        
    # Calculate new location
    # World_Ref_Point = World_Location + World_Rotation * Local_Ref_Point
    # Assuming no rotation for now (aligned to world Z):
    # World_Ref_Point = World_Location + Local_Ref_Point
    # World_Location = World_Ref_Point - Local_Ref_Point
    
    new_loc = target_loc - ref_local
    
    # 3. Apply Offset
    # Offset is in reference local axes. Since aligned to world, it's world offset.
    final_loc = new_loc + Vector(offset)
    
    obj.location = final_loc
    obj.rotation_euler = (0, 0, 0) # Ensure orientation is maintained

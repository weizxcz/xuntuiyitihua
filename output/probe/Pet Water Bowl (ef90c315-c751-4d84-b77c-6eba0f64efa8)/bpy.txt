import bpy
import bmesh

# --- Helper Functions ---

def make_material(name, diffuse_color=(0.95, 0.95, 0.95, 1.0)):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def apply_material(obj, mat):
    """Assign material to object."""
    if obj.data.materials:
        obj.data.materials[0] = mat
    else:
        obj.data.materials.append(mat)

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder mesh object."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cube(name, size):
    """Add a cube mesh object."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_cone(name, radius_bottom, radius_top, depth, segments=32):
    """Add a cone/tapered cylinder."""
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, segments=segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_uvsphere(name, radius, width_segments=32, height_segments=16):
    """Add a UV sphere."""
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, segments=width_segments, ring_count=height_segments, location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    return obj

def add_bevel(obj, width, segments=5):
    """Add a bevel modifier to an object."""
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    return mod

def align_axis_to_vector(obj, axis='Z', vector=Vector((0, 0, 1))):
    """Rotate object so its local axis aligns with the given world vector."""
    if axis == 'X':
        local_axis = Vector((1, 0, 0))
    elif axis == 'Y':
        local_axis = Vector((0, 1, 0))
    else: # Z
        local_axis = Vector((0, 0, 1))
    
    # Calculate rotation
    rot_quat = local_axis.rotation_difference(vector.normalized())
    obj.rotation_euler = rot_quat.to_euler()

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to find centers and faces of objects."""
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        self.center_world = obj.matrix_world.to_translation()
        
        # Calculate face centers based on bounding box in world space
        # Assuming axis-aligned for simplicity unless rotated, but we use world matrix
        verts_world = [self.obj.matrix_world @ v for v in self.bound_box]
        
        # Find min/max for each axis
        xs = [v.x for v in verts_world]
        ys = [v.y for v in verts_world]
        zs = [v.z for v in verts_world]
        
        self.min_x = min(xs); self.max_x = max(xs)
        self.min_y = min(ys); self.max_y = max(ys)
        self.min_z = min(zs); self.max_z = max(zs)
        
        # Define face centers
        self.center_x_min = Vector((self.min_x, self.center_world.y, self.center_world.z))
        self.center_x_max = Vector((self.max_x, self.center_world.y, self.center_world.z))
        self.center_y_min = Vector((self.center_world.x, self.min_y, self.center_world.z))
        self.center_y_max = Vector((self.center_world.x, self.max_y, self.center_world.z))
        self.center_z_min = Vector((self.center_world.x, self.center_world.y, self.min_z))
        self.center_z_max = Vector((self.center_world.x, self.center_world.y, self.max_z))

    def face_center_world(self, face_type):
        """Return world location of a specific face center."""
        if face_type == 'x_min': return self.center_x_min
        if face_type == 'x_max': return self.center_x_max
        if face_type == 'y_min': return self.center_y_min
        if face_type == 'y_max': return self.center_y_max
        if face_type == 'z_min': return self.center_z_min
        if face_type == 'z_max': return self.center_z_max
        return self.center_world

    def radial_point_world(self, direction, radius):
        """Get a point on the surface in a given direction."""
        # Simplified: just return center + direction * radius
        return self.center_world + direction.normalized() * radius

def boolean_subtract(cutter_name, target_name):
    """Perform boolean subtract using modifier stack."""
    target = bpy.data.objects.get(target_name)
    cutter = bpy.data.objects.get(cutter_name)
    
    if not target or not cutter:
        print(f"Error: Object {target_name} or {cutter_name} not found.")
        return

    # Ensure cutter is a mesh
    if cutter.type != 'MESH':
        bpy.ops.object.convert(target='MESH')
        cutter = bpy.context.active_object
        cutter.name = cutter_name # Re-assign name if changed

    # Add Boolean Modifier to Target
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove cutter
    bpy.data.objects.remove(cutter, do_unlink=True)

def boolean_union(union_obj_name, target_name):
    """Perform boolean union."""
    target = bpy.data.objects.get(target_name)
    union_obj = bpy.data.objects.get(union_obj_name)
    
    if not target or not union_obj:
        print(f"Error: Object {target_name} or {union_obj_name} not found.")
        return

    if union_obj.type != 'MESH':
        bpy.ops.object.convert(target='MESH')
        union_obj = bpy.context.active_object
        union_obj.name = union_obj_name

    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = union_obj
    
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    bpy.data.objects.remove(union_obj, do_unlink=True)

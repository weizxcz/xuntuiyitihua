def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def average(points):
    """Calculate the average of a list of vectors."""
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    """Helper class to calculate positions based on object geometry."""
    def __init__(self, obj):
        self.obj = obj
        self.bound_box = obj.bound_box
        # Convert local bound box to world space
        self.world_box = [obj.matrix_world @ Vector((v[0], v[1], v[2])) for v in self.bound_box]
        
    def center_world(self):
        """Return the world center of the object."""
        return average(self.world_box)
    
    def face_center_world(self, face='XY'):
        """Return the center of a specific face in world space."""
        # Simplified: assumes axis-aligned bounding box for simplicity in this generator
        # Find min/max for each axis
        xs = [v.x for v in self.world_box]
        ys = [v.y for v in self.world_box]
        zs = [v.z for v in self.world_box]
        
        cx = sum(xs) / 2
        cy = sum(ys) / 2
        cz = sum(zs) / 2
        
        if face == 'XY':
            # Top face (max Z) or Bottom face (min Z)? 
            # Convention: XY_face usually implies the face normal to Z.
            # Let's return the center of the top face (max Z) by default for "XY_face" alignment context
            # unless specified otherwise. However, "Align XY_face to origin" usually means center the projection.
            # Let's return the center of the face at max Z.
            return Vector((cx, cy, max(zs)))
        elif face == 'XZ':
            return Vector((cx, max(ys), cz))
        elif face == 'YZ':
            return Vector((max(xs), cy, cz))
        return Vector((cx, cy, cz))

    def radial_point_world(self, direction='X'):
        """Return a point on the radial edge."""
        xs = [v.x for v in self.world_box]
        ys = [v.y for v in self.world_box]
        zs = [v.z for v in self.world_box]
        
        cx = sum(xs) / 2
        cy = sum(ys) / 2
        cz = sum(zs) / 2
        
        if direction == 'X':
            return Vector((max(xs), cy, cz))
        elif direction == 'Y':
            return Vector((cx, max(ys), cz))
        elif direction == 'Z':
            return Vector((cx, cy, max(zs)))
        return Vector((cx, cy, cz))

def get_face_center(obj, face_type='XY', side='top'):
    """
    Helper to get the center of a specific face for alignment.
    face_type: 'XY', 'XZ', 'YZ'
    side: 'top' (max), 'bottom' (min), 'front', 'back', 'left', 'right'
    """
    xs = [v.x for v in obj.bound_box]
    ys = [v.y for v in obj.bound_box]
    zs = [v.z for v in obj.bound_box]
    
    cx = sum(xs) / 2
    cy = sum(ys) / 2
    cz = sum(zs) / 2
    
    if face_type == 'XY':
        if side == 'top':
            return Vector((cx, cy, max(zs)))
        elif side == 'bottom':
            return Vector((cx, cy, min(zs)))
    elif face_type == 'XZ':
        if side == 'front': # +Y
            return Vector((cx, max(ys), cz))
        elif side == 'back': # -Y
            return Vector((cx, min(ys), cz))
    elif face_type == 'YZ':
        if side == 'right': # +X
            return Vector((max(xs), cy, cz))
        elif side == 'left': # -X
            return Vector((min(xs), cy, cz))
            
    return Vector((cx, cy, cz))

def align_and_offset(obj, align_to_obj, align_face_type='XY', align_side='bottom', 
                     offset_obj, offset_vector):
    """
    Aligns obj's specified face to align_to_obj's specified face,
    then applies an offset relative to the offset_obj's local axes.
    """
    # 1. Align
    # Get target face center
    target_center = get_face_center(align_to_obj, align_face_type, align_side)
    
    # Get source face center
    source_center = get_face_center(obj, align_face_type, align_side)
    
    # Calculate translation to align faces
    # We want source_center to move to target_center
    # But we also need to consider the offset direction.
    # The prompt says: "Align ... Then move by offset".
    # So first, snap the face to the other face.
    
    # Current location of obj
    current_loc = obj.location
    
    # Vector from source center to target center
    align_vec = target_center - source_center
    
    # Apply alignment
    obj.location += align_vec
    
    # 2. Offset
    # The offset is in reference local axes of the offset_obj.
    # We need to transform the offset_vector from offset_obj's local space to world space.
    if offset_obj:
        # Get the rotation matrix of the offset object
        rot_matrix = offset_obj.matrix_world.to_3x3()
        # Transform offset vector
        world_offset = rot_matrix @ offset_vector
        obj.location += world_offset
    else:
        # If no offset object, assume world space? 
        # Prompt says "offset is in reference local axes". 
        # If no object provided, we might assume the aligned object's local axes or world.
        # Given the examples, offset_obj is often the base or the aligned object.
        # Let's assume world space if no specific offset object is passed, 
        # but usually, we pass the object we just aligned or a base.
        # For this script, I will pass the object being aligned as the reference for local axes if not specified.
        # However, the prompt implies "offset(0,0,0.010)" is relative to the object's orientation.
        # Since all objects are aligned to Z-up, local Z is world Z.
        obj.location += offset_vector

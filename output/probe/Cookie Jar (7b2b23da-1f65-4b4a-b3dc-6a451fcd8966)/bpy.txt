def make_material(name, diffuse_color):
    """Create a Principled BSDF material with specified diffuse color."""
    if name in bpy.data.materials:
        mat = bpy.data.materials[name]
    else:
        mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (*diffuse_color[:3], 1.0)
    # Set roughness to 1.0 for pure diffuse look if desired, default is 0.4
    bsdf.inputs['Roughness'].default_value = 1.0
    return mat

def add_cylinder(name, radius, depth, segments=32):
    """Add a cylinder primitive, centered at origin, aligned to Z."""
    bpy.ops.mesh.primitive_cylinder_add(vertices=segments, radius=radius, depth=depth, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.data.name = name + "_data"
    return obj

def add_cube(name, size):

    bpy.ops.mesh.primitive_cube_add(size=size, align='WORLD', location=(0, 0, 0))
    obj = bpy.context.active_object
    obj.name = name
    obj.data.name = name + "_data"
    return obj

def add_bevel(obj, width, segments=2, profile=0.5, limit_method='ANGLE'):

    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.profile = profile
    mod.limit_method = limit_method
    mod.angle_limit = math.radians(45) # Default angle limit

def boolean_subtract(target_obj, cutter_obj, keep_target=True):
    """
    Perform a boolean subtract operation.
    target_obj: The object to keep (the one being cut).
    cutter_obj: The object to use as the cutter.
    """
    # Ensure target has a boolean modifier
    target_mod = target_obj.modifiers.new(name="BooleanSub", type='BOOLEAN')
    target_mod.operation = 'DIFFERENCE'
    target_mod.object = cutter_obj
    
    # Apply modifier
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=target_mod.name)
    
    # Remove cutter object from scene
    if cutter_obj in bpy.context.scene.objects:
        bpy.data.objects.remove(cutter_obj, do_unlink=True)
        
    return target_obj

def boolean_union(target_obj, cutter_obj):
    """
    Perform a boolean union operation.
    """
    target_mod = target_obj.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    target_mod.operation = 'UNION'
    target_mod.object = cutter_obj
    
    bpy.context.view_layer.objects.active = target_obj
    bpy.ops.object.modifier_apply(modifier=target_mod.name)
    
    if cutter_obj in bpy.context.scene.objects:
        bpy.data.objects.remove(cutter_obj, do_unlink=True)
        
    return target_obj

def align_axis_to_vector(obj, axis='Z', vector=(0, 0, 1)):
    """
    Rotate object so its local axis aligns with the given world vector.
    axis: 'X', 'Y', or 'Z'
    vector: Target world direction vector
    """
    vec = mathutils.Vector(vector).normalized()
    local_axis = mathutils.Vector([1 if axis == 'X' else 0 if axis == 'Y' else 0,
                                   0 if axis == 'X' else 1 if axis == 'Y' else 0,
                                   0 if axis == 'X' else 0 if axis == 'Y' else 1])
    
    # Calculate rotation matrix to align local_axis to vec
    rot_mat = local_axis.rotation_difference(vec).to_matrix().to_4x4()
    
    # Apply rotation
    obj.rotation_euler = rot_mat.to_euler()

def average(points):
    """Calculate the average position of a list of vectors."""
    if not points:
        return Vector((0, 0, 0))
    avg = Vector((0, 0, 0))
    for p in points:
        avg += p
    return avg / len(points)

def get_face_center(obj, face_type='bottom'):
    """
    Calculate the world center of a specific face for a primitive object.
    Assumes standard orientation (Z-up for cylinders/spheres).
    """
    obj_mat = obj.matrix_world
    loc = obj_mat.to_translation()
    rot = obj_mat.to_euler()
    
    # If object is rotated, we need to transform the local face vector.
    # However, the prompt implies standard Z-up orientation for creation.
    # Let's assume the object is aligned to Z as per instructions.
    
    if obj.type == 'MESH':
        # Get bounding box to find extents
        # Or use geometry if available. 
        # Simple approach: Use object dimensions.
        dim = obj.dimensions
        # Local face centers
        if face_type == 'bottom':
            local_center = Vector((0, 0, -dim[2]/2))
        elif face_type == 'top':
            local_center = Vector((0, 0, dim[2]/2))
        elif face_type == 'center':
            local_center = Vector((0, 0, 0))
        else:
            local_center = Vector((0, 0, 0))
            
        # Transform to world
        world_center = obj_mat @ local_center
        return world_center
    return loc

def anchor_to_face(target_obj, anchor_obj, anchor_face_type):
    """
    Anchor target_obj to anchor_obj's face.
    Moves target_obj so its bottom face (default) touches the anchor face.
    """
    # 1. Get the world position of the anchor face
    anchor_face_pos = get_face_center(anchor_obj, anchor_face_type)
    
    # 2. Get the local bottom face center of the target object
    # We need to move the target so its bottom face is at anchor_face_pos
    target_dim = target_obj.dimensions
    # Local bottom center relative to object origin
    local_bottom_center = Vector((0, 0, -target_dim[2]/2))
    
    # 3. Calculate the offset needed
    # Current world position of target's bottom center
    current_target_bottom = target_obj.matrix_world @ local_bottom_center
    
    # Difference vector
    diff = anchor_face_pos - current_target_bottom
    
    # 4. Apply translation
    target_obj.location += diff

def apply_material(obj, mat_name):
    """Apply a material to an object."""
    if mat_name in bpy.data.materials:
        mat = bpy.data.materials[mat_name]
        if obj.data.materials:
            obj.data.materials[0] = mat
        else:
            obj.data.materials.append(mat)
    else:
        print(f"Material {mat_name} not found!")

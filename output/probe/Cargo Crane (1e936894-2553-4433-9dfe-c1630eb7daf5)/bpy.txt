import bpy

from mathutils import Vector, Matrix

# --- Helper Functions ---

def make_material(name, diffuse_color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    # size is (x,y,z)
    half_size = (size[0]/2, size[1]/2, size[2]/2)
    bpy.ops.mesh.primitive_cube_add(size=1, location=location, rotation=rotation)
    obj = bpy.context.object
    obj.name = name
    obj.scale = half_size
    if material:
        obj.data.materials.append(material)
    return obj

def add_cone(name, radius_bottom, radius_top, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    bpy.ops.mesh.primitive_cone_add(radius1=radius_bottom, radius2=radius_top, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_uvsphere(name, radius, location=(0,0,0), rotation=(0,0,0), material=None):

    obj = bpy.context.object
    obj.name = name

def add_bevel(obj, width=0.05, segments=2):
    # Add bevel modifier
    mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = width
    mod.segments = segments
    mod.limit_method = 'ANGLE'
    # Apply immediately for simplicity in this script context
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=mod.name)

def boolean_subtract(target, cutter):
    # Add boolean modifier to target using cutter
    mod = target.modifiers.new(name="BooleanSub", type='BOOLEAN')
    mod.operation = 'DIFFERENCE'
    mod.object = cutter
    bpy.context.view_layer.objects.active = target
    bpy.ops.object.modifier_apply(modifier=mod.name)

def align_axis_to_vector(obj, axis='Z', vector=(0,0,1)):
    """Align object's local axis to a world vector."""
    if axis == 'Z':
        local_axis = Vector((0,0,1))
    elif axis == 'Y':
        local_axis = Vector((0,1,0))
    elif axis == 'X':
        local_axis = Vector((1,0,0))
    else:
        return

    target_vec = vector.normalized()
    if local_axis.normalized().dot(target_vec) < -0.999:
        rot = Matrix.Rotation(math.pi, 4, local_axis.cross(target_vec).normalized() if local_axis.cross(target_vec).length > 0 else Vector((1,0,0)))
    else:
        rot = Matrix.Rotation(math.acos(max(-1, min(1, local_axis.dot(target_vec)))), 4, local_axis.cross(target_vec).normalized() if local_axis.cross(target_vec).length > 0 else Vector((1,0,0)))
    
    # Convert matrix to quaternion and apply
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = rot.to_quaternion()

def average(points):
    if not points:
        return Vector((0,0,0))
    avg = Vector((0,0,0))
    for p in points:
        avg += p
    return avg / len(points)

class Locator:
    def __init__(self, obj):
        self.obj = obj
        self.update()

    def update(self):
        self.center_world = self.obj.location.copy()
        # Calculate face centers based on rotation and scale
        # Assuming standard orientation for simplicity unless rotated
        # For rotated objects, we need to transform the local face centers
        rot_mat = self.obj.matrix_world.to_3x3()
        scale = self.obj.scale
        
        # Local face centers (assuming origin at center)
        # Z+ face
        z_plus_local = Vector((0,0, self.obj.dimensions[2]/2))
        # Z- face
        z_minus_local = Vector((0,0, -self.obj.dimensions[2]/2))
        # X+ face
        x_plus_local = Vector((self.obj.dimensions[0]/2, 0, 0))
        # X- face
        x_minus_local = Vector((-self.obj.dimensions[0]/2, 0, 0))
        # Y+ face
        y_plus_local = Vector((0, self.obj.dimensions[1]/2, 0))
        # Y- face
        y_minus_local = Vector((0, -self.obj.dimensions[1]/2, 0))
        
        self.z_plus_face_world = self.center_world + (rot_mat @ z_plus_local)
        self.z_minus_face_world = self.center_world + (rot_mat @ z_minus_local)
        self.x_plus_face_world = self.center_world + (rot_mat @ x_plus_local)
        self.x_minus_face_world = self.center_world + (rot_mat @ x_minus_local)
        self.y_plus_face_world = self.center_world + (rot_mat @ y_plus_local)
        self.y_minus_face_world = self.center_world + (rot_mat @ y_minus_local)

    def center_world(self):
        return self.obj.location

    def face_center_world(self, face):
        if face == 'Z+': return self.z_plus_face_world
        if face == 'Z-': return self.z_minus_face_world
        if face == 'X+': return self.x_plus_face_world
        if face == 'X-': return self.x_minus_face_world
        if face == 'Y+': return self.y_plus_face_world
        if face == 'Y-': return self.y_minus_face_world
        return self.center_world()

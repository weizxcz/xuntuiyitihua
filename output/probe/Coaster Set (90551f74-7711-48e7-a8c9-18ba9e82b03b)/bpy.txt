import bpy

# --- Helper Functions ---

def make_material(name, diffuse_color):
    """Create a new Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = diffuse_color
    return mat

def add_cylinder(name, radius, depth, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cylinder primitive, oriented along Z by default."""
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def add_cube(name, size, location=(0,0,0), rotation=(0,0,0), material=None):
    """Add a cube primitive."""
    bpy.ops.mesh.primitive_cube_add(size=size, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    if material:
        obj.data.materials.append(material)
    return obj

def align_to_vector(obj, vector):
    """Align object's local Z axis to a world vector."""
    vec = vector.normalized()
    # Create rotation matrix that aligns Z (0,0,1) to vec
    z_axis = Vector((0, 0, 1))
    if z_axis.angle(vec) < 1e-6:
        rot_quat = Matrix.Identity(3).to_quaternion()
    else:
        rot_quat = z_axis.rotation_difference(vec)
    
    # Apply rotation
    obj.rotation_euler = rot_quat.to_euler()

def align_centers_xy(obj_target, obj_source):
    """Align XY center of obj_source to obj_target."""
    obj_source.location.x = obj_target.location.x
    obj_source.location.y = obj_target.location.y

def align_bottom_to_top_z(obj_bottom, obj_top):
    """Align bottom face of obj_bottom to top face of obj_top along Z."""
    # Calculate top Z of bottom object
    # Assuming cylinder/cube centered at 0,0,0 in local space
    # We need to move obj_bottom so its bottom Z equals obj_top's top Z
    # obj_top top Z = obj_top.location.z + (obj_top.dimensions.z / 2)
    # obj_bottom bottom Z = obj_bottom.location.z - (obj_bottom.dimensions.z / 2)
    
    top_z = obj_top.location.z + (obj_top.dimensions.z / 2)
    half_h_bottom = obj_bottom.dimensions.z / 2
    
    obj_bottom.location.z = top_z - half_h_bottom

def boolean_union(target, operand):
    """Perform boolean union of operand into target, then remove operand."""
    # Select target
    bpy.ops.object.select_all(action='DESELECT')
    target.select_set(True)
    bpy.context.view_layer.objects.active = target
    
    # Add Boolean Modifier
    mod = target.modifiers.new(name="BooleanUnion", type='BOOLEAN')
    mod.operation = 'UNION'
    mod.object = operand
    
    # Apply modifier
    bpy.ops.object.modifier_apply(modifier=mod.name)
    
    # Remove operand
    bpy.data.objects.remove(operand, do_unlink=True)

def create_locator(name, location):
    """Create a simple empty locator."""
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=location)
    loc = bpy.context.active_object
    loc.name = name
    return loc

def get_world_center(obj):
    """Get the world location of an object's center."""
    return obj.matrix_world.to_translation()

def get_face_center_world(obj, face_normal_axis='Z'):
    """Get world location of the face defined by normal axis (e.g., +Z is top)."""
    # For primitives centered at origin, this is just center + half_dim * axis
    # But we must use world matrix
    half_dim = obj.dimensions / 2.0
    local_center = Vector((0,0,0))
    
    if face_normal_axis == 'Z':
        local_offset = Vector((0, 0, half_dim.z))
    elif face_normal_axis == '-Z':
        local_offset = Vector((0, 0, -half_dim.z))
    elif face_normal_axis == 'X':
        local_offset = Vector((half_dim.x, 0, 0))
    elif face_normal_axis == '-X':
        local_offset = Vector((-half_dim.x, 0, 0))
    elif face_normal_axis == 'Y':
        local_offset = Vector((0, half_dim.y, 0))
    elif face_normal_axis == '-Y':
        local_offset = Vector((0, -half_dim.y, 0))
    else:
        local_offset = Vector((0,0,0))
        
    return obj.matrix_world @ (local_center + local_offset)

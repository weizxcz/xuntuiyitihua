## 创建当前段下的三角面

**类型：** 建模命令

**说明：** 在当前段下创建三角面

**示例：**

```python
ptList = [NCTI.Point(0,0,0), NCTI.Point(10,0,0), NCTI.Point(10,20,0)]
normalList = [NCTI.Vector(0,0,1), NCTI.Vector(0,0,1), NCTI.Vector(0,0,1)]
indexList = [0, 1, 2]
seg = doc.Scene()
segChild = seg.Subsegment("Axis")
segChild.InsertTriangle(ptList, indexList, normalList)
```

## 插入多段线

**类型：** 建模命令

**说明：** 在当前段下创建多段线

**示例：**

```python
doc.Clear()
doc.ResetCaseResult()
doc.RunCommand("cmd_ncti_create_cylinder", "c1", 5, 8)
sel = NCTI.SelectionManager(doc)
sel.ObjectNames = ["c1"]
sel.SetSelected()
box = doc.GetBoundingBox(sel.ObjectNames)
if len(box) > 5:
    pt1 = NCTI.Point(box[0] , box[1],  box[2])
    pt2 = NCTI.Point(box[0] , box[1],  box[5])
    pt3 = NCTI.Point(box[0] , box[4],  box[5])
    pt4 = NCTI.Point(box[0] , box[4],  box[2])
    pt5 = NCTI.Point(box[3] , box[4],  box[2])
    pt6 = NCTI.Point(box[3] , box[4],  box[5])
    pt7 = NCTI.Point(box[3] , box[1],  box[5])
    pt8 = NCTI.Point(box[3] , box[1],  box[2])

    seg = doc.Scene()
    segChild = seg.Subsegment("Axis")
    #设置多段线的颜色，此处设置为绿色
    seg.SetPolylineColor(NCTI.Color(0,1,0,1))
    pts1 = [pt1, pt2, pt3, pt4, pt1]
    pts2 = [pt5, pt6, pt7, pt8, pt5]
    pts3 = [pt1, pt8]
    pts4 = [pt2, pt7]
    pts5 = [pt3, pt6]
    pts6 = [pt4, pt5]
    #在当前段下创建多段线
    segChild.InsertPolyline(pts1)
    segChild.InsertPolyline(pts2)
    segChild.InsertPolyline(pts3)
    segChild.InsertPolyline(pts4)
    segChild.InsertPolyline(pts5)
    segChild.InsertPolyline(pts6)
```

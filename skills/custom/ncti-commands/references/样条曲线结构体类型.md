## 创建样条曲线所需的结构体

**类型：** 约束命令

**说明：** 创建BSData对象，用于创建约束样条曲线所需的结构体

**示例：**

```python
import math
PI = 3.141592653589793238462643e0
np = 10
degree = 3
ptVec = []
vertexVec = []
dcmPtVec = []
weights = []
doc.RunCommand("cmd_ncti_create_dim_sys", "dcmsys1")
for i in range(np):
    pt = NCTI.Point(i * 20, 20 * math.sin(3.5 * PI * i / np), 0.0)
    ptName = "pt" + str(i)
    doc.RunCommand("cmd_ncti_create_vertex", ptName, pt)
    ptVec.append(pt)
    vertexVec.append(ptName)
    dcmPtName = "dcmPt" + str(i)
    doc.RunCommand("cmd_ncti_create_dcm_point", dcmPtName, "dcmsys1", pt, ptName)
    dcmPtVec.append(dcmPtName)
    weights.append(1.0)
kv = 0
knot_vector = []
for ii in range(degree):
    knot_vector.append(kv)
for ii in range(degree, np):
    knot_vector.append(kv)
    kv += 1
for ii in range(np, np + degree + 1):
    knot_vector.append(kv)
bs_data = NCTI.BSData()
bs_data.data_mask = [DCM_BS_RIGIDITY, DCM_BS_PERIODICITY, DCM_BS_RATIONALITY, DCM_BS_DEGREE, DCM_BS_CP_N,DCM_BS_KNOT_VECTOR, DCM_BS_CP_G_NODES]
bs_data.rigidity = DCM_BS_RIGIDITY_FLEXIBLE
bs_data.periodicity = DCM_BS_PERIODICITY_NON_PER
bs_data.rationality = DCM_BS_RATIONALITY_NON_RAT
bs_data.degree = degree
bs_data.cp_g_nodes = dcmPtVec
bs_data.knot_vector = knot_vector
doc.RunCommand("cmd_ncti_bspline_curve", "bs", 3, [0,1,2,3,4,5,6,7], [4,1,1,1,1,1,1,4], ptVec,weights)
doc.RunCommand("cmd_ncti_create_dcm_spline","dcmBs","dcmsys1", dcmPtVec, DCM_BS_PERIODICITY_NON_PER, [], bs_data,DCM_BS_STATUS_NOT_SOLVED,"bs")
```

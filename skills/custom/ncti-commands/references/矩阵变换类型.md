## 倍增变换

**说明：** 倍增变换

**示例：**

```python
pt = NCTI.Point(0, 0, 0)
vec = NCTI.Vector(0, 0, 1)

matrix = NCTI.Transform(vec)
print(matrix)
t1 = matrix.Inverse()
print(t1)
t2 = matrix.MultiplyMe(t1)

```

## 创建3D样条曲线所需的结构体

**类型：** 基础命令

**说明：** 创建3D样条曲线所需的结构体

**示例：**

```python
import math
PI = 3.141592653589793238462643e0
np = 10
degree = 3
ptVec = []
vertexVec = []
dcmPtVec = []
weights = []

doc.RunCommand("cmd_ncti_create_dcm3_dim_sys", "dcmsys")

for i in range(np):
    pt = NCTI.Point(i * 20, 20 * math.sin(3.5 * PI * i / np), 0.0)
    ptName = "pt" + str(i)
    doc.RunCommand("cmd_ncti_create_vertex", ptName, pt)
    ptVec.append(pt)
    vertexVec.append(ptName)

    dcmPtName = "dcm_pt" + str(i)
    doc.RunCommand("cmd_ncti_create_dcm3_point", dcmPtName, "dcmsys", pt, ptName)
    dcmPtVec.append(dcmPtName)

    weights.append(1.0)

knot_vector = []
knots = []
multiplicities = []
kv = 0
for ii in range(degree + 1):
    knot_vector.append(0)
    kv+=1
knots.append(0)
multiplicities.append(degree + 1)
for ii in range(degree + 1, np):
    knot_vector.append(kv)
    knots.append(kv)
    kv += 1
    multiplicities.append(1)
for ii in range(np, np + degree + 1):
    knot_vector.append(kv + degree)
knots.append(kv + degree)
multiplicities.append(4)

bs_data = NCTI.BSData3D()
bs_data.data_mask = [DCM3_BS_RIGIDITY, DCM3_BS_PERIODICITY, DCM3_BS_RATIONALITY, DCM3_BS_DEGREE,
                     DCM3_BS_CP_N, DCM3_BS_KNOT_VECTOR, DCM3_BS_CP_G_NODES]
bs_data.rigidity = DCM3_BS_RIGIDITY_FLEXIBLE
bs_data.periodicity = DCM3_BS_PERIODICITY_NON_PER
bs_data.rationality = DCM3_BS_RATIONALITY_NON_RAT
bs_data.degree = degree
bs_data.cp_g_nodes = dcmPtVec
bs_data.knot_vector = knot_vector

doc.RunCommand("cmd_ncti_bspline_curve", "bs", degree, knots, multiplicities, ptVec, weights)
doc.RunCommand("cmd_ncti_create_dcm3_spline_curve", "dcm_bs", "dcmsys", dcmPtVec, bs_data, DCM3_BS_PERIODICITY_NON_PER, degree,"bs")

#无效数据掩码：
print(bs_data.bad_data_mask)
#控制点节点：
print(bs_data.cp_g_nodes)
#控制点向量：
print(bs_data.cp_vectors)
#控制点权重：
print(bs_data.cp_weights)
#数据有效掩码：
print(bs_data.data_mask)
#曲线次数：
print(bs_data.degree)
#依赖关系：
print(bs_data.dependence)
#插值持续时间：
print(bs_data.interp_durations)
#插值点节点：
print(bs_data.interp_g_nodes)
#插值点数量：
print(bs_data.interp_n)
#插值参数：
print(bs_data.interp_parameters)
#插值类型：
print(bs_data.interp_types)
#插值点向量：
print(bs_data.interp_vectors)
#节点向量：
print(bs_data.knot_vector)
#参数化方式：
print(bs_data.parameterisation)
#周期性标志：
print(bs_data.periodicity)
#有理标志：
print(bs_data.rationality)
#刚度系数：
print(bs_data.rigidity)
#求解模式：
print(bs_data.solving_mode)
```

print(t4)

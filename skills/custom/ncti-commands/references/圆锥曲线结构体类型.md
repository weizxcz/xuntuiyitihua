## 创建圆锥曲线所需的结构体

**类型：** 约束命令

**说明：** 根据COData对象，用于创建圆锥曲线约束对象所需的结构体

**示例：**

```python
cp_vec = []
cp_vec.append(0.01)
cp_vec.append(1.01)
cp_vec.append(0.51)
cp_vec.append(1.51)
cp_vec.append(1.01)
cp_vec.append(0.01)
doc.RunCommand("cmd_ncti_create_dim_sys", "dcmsys")
dim = 2;
cp_n = 3;
dcmpt_list = []
bver_list = []
for i in range(cp_n):
    print(cp_vec[i*dim])
    pt = NCTI.Point(cp_vec[i*dim], cp_vec[i*dim + 1], 0)
    doc.RunCommand("cmd_ncti_create_vertex", f"bver{i+1}", pt)
    bver_list.append(f"bver{i+1}")
    doc.RunCommand("cmd_ncti_create_dcm_point",f"dcmpt{i+1}","dcmsys", pt, f"bver{i+1}")
    dcmpt_list.append(f"dcmpt{i+1}")
rho = 0.4
codata = NCTI.COData()
codata.data_mask = [DCM_CONIC_TYPE, DCM_CONIC_RIGIDITY, DCM_CONIC_BEZIER_RHO, DCM_CONIC_CP_N, DCM_CONIC_CP_G_NODES]
codata.type = DCM_CONIC_TYPE_BEZIER
codata.rigidity = DCM_CONIC_RIGIDITY_FLEXIBLE
codata.bezier_rho = rho
codata.cp_g_nodes = dcmpt_list
doc.RunCommand("cmd_ncti_create_circle","bcir", NCTI.Point(0,0,0), 1)
doc.RunCommand("cmd_ncti_create_dcm_conic", "dcmconic", "dcmsys", dcmpt_list, rho, codata,DCM_CONIC_STATUS_OK, "bcir")
```

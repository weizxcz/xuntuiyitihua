# 中点约束结构体类型参考文档

共 3 个命令

---

## 定义中点约束结构体对象

**类型：** 约束命令

**说明：** 定义中点约束结构体对象

**示例：**

```python
pt = NCTI.Point(0,0,0)
vn = NCTI.Vector(0,0,1)
vu = NCTI.Vector(1,0,0)
maj_rad = 10.0
min_rad = 5.0
doc.RunCommand("cmd_ncti_create_ellipse", "Bellipse", pt, vn, vu, maj_rad, min_rad)

doc.RunCommand("cmd_ncti_create_dim_sys", "dcmsys")
doc.RunCommand("cmd_ncti_create_dcm_ellipse", "Dellipse", "dcmsys", pt, vu, maj_rad, min_rad,"Bellipse")

doc.RunCommand("cmd_ncti_create_dcm_dimension", "dimLen", DCM_CURVE_LENGTH, 10)
doc.RunCommand("cmd_ncti_edit_dcm_dim_help_parameter", "dimLen", 0, 3.14)
adddData = NCTI.AdddData()
adddData.m_GNodeNameOne = "Dellipse"
adddData.m_GNodeNameTwo = "Dellipse"
adddData.m_DNodeName = "len_node" #定义d_node名称
doc.RunCommand("cmd_ncti_dcm_addd", "dcmsys", "dimLen", adddData)

#fix parameter node
doc.RunCommand("cmd_ncti_dcm_parameter_node","dcmsys", "dimLen", "len_node", DCM_CONSTRAINT_END_ONE, "p_node1")
doc.RunCommand("cmd_ncti_dcm_parameter_node","dcmsys", "dimLen", "len_node", DCM_CONSTRAINT_END_TWO, "p_node2")
doc.RunCommand("cmd_ncti_dcm_fix", "dcmsys", "p_node1", 1)
doc.RunCommand("cmd_ncti_dcm_fix", "dcmsys", "p_node2", 1)

doc.RunCommand("cmd_ncti_dcm_evaluate", "dcmsys")
params = doc.dcm_help_parameters("dcmsys", "len_node")
print(params)
```

## 设置约束参数

**类型：** 约束命令

**说明：** 设置约束参数

**示例：**

```python
pt1 = NCTI.Point(0,0,0)
pt2 = NCTI.Point(0,1,0)
pt3 = NCTI.Point(1,2,0)
vec = NCTI.Vector(10,0,0)
doc.RunCommand("cmd_ncti_create_line","bline1", pt1, vec)
doc.RunCommand("cmd_ncti_create_vertex","bver1", pt2)
doc.RunCommand("cmd_ncti_create_vertex","bver2", pt3)
doc.RunCommand("cmd_ncti_create_dim_sys", "dcmsys1")
doc.RunCommand("cmd_ncti_create_dcm_line", "dcmline1", "dcmsys1", pt1, vec, "bline1")
doc.RunCommand("cmd_ncti_create_dcm_point", "dcmpt1", "dcmsys1", pt2, "bver1")
doc.RunCommand("cmd_ncti_create_dcm_point", "dcmpt2", "dcmsys1", pt3, "bver2")
doc.RunCommand("cmd_ncti_create_dcm_dimension", "dcmeqdist", DCM_EQUAL_DISTANCE)
adata = NCTI.AdddData()
adata.m_GNodeNameOne = "dcmline1"
adata.m_GNodeNameTwo = "dcmpt1"
adata.m_GNodeNameThree = "dcmline1" #非必须
adata.m_GNodeNameFour = "dcmpt2"#非必须
adata.m_DNodeName = "d_eqdist" #执行cmd_ncti_dcm_addd命令后，约束尺寸对象dcmeqdist会生产一个d_node对象，该名称为生产的d_node对象的名称
doc.RunCommand("cmd_ncti_dcm_addd", "dcmsys1",  "dcmeqdist", adata)
doc.RunCommand("cmd_ncti_dcm_evaluate", "dcmsys1")
```
